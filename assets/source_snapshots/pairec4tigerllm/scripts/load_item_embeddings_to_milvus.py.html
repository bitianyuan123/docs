1#!/usr/bin/env python3
2# scripts/load_item_embeddings_to_milvus.py
3#
4# 把 DSSM 导出的 item 向量灌入 Milvus.
5#
6# 用法:
7#   python3 scripts/load_item_embeddings_to_milvus.py \
8#     --milvus_host 127.0.0.1 --milvus_port 19530 \
9#     --export_dir checkpoints/dssm/export \
10#     --collection dssm_item_vectors [--drop]
11#
12# 需要 pymilvus: pip install pymilvus
13
14import argparse
15import json
16import os
17
18import numpy as np
19from pymilvus import (Collection, CollectionSchema, DataType, FieldSchema,
20                      connections, utility)
21
22
23def parse_args():
24    p = argparse.ArgumentParser()
25    p.add_argument("--milvus_host", default="127.0.0.1")
26    p.add_argument("--milvus_port", default="19530")
27    p.add_argument("--export_dir", default="checkpoints/dssm/export")
28    p.add_argument("--collection", default="dssm_item_vectors")
29    p.add_argument("--batch_size", type=int, default=5000)
30    p.add_argument("--nlist", type=int, default=1024)
31    p.add_argument("--drop", action="store_true", help="先删除已有同名 collection")
32    return p.parse_args()
33
34
35def main():
36    args = parse_args()
37    connections.connect(host=args.milvus_host, port=args.milvus_port)
38
39    if args.drop and utility.has_collection(args.collection):
40        utility.drop_collection(args.collection)
41        print(f"[milvus] dropped existing collection {args.collection}")
42
43    vectors = np.load(os.path.join(args.export_dir, "item_vectors.npy"))
44    with open(os.path.join(args.export_dir, "item_ids.json")) as f:
45        item_ids = json.load(f)
46    assert len(item_ids) == vectors.shape[0], "item_ids 与 vectors 行数不一致"
47    dim = vectors.shape[1]
48    print(f"[milvus] {len(item_ids)} vectors, dim={dim}")
49
50    if utility.has_collection(args.collection):
51        collection = Collection(args.collection)
52    else:
53        fields = [
54            FieldSchema(name="item_id", dtype=DataType.INT64, is_primary=True,
55                        auto_id=False),
56            FieldSchema(name="vector", dtype=DataType.FLOAT_VECTOR, dim=dim),
57        ]
58        schema = CollectionSchema(fields, description="DSSM item vectors")
59        collection = Collection(args.collection, schema)
60        print(f"[milvus] created collection {args.collection}")
61
62    for start in range(0, len(item_ids), args.batch_size):
63        ids_part = item_ids[start:start + args.batch_size]
64        vec_part = vectors[start:start + args.batch_size].tolist()
65        collection.insert([ids_part, vec_part])
66        if (start // args.batch_size) % 20 == 0:
67            print(f"[milvus] inserted {start + len(ids_part)}/{len(item_ids)}")
68    collection.flush()
69    print("[milvus] insert done, creating index...")
70
71    collection.create_index(
72        field_name="vector",
73        index_params={"index_type": "IVF_FLAT", "metric_type": "IP",
74                      "params": {"nlist": args.nlist}})
75    collection.load()
76    print(f"[milvus] index created, entities={collection.num_entities}")
77
78    # smoke: 用第 0 条向量自查询, 第一名应该是它自己
79    if len(item_ids) > 0:
80        res = collection.search(
81            data=[vectors[0].tolist()], anns_field="vector",
82            param={"metric_type": "IP", "params": {"nprobe": 16}},
83            limit=3, output_fields=["item_id"])
84        top = [(h.entity.get("item_id"), round(float(h.distance), 4))
85               for h in res[0]]
86        print(f"[milvus] self-search smoke item_id={item_ids[0]} top3={top}")
87        assert top and top[0][0] == item_ids[0], "self-search 第一名不是自身, 灌库异常"
88        print("[milvus] PASS")
89
90
91if __name__ == "__main__":
92    main()
