289std::string Trim(const std::string& input) {
290  size_t begin = 0;
291  while (begin < input.size() &&
292         std::isspace(static_cast<unsigned char>(input[begin]))) {
293    ++begin;
294  }
295  size_t end = input.size();
296  while (end > begin &&
297         std::isspace(static_cast<unsigned char>(input[end - 1]))) {
298    --end;
299  }
300  return input.substr(begin, end - begin);
301}

400class SemanticMapParser {
401 public:
402  explicit SemanticMapParser(std::string text) : text_(std::move(text)) {}
403
404  bool Parse(std::vector<Candidate>* candidates, std::string* error) {
405    SkipWs();
406    if (!Consume('{')) {
407      *error = "semantic map must start with object";
408      return false;
409    }
410    SkipWs();
411    if (Consume('}')) {
412      return true;
413    }
414    while (true) {
415      std::string item_id_text;
416      if (!ParseString(&item_id_text)) {
417        *error = "failed to parse item id string";
418        return false;
419      }
420      SkipWs();
421      if (!Consume(':')) {
422        *error = "expected ':' after item id";
423        return false;
424      }
425      Candidate candidate;
426      candidate.item_id = std::atoi(item_id_text.c_str());
427      if (!ParseIntArray(&candidate.semantic_id)) {
428        *error = "failed to parse semantic id array for item " + item_id_text;
429        return false;
430      }
431      if (!candidate.semantic_id.empty()) {
432        candidates->push_back(std::move(candidate));
433      }
434      SkipWs();
435      if (Consume('}')) {
436        return true;
437      }
438      if (!Consume(',')) {
439        *error = "expected ',' or '}' after semantic id array";
440        return false;
441      }
442      SkipWs();
443    }
444  }
445
446 private:
447  void SkipWs() {
448    while (pos_ < text_.size() &&
449           std::isspace(static_cast<unsigned char>(text_[pos_]))) {
450      ++pos_;
451    }
452  }
453
454  bool Consume(char expected) {
455    SkipWs();
456    if (pos_ >= text_.size() || text_[pos_] != expected) {
457      return false;
458    }
459    ++pos_;
460    return true;
461  }
462
463  bool ParseString(std::string* output) {
464    SkipWs();
465    if (pos_ >= text_.size() || text_[pos_] != '"') {
466      return false;
467    }
468    ++pos_;
469    output->clear();
470    while (pos_ < text_.size()) {
471      char ch = text_[pos_++];
472      if (ch == '"') {
473        return true;
474      }
475      if (ch == '\\') {
476        if (pos_ >= text_.size()) {
477          return false;
478        }
479        ch = text_[pos_++];
480      }
481      output->push_back(ch);
482    }
483    return false;
484  }
485
486  bool ParseInt(int* output) {
487    SkipWs();
488    bool negative = false;
489    if (pos_ < text_.size() && text_[pos_] == '-') {
490      negative = true;
491      ++pos_;
492    }
493    if (pos_ >= text_.size() ||
494        !std::isdigit(static_cast<unsigned char>(text_[pos_]))) {
495      return false;
496    }
497    int value = 0;
498    while (pos_ < text_.size() &&
499           std::isdigit(static_cast<unsigned char>(text_[pos_]))) {
500      value = value * 10 + (text_[pos_] - '0');
501      ++pos_;
502    }
503    *output = negative ? -value : value;
504    return true;
505  }
506
507  bool ParseIntArray(std::vector<int>* values) {
508    values->clear();
509    if (!Consume('[')) {
510      return false;
511    }
512    SkipWs();
513    if (Consume(']')) {
514      return true;
515    }
516    while (true) {
517      int value = 0;
518      if (!ParseInt(&value)) {
519        return false;
520      }
521      values->push_back(value);
522      SkipWs();
523      if (Consume(']')) {
524        return true;
525      }
526      if (!Consume(',')) {
527        return false;
528      }
529    }
530  }
531
532  std::string text_;
533  size_t pos_ = 0;
534};

562struct CppTokenizerConfig {
563  int eos_id = -1;
564  int pad_id = -1;
565  std::vector<int> prefix_tokens;
566  std::vector<int> separator_tokens;
567  std::vector<int> suffix_tokens;
568  std::vector<std::vector<int>> semantic_token_ids;
569  std::unordered_map<int, std::pair<int, int>> id_to_semantic;
570
571  bool Load(const std::string& path, std::string* error) {
572    std::ifstream in(path);
573    if (!in) {
574      *error = "failed to open C++ tokenizer config: " + path;
575      return false;
576    }
577
578    semantic_token_ids.assign(4, std::vector<int>(256, -1));
579    std::string line;
580    int line_no = 0;
581    while (std::getline(in, line)) {
582      ++line_no;
583      line = Trim(line);
584      if (line.empty() || line[0] == '#') {
585        continue;
586      }
587      std::istringstream iss(line);
588      std::string key;
589      iss >> key;
590      if (key == "eos_id") {
591        iss >> eos_id;
592      } else if (key == "pad_id") {
593        iss >> pad_id;
594      } else if (key == "prefix") {
595        if (!ParseTokenList(iss, &prefix_tokens)) {
596          *error = "failed to parse prefix at line " + std::to_string(line_no);
597          return false;
598        }
599      } else if (key == "separator") {
600        if (!ParseTokenList(iss, &separator_tokens)) {
601          *error = "failed to parse separator at line " + std::to_string(line_no);
602          return false;
603        }
604      } else if (key == "suffix") {
605        if (!ParseTokenList(iss, &suffix_tokens)) {
606          *error = "failed to parse suffix at line " + std::to_string(line_no);
607          return false;
608        }
609      } else if (key == "semantic") {
610        int layer = -1;
611        int value = -1;
612        int token_id = -1;
613        iss >> layer >> value >> token_id;
614        if (layer < 0 || value < 0 || token_id < 0) {
615          *error = "invalid semantic token entry at line " + std::to_string(line_no);
616          return false;
617        }
618        if (static_cast<size_t>(layer) >= semantic_token_ids.size()) {
619          semantic_token_ids.resize(layer + 1);
620        }
621        if (static_cast<size_t>(value) >= semantic_token_ids[layer].size()) {
622          semantic_token_ids[layer].resize(value + 1, -1);
623        }
624        semantic_token_ids[layer][value] = token_id;
625        id_to_semantic[token_id] = std::make_pair(layer, value);
626      } else {
627        *error = "unknown tokenizer config key at line " +
628                 std::to_string(line_no) + ": " + key;
629        return false;
630      }
631    }
632
633    if (eos_id < 0 || pad_id < 0) {
634      *error = "tokenizer config must define eos_id and pad_id";
635      return false;
636    }
637    if (prefix_tokens.empty() || suffix_tokens.empty()) {
638      *error = "tokenizer config must define non-empty prefix and suffix";
639      return false;
640    }
641    if (semantic_token_ids.size() < 4 || id_to_semantic.empty()) {
642      *error = "tokenizer config has no semantic token mappings";
643      return false;
644    }
645    return true;
646  }
647
648 private:
649  static bool ParseTokenList(std::istringstream& iss, std::vector<int>* tokens) {
650    tokens->clear();
651    int token = 0;
652    while (iss >> token) {
653      tokens->push_back(token);
654    }
655    return !tokens->empty();
656  }

1172    std::vector<std::vector<int>> semantic_candidates = ParseOutputTokens(sampled_tokens);
1173    parse_timer.stop();
1174    trace->set_parse_combo_ms(parse_timer.m_elapsed());
1175
1176    butil::Timer map_timer;
1177    map_timer.start();
1178    FillRecommendations(request, semantic_candidates, response);
1179    map_timer.stop();
1180    trace->set_map_item_ms(map_timer.m_elapsed());
1181
1182    if (config_.trt_datasystem_mget_probe) {
1183#if defined(PAIREC_ENABLE_DATASYSTEM_KV_PROBE) && PAIREC_ENABLE_DATASYSTEM_KV_PROBE
1184      std::string kv_error;
1185      if (!RunDatasystemSetGetProbe(request, trace, &kv_error)) {
1186        response->clear_recommendations();
1187        response->set_error(kv_error);
1188        return;
1189      }
1190#else
1191      response->clear_recommendations();
1192      response->set_error(
1193          "trt_datasystem_mget_probe requires PAIREC_ENABLE_DATASYSTEM_KV_PROBE");
1194      return;
1195#endif
1196    }
1197
1198    trace->set_backend_total_ms(
1199        trace->prompt_ms() + trace->runner_generate_ms() +
1200        trace->parse_combo_ms() + trace->map_item_ms() +
1201        trace->kv_write_ms() + trace->kv_lookup_ms());

1221  bool BuildPromptTokens(
1222      const pairec::inference::RecommendRequest& request,
1223      std::vector<int>* prompt_tokens,
1224      std::string* error) const {
1225    std::vector<std::vector<int>> history;
1226    for (const auto& semantic_id : request.history()) {
1227      std::vector<int> item;
1228      bool all_zero = true;
1229      for (int value : semantic_id.value()) {
1230        item.push_back(value);
1231        if (value != 0) {
1232          all_zero = false;
1233        }
1234      }
1235      if (!all_zero && item.size() >= tokenizer_.semantic_token_ids.size()) {
1236        history.push_back(std::move(item));
1237      }
1238    }
1239    if (history.empty()) {
1240      *error = "empty request history";
1241      return false;
1242    }
1243
1244    while (true) {
1245      std::vector<int> built;
1246      std::string build_error;
1247      if (!BuildPromptTokensForHistory(history, &built, &build_error)) {
1248        *error = build_error;
1249        return false;
1250      }
1251      if (static_cast<int>(built.size()) <= config_.trt_max_input_len ||
1252          history.size() <= 1) {
1253        if (static_cast<int>(built.size()) > config_.trt_max_input_len) {
1254          built.erase(built.begin(),
1255                      built.begin() + (built.size() - config_.trt_max_input_len));
1256        }
1257        *prompt_tokens = std::move(built);
1258        return true;
1259      }
1260      history.erase(history.begin());
1261    }
1262  }

1264  bool BuildPromptTokensForHistory(
1265      const std::vector<std::vector<int>>& history,
1266      std::vector<int>* prompt_tokens,
1267      std::string* error) const {
1268    prompt_tokens->clear();
1269    prompt_tokens->insert(prompt_tokens->end(),
1270                          tokenizer_.prefix_tokens.begin(),
1271                          tokenizer_.prefix_tokens.end());
1272    for (size_t i = 0; i < history.size(); ++i) {
1273      if (i != 0) {
1274        prompt_tokens->insert(prompt_tokens->end(),
1275                              tokenizer_.separator_tokens.begin(),
1276                              tokenizer_.separator_tokens.end());
1277      }
1278      const auto& item = history[i];
1279      for (size_t layer = 0; layer < tokenizer_.semantic_token_ids.size(); ++layer) {
1280        const int value = item[layer];
1281        if (value < 0 ||
1282            static_cast<size_t>(value) >= tokenizer_.semantic_token_ids[layer].size()) {
1283          *error = "semantic id value out of tokenizer range";
1284          return false;
1285        }
1286        const int token_id = tokenizer_.semantic_token_ids[layer][value];
1287        if (token_id < 0) {
1288          *error = "missing semantic token id in C++ tokenizer config";
1289          return false;
1290        }
1291        prompt_tokens->push_back(token_id);
1292      }
1293    }
1294    prompt_tokens->insert(prompt_tokens->end(),
1295                          tokenizer_.suffix_tokens.begin(),
1296                          tokenizer_.suffix_tokens.end());
1297    return true;
1298  }

1300  bool RunExecutor(
1301      const std::vector<int>& prompt_tokens,
1302      uint64_t seed,
1303      std::optional<uint64_t> datasystem_correlation_id,
1304      std::vector<int>* output_tokens,
1305      double* elapsed_ms,
1306      ExecutorPhaseTiming* phase_timing,
1307      bool* executor_enqueued,
1308      std::string* error) {
1309    namespace texec = tensorrt_llm::executor;
1310    const auto executor_started = std::chrono::steady_clock::now();
1311    const auto request_setup_started = phase_timing
1312        ? std::chrono::steady_clock::now()
1313        : std::chrono::steady_clock::time_point{};
1314    if (phase_timing) {
1315      *phase_timing = ExecutorPhaseTiming{};
1316    }
1317    *executor_enqueued = false;
1318    try {
1319      texec::SamplingConfig sampling_config(1);
1320      if (config_.trt_top_k > 0) {
1321        sampling_config.setTopK(
1322            std::optional<texec::SizeType32>(
1323                static_cast<texec::SizeType32>(config_.trt_top_k)));
1324      }
1325      if (config_.trt_temperature > 0.0) {
1326        sampling_config.setTemperature(
1327            std::optional<texec::FloatType>(
1328                static_cast<texec::FloatType>(config_.trt_temperature)));
1329      }
1330      sampling_config.setSeed(
1331          std::optional<texec::RandomSeedType>(
1332              static_cast<texec::RandomSeedType>(seed)));
1333      texec::OutputConfig output_config(false, false, false, true);
1334      texec::Request executor_request(
1335          texec::VecTokens(prompt_tokens.begin(), prompt_tokens.end()),
1336          static_cast<texec::SizeType32>(config_.trt_max_new_tokens),
1337          false,
1338          sampling_config,
1339          output_config,
1340          std::optional<texec::SizeType32>(
1341              static_cast<texec::SizeType32>(tokenizer_.eos_id)),
1342          std::optional<texec::SizeType32>(
1343              static_cast<texec::SizeType32>(tokenizer_.pad_id)));
1344      if (datasystem_correlation_id) {
1345        executor_request.setClientId(*datasystem_correlation_id);
1346        tensorrt_llm::batch_manager::kv_cache_manager::
1347            markDataSystemRequestExecutorEnqueued(*datasystem_correlation_id);
1348      }
1349      if (phase_timing) {
1350        phase_timing->request_setup_us =
1351            std::chrono::duration_cast<std::chrono::microseconds>(
1352                std::chrono::steady_clock::now() - request_setup_started)
1353                .count();
1354      }
1355      const auto enqueue_started = phase_timing
1356          ? std::chrono::steady_clock::now()
1357          : std::chrono::steady_clock::time_point{};
1358      const auto request_id = executor_->enqueueRequest(executor_request);
1359      if (phase_timing) {
1360        phase_timing->enqueue_call_us =
1361            std::chrono::duration_cast<std::chrono::microseconds>(
1362                std::chrono::steady_clock::now() - enqueue_started)
1363                .count();
1364      }
1365      *executor_enqueued = true;
1366      const auto await_started = phase_timing
1367          ? std::chrono::steady_clock::now()
1368          : std::chrono::steady_clock::time_point{};
1369      const auto deadline = std::chrono::steady_clock::now() +
1370                            std::chrono::milliseconds(config_.trt_request_timeout_ms);
1371
1372      while (std::chrono::steady_clock::now() < deadline) {
1373        auto responses =
1374            executor_->awaitResponses(
1375                request_id,
1376                std::optional<std::chrono::milliseconds>(
1377                    std::chrono::milliseconds(10)));
1378        for (const auto& response : responses) {
1379          if (response.hasError()) {
1380            *error = "TensorRT-LLM Executor error: " + response.getErrorMsg();
1381            executor_->cancelRequest(request_id);
1382            return false;
1383          }
1384          const auto& result = response.getResult();
1385          if (result.isFinal) {
1386            const auto response_extract_started = phase_timing
1387                ? std::chrono::steady_clock::now()
1388                : std::chrono::steady_clock::time_point{};
1389            if (phase_timing) {
1390              phase_timing->await_final_us =
1391                  std::chrono::duration_cast<std::chrono::microseconds>(
1392                      response_extract_started - await_started)
1393                      .count();
1394            }
1395            output_tokens->clear();
1396            if (!result.outputTokenIds.empty()) {
1397              output_tokens->assign(result.outputTokenIds[0].begin(),
1398                                    result.outputTokenIds[0].end());
1399            }
1400            if (phase_timing) {
1401              phase_timing->response_extract_us =
1402                  std::chrono::duration_cast<std::chrono::microseconds>(
1403                      std::chrono::steady_clock::now() - response_extract_started)
1404                      .count();
1405            }
1406            const int64_t total_us =
1407                std::chrono::duration_cast<std::chrono::microseconds>(
1408                    std::chrono::steady_clock::now() - executor_started)
1409                    .count();
1410            if (phase_timing) {
1411              phase_timing->total_us = total_us;
1412            }
1413            *elapsed_ms = static_cast<double>(total_us) / 1000.0;
1414            return true;
1415          }
1416        }
1417      }
1418      executor_->cancelRequest(request_id);
1419      *error = "TensorRT-LLM Executor request timed out";
1420      return false;
1421    } catch (const std::exception& e) {
1422      *error = std::string("TensorRT-LLM Executor exception: ") + e.what();
1423      return false;
1424    }
1425  }

1427  std::vector<std::vector<int>> ParseOutputTokens(const std::vector<int>& token_ids) const {
1428    std::vector<std::vector<int>> layer_values(tokenizer_.semantic_token_ids.size());
1429    std::vector<std::unordered_set<int>> seen_values(tokenizer_.semantic_token_ids.size());
1430    for (int token_id : token_ids) {
1431      auto it = tokenizer_.id_to_semantic.find(token_id);
1432      if (it == tokenizer_.id_to_semantic.end()) {
1433        continue;
1434      }
1435      const int layer = it->second.first;
1436      const int value = it->second.second;
1437      if (layer < 0 || static_cast<size_t>(layer) >= layer_values.size()) {
1438        continue;
1439      }
1440      if (seen_values[layer].insert(value).second) {
1441        layer_values[layer].push_back(value);
1442      }
1443    }
1444
1445    for (auto& values : layer_values) {
1446      if (values.empty()) {
1447        values.push_back(0);
1448      }
1449      std::sort(values.begin(), values.end());
1450    }
1451
1452    std::vector<std::vector<int>> combinations;
1453    std::vector<int> current;
1454    BuildCartesian(layer_values, 0, &current, &combinations);
1455    return combinations;
1456  }
1457
1458  static void BuildCartesian(
1459      const std::vector<std::vector<int>>& layers,
1460      size_t layer,
1461      std::vector<int>* current,
1462      std::vector<std::vector<int>>* combinations) {
1463    if (layer == layers.size()) {
1464      combinations->push_back(*current);
1465      return;
1466    }
1467    for (int value : layers[layer]) {
1468      current->push_back(value);
1469      BuildCartesian(layers, layer + 1, current, combinations);
1470      current->pop_back();
1471    }
1472  }

1474  void FillRecommendations(
1475      const pairec::inference::RecommendRequest& request,
1476      const std::vector<std::vector<int>>& semantic_candidates,
1477      pairec::inference::RecommendResponse* response) const {
1478    std::unordered_set<std::string> history_keys;
1479    for (const auto& history : request.history()) {
1480      history_keys.insert(SemanticKey(history));
1481    }
1482    std::unordered_set<int> emitted_items;
1483    const int topk = request.has_topk() && request.topk() > 0 ? request.topk() : 10;
1484    for (const auto& semantic_id : semantic_candidates) {
1485      if (response->recommendations_size() >= topk) {
1486        break;
1487      }
1488      const std::string key = SemanticKey(semantic_id);
1489      if (history_keys.find(key) != history_keys.end()) {
1490        continue;
1491      }
1492      auto it = semantic_to_candidate_.find(key);
1493      if (it == semantic_to_candidate_.end()) {
1494        continue;
1495      }
1496      const Candidate* candidate = it->second;
1497      if (!emitted_items.insert(candidate->item_id).second) {
1498        continue;
1499      }
1500      auto* rec = response->add_recommendations();
1501      rec->set_item_id(candidate->item_id);
1502      rec->set_score(1.0);
1503      for (int value : candidate->semantic_id) {
1504        rec->add_semantic_id(value);
1505      }
1506    }
1507  }

1558  static std::string SanitizeKeyComponent(const std::string& input) {
1559    std::string out;
1560    out.reserve(std::min<size_t>(input.size(), 96));
1561    for (char ch : input) {
1562      const unsigned char c = static_cast<unsigned char>(ch);
1563      if (std::isalnum(c) || ch == '~' || ch == '!' || ch == '@' ||
1564          ch == '#' || ch == '$' || ch == '%' || ch == '^' || ch == '&' ||
1565          ch == '*' || ch == '.' || ch == '-' || ch == '_') {
1566        out.push_back(ch);
1567      } else {
1568        out.push_back('_');
1569      }
1570      if (out.size() >= 96) {
1571        break;
1572      }
1573    }
1574    if (out.empty()) {
1575      out = "empty";
1576    }
1577    return out;
1578  }

1599  bool RunDatasystemSetGetProbe(
1600      const pairec::inference::RecommendRequest& request,
1601      pairec::inference::TraceInfo* trace,
1602      std::string* error) {
1603    if (!datasystem_probe_client_) {
1604      *error = "trtllm DataSystem Set/Get probe was not initialized";
1605      return false;
1606    }
1607
1608    std::vector<std::string> keys = BuildDatasystemProbeKeys(request);
1609
1610    datasystem::SetParam set_param;
1611    set_param.writeMode = datasystem::WriteMode::NONE_L2_CACHE_EVICT;
1612    set_param.ttlSecond = static_cast<uint32_t>(std::max(0, config_.kvc_probe_ttl_sec));
1613    set_param.existence = datasystem::ExistenceOpt::NONE;
1614    set_param.cacheType = datasystem::CacheType::MEMORY;
1615
1616    std::vector<char> payload(static_cast<size_t>(config_.kvc_probe_object_bytes));
1617    const uint64_t hash = StableRequestHash(request);
1618    for (size_t i = 0; i < payload.size(); ++i) {
1619      payload[i] = static_cast<char>((i + hash) & 0xff);
1620    }
1621
1622    const uint64_t object_bytes =
1623        static_cast<uint64_t>(config_.kvc_probe_object_bytes);
1624    std::vector<uint64_t> sizes(keys.size(), object_bytes);
1625    std::vector<std::shared_ptr<datasystem::Buffer>> buffers;
1626
1627    butil::Timer create_timer;
1628    create_timer.start();
1629    datasystem::Status create_status =
1630        datasystem_probe_client_->MCreate(keys, sizes, set_param, buffers);
1631    create_timer.stop();
1632    const double create_ms = create_timer.m_elapsed();
1633    const size_t create_call_count = 1;
1634    if (create_status.IsError()) {
1635      *error = "trtllm DataSystem MCreate failed: " + create_status.ToString();
1636      return false;
1637    }
1638    if (buffers.size() != keys.size()) {
1639      *error = "trtllm DataSystem MCreate returned unexpected buffer count";
1640      return false;
1641    }
1642
1643    butil::Timer fill_timer;
1644    fill_timer.start();
1645    for (auto& buffer : buffers) {
1646      if (!buffer) {
1647        *error = "trtllm DataSystem MCreate returned null buffer";
1648        return false;
1649      }
1650      datasystem::Status copy_status =
1651          buffer->MemoryCopy(payload.data(), static_cast<uint64_t>(payload.size()));
1652      if (copy_status.IsError()) {
1653        *error = "trtllm DataSystem probe buffer copy failed: " +
1654                 copy_status.ToString();
1655        return false;
1656      }
1657    }
1658    fill_timer.stop();
1659    const double fill_ms = fill_timer.m_elapsed();
1660
1661    butil::Timer set_timer;
1662    set_timer.start();
1663    datasystem::Status set_status = datasystem_probe_client_->MSet(buffers);
1664    set_timer.stop();
1665    const double set_ms = set_timer.m_elapsed();
1666    const size_t set_call_count = 1;
1667    if (set_status.IsError()) {
1668      *error = "trtllm DataSystem MSet failed: " + set_status.ToString();
1669      return false;
1670    }
1671
1672    butil::Timer get_timer;
1673    get_timer.start();
1674    std::vector<datasystem::Optional<datasystem::Buffer>> out_buffers;
1675    datasystem::Status get_status =
1676        datasystem_probe_client_->Get(keys, out_buffers, config_.kvc_probe_timeout_ms);
1677    get_timer.stop();
1678    const double get_ms = get_timer.m_elapsed();
1679    const size_t get_call_count = 1;
1680    if (get_status.IsError()) {
1681      *error = "trtllm DataSystem MGet failed: " + get_status.ToString();
1682      return false;
1683    }
1684
1685    size_t found_count = 0;
1686    uint64_t found_bytes = 0;
1687    for (const auto& buffer : out_buffers) {
1688      if (!buffer) {
1689        continue;
1690      }
1691      ++found_count;
1692      found_bytes += static_cast<uint64_t>(buffer->GetSize());
1693    }
1694    if (found_count != keys.size()) {
1695      std::ostringstream out;
1696      out << "trtllm DataSystem MGet returned " << found_count << "/"
1697          << keys.size() << " buffers";
1698      *error = out.str();
1699      return false;
1700    }
1701
1702    const double kv_write_ms = create_ms + fill_ms + set_ms;
1703    trace->set_kv_write_ms(kv_write_ms);
1704    trace->set_kv_lookup_ms(get_ms);
1705    trace->set_kv_source("trtllm_cpp_datasystem_mset_mget");
1706
1707    std::cout << "[brpc-inference] method=TrtllmDatasystemMSetMGet"
1708              << " request_id=" << request.request_id()
1709              << " user=" << request.user_id()
1710              << " object_count=" << keys.size()
1711              << " object_bytes=" << config_.kvc_probe_object_bytes
1712              << " total_bytes=" << found_bytes
1713              << " create_call_count=" << create_call_count
1714              << " mcreate_call_count=" << create_call_count
1715              << " set_call_count=" << set_call_count
1716              << " get_call_count=" << get_call_count
1717              << " mset_call_count=" << set_call_count
1718              << " mget_call_count=" << get_call_count
1719              << " set_buffer_count=" << buffers.size()
1720              << " get_key_count=" << keys.size()
1721              << " out_buffer_count=" << out_buffers.size()
1722              << " create_ms=" << create_ms
1723              << " fill_ms=" << fill_ms
1724              << " set_ms=" << set_ms
1725              << " get_ms=" << get_ms
1726              << " found=" << found_count

1780      google::protobuf::RpcController* controller,
1781      const pairec::inference::RecommendRequest* request,
1782      pairec::inference::RecommendResponse* response,
1783      google::protobuf::Closure* done) override {
1784    brpc::ClosureGuard done_guard(done);

1792

1794    response->set_inference_time_ms(timer.m_elapsed());

1796    trace->set_total_ms(timer.m_elapsed());
1797    trace->set_infer_ms(timer.m_elapsed());
1798    trace->set_backend(backend_->Name());
1799    trace->set_request_id(request->request_id());

1801        trace->kv_source() == "datasystem_mset_mget_probe" ||
1802        trace->kv_source() == "trtllm_cpp_datasystem_mset_mget";
1803    trace->set_datasystem_expected(exact_datasystem_probe || std::getenv("DATASYSTEM_HOST") != nullptr);

1805    if (exact_datasystem_probe) {
1806      trace->set_datasystem_sync_get_count(1);
1807      trace->set_datasystem_sync_set_count(1);
1808      trace->set_datasystem_sync_get_us(static_cast<int64_t>(trace->kv_lookup_ms() * 1000.0));
1809      trace->set_datasystem_sync_set_us(static_cast<int64_t>(trace->kv_write_ms() * 1000.0));
1810      trace->set_datasystem_attribution_reason("request_correlated_explicit_probe");
1811    } else if (trace->datasystem_expected()) {
1812#if defined(PAIREC_ENABLE_TRTLLM_CPP) && PAIREC_ENABLE_TRTLLM_CPP
1813      if (tensorrt_llm::batch_manager::kv_cache_manager::
1814              dataSystemRequestAttributionEnabled()) {
1815        trace->set_datasystem_attribution_reason(
1816            "native_datasystem_completion_event_pending");
1817      } else {
1818        trace->set_datasystem_attribution_reason(
1819            "native_trt_kvc_request_identity_not_propagated");
1820      }
1821#else
1822      trace->set_datasystem_attribution_reason("native_trt_kvc_request_identity_not_propagated");
1823#endif
1824    } else {
1825      trace->set_datasystem_attribution_reason("datasystem_not_expected");
1826    }
1827
1828    if (response->recommendations_size() == 0) {
1829      response->set_code(299);
1830      if (!response->has_error() || response->error().empty()) {
1831        response->set_error("items size not enough");
1832      }
1833      cntl->SetFailed(response->error());
1834    }
1835
1836    std::cout << "[brpc-inference] method=Recommend user=" << request->user_id()
1837              << " request_id=" << request->request_id()
1838              << " code=" << response->code()
1839              << " items=" << response->recommendations_size()
1840              << " latency_ms=" << timer.m_elapsed()
1841              << " backend=" << backend_->Name() << std::endl;

1888  }
1889
1890  std::unique_ptr<InferenceBackend> backend = CreateBackend(config.backend);
1891  if (!backend) {

1907    return 1;
1908  }
1909

1911  options.idle_timeout_sec = config.idle_timeout_sec;

1917
