1# training/dssm/export_embeddings.py
2#
3# 导出 DSSM item 向量 + user 画像, 供 Milvus 灌库和 user tower 服务使用.
4#
5# 用法:
6#   python -m training.dssm.export_embeddings \
7#     --checkpoint checkpoints/dssm/dssm_model.pt \
8#     --vocab_path checkpoints/dssm/vocab.json \
9#     --csv_path /home/vivwimp/Tenrec/ctr_data_1M.csv \
10#     --out_dir checkpoints/dssm/export \
11#     --profile_rows 5000000
12#
13# 产物:
14#   item_vectors.npy    (N, out_dim) float32, 已 L2 归一化
15#   item_ids.json       与 item_vectors 行序一致的 item_id 列表 (int)
16#   user_profiles.json  {user_id: {gender, age, hist: [...]}}
17
18import argparse
19import json
20import os
21
22import numpy as np
23import torch
24
25from training.dssm.dataset import (export_item_categories,
26                                   export_user_profiles)
27from training.dssm.model import DSSM
28
29
30def parse_args():
31    p = argparse.ArgumentParser()
32    p.add_argument("--checkpoint", default="checkpoints/dssm/dssm_model.pt")
33    p.add_argument("--vocab_path", default="checkpoints/dssm/vocab.json")
34    p.add_argument("--csv_path", default="/home/vivwimp/Tenrec/ctr_data_1M.csv")
35    p.add_argument("--out_dir", default="checkpoints/dssm/export")
36    p.add_argument("--profile_rows", type=int, default=5_000_000)
37    p.add_argument("--batch_size", type=int, default=8192)
38    p.add_argument("--device", default="cuda" if torch.cuda.is_available() else "cpu")
39    return p.parse_args()
40
41
42def main():
43    args = parse_args()
44    os.makedirs(args.out_dir, exist_ok=True)
45
46    with open(args.vocab_path) as f:
47        vocab = json.load(f)
48    state = torch.load(args.checkpoint, map_location=args.device)
49    sizes = state["config"]["vocab_sizes"]
50    model = DSSM(sizes,
51                 embed_dim=state["config"]["embed_dim"],
52                 out_dim=state["config"]["out_dim"])
53    model.load_state_dict(state["model"])
54    model.to(args.device).eval()
55
56    # item idx -> 原始 item_id (idx 0 为 padding, 跳过)
57    idx2item = {idx: int(item_id) for item_id, idx in vocab["item2idx"].items()}
58    idxs = sorted(idx2item.keys())
59    print(f"[export] {len(idxs)} items to export")
60
61    vectors = np.zeros((len(idxs), state["config"]["out_dim"]), dtype=np.float32)
62
63    # item -> category 映射 (与训练时的 category 输入一致)
64    cat_map_path = os.path.join(args.out_dir, "item_categories.json")
65    if os.path.exists(cat_map_path):
66        with open(cat_map_path) as f:
67            item_cat = json.load(f)
68    else:
69        n = export_item_categories(args.csv_path, args.profile_rows, cat_map_path)
70        print(f"[export] wrote item_categories.json ({n} items)")
71        with open(cat_map_path) as f:
72            item_cat = json.load(f)
73    cat_lookup = vocab["cat2idx"]
74
75    with torch.no_grad():
76        for start in range(0, len(idxs), args.batch_size):
77            part = idxs[start:start + args.batch_size]
78            item_t = torch.tensor(part, dtype=torch.long, device=args.device)
79            cat_t = torch.tensor(
80                [cat_lookup.get(str(item_cat.get(str(idx2item[i]), "")), 0)
81                 for i in part],
82                dtype=torch.long, device=args.device)
83            vec = model.item_tower(item_t, cat_t)
84            vectors[start:start + len(part)] = vec.cpu().numpy()
85
86    # L2 归一化 (IP 距离等价 cosine)
87    norm = np.linalg.norm(vectors, axis=1, keepdims=True)
88    vectors = vectors / np.clip(norm, 1e-12, None)
89
90    np.save(os.path.join(args.out_dir, "item_vectors.npy"), vectors)
91    with open(os.path.join(args.out_dir, "item_ids.json"), "w") as f:
92        json.dump([idx2item[i] for i in idxs], f)
93    print(f"[export] wrote item_vectors.npy ({vectors.shape}) and item_ids.json")
94
95    n = export_user_profiles(
96        args.csv_path, vocab, args.profile_rows,
97        os.path.join(args.out_dir, "user_profiles.json"))
98    print(f"[export] wrote user_profiles.json ({n} users)")
99
100
101if __name__ == "__main__":
102    main()
