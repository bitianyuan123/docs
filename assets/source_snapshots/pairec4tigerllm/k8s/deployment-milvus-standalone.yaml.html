1# ============================================
2# Milvus standalone with embedded etcd (single Pod)
3# ============================================
4#
5# 使用官方 Milvus 镜像内置的 embedded etcd + local storage。
6# 通过 ConfigMap 注入 embedEtcd.yaml 和 user.yaml，把 etcd 端点固定到
7# 127.0.0.1:2379，避免默认 "localhost" 被解析到 IPv6 [::1] 导致连接失败。
8# 数据持久化到 master 节点的 hostPath /home/zcx/milvus-data。
9
10apiVersion: v1
11kind: ConfigMap
12metadata:
13  name: milvus-embed-config
14  namespace: pairec
15data:
16  embedEtcd.yaml: |
17    listen-client-urls: http://127.0.0.1:2379
18    advertise-client-urls: http://127.0.0.1:2379
19    quota-backend-bytes: 4294967296
20    auto-compaction-mode: revision
21    auto-compaction-retention: '1000'
22  user.yaml: |
23    etcd:
24      endpoints:
25        - 127.0.0.1:2379
26
27---
