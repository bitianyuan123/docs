1# ============================================
2# DataSystem worker pool with shared ETCD
3# ============================================
4#
5# This replaces the single-worker verification Deployment. It runs one shared
6# ETCD instance and one DataSystem worker per node via hostNetwork. Workers
7# register into the same ETCD table and report HOST_IP as host_id, so clients
8# can use ServiceDiscovery with same-node affinity.
9#
10# Defaults match the current ARM K8s lab:
11#   ETCD:      141.61.91.188:12379
12#   workers:   <node-host-ip>:18481
13#   cluster:   pairec
14#
15# Before applying, stop the legacy `datasystem` Deployment because it uses the
16# same host ports 12379/12380/18481.
17

23  labels:
24    app: datasystem-pool-etcd

59            - name: ETCD_BIN
60              value: /host-etcd/etcd
61            - name: ETCDCTL_BIN
62              value: /host-etcd/etcdctl

64          args:
65            - |
66              set -euo pipefail
67              unset LD_PRELOAD

83                --advertise-client-urls "http://${HOST_IP}:${ETCD_CLIENT_PORT}" \
84                --listen-peer-urls "http://0.0.0.0:${ETCD_PEER_PORT}" \
85                --initial-advertise-peer-urls "http://${HOST_IP}:${ETCD_PEER_PORT}" \
86                --initial-cluster "etcd-yuanrong-pool=http://${HOST_IP}:${ETCD_PEER_PORT}"
87          ports:
88            - name: etcd-client
89              containerPort: 12379
90              hostPort: 12379
91            - name: etcd-peer
92              containerPort: 12380
93              hostPort: 12380
94          resources:
95            requests:
96              cpu: "250m"
97              memory: "512Mi"
98            limits:
99              cpu: "1"
100              memory: "1Gi"
101          volumeMounts:
102            - name: etcd-bin
103              mountPath: /host-etcd
104              readOnly: true
