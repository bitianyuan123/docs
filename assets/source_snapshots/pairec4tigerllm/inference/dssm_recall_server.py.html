46def load_resources():
47    global model, vocab, profiles, item_vectors, item_ids
48    global milvus_collection, milvus_err
49
50    with open(VOCAB_PATH) as f:
51        vocab = json.load(f)
52    state = torch.load(CHECKPOINT, map_location=device)
53    model = DSSM(state["config"]["vocab_sizes"],
54                 embed_dim=state["config"]["embed_dim"],
55                 out_dim=state["config"]["out_dim"])
56    model.load_state_dict(state["model"])
57    model.to(device).eval()
58
59    with open(os.path.join(EXPORT_DIR, "user_profiles.json")) as f:
60        profiles = json.load(f)
61
62    if MILVUS_HOST:
63        try:
64            from pymilvus import Collection, connections
65            connections.connect(host=MILVUS_HOST, port=str(MILVUS_PORT))
66            milvus_collection = Collection(MILVUS_COLLECTION)
67            milvus_collection.load()
68            print(f"[dssm-recall] milvus connected {MILVUS_HOST}:{MILVUS_PORT} "
69                  f"collection={MILVUS_COLLECTION}")
70        except Exception as e:  # noqa: BLE001
71            milvus_err = str(e)
72            print(f"[dssm-recall] milvus connect failed: {e}; fallback to local search")
73    if milvus_collection is None:
74        item_vectors = np.load(os.path.join(EXPORT_DIR, "item_vectors.npy"))
75        with open(os.path.join(EXPORT_DIR, "item_ids.json")) as f:
76            item_ids = json.load(f)
77        print(f"[dssm-recall] local search fallback: {len(item_ids)} items")

80def build_user_vector(user_id):
81    """返回 (user_vec_np, error_msg)."""
82    uid = str(user_id)
83    user2idx = vocab["user2idx"]
84    profile = profiles.get(uid)
85    if profile is None:
86        return None, f"user {uid} not found in profiles"
87    item2idx = vocab["item2idx"]
88    hist = [item2idx.get(str(int(v)), 0) for v in profile["hist"]]
89    gender2idx = vocab["gender2idx"]
90    age2idx = vocab["age2idx"]
91    feats = {
92        "user_id": torch.tensor([user2idx.get(uid, 0)], device=device),
93        "gender": torch.tensor([gender2idx.get(str(profile["gender"]), 0)], device=device),
94        "age": torch.tensor([age2idx.get(str(profile["age"]), 0)], device=device),
95        "hist": torch.tensor([hist], device=device),
96    }
97    with torch.no_grad():
98        vec = model.user_tower(feats["user_id"], feats["gender"],
99                               feats["age"], feats["hist"])
100    vec = vec.cpu().numpy()[0]
101    vec = vec / max(np.linalg.norm(vec), 1e-12)
102    return vec.astype(np.float32), None

105def search_milvus(vec, topk):
106    res = milvus_collection.search(
107        data=[vec.tolist()], anns_field="vector",
108        param={"metric_type": "IP", "params": {"nprobe": 16}},
109        limit=topk, output_fields=["item_id"])
110    hits = res[0]
111    return [{"item_id": str(h.entity.get("item_id")), "score": float(h.distance)}
112            for h in hits]
113

115def search_local(vec, topk):
116    scores = item_vectors @ vec
117    k = min(topk, len(item_ids))
118    idx = np.argpartition(-scores, k - 1)[:k]
119    idx = idx[np.argsort(-scores[idx])]
120    return [{"item_id": str(item_ids[i]), "score": float(scores[i])} for i in idx]

134@app.route("/recall", methods=["POST"])
135def recall():
136    t0_ns = time.perf_counter_ns()
137    payload = request.get_json(force=True, silent=True) or {}
138    user_id = payload.get("user_id")
139    topk = int(payload.get("topk", 20))
140    trace_context = payload.get("context") or {}
141    request_id = str(trace_context.get("request_id") or payload.get("request_id") or "")
142    if user_id is None:
143        return jsonify({"code": 400, "msg": "user_id required", "items": []})
144
145    feature_started_ns = time.perf_counter_ns()
146    vec, err = build_user_vector(user_id)
147    feature_us = (time.perf_counter_ns() - feature_started_ns) // 1000
148    if err:
149        return jsonify({"code": 404, "msg": err, "items": []})
150
151    search_started_ns = time.perf_counter_ns()
152    if milvus_collection is not None:
153        items = search_milvus(vec, topk)
154        source = "milvus"
155    else:
156        items = search_local(vec, topk)
157        source = "local"
158    search_us = (time.perf_counter_ns() - search_started_ns) // 1000
159    total_us = (time.perf_counter_ns() - t0_ns) // 1000
160    response_context = {
161        "request_id": request_id,
162        "span_id": str(trace_context.get("span_id") or "vector-backend"),
163        "parent_span_id": str(trace_context.get("parent_span_id") or ""),
164        "sampled": bool(trace_context.get("sampled", True)),
165        "contract_version": str(
166            trace_context.get("contract_version") or "pairec.pipeline_trace.v1"),
167    }
168    return jsonify({
169        "code": 200,
170        "msg": "success",
171        "message": "success",
172        "request_id": request_id,
173        "items": items,
174        "source": source,
175        "latency_ms": round(total_us / 1000, 3),
176        "trace": {
177            "context": response_context,
178            "component": "dssm_recall_backend",
179            "protocol": "http_localhost",
180            "status": "ok",
181            "backend": source,
182            "feature_us": feature_us,
183            "compute_us": search_us,
184            "backend_total_us": total_us,
185            "total_us": total_us,
186            "attribution_complete": True,
187            "asynchronous": False,
188        },
189    })
