1// 稀疏参数服务器（PS）数据面契约（fabric ①）。
2//
3// 独立 PS 服务承载大规模稀疏 embedding 表，供 Nearline（S 特征）与 Online（NS 特征）
4// 共享查表，避免每个精排服务实例各自维护一份稀疏表（内存/存储浪费）。
5//
6// 与 UserKV（fabric ②，datasystem）与元数据（fabric ③，Redis Cluster）分离。
7// 协议字段与 Python 侧 EmbeddingPSClient 的 wire 契约对齐。
8
9syntax = "proto3";
10
11package onetrans.ps;
12
13// 显式启用 C++ 泛型服务代码生成（否则 protoc/protobuf 3.14+ 默认不生成 Service 基类，
14// brpc 的 AddService(google::protobuf::Service*) 将无法接收 EmbeddingServiceImpl）。
15option cc_generic_services = true;
16
17// 单次查表：给定表名（含模型版本）与稀疏特征 ID 列表。
18message EmbeddingLookupRequest {
19  string table = 1;          // 表名/模型版本，如 "onetrans@2026-08-25"
20  repeated int64 ids = 2;    // 稀疏特征 ID（去重后最多 N 个）
21}
22
23// 查表响应：逐 id 展平的向量 weights = [ids.size() * dim]。
24message EmbeddingLookupResponse {
25  string table = 1;
26  int32 dim = 2;             // 每 id 向量维度
27  repeated int64 ids = 3;    // 回显 id（与请求对齐）
28  repeated float weights = 4;  // 展平 [ids.size() * dim]，miss 填 0（seed 兜底在上层）
29  int64 version = 5;         // 表权重版本（权重版本化加载 / 失效校验）
30  int32 shard_id = 6;        // 命中分片（调试/指标，可忽略）
31}
32
33// 批量：客户端攒批推送多个子请求，服务端按分片并行 gather（动态 batching）。
34message EmbeddingBatchRequest {
35  repeated EmbeddingLookupRequest requests = 1;
36}
37
38message EmbeddingBatchResponse {
39  repeated EmbeddingLookupResponse responses = 1;
40}
41
42// ---- 写路径（阶段 B）：权重版本化加载 / 离线灌表注入 ---- //
43// 与 Python 侧 ShardedEmbeddingTable.set / LocalEmbeddingPS.set_many 语义对齐：
44// 对同一 table 按 (id, weights) 逐行写入，每写触发表版本 version 递增。
45
46// 单表写入：一次注入该表的一批行 weights = [ids.size() * dim]（行主序）。
47message EmbeddingSetRequest {
48  string table = 1;          // 表名/模型版本，如 "onetrans@2026-08-25"
49  repeated int64 ids = 2;    // 行号/特征 ID
50  repeated float weights = 3; // 平铺 [ids.size() * dim]
51  int32 dim = 4;             // 每行向量维度（须与表维度一致，不一致则整批拒绝）
52}
53
54// 写后回执：updated 为实际写入行数，version 为写后表版本（调用方可据此校验）。
55message EmbeddingSetResponse {
56  string table = 1;
57  int32 dim = 2;
58  int64 version = 3;
59  int32 updated = 4;
60}
61
62// 批量写：客户端按表攒批（多表/多目标源同一批灌入）。
63message EmbeddingBatchSetRequest {
64  repeated EmbeddingSetRequest sets = 1;
65}
66
67message EmbeddingBatchSetResponse {
68  repeated EmbeddingSetResponse responses = 1;
69}
70
71service EmbeddingService {
72  rpc Lookup(EmbeddingLookupRequest) returns (EmbeddingLookupResponse);
73  rpc BatchLookup(EmbeddingBatchRequest) returns (EmbeddingBatchResponse);
74  rpc Set(EmbeddingSetRequest) returns (EmbeddingSetResponse);
75  rpc BatchSet(EmbeddingBatchSetRequest) returns (EmbeddingBatchSetResponse);
76}
