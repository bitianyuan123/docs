1// 独立稀疏参数服务器（PS）——brpc + bthread（M:N）参考实现。
2//
3// 设计要点（对应 docs/detailed_design.md §4.5「稀疏 PS 数据面契约」与 §7.4「线程与并发模型」）：
4// - 稀疏表按 id 稳定哈希切分为 N 个分片，每分片一把细粒度锁（无全局锁，QPS 随核扩展）；
5// - 分片路由统一 Knuth 乘法哈希（detail::ShardOf 为唯一标准，Python 侧
6//   embedding_ps_client.py 的 ShardedEmbeddingTable.shard_of 与其逐位对齐，G10）；
7// - 多表支持：table 名 -> ShardedEmbeddingTable 注册中心（按需懒建 + 淘汰），
8//   DoLookup 按 req.table() 路由，支撑多模型版本并存与灰度（G11）；
9// - brpc::Server 默认 bthread M:N 调度，Lookup 为阻塞式 handler，由 bthread 承载高并发；
10// - 未命中返回 0 向量，由客户端按 seed 兜底哈希嵌入重建（权重版本化最差路径）；
11// - 表版本（version）随写递增，供在线侧权重版本化加载与失效校验。
12//
13// 构建：见 BUILD（bazel + brpc/protobuf dep）。

38namespace detail {
39// 稳定乘法哈希（Knuth）：**分片路由唯一标准**（G10）。
40// Python 侧 ShardedEmbeddingTable.shard_of 用同一常数、同一二补码回绕与 2^64 截断语义，
41// 保证读写两侧对同一 id 落同一分片。
42inline std::size_t ShardOf(int64_t id, int num_shards) {
43  constexpr uint64_t kKnuth = 11400714819323198485ULL;  // 0x9E3779B97F4A7C15
44  return static_cast<std::size_t>((static_cast<uint64_t>(id) * kKnuth) % num_shards);
45}
46}  // namespace detail

48// 分片嵌入表：`num_shards` 个独立 HashMap + 每分片锁（细粒度并发）。
49class ShardedEmbeddingTable {
50 public:
51  using Row = std::vector<float>;
52
53  ShardedEmbeddingTable(int num_shards, int dim)
54      : num_shards_(num_shards), dim_(dim), version_(0),
55        shards_(num_shards), locks_(num_shards) {}
56
57  int dim() const { return dim_; }
58  int64_t version() const { return version_.load(std::memory_order_acquire); }
59
60  void Set(int64_t id, Row weights) {
61    const std::size_t s = detail::ShardOf(id, num_shards_);
62    std::lock_guard<std::mutex> lk(locks_[s]);
63    shards_[s].map[id] = std::move(weights);
64    version_.fetch_add(1, std::memory_order_acq_rel);
65  }
66
67  // 查表：命中写回 out 返回 true；未命中返回 false（调用方填 0）。
68  bool Get(int64_t id, Row* out) const {
69    const std::size_t s = detail::ShardOf(id, num_shards_);
70    std::lock_guard<std::mutex> lk(locks_[s]);
71    const auto it = shards_[s].map.find(id);
72    if (it == shards_[s].map.end()) return false;
73    *out = it->second;
74    return true;
75  }
76
77 private:
78  struct Shard {
79    std::unordered_map<int64_t, Row> map;
80  };
81
82  int num_shards_;
83  int dim_;
84  std::atomic<int64_t> version_;
85  std::vector<Shard> shards_;
86  mutable std::vector<std::mutex> locks_;
87};

89// 多表注册中心：table 名 -> ShardedEmbeddingTable（G11）。
90//
91// 与 Python 侧 LocalEmbeddingPS 的多表语义对齐：
92// - 表按需懒建（Get 首次触达即注册），返回 shared_ptr 快照，读路径持锁仅覆盖注册动作；
93// - Erase 淘汰整表（模型版本下线/回滚）；淘汰后进行中的旧请求仍持 shared_ptr 安全读；
94// - 每表独立 version（写递增），供「多模型版本并存 + 按版本查表 + 权重版本化失效」。
95class TableRegistry {
96 public:
97  using TablePtr = std::shared_ptr<ShardedEmbeddingTable>;
98
99  TableRegistry(int num_shards, int dim) : num_shards_(num_shards), dim_(dim) {}
100
101  // 取表；不存在则注册新表（懒建，对齐 Python LocalEmbeddingPS.table()）。
102  TablePtr Get(const std::string& name) {
103    std::lock_guard<std::mutex> lk(mu_);
104    const auto it = tables_.find(name);
105    if (it != tables_.end()) return it->second;
106    auto t = std::make_shared<ShardedEmbeddingTable>(num_shards_, dim_);
107    tables_.emplace(name, t);
108    return t;
109  }
110
111  // 只读查找（不注册）：不存在的表返回 false（调用方走 miss/兜底路径）。
112  bool Find(const std::string& name, TablePtr* out) const {
113    std::lock_guard<std::mutex> lk(mu_);
114    const auto it = tables_.find(name);
115    if (it == tables_.end()) return false;
116    *out = it->second;
117    return true;
118  }
119
120  // 淘汰整表（版本下线/回滚）；返回是否确实删除。
121  bool Erase(const std::string& name) {
122    std::lock_guard<std::mutex> lk(mu_);
123    return tables_.erase(name) > 0;
124  }
125
126  size_t Size() const {
127    std::lock_guard<std::mutex> lk(mu_);
128    return tables_.size();
129  }
130
131 private:
132  const int num_shards_;
133  const int dim_;
134  mutable std::mutex mu_;
135  std::unordered_map<std::string, TablePtr> tables_;
136};

138// RPC 服务实现（brpc：handler 运行于 bthread，天然 M:N）。
139class EmbeddingServiceImpl : public EmbeddingService {
140 public:
141  explicit EmbeddingServiceImpl(TableRegistry* registry) : registry_(registry) {}
142
143  void Lookup(google::protobuf::RpcController* /*cntl*/,
144              const EmbeddingLookupRequest* req,
145              EmbeddingLookupResponse* resp,
146              google::protobuf::Closure* done) override {
147    brpc::ClosureGuard done_guard(done);
148    DoLookup(*req, resp);
149  }
150
151  void BatchLookup(google::protobuf::RpcController* /*cntl*/,
152                   const EmbeddingBatchRequest* req,
153                   EmbeddingBatchResponse* resp,
154                   google::protobuf::Closure* done) override {
155    brpc::ClosureGuard done_guard(done);
156    for (const auto& sub : req->requests()) {
157      DoLookup(sub, resp->add_responses());
158    }
159  }
160
161  // ---- 写路径（阶段 B）：权重版本化加载 / 离线灌表注入 ---- //
162  void Set(google::protobuf::RpcController* /*cntl*/,
163           const EmbeddingSetRequest* req,
164           EmbeddingSetResponse* resp,
165           google::protobuf::Closure* done) override {
166    brpc::ClosureGuard done_guard(done);
167    DoSet(*req, resp);
168  }
169
170  void BatchSet(google::protobuf::RpcController* /*cntl*/,
171                const EmbeddingBatchSetRequest* req,
172                EmbeddingBatchSetResponse* resp,
173                google::protobuf::Closure* done) override {
174    brpc::ClosureGuard done_guard(done);
175    for (const auto& sub : req->sets()) {
176      DoSet(sub, resp->add_responses());
177    }
178  }
179
180 private:
181  void DoLookup(const EmbeddingLookupRequest& req, EmbeddingLookupResponse* resp) const {
182    // G11：按 req.table() 路由到对应分片表（多模型版本/灰度）；懒建后未写入即全 miss。
183    const TableRegistry::TablePtr table = registry_->Get(req.table());
184    const int dim = table->dim();
185    resp->set_table(req.table());
186    resp->set_dim(dim);
187    resp->set_version(table->version());
188    resp->mutable_weights()->Reserve(req.ids_size() * dim);
189    for (const int64_t id : req.ids()) {
190      ShardedEmbeddingTable::Row row;
191      const bool hit = table->Get(id, &row);
192      resp->add_ids(id);
193      for (int d = 0; d < dim; ++d) {
194        // 未命中回 0：客户端 / 上层用 seed 兜底哈希嵌入重建（权重版本化最差路径）。
195        resp->add_weights(hit ? row[d] : 0.0f);
196      }
197    }
198  }
199
200  // 写路径：按表注入一批行 weights=[ids*dim]（行主序）；每写触发行所在分片表版本递增。
201  // 维度不匹配整批拒绝（updated=0，返回当前版本），避免半写坏数据。
202  void DoSet(const EmbeddingSetRequest& req, EmbeddingSetResponse* resp) {
203    const TableRegistry::TablePtr table = registry_->Get(req.table());
204    const int dim = table->dim();
205    resp->set_table(req.table());
206    resp->set_dim(dim);
207    if (req.weights_size() != req.ids_size() * dim) {
208      resp->set_updated(0);
209      resp->set_version(table->version());
210      LOG(ERROR) << "Set 维度不匹配 table=" << req.table()
211                 << " ids=" << req.ids_size() << " dim=" << dim
212                 << " weights=" << req.weights_size();
213      return;
214    }
215    for (int i = 0; i < req.ids_size(); ++i) {
216      ShardedEmbeddingTable::Row row(dim);
217      const int base = i * dim;
218      for (int d = 0; d < dim; ++d) row[d] = req.weights(base + d);
219      table->Set(req.ids(i), std::move(row));
220    }
221    resp->set_updated(req.ids_size());
222    resp->set_version(table->version());
223  }
224
225  TableRegistry* registry_;  // 不拥有（由 main 持有）
226};

231int main(int argc, char** argv) {
232  google::ParseCommandLineFlags(&argc, &argv, true);
233
234  onetrans::ps::TableRegistry registry(FLAGS_num_shards, FLAGS_dim);
235  onetrans::ps::EmbeddingServiceImpl impl(&registry);
236
237  brpc::Server server;
238  if (server.AddService(&impl, brpc::SERVER_DOESNT_OWN_SERVICE) != 0) {
239    LOG(ERROR) << "AddService 失败";
240    return 1;
241  }
242
243  brpc::ServerOptions options;
244  if (FLAGS_num_bthreads > 0) {
245    options.num_threads = FLAGS_num_bthreads;
246  }
247
248  if (server.Start(FLAGS_port, &options) != 0) {
249    LOG(ERROR) << "PS 启动失败，端口 " << FLAGS_port;
250    return 1;
251  }
252
253  LOG(INFO) << "embedding PS 监听 :" << FLAGS_port
254            << " (shards=" << FLAGS_num_shards << ", dim=" << FLAGS_dim << ")";
255  server.RunUntilAskedToQuit();
256  return 0;
257}
