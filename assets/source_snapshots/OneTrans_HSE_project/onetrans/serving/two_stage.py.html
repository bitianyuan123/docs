1"""两阶段推理引擎（Nearline prefill / Online 交叉打分）。
2
3把 OneTrans backbone 的 mixed causal self-attention 拆成：
4
5- :meth:`TwoStageRunner.encode_s`（Stage I / nearline）：只编码用户历史 S 序列，
6  逐层缓存 ``K_s^l / V_s^l``（与 LLM prefill 对应）。
7- :meth:`TwoStageRunner.score_ns`（Stage II / online）：读缓存 K/V，对候选 NS token
8  做跨层交叉注意力打分（与 LLM decode 对应，但并行非自回归）。
9
10拆分成立的前提：S token 位于序列前段且严格因果，其隐藏态与 K/V 与 NS 无关。
11故 S 侧可安全预计算；两阶段拼接后与单前向**数值等价**（见 demo 校验）。
12
13对应设计文档 §4.1。pyramid 缩层方向已修正为「保留尾部（最新 token）」，
14S 侧统一采用左 padding（有效 token 靠后），见 ``docs/detailed_design.md`` §4.2 仓库修正。
15"""
16

28@dataclass
29class UserKV:
30    """Stage I 产出：逐层 S 侧 K/V（未序列化的内存形态，B=1 单用户）。"""
31
32    per_layer: list[tuple[Tensor, Tensor]]  # (K_s^l, V_s^l)，[1, S_l, H, d]
33    per_layer_len: list[int]  # 每层**有效** S token 数（左 padding 语义）
34    s_len: int  # 原始有效历史长度（append offset 语义）
35
36
37def _s_attn_mask(s_mask: Tensor) -> Tensor:
38    """S 侧 causal 自注意力掩码：padding（列）+ 上三角因果（左 padding）。"""
39    B, S = s_mask.shape
40    device = s_mask.device
41    m = torch.zeros(B, 1, S, S, device=device)
42    m = m.masked_fill((~s_mask)[:, None, None, :], float("-inf"))
43    causal = torch.triu(torch.ones(S, S, dtype=torch.bool, device=device), diagonal=1)
44    return m.masked_fill(causal[None, None, :, :], float("-inf"))
45
46
47def _cross_attn_mask(s_mask: Tensor, ns_len: int) -> Tensor:
48    """NS→(S|NS) 交叉注意力掩码：S 列只按 padding 掩码，NS 列按因果掩码。"""
49    B, S = s_mask.shape
50    device = s_mask.device
51    total = S + ns_len
52    m = torch.zeros(B, 1, ns_len, total, device=device)
53    m[:, :, :, :S] = m[:, :, :, :S].masked_fill((~s_mask)[:, None, None, :], float("-inf"))
54    causal = torch.triu(torch.ones(ns_len, ns_len, dtype=torch.bool, device=device), diagonal=1)
55    m[:, :, :, S:] = m[:, :, :, S:].masked_fill(causal[None, None, :, :], float("-inf"))
56    return m

59class TwoStageRunner:
60    """复用已训练 OneTrans 权重的两阶段推理引擎。
61
62    :param backbone: 一个已构建（可加载权重）的 :class:`OneTrans` 实例。
63    """
64
65    def __init__(self, backbone: OneTrans) -> None:
66        self.backbone = backbone
67        self.blocks = backbone.blocks
68        self.linear = backbone.linear
69        self.d_model = backbone.d_model
70        self.ns_tokens_num = backbone.ns_tokens_num
71        self.use_cls_token = backbone.use_cls_token
72        self.dims = backbone.dims  # pyramid 逐层 S 长度（int tensor）
73
74    # ------------------------------------------------------------------ #
75    # Stage I：S 侧编码（prefill，单用户 B=1）
76    # ------------------------------------------------------------------ #
77    @torch.no_grad()
78    def encode_s(self, s_emb: Tensor, s_mask: Tensor | None = None) -> UserKV:
79        """编码用户历史，逐层缓存 K_s/V_s。
80
81        :param s_emb: 已 tokenize + pos + RMSNorm 的 S 序列，[1, S0, D]，S0=dims[0]
82        :param s_mask: [1, S0] bool 有效掩码（左 padding，有效 token 靠后）
83        """
84        if s_emb.shape[0] != 1:
85            raise ValueError("encode_s 只支持单用户 B=1（nearline 按 user 分区）")
86        B, S, D = s_emb.shape
87        if s_mask is None:
88            s_mask = torch.ones(B, S, dtype=torch.bool, device=s_emb.device)
89
90        s = s_emb
91        smask = s_mask
92        per_layer: list[tuple[Tensor, Tensor]] = []
93        per_layer_len: list[int] = []
94        s_len = int(smask.sum().item())
95
96        for block in self.blocks:
97            per_layer_len.append(int(smask.sum().item()))
98            h = block.norm(s)
99            q_s, k_s, v_s = _project_s(block.mixed_attn, h)
100            per_layer.append((k_s, v_s))  # 缓存输入长度 dims[l] 的 K/V
101
102            attn = F.scaled_dot_product_attention(
103                q_s.transpose(1, 2), k_s.transpose(1, 2), v_s.transpose(1, 2),
104                attn_mask=_s_attn_mask(smask), dropout_p=0.0,
105            )
106            attn = attn.transpose(1, 2).reshape(B, S, D)
107            attn = block.mixed_attn.final_proj(attn)
108            z = attn + s
109            z = z + block.mixed_ffn.network_s(block.norm(z))
110
111            # pyramid 降层：保留「最新（尾部）」的 S token（论文 §3.4 tail，配合左 padding）
112            s_in = s.shape[1]
113            s = z[:, s_in - block.out_seq_num : s_in, :]
114            smask = smask[:, s_in - block.out_seq_num : s_in]
115            S = block.out_seq_num
116
117        return UserKV(per_layer=per_layer, per_layer_len=per_layer_len, s_len=s_len)
118
119    # ------------------------------------------------------------------ #
120    # Stage II：NS 侧交叉打分（decode，并行非自回归，B=M 候选）
121    # ------------------------------------------------------------------ #
122    @torch.no_grad()
123    def score_ns(self, kv: UserKV, ns_emb: Tensor) -> Tensor:
124        """读缓存 K/V，对候选 NS token 逐层交叉注意力打分。
125
126        :param kv: :meth:`encode_s` 的产出（单用户，K/V 为 [1, S_l, H, d]）
127        :param ns_emb: 已 tokenize + RMSNorm 的 NS 序列，[B, Ns, D]（B=M 候选数）
128        :return: logits [B, T]（T=任务数，此处 2）
129        """
130        B, Ns, D = ns_emb.shape
131        ns = ns_emb
132
133        for l, block in enumerate(self.blocks):
134            k_s, v_s = kv.per_layer[l]  # [1, S_l, H, d]
135            S_l = k_s.shape[1]
136            # 广播单用户 K/V 到全部候选
137            k_s = k_s.expand(B, -1, -1, -1)
138            v_s = v_s.expand(B, -1, -1, -1)
139
140            valid_l = kv.per_layer_len[l]
141            s_mask = torch.cat(
142                [
143                    torch.zeros(B, S_l - valid_l, dtype=torch.bool, device=ns.device),
144                    torch.ones(B, valid_l, dtype=torch.bool, device=ns.device),
145                ],
146                dim=1,
147            )
148
149            h = block.norm(ns)
150            q_ns, k_ns, v_ns = _project_ns(block.mixed_attn, h)  # [B, Ns, H, d]
151
152            k = torch.cat([k_s, k_ns], dim=1)  # [B, S_l+Ns, H, d]
153            v = torch.cat([v_s, v_ns], dim=1)
154            attn = F.scaled_dot_product_attention(
155                q_ns.transpose(1, 2), k.transpose(1, 2), v.transpose(1, 2),
156                attn_mask=_cross_attn_mask(s_mask, Ns), dropout_p=0.0,
157            )
158            attn = attn.transpose(1, 2).reshape(B, Ns, D)
159            attn = block.mixed_attn.final_proj(attn)
160            z = attn + ns
161            z = z + _apply_ns_ffn(block.mixed_ffn, block.norm(z), Ns)
162            ns = z
163
164        if self.use_cls_token:
165            out = ns[:, -1, :]
166        else:
167            out = ns[:, -self.ns_tokens_num:, :].mean(dim=1)
168        return self.linear(out)
169
170    # ------------------------------------------------------------------ #
171    # Stage II（批量）：B 个（user, 候选）对打包一次前向（动态 batching）
172    # ------------------------------------------------------------------ #
173    @torch.no_grad()
174    def score_ns_batch(self, kvs: list[UserKV], ns_emb: Tensor) -> Tensor:
175        """批量交叉打分：把 ``B`` 个（用户，候选）对打包为单次前向。
176
177        实现前提（见设计文档 §4/§7）：pyramid 每层 K/V 宽度 ``S_l = dims[l]`` 对
178        所有用户恒定，可变的是 ``per_layer_len[l]``（有效 token 数）。故逐层把各用户
179        K/V 直接 ``stack``，有效掩码用「左 padding（有效靠后）」逐行构造，与
180        :meth:`score_ns` 数值等价（B=1 时逐位一致）。
181
182        :param kvs: ``B`` 个 :class:`UserKV`（同一用户多候选时重复传入）
183        :param ns_emb: ``[B, Ns, D]``
184        :return: logits ``[B, T]``
185        """
186        B = ns_emb.shape[0]
187        if B != len(kvs):
188            raise ValueError("kvs 数量须等于 ns_emb batch")
189        ns = ns_emb
190
191        for l, block in enumerate(self.blocks):
192            # 逐层 stack：各用户 K/V 宽度均为 dims[l]（fixed），故可直接打包
193            k_s = torch.stack([kv.per_layer[l][0][0] for kv in kvs], dim=0)  # [B, S_l, H, d]
194            v_s = torch.stack([kv.per_layer[l][1][0] for kv in kvs], dim=0)
195            S_l = k_s.shape[1]
196            valid = torch.tensor(
197                [kv.per_layer_len[l] for kv in kvs], dtype=torch.long, device=ns.device
198            )  # [B]
199            # 左 padding 掩码：每行后 valid 列为有效（与 score_ns 的 s_mask 构造等价）
200            s_mask = (
201                torch.arange(S_l, device=ns.device)[None, :] >= (S_l - valid)[:, None].to(ns.device)
202            )
203
204            h = block.norm(ns)
205            q_ns, k_ns, v_ns = _project_ns(block.mixed_attn, h)  # [B, Ns, H, d]
206
207            k = torch.cat([k_s, k_ns], dim=1)  # [B, S_l+Ns, H, d]
208            v = torch.cat([v_s, v_ns], dim=1)
209            attn = F.scaled_dot_product_attention(
210                q_ns.transpose(1, 2), k.transpose(1, 2), v.transpose(1, 2),
211                attn_mask=_cross_attn_mask(s_mask, ns.shape[1]), dropout_p=0.0,
212            )
213            attn = attn.transpose(1, 2).reshape(B, ns.shape[1], self.d_model)
214            attn = block.mixed_attn.final_proj(attn)
215            z = attn + ns
216            z = z + _apply_ns_ffn(block.mixed_ffn, block.norm(z), ns.shape[1])
217            ns = z
218
219        if self.use_cls_token:
220            out = ns[:, -1, :]
221        else:
222            out = ns[:, -self.ns_tokens_num:, :].mean(dim=1)
223        return self.linear(out)

226# --------------------------------------------------------------------------- #
227# 投影辅助（复用 mixed_attn / mixed_ffn 权重）
228# --------------------------------------------------------------------------- #
229def _project_s(mixed_attn, h: Tensor) -> tuple[Tensor, Tensor, Tensor]:
230    B, S, D = h.shape
231    H = mixed_attn.n_heads
232    d = mixed_attn.head_dim
233    qkv = mixed_attn.W_s(h).reshape(B, S, 3, H, d)
234    q, k, v = qkv.unbind(dim=2)  # 各自 [B, S, H, d]
235    return q, k, v
236
237
238def _project_ns(mixed_attn, ns: Tensor) -> tuple[Tensor, Tensor, Tensor]:
239    B, Ns, D = ns.shape
240    H = mixed_attn.n_heads
241    d = mixed_attn.head_dim
242    qs, ks, vs = [], [], []
243    for i in range(Ns):
244        qkv = mixed_attn.W_ns_list[i](ns[:, i, :]).reshape(B, 3, H, d)
245        q, k, v = qkv.unbind(dim=1)  # 各自 [B, H, d]
246        qs.append(q.unsqueeze(1))
247        ks.append(k.unsqueeze(1))
248        vs.append(v.unsqueeze(1))
249    return torch.cat(qs, dim=1), torch.cat(ks, dim=1), torch.cat(vs, dim=1)
250
251
252def _apply_ns_ffn(mixed_ffn, h: Tensor, Ns: int) -> Tensor:
253    out = [mixed_ffn.networks_ns_list[i](h[:, i : i + 1, :]) for i in range(Ns)]
254    return torch.cat(out, dim=1)
