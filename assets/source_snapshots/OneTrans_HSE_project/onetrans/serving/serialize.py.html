1"""UserKV payload 序列化（零拷贝数据面）：tensor <-> 连续字节 blob。
2
3遵循设计文档 §2.3 的内存布局：
4
5    <magic:12B> <header_len:4B> <header_json> <raw_bytes>
6    raw_bytes = concat(K_s^0, V_s^0, K_s^1, V_s^1, ..., K_s^{L-1}, V_s^{L-1})  # 均 bshd 连续
7
8零拷贝要点（P1：消除 UserKV 数据面的冗余拷贝）：
9
10- **读侧**：``deserialize`` 用 ``torch.frombuffer(buf, offset=...)`` 直接视图底层字节
11  （``bytes`` / ``bytearray`` / ``memoryview`` / ``mmap``），不再 ``bytearray(buf)`` 切出副本。
12  对 mmap 后端，反序列化的 K/V 张量直接指向共享内存，无 CPU RAM 二次拷贝。
13- **写侧**：``serialize`` 预分配单一 ``bytearray``，用 ``ctypes.memmove`` 从张量
14  ``data_ptr`` 一次拷贝到底层缓冲，避免逐层 ``+`` 拼接的 O(n²) 与中间 ``bytes`` 对象。
15
16禁止 pickle 任意对象；仅写 dtype/形状元信息 + 连续字节。
17"""

33def _nbytes(shape: Sequence[int], dtype: torch.dtype) -> int:
34    n = 1
35    for s in shape:
36        n *= s
37    return n * dtype.itemsize

40def serialize(
41    per_layer: Sequence[tuple[Tensor, Tensor]],
42    s_len: int | None = None,
43    per_layer_len: Sequence[int] | None = None,
44) -> bytes:
45    """把逐层 ``(K_s^l, V_s^l)`` 序列化为单个连续字节 blob。
46
47    预分配总字节数，一次 memmove 拷贝每张量底层字节，避免重复分配。
48
49    ``s_len``/``per_layer_len`` 为**有效长度元数据**（左 padding 语义下 ``K_s`` 的
50    shape[1] 是 pyramid 该层满宽 ``dims[l]``，需显式区分有效 token 数，见
51    ``docs/gap_analysis.md`` G1）。不传时回退为满宽（兼容无 padding 场景与旧调用）。
52    """
53    if per_layer_len is None:
54        per_layer_len = [k.shape[1] for k, _ in per_layer]
55    s_len = s_len if s_len is not None else (per_layer_len[0] if per_layer_len else 0)
56
57    flat: list[Tensor] = []
58    meta: list[tuple[int, str]] = []  # (element_size, dtype_str)
59    for k, v in per_layer:
60        k = k.detach().to("cpu").contiguous()
61        v = v.detach().to("cpu").contiguous()
62        assert k.dtype == v.dtype, "每层 K/V dtype 必须一致"
63        flat.append(k)
64        flat.append(v)
65        meta.append((k.element_size(), str(k.dtype).replace("torch.", "")))
66        meta.append((v.element_size(), str(v.dtype).replace("torch.", "")))
67
68    header = {
69        "dtype": meta[0][1] if meta else "float32",
70        "n_layers": len(per_layer),
71        "s_len": s_len,
72        "layers": [
73            {"l": i, "k_shape": list(k.shape), "v_shape": list(v.shape), "len": per_layer_len[i]}
74            for i, (k, v) in enumerate(per_layer)
75        ],
76    }
77    header_bytes = json.dumps(header).encode("utf-8")
78
79    body_bytes = sum(t.numel() * t.element_size() for t in flat)
80    total = len(_MAGIC) + 4 + len(header_bytes) + body_bytes
81    buf = bytearray(total)
82    pos = 0
83    buf[pos : pos + len(_MAGIC)] = _MAGIC
84    pos += len(_MAGIC)
85    struct.pack_into(_HEADER_FMT, buf, pos, len(header_bytes))
86    pos += 4
87    buf[pos : pos + len(header_bytes)] = header_bytes
88    pos += len(header_bytes)
89
90    # 单次 memmove 拷贝每张量底层字节到预分配缓冲
91    carr = (ctypes.c_char * len(buf)).from_buffer(buf)
92    dst_base = ctypes.addressof(carr)
93    for t in flat:
94        n = t.numel() * t.element_size()
95        ctypes.memmove(dst_base + pos, t.data_ptr(), n)
96        pos += n
97
98    return bytes(buf)

101def _parse_header(payload: bytes | bytearray | memoryview) -> tuple[dict, int]:
102    """解析并校验 ``<magic><header_len><header_json>``，返回 ``(header, 其后偏移)``。"""
103    if bytes(payload[: len(_MAGIC)]) != _MAGIC:
104        raise ValueError("未知 payload 魔数")
105    pos = len(_MAGIC)
106    (hlen,) = struct.unpack_from(_HEADER_FMT, payload, pos)
107    pos += 4
108    header = json.loads(bytes(payload[pos : pos + hlen]).decode("utf-8"))
109    pos += hlen
110    return header, pos

113def read_header(payload: bytes | bytearray | memoryview) -> dict:
114    """仅解析 header，返回 ``dict``（含 ``s_len`` 与各层 ``layers[].len``）。
115
116    供读侧只需有效长度元数据、而不反序列化整对象张量的场景（如 datasystem
117    ``get``/``append``，见 ``docs/gap_analysis.md`` G1）。
118    """
119    header, _ = _parse_header(payload)
120    return header

123def deserialize_with_meta(
124    payload: bytes | bytearray | memoryview,
125) -> tuple[list[tuple[Tensor, Tensor]], int, list[int]]:
126    """反序列化并返回 ``(per_layer, s_len, per_layer_len)``。
127
128    有效长度来自 header（``s_len`` / ``layers[].len``）；旧数据（无这些字段）回退为
129    「满宽有效」，与旧语义一致（向后兼容，见 ``docs/gap_analysis.md`` G1）。
130    """
131    header, _ = _parse_header(payload)
132    per_layer = deserialize(payload)
133    s_len = header.get("s_len")
134    if s_len is None:
135        s_len = per_layer[0][0].shape[1] if per_layer else 0
136    per_layer_len = [m.get("len", m["k_shape"][1]) for m in header["layers"]]
137    return per_layer, s_len, per_layer_len

140def deserialize(payload: bytes | bytearray | memoryview) -> list[tuple[Tensor, Tensor]]:
141    """把字节 blob 还原为逐层 ``(K_s^l, V_s^l)``（零拷贝视图）。
142
143    :param payload: 底层缓冲；``bytes`` 得到只读视图，``bytearray``/``memoryview``/``mmap``
144        得到可直接写回的视图（零拷贝）。返回张量与底层缓冲共享存储。
145    """
146    header, pos = _parse_header(payload)
147    dtype = getattr(torch, header["dtype"])
148    out: list[tuple[Tensor, Tensor]] = []
149    for meta in header["layers"]:
150        k = _view_tensor(payload, pos, meta["k_shape"], dtype)
151        pos += _nbytes(meta["k_shape"], dtype)
152        v = _view_tensor(payload, pos, meta["v_shape"], dtype)
153        pos += _nbytes(meta["v_shape"], dtype)
154        out.append((k, v))
155    return out

158def _view_tensor(
159    payload: bytes | bytearray | memoryview,
160    pos: int,
161    shape: Sequence[int],
162    dtype: torch.dtype,
163) -> Tensor:
164    numel = 1
165    for s in shape:
166        numel *= s
167    # frombuffer 直接视图底层缓冲（offset 指向各层偏移），无需副本
168    return torch.frombuffer(payload, dtype=dtype, count=numel, offset=pos).reshape(shape)

171def per_layer_offsets(payload: bytes | bytearray | memoryview) -> list[tuple[int, int]]:
172    """返回每层 ``(K 偏移, V 偏移)``（用于按层抽取而不反序列化整对象，见设计文档 §2.3）。"""
173    header, pos = _parse_header(payload)
174    dtype = getattr(torch, header["dtype"])
175    offsets: list[tuple[int, int]] = []
176    for meta in header["layers"]:
177        k_pos = pos
178        pos += _nbytes(meta["k_shape"], dtype)
179        v_pos = pos
180        pos += _nbytes(meta["v_shape"], dtype)
181        offsets.append((k_pos, v_pos))
182    return offsets

185def iterate_tensors(
186    payload: bytes | bytearray | memoryview,
187) -> Iterable[Tensor]:
188    """惰性迭代全部逐层张量（零拷贝视图），供逐层处理用。"""
189    yield from (t for pair in deserialize(payload) for t in pair)
