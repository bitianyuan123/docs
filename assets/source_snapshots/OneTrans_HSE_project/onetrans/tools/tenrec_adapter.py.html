1"""Tenrec 域特征适配器：Tenrec 原生 ID + Tenrec 抽取特征 → 现有音乐域模型契约。
2
3背景：C++ 主链与权重冻结在音乐域契约（NS 组：dense 240 / uid / item / artist bag /
4album bag；dense 15+15 分桶；S 段 5..50）。本适配器让 Tenrec 真实词表规模的数据
5跑通该模型（跑不准没关系：稀疏表随机/未训练、分桶权重按音乐域值域训练）。
6
7原生 ID 映射（查 PS tenrec/ 前缀表，与词表规模无关，PS miss 返 0 向量）：
8  item_id      → int(item_id)      十rec 原生（PS tenrec/item，231 万行）
9  uid_sparse   → int(user_id)      Tenrec 原生（PS tenrec/user，99.9 万行）
10  artist_ids   → [video_category]  类别即 artist（PS tenrec/artist，3 行）
11  album_ids    → []                Tenrec 无对应列 → 空 bag = 0 向量
12  S 段         → hist（10 item）原生 ID；时间戳合成升序（S tokenizer 与时间无关）
13  cand dense15 → item 15 统计列 z-score（item 全体分布），clip ±5
14  user dense15 → [z(gender), z(age)] + hist item 前 13 统计列 z 均值
15
16用法：
17  生成：.venv/bin/python -m onetrans.tools.tenrec_adapter \
18          --feat-dir artifacts/domain_features --out-dir artifacts/tenrec_cases \
19          --users 8 --cands 4
20  回放：.venv/bin/python -m onetrans.tools.tenrec_adapter --replay \
21          --cases artifacts/tenrec_cases/cases.jsonl --server http://127.0.0.1:19099
22（服务器需 --model-version tenrec --embedding-source ps --ps-server <ps>）
23"""
24from __future__ import annotations

26import argparse
27import json
28import math
29import random
30import sys

34# item_features.tsv 第 2..16 列（15 个统计特征）
35ITEM_COLS = ["n", "click", "like", "follow", "share", "like_rate", "follow_rate",
36             "share_rate", "ctr", "watch_mean", "watch_max", "watch_sum",
37             "male_ratio", "avg_age", "log1p_watch_sum"]
38USER_HIST_COLS = ITEM_COLS[:13]  # user dense 槽位 2..14 的来源列
39TS_BASE = 1700000000
40CLIP = 5.0

43def zval(x: float, mean: float, std: float) -> float:
44    return max(-CLIP, min(CLIP, (x - mean) / std)) if std > 1e-6 else 0.0
45

47class ColStats:
48    """流式列统计（count/sum/sumsq → mean/std，std 下限 1e-6）。"""
49
50    def __init__(self, n: int) -> None:
51        self.n = [0] * n
52        self.sum = [0.0] * n
53        self.sq = [0.0] * n
54
55    def add(self, vals: list[float]) -> None:
56        for i, v in enumerate(vals):
57            self.n[i] += 1
58            self.sum[i] += v
59            self.sq[i] += v * v
60
61    def mean(self, i: int) -> float:
62        return self.sum[i] / self.n[i] if self.n[i] else 0.0
63
64    def std(self, i: int) -> float:
65        if self.n[i] < 2:
66            return 1e9  # 单样本视作无变异 → z=0
67        var = max(0.0, self.sq[i] / self.n[i] - (self.sum[i] / self.n[i]) ** 2)
68        return max(1e-6, math.sqrt(var))

71def reservoir(rng: random.Random, cap: int):
72    """返回 (push(item), finalize()->list)。标准蓄水池采样。"""
73    picked: list = []
74
75    def push(item) -> None:
76        if len(picked) < cap:
77            picked.append(item)
78        else:
79            j = rng.randrange(len(picked) + 1)
80            if j < len(picked):
81                picked[j] = item
82
83    return push, lambda: picked

86def generate(feat_dir: Path, out_dir: Path, k_users: int, m_cands: int,
87             seed: int) -> int:
88    rng = random.Random(seed)
89    item_tsv = feat_dir / "item_features.tsv"
90    user_tsv = feat_dir / "user_features.tsv"
91
92    # -- pass A1：item 全体列统计 + 候选池蓄水池采样（K*M 个 item_id） -------
93    stats = ColStats(len(ITEM_COLS))
94    push_cand, finalize_cand = reservoir(rng, k_users * m_cands)
95    cat_of: dict[int, int] = {}
96    with item_tsv.open(encoding="utf-8") as f:
97        next(f)  # header
98        for line in f:
99            p = line.rstrip("\n").split("\t")
100            if len(p) < 2 + len(ITEM_COLS):
101                continue
102            iid = int(p[0])
103            cat_of[iid] = int(p[1])
104            stats.add([float(x) for x in p[2:2 + len(ITEM_COLS)]])
105            push_cand(iid)
106    cand_pool = [i for i in finalize_cand() if i in cat_of]
107
108    # -- pass A2：user 全体 gender/age 统计 + 用户蓄水池采样 -------------------
109    ustat = ColStats(2)
110    push_user, finalize_user = reservoir(rng, k_users)
111    with user_tsv.open(encoding="utf-8") as f:
112        next(f)
113        for line in f:
114            p = line.rstrip("\n").split("\t")
115            if len(p) < 4 or not p[3]:
116                continue
117            ustat.add([float(p[1]), float(p[2])])
118            push_user((int(p[0]), float(p[1]), float(p[2]), p[3].split(",")))
119    users = finalize_user()
120
121    # -- pass B：按需取回 item 统计行（hist item ∪ 候选池） --------------------
122    need: set[int] = set(cand_pool)
123    for _, _, _, hist in users:
124        need.update(int(h) for h in hist if h)
125    rows: dict[int, list[float]] = {}
126    with item_tsv.open(encoding="utf-8") as f:
127        next(f)
128        for line in f:
129            p = line.rstrip("\n").split("\t")
130            if len(p) < 2 + len(ITEM_COLS):
131                continue
132            iid = int(p[0])
133            if iid in need:
134                rows[iid] = [float(x) for x in p[2:2 + len(ITEM_COLS)]]
135    print(f"[adapt] item 统计行 {len(rows)}/{len(need)}，候选池 {len(cand_pool)}，用户 {len(users)}")
136
137    # -- 组装 cases ------------------------------------------------------------
138    out_dir.mkdir(parents=True, exist_ok=True)
139    out_path = out_dir / "cases.jsonl"
140    n_lines = 0
141    with out_path.open("w", encoding="utf-8") as out:
142        for uid, gender, age, hist in users:
143            hist_ids = [int(h) for h in hist if h][:50]
144            if len(hist_ids) < 5:
145                continue  # S 段下限 5
146            uid_str = f"tenrec-{uid}"
147
148            # ingest：S 段 = hist 原生 item id + 合成升序时间戳
149            out.write(json.dumps({
150                "kind": "ingest", "user_id": uid_str,
151                "item_ids": hist_ids,
152                "timestamps": [TS_BASE + i * 60 for i in range(len(hist_ids))],
153            }, ensure_ascii=False) + "\n")
154
155            # user dense：z(gender), z(age) + hist item 前 13 列 z 均值
156            user_dense = [zval(gender, ustat.mean(0), ustat.std(0)),
157                          zval(age, ustat.mean(1), ustat.std(1))]
158            for c in range(len(USER_HIST_COLS)):
159                zs = [zval(rows[h][c], stats.mean(c), stats.std(c))
160                      for h in hist_ids if h in rows]
161                user_dense.append(max(-CLIP, min(CLIP, sum(zs) / len(zs))) if zs else 0.0)
162
163            # score：每用户独立采样 M 个候选（含 item 画像 + category bag）
164            cands = rng.sample(cand_pool, min(m_cands, len(cand_pool)))
165            cand_json = []
166            for cid in cands:
167                st = rows[cid]
168                cand_json.append({
169                    "item_id": cid,                    # Tenrec 原生 id
170                    "artist_ids": [cat_of[cid]],       # video_category 即 artist
171                    "album_ids": [],
172                    "dense": [zval(st[c], stats.mean(c), stats.std(c))
173                              for c in range(len(ITEM_COLS))],
174                })
175            out.write(json.dumps({
176                "kind": "score", "user_id": uid_str,
177                "uid_sparse": uid,                     # Tenrec 原生 user_id
178                "user_dense": user_dense,
179                "candidates": cand_json,
180            }, ensure_ascii=False) + "\n")
181            n_lines += 2
182    print(f"[adapt] 写出 {out_path}（{n_lines} 行 = ingest+score × {n_lines // 2} 用户）")
183    return 0

186def http_json(base: str, path: str, body: dict) -> tuple[int, dict]:
187    req = urllib.request.Request(base + path, data=json.dumps(body).encode(),
188                                 method="POST",
189                                 headers={"Content-Type": "application/json"})
190    with urllib.request.urlopen(req, timeout=30) as resp:
191        return resp.status, json.loads(resp.read().decode())

194def replay(cases: Path, server: str) -> int:
195    lines = [json.loads(l) for l in cases.read_text(encoding="utf-8").splitlines() if l]
196    first_score = next(l for l in lines if l["kind"] == "score")
197    failures = 0
198    n_user = 0
199
200    def check(name: str, ok: bool, detail: str = "") -> None:
201        nonlocal failures
202        print(f"{'PASS' if ok else 'FAIL'}  {name}" + (f"  {detail}" if detail else ""))
203        if not ok:
204            failures += 1
205
206    for case in lines:
207        if case["kind"] == "ingest":
208            code, j = http_json(server, "/ingest", case)
209            check(f"ingest {case['user_id']}",
210                  code == 200 and j.get("accepted") is True)
211        else:
212            m = len(case["candidates"])
213            code, j = http_json(server, "/score", case)
214            logits = j.get("logits", [])
215            finite = all(math.isfinite(v) for row in logits for v in row)
216            check(f"score {case['user_id']}",
217                  code == 200 and j.get("kv_hit") is True
218                  and len(logits) == m and all(len(r) == 2 for r in logits) and finite,
219                  f"M={m} kv_hit={j.get('kv_hit')} "
220                  f"top_logit={max((v for r in logits for v in r), default=float('nan')):.4f}")
221            n_user += 1
222
223    # 确定性：同一 score 请求重发两次 → logits 逐位一致（哈希折叠 + 查表全确定）
224    _, r1 = http_json(server, "/score", first_score)
225    _, r2 = http_json(server, "/score", first_score)
226    check("score/determinism", r1.get("logits") == r2.get("logits"))
227
228    print(f"\n[replay] {n_user} 用户完成，"
229          f"{'全部通过' if failures == 0 else f'{failures} 项失败'}")
230    return 0 if failures == 0 else 1

233def main() -> int:
234    ap = argparse.ArgumentParser(description="Tenrec → 音乐域契约特征适配器")
235    ap.add_argument("--feat-dir", type=Path, default=Path("artifacts/domain_features"))
236    ap.add_argument("--out-dir", type=Path, default=Path("artifacts/tenrec_cases"))
237    ap.add_argument("--users", type=int, default=8, help="采样用户数")
238    ap.add_argument("--cands", type=int, default=4, help="每用户候选数")
239    ap.add_argument("--seed", type=int, default=2026)
240    ap.add_argument("--replay", action="store_true", help="回放模式")
241    ap.add_argument("--cases", type=Path, default=Path("artifacts/tenrec_cases/cases.jsonl"))
242    ap.add_argument("--server", default="http://127.0.0.1:19099")
243    args = ap.parse_args()
244
245    if args.replay:
246        return replay(args.cases, args.server.rstrip("/"))
247    return generate(args.feat_dir, args.out_dir, args.users, args.cands, args.seed)
