7EmbeddingTables EmbeddingTables::load(const ArtifactStore& store) {
8    EmbeddingTables t;
9    t.d_model_ = store.config().d_model;
10
11    struct Spec {
12        const char* name;
13        const char* tensor;
14    };
15    const Spec specs[] = {
16        {"item", "embedder.item_embedding.weight"},
17        {"user", "embedder.user_embedding.weight"},
18        {"artist", "embedder.artist_embedding.embedding.weight"},
19        {"album", "embedder.album_embedding.embedding.weight"},
20    };
21    for (const auto& s : specs) {
22        Tensor w = store.get(s.tensor);
23        if (w.shape.size() != 2 || w.shape[1] != t.d_model_)
24            throw std::runtime_error(std::string("embedding 表形状异常: ") + s.tensor);
25        t.tables_[s.name] = std::move(w.data);
26    }
27    return t;
28}

30std::vector<float> EmbeddingTables::lookup(const std::string& table,
31                                           const std::vector<int64_t>& ids) const {
32    auto it = tables_.find(table);
33    if (it == tables_.end()) throw std::runtime_error("未知 embedding 表: " + table);
34    const auto& tab = it->second;
35    const int64_t rows = static_cast<int64_t>(tab.size()) / d_model_;
36
37    std::vector<float> out;
38    out.reserve(ids.size() * static_cast<size_t>(d_model_));
39    for (int64_t id : ids) {
40        if (id < 0 || id >= rows)
41            throw std::runtime_error("embedding id 越界: table=" + table + " id=" + std::to_string(id));
42        const float* row = tab.data() + static_cast<size_t>(id) * static_cast<size_t>(d_model_);
43        out.insert(out.end(), row, row + d_model_);
44    }
45    return out;
46}
