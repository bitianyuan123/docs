8ScoreFlow::ScoreFlow(const EmbeddingFrontend& frontend, const TwoStageRunner& runner,
9                     KVStore& store, const LookupFn& lookup, Metrics& metrics,
10                     PythonComputeBridge* bridge, std::string model_version, Config cfg)
11    : frontend_(frontend),
12      runner_(runner),
13      store_(store),
14      lookup_(lookup),
15      metrics_(metrics),
16      bridge_(bridge),
17      model_version_(std::move(model_version)),
18      cfg_(cfg) {
19    if (cfg_.compute_threads <= 0) {
20        unsigned int hw = std::thread::hardware_concurrency();
21        cfg_.compute_threads = hw > 0 ? static_cast<int>(hw) : 4;
22    }
23    // 功能分池（§7.4.2）：IO 池承接阻塞式网络/外部调用，CPU 池承接计算。
24    lookup_pool_ = executors_.make("embed_lookup", cfg_.lookup_threads, cfg_.queue_cap, true);
25    encode_pool_ = executors_.make("frontend_encode", cfg_.encode_threads, cfg_.queue_cap, false);
26    kv_pool_ = executors_.make("kv_io", cfg_.kv_threads, cfg_.queue_cap, true);
27    compute_pool_ = executors_.make("compute_cpp", cfg_.compute_threads, cfg_.queue_cap, false);
28}

32void ScoreFlow::start() {
33    if (running_.exchange(true)) return;
34    batch_thread_ = std::thread([this] { batch_loop(); });
35}

37void ScoreFlow::stop() {
38    if (!running_.exchange(false)) return;
39    {
40        std::lock_guard<std::mutex> lk(batch_mu_);
41        batch_stop_ = true;
42    }
43    batch_cv_.notify_all();
44    if (batch_thread_.joinable()) batch_thread_.join();
45    executors_.stop_all();
46}

60std::future<ScoreOutcome> ScoreFlow::submit(ScoreInput input) {
61    auto promise = std::make_shared<std::promise<ScoreOutcome>>();
62    auto fut = promise->get_future();
63    submit(std::move(input), [promise](ScoreOutcome out, const std::string& err) {
64        if (err.empty())
65            promise->set_value(std::move(out));
66        else
67            promise->set_exception(
68                std::make_exception_ptr(std::runtime_error("flow: " + err)));
69    });
70    return fut;
71}

78// ---- 阶段 1：EmbedLookup Pool（IO）——查表 + mean-bag ---------------------- //
79void ScoreFlow::stage_lookup(const CtxPtr& ctx) {
80    try {
81        lookup_pool_->add([this, ctx] {
82            try {
83                {
84                    Metrics::Timer t(metrics_, "flow.stage_lookup");
85                    auto _t0 = std::chrono::steady_clock::now();
86                    ctx->emb = frontend_.lookup_ns(ctx->input, lookup_);
87                    auto look_us = std::chrono::duration_cast<std::chrono::microseconds>(
88                        std::chrono::steady_clock::now() - _t0).count();
89                    ctx->lookup_us += look_us;
90                    ctx->feature_us += look_us;
91                }
92                stage_encode(ctx);
93            } catch (const std::exception& e) {
94                fail_ctx(ctx, e.what());
95            }
96        });
97    } catch (const ExecutorOverloaded& e) {
98        metrics_.count("flow.overload_lookup");
99        fail_ctx(ctx, e.what());
100    }
101}

103// ---- 阶段 2：Frontend Encode Pool（CPU）——piecewise + MLP + RMSNorm -------- //
104void ScoreFlow::stage_encode(const CtxPtr& ctx) {
105    try {
106        encode_pool_->add([this, ctx] {
107            try {
108                {
109                    Metrics::Timer t(metrics_, "flow.stage_encode");
110                    auto _t0 = std::chrono::steady_clock::now();
111                    ctx->ns_emb = frontend_.encode_ns_with(ctx->input, ctx->emb);
112                    ctx->emb = NsEmbeddings{};  // 释放查表中间态
113                    auto enc_us = std::chrono::duration_cast<std::chrono::microseconds>(
114                        std::chrono::steady_clock::now() - _t0).count();
115                    ctx->encode_us += enc_us;
116                    ctx->feature_us += enc_us;
117                }
118                stage_kv(ctx, std::move(ctx->ns_emb));
119            } catch (const std::exception& e) {
120                fail_ctx(ctx, e.what());
121            }
122        });
123    } catch (const ExecutorOverloaded& e) {
124        metrics_.count("flow.overload_encode");
125        fail_ctx(ctx, e.what());
126    }
127}

129// ---- 阶段 3：KV I/O Pool（IO）——mget + 攒批入队 ----------------------------- //
130void ScoreFlow::stage_kv(const CtxPtr& ctx, Tensor ns_emb) {
131    try {
132        kv_pool_->add([this, ctx, ns_emb = std::move(ns_emb)]() mutable {
133            try {
134                {
135                    Metrics::Timer t(metrics_, "flow.stage_kv");
136                    auto _t0 = std::chrono::steady_clock::now();
137                    ctx->rec = store_.get(KVKey{model_version_, ctx->input.user_id});
138                    auto kv_us_ = std::chrono::duration_cast<std::chrono::microseconds>(
139                        std::chrono::steady_clock::now() - _t0).count();
140                    ctx->kv_us += kv_us_;
141                    ctx->backend_rpc_us += kv_us_;
142                }
143                ctx->ns_emb = std::move(ns_emb);
144                {
145                    std::unique_lock<std::mutex> lk(batch_mu_);
146                    if (batch_stop_) {
147                        lk.unlock();
148                        fail_ctx(ctx, "shutting down");
149                        return;
150                    }
151                    ctx->enqueued_at = std::chrono::steady_clock::now();  // batch_wait 基准
152                    batch_q_.push_back(ctx);
153                }
154                batch_cv_.notify_one();
155            } catch (const std::exception& e) {
156                fail_ctx(ctx, e.what());
157            }
158        });
159    } catch (const ExecutorOverloaded& e) {
160        metrics_.count("flow.overload_kv");
161        fail_ctx(ctx, e.what());
162    }
163}

165// ---- 攒批线程：等首条（无限期）→ max_wait 窗口 → 满 max_batch 出批 ---------- //
166void ScoreFlow::batch_loop() {
167    while (running_) {
168        auto jobs = take_batch();
169        if (jobs.empty()) break;  // stop 唤醒
170        on_batch(std::move(jobs));
171    }
172}

174std::vector<ScoreFlow::CtxPtr> ScoreFlow::take_batch() {
175    std::unique_lock<std::mutex> lk(batch_mu_);
176    batch_cv_.wait(lk, [&] { return !batch_q_.empty() || batch_stop_; });
177    if (batch_q_.empty()) return {};
178    auto deadline = std::chrono::steady_clock::now() +
179                    std::chrono::milliseconds(cfg_.batch_wait_ms);
180    while (batch_q_.size() < cfg_.max_batch) {
181        if (batch_cv_.wait_until(lk, deadline) == std::cv_status::timeout) break;
182    }
183    std::vector<CtxPtr> batch;
184    batch.reserve(std::min(cfg_.max_batch, batch_q_.size()));
185    while (!batch_q_.empty() && batch.size() < cfg_.max_batch) {
186        batch.push_back(std::move(batch_q_.front()));
187        batch_q_.pop_front();
188    }
189    return batch;
190}

192void ScoreFlow::on_batch(std::vector<CtxPtr> jobs) {
193    Metrics::Timer t(metrics_, "flow.batch");
194    const int64_t T = runner_.model().head_w.shape[0];
195
196    // 全批统一上报 batch 维度（batch_size / batch_wait_us），miss 也一并覆盖
197    const auto batch_now = std::chrono::steady_clock::now();
198    for (auto& ctx : jobs) {
199        ctx->batch_size = static_cast<int32_t>(jobs.size());
200        ctx->batch_wait_us = std::chrono::duration_cast<std::chrono::microseconds>(
201            batch_now - ctx->enqueued_at).count();
202        if (ctx->batch_wait_us < 0) ctx->batch_wait_us = 0;
203    }
204
205    // 1. miss 前置回填（不进计算批，§7.4.8：miss 行无需前向）
206    std::vector<CtxPtr> hits;
207    hits.reserve(jobs.size());
208    for (auto& ctx : jobs) {
209        if (ctx->rec) {
210            metrics_.count("kv.hit");
211            hits.push_back(std::move(ctx));
212        } else {
213            metrics_.count("kv.miss");
214            ScoreOutcome out;
215            out.kv_hit = false;
216            out.logits = Tensor::zeros(
217                {static_cast<int64_t>(ctx->input.candidates.size()), T});
218            out.feature_us = ctx->feature_us;
219            out.lookup_us = ctx->lookup_us;
220            out.encode_us = ctx->encode_us;
221            out.kv_us = ctx->kv_us;
222            out.batch_wait_us = ctx->batch_wait_us;
223            out.batch_size = ctx->batch_size;
224            if (ctx->done) ctx->done(std::move(out), "");
225            inflight_.fetch_sub(1, std::memory_order_relaxed);
226        }
227    }
228    if (hits.empty()) return;
229
230    // 2. 拼接命中行：ns_blob fp32 [rows, Ns, D] + KV payload 去重 + row→payload
231    int64_t rows = 0;
232    for (auto& c : hits) rows += c->ns_emb.shape[0];
233    const int64_t Ns = hits[0]->ns_emb.shape[1];
234    const int64_t D = hits[0]->ns_emb.shape[2];
235
236    BridgeBatch bb;
237    bb.n_rows = rows;
238    bb.ns = Ns;
239    bb.d_model = D;
240    bb.ns_blob.resize(static_cast<size_t>(rows * Ns * D * sizeof(float)));
241
242    std::unordered_map<std::string, int32_t> dedup;  // payload → 下标
243    std::vector<int64_t> row0s;                       // 每命中请求在批内的首行
244    size_t off = 0;
245    int64_t row0 = 0;
246    for (auto& c : hits) {
247        std::memcpy(bb.ns_blob.data() + off, c->ns_emb.data.data(),
248                    c->ns_emb.data.size() * sizeof(float));
249        off += c->ns_emb.data.size() * sizeof(float);
250        row0s.push_back(row0);
251        row0 += c->ns_emb.shape[0];
252        auto it = dedup.find(c->rec->payload);
253        if (it == dedup.end()) {
254            it = dedup.emplace(c->rec->payload, static_cast<int32_t>(bb.kv_payloads.size()))
255                     .first;
256            bb.kv_payloads.push_back(c->rec->payload);
257        }
258        for (int64_t m = 0; m < c->ns_emb.shape[0]; ++m) bb.row_kv_idx.push_back(it->second);
259        c->kv_idx = it->second;  // C++ 降级路径复用去重结果
260    }
261
262    metrics_.count("online.qps", static_cast<double>(jobs.size()));
263    metrics_.count("online.candidate_throughput", static_cast<double>(rows));
264    metrics_.gauge("online.batch_size", static_cast<double>(rows));
265
266    // 3. 计算后端选择：bridge 可用 → Python（PyTorch 算子下发）；否则 C++ CPU
267    if (bridge_ && bridge_->available()) {
268        bb.on_ok = [this, hits, row0s, T](std::string blob) {
269            fill_outcomes(hits, row0s, blob, T);
270        };
271        bb.on_fail = [this, hits](const std::string& err) {
272            for (auto& c : hits) fail_ctx(c, "bridge: " + err);
273        };
274        if (bridge_->submit(std::move(bb))) {
275            metrics_.count("flow.backend.python");
276            return;
277        }
278        metrics_.count("flow.bridge_overflow");  // 桥队列满 → 降级 C++
279    }
280    (void)row0s;  // C++ 路径自行重算行基（dispatch_cpp 内重建）
281
282    dispatch_cpp(std::move(hits));
283}

285// C++ CPU 降级：Compute Pool 上一次 score_ns_batch（数值与 Python 等价，golden 对拍）
286void ScoreFlow::dispatch_cpp(std::vector<CtxPtr> hits) {
287    metrics_.count("flow.backend.cpp");
288    const int64_t T = runner_.model().head_w.shape[0];
289    compute_pool_->add([this, hits = std::move(hits), T] {
290        std::unordered_map<int32_t, std::unique_ptr<UserKV>> kvs;  // kv_idx → UserKV
291        std::vector<int64_t> row0s;
292        int64_t rows = 0;
293        try {
294            std::vector<const UserKV*> kv_rows;
295            std::vector<Tensor> embs;
296            for (auto& c : hits) {
297                auto it = kvs.find(c->kv_idx);
298                if (it == kvs.end()) {
299                    it = kvs.emplace(c->kv_idx,
300                                     std::make_unique<UserKV>(kv_deserialize(c->rec->payload)))
301                             .first;
302                }
303                row0s.push_back(rows);
304                for (int64_t m = 0; m < c->ns_emb.shape[0]; ++m) kv_rows.push_back(it->second.get());
305                embs.push_back(c->ns_emb);
306                rows += c->ns_emb.shape[0];
307            }
308            // 拼接 [rows, Ns, D]
309            const int64_t Ns = embs[0].shape[1], D = embs[0].shape[2];
310            Tensor ns_cat({rows, Ns, D});
311            int64_t off = 0;
312            for (auto& e : embs) {
313                std::copy(e.data.begin(), e.data.end(), ns_cat.data.begin() + off);
314                off += e.data.size();
315            }
316            Tensor logits;
317            int64_t comp_us = 0;
318            {
319                Metrics::Timer t(metrics_, "flow.score_cpp");
320                auto _t0 = std::chrono::steady_clock::now();
321                logits = runner_.score_ns_batch(kv_rows, ns_cat);
322                comp_us = std::chrono::duration_cast<std::chrono::microseconds>(
323                              std::chrono::steady_clock::now() - _t0).count();
324            }
325            // compute 耗时：批次真实总耗时上报 + 单请求均摊（Ns 固定，近似相等）
326            const int64_t per = hits.empty() ? comp_us : comp_us / (int64_t)hits.size();
327            for (auto& c : hits) {
328                c->compute_us += per;
329                c->batch_compute_us += comp_us;  // 批次真实前向总耗时
330            }
331            std::string blob(reinterpret_cast<const char*>(logits.data.data()),
332                             logits.data.size() * sizeof(float));
333            fill_outcomes(hits, row0s, blob, T);
334        } catch (const std::exception& e) {
335            for (auto& c : hits) fail_ctx(c, e.what());
336        }
337    });
338}

340void ScoreFlow::fill_outcomes(const std::vector<CtxPtr>& hits,
341                              const std::vector<int64_t>& row0s, const std::string& logits_blob,
342                              int64_t T) {
343    const float* lg = reinterpret_cast<const float*>(logits_blob.data());
344    for (size_t i = 0; i < hits.size(); ++i) {
345        const CtxPtr& c = hits[i];
346        const int64_t M = c->ns_emb.shape[0];
347        ScoreOutcome out;
348        out.kv_hit = true;
349        out.feature_us = c->feature_us;
350        out.compute_us = c->compute_us;
351        out.backend_rpc_us = c->backend_rpc_us;
352        out.lookup_us = c->lookup_us;
353        out.encode_us = c->encode_us;
354        out.kv_us = c->kv_us;
355        out.batch_wait_us = c->batch_wait_us;
356        out.batch_compute_us = c->batch_compute_us;
357        out.batch_size = c->batch_size;
358        out.logits = Tensor({M, T});
359        std::memcpy(out.logits.data.data(), lg + row0s[i] * T,
360                    static_cast<size_t>(M * T) * sizeof(float));
361        if (c->done) c->done(std::move(out), "");
362        inflight_.fetch_sub(1, std::memory_order_relaxed);
363    }
