1// SEDA 分阶段在线打分流水线（§7.4.3）。
2//
3// 阶段链（每阶段独立线程池，有界队列背压）：
4//   EmbedLookup Pool → Frontend Encode Pool → KV I/O Pool → 攒批线程
5//   → PythonComputeBridge（持 GIL 的 PyTorch 前向）或 Compute Pool（C++ CPU 降级）
6//
7// GIL 边界：除最终计算批投递给桥线程外，全部阶段在纯 C++ 线程池上运行
8// （docs/detailed_design.md §7.4.2 线程池清单 / §7.4.9 锁分析）。

29namespace onetrans {
30
31class ScoreFlow {
32public:
33    struct Config {
34        int lookup_threads = 4;    // EmbedLookup Pool（IO）
35        int encode_threads = 2;    // Frontend Encode Pool（CPU）
36        int kv_threads = 4;        // KV I/O Pool（IO）
37        int compute_threads = 0;   // Compute Pool（C++ 降级路径；≤0 → 硬件核数）
38        size_t queue_cap = 1024;   // 各池入口有界队列容量
39        size_t max_batch = 16;     // 攒批上限（请求数）
40        int batch_wait_ms = 5;     // 攒批窗口
41    };
42
43    // bridge 可为 nullptr（强制 C++ 路径）
44    ScoreFlow(const EmbeddingFrontend& frontend, const TwoStageRunner& runner,
45              KVStore& store, const LookupFn& lookup, Metrics& metrics,
46              PythonComputeBridge* bridge, std::string model_version, Config cfg);
47    ~ScoreFlow();
48
49    void start();
50    void stop();
51
52    // 回调式提交（生产路径）：完成线程直接调用 done
53    // （err 非空 = 流水线失败，outcome 无效）。done 保证恰好调用一次。
54    using ScoreDone = std::function<void(ScoreOutcome out, const std::string& err)>;
55    void submit(ScoreInput input, ScoreDone done);
56
57    // future 便利版（测试/工具）：内部包装回调
58    std::future<ScoreOutcome> submit(ScoreInput input);
59
60    // 当前实际生效的计算后端（观测 / healthz）
61    std::string backend() const;
62    size_t inflight() const { return inflight_.load(std::memory_order_relaxed); }
63
64private:
65    struct Ctx {  // 单请求贯穿流水线的共享状态（阶段间 move 传递所有权）
66        ScoreInput input;
67        ScoreDone done;                          // 完成回调（恰好一次）
68        NsEmbeddings emb;                         // lookup 阶段产出
69        Tensor ns_emb;                           // encode 阶段产出 [M, Ns, D]
70        std::shared_ptr<const UserKVRecord> rec;  // kv 阶段产出（miss 为 null）
71        int32_t kv_idx = -1;                      // 批内 KV payload 去重下标
72        // 单请求 stage 耗时(us)，供 /rank trace 与 PaiRec service_phase 观测
73        int64_t feature_us = 0, compute_us = 0, backend_rpc_us = 0;
74        int64_t lookup_us = 0, encode_us = 0, kv_us = 0;
75        int64_t batch_wait_us = 0, batch_compute_us = 0;  // 攒批等待 + 批次真实前向
76        int32_t batch_size = 0;                            // 所属批次的请求数
77        std::chrono::steady_clock::time_point enqueued_at; // 进入攒批队列时刻（batch_wait 基）
78    };
79    using CtxPtr = std::shared_ptr<Ctx>;
80
81    // 阶段（各自 add 到对应池；满时快速失败回填 503 语义异常）
82    void stage_lookup(const CtxPtr& ctx);
83    void stage_encode(const CtxPtr& ctx);
84    void stage_kv(const CtxPtr& ctx, Tensor ns_emb);
85
86    // 攒批线程主循环
87    void batch_loop();
88    std::vector<CtxPtr> take_batch();
89    // 出批处理：拼 ns_blob + KV 去重 + 槽位表 → bridge / compute pool
90    void on_batch(std::vector<CtxPtr> jobs);
91    // C++ CPU 降级路径（在 Compute Pool 上执行 score_ns_batch）
92    void dispatch_cpp(std::vector<CtxPtr> hits);
93    // 回填：logits blob fp32 [rows, T] 按槽位切分（miss 已在 on_batch 前置回填）
94    void fill_outcomes(const std::vector<CtxPtr>& hits, const std::vector<int64_t>& row0s,
95                       const std::string& logits_blob, int64_t T);
96    void fail_ctx(const CtxPtr& ctx, const std::string& err);
97
98    const EmbeddingFrontend& frontend_;
99    const TwoStageRunner& runner_;
100    KVStore& store_;
101    LookupFn lookup_;
102    Metrics& metrics_;
103    PythonComputeBridge* bridge_;
104    std::string model_version_;
105    Config cfg_;
106
107    ExecutorSet executors_;
108    std::shared_ptr<Executor> lookup_pool_;
109    std::shared_ptr<Executor> encode_pool_;
110    std::shared_ptr<Executor> kv_pool_;
111    std::shared_ptr<Executor> compute_pool_;
112
113    // 攒批队列（独立于各池：批线程单消费者）
114    std::mutex batch_mu_;
115    std::condition_variable batch_cv_;
116    std::deque<CtxPtr> batch_q_;
117    bool batch_stop_ = false;
118    std::thread batch_thread_;
119
120    std::atomic<size_t> inflight_{0};
121    std::atomic<bool> running_{false};
122};
123
124}  // namespace onetrans
