1// Python 计算桥（§7.4.5）：嵌入式解释器 + 专用线程持 GIL 下发 PyTorch 算子。
2//
3// 边界契约（与 docs/detailed_design.md §7.4 一致）：
4// - C++ 编排（lookup/encode/KV mget/攒批/回填）全部在标准库线程池
5//   （common/executor.h）上运行，不触碰 GIL；
6// - 唯一持 GIL 的是本桥的专用线程：从有界队列取批 → PyGILState_Ensure →
7//   调 bridge_score.score_batch（PyTorch CUDA/CPU 前向）→ 回填回调 → Release；
8// - 队列满 / 解释器不可用 / torch 缺失 → available()==false 或 submit 返回
9//   false，上层降级为 C++ CPU score_ns_batch（数值等价，性能降级）；
10// - 多 GPU 扩展：每 GPU 一个桥实例（互不共享队列）。

25// 一个待计算批（C++ 侧已完成全部前处理）
26struct BridgeBatch {
27    std::string ns_blob;                     // fp32 LE [n_rows, ns, d]
28    std::vector<std::string> kv_payloads;     // 去重后的 UserKV payload（C++ kv_serialize）
29    std::vector<int32_t> row_kv_idx;          // 每行 → payload 下标（长度 n_rows）
30    int64_t n_rows = 0, ns = 0, d_model = 0;
31
32    // 回调（桥线程调用；on_ok 的 blob 为 fp32 LE [n_rows, T]）
33    std::function<void(std::string logits_blob)> on_ok;
34    std::function<void(const std::string& err)> on_fail;
35};
36
37class PythonComputeBridge {
38public:
39    // module_dir：bridge_score.py 所在目录（sys.path 注入）
40    // weights_dir：weights.bin/manifest.json 所在目录
41    // init_timeout_s：import torch + 权重上载的最长等待（超时视为不可用，降级）
42    bool start(const std::string& module_dir, const std::string& weights_dir,
43               int init_timeout_s = 120);
44    void stop();
45
46    bool available() const { return ready_.load(std::memory_order_acquire); }
47    const std::string& last_error() const { return last_error_; }
48
49    // 投递一批（队列满返回 false → 调用方降级）
50    bool submit(BridgeBatch batch);
51
52    size_t queued() const;
53
54private:
55    void run();
56    bool init_python(const std::string& module_dir, const std::string& weights_dir);
57    // 持 GIL 执行一次前向；返回 logits blob，失败抛异常
58    std::string call_score(BridgeBatch& b);
59
60    mutable std::mutex mu_;
61    std::condition_variable cv_;
62    std::deque<BridgeBatch> q_;
63    size_t cap_ = 16;
64    bool stopped_ = true;
65    std::atomic<bool> ready_{false};
66    std::string last_error_;
67    std::thread thread_;
68    // Python 对象句柄（实现细节，void* 隔离 Python.h 泄漏到头文件）
69    void* py_module_ = nullptr;
70    void* py_score_fn_ = nullptr;
71    void* py_meta_fn_ = nullptr;
72    // 首轮 init 后已 PyEval_SaveThread（重试时需走 Ensure/Release 而非再 Save）
73    bool py_thread_saved_ = false;
74};
