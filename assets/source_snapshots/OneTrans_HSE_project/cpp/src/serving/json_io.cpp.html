12IngestInput parse_ingest_input(const std::string& body) {
13    JsonValue root = json_parse(body);
14    IngestInput in;
15    in.user_id = root.at("user_id").as_string();
16    for (const auto& v : root.at("item_ids").as_array()) in.item_ids.push_back(v.as_int());
17    for (const auto& v : root.at("timestamps").as_array()) in.timestamps.push_back(v.as_int());
18    if (in.item_ids.size() != in.timestamps.size())
19        throw std::runtime_error("item_ids 与 timestamps 必须等长");
20    return in;
21}

23ScoreInput parse_score_input(const std::string& body) {
24    JsonValue root = json_parse(body);
25    ScoreInput in;
26    in.user_id = root.at("user_id").as_string();
27    in.uid_sparse = root.at("uid_sparse").as_int();
28    for (const auto& v : root.at("user_dense").as_array())
29        in.user_dense.push_back(static_cast<float>(v.as_number()));
30    for (const auto& c : root.at("candidates").as_array()) {
31        CandidateInput cand;
32        cand.item_id = c.at("item_id").as_int();
33        for (const auto& v : c.at("artist_ids").as_array()) cand.artist_ids.push_back(v.as_int());
34        for (const auto& v : c.at("album_ids").as_array()) cand.album_ids.push_back(v.as_int());
35        for (const auto& v : c.at("dense").as_array())
36            cand.dense.push_back(static_cast<float>(v.as_number()));
37        in.candidates.push_back(std::move(cand));
38    }
39    if (in.candidates.empty()) throw std::runtime_error("candidates 为空");
40    return in;
41}

43std::string logits_to_json(const Tensor& logits) {
44    const int64_t M = logits.shape[0], T = logits.shape[1];
45    std::ostringstream os;
46    os << std::setprecision(9) << "[";
47    for (int64_t m = 0; m < M; ++m) {
48        if (m) os << ",";
49        os << "[";
50        for (int64_t t = 0; t < T; ++t) {
51            if (t) os << ",";
52            os << logits.at(m * T + t);
53        }
54        os << "]";
55    }
56    os << "]";
57    return os.str();
58}

60PaiRecRankRequest parse_rank_request(const std::string& body) {
61    JsonValue root = json_parse(body);
62    PaiRecRankRequest req;
63    req.user_id = root.at("user_id").as_string();
64    if (root.as_object().count("request_id")) req.request_id = root.at("request_id").as_string();
65    for (const auto& it : root.at("items").as_array()) {
66        // PaiRec 契约 item_id 为 string（"101"）；也兼容 number（101）型。
67        const JsonValue& v = it.at("item_id");
68        int64_t id = 0;
69        if (v.is_number()) {
70            id = v.as_int();
71        } else {
72            const std::string& s = v.as_string();
73            id = s.empty() ? 0 : std::strtoll(s.c_str(), nullptr, 10);
74        }
75        req.item_ids.push_back(id);
76    }
77    if (req.item_ids.empty()) throw std::runtime_error("rank items 为空");
78    return req;
79}
