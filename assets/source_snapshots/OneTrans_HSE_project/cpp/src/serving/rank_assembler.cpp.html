7// UTF-8 BOM 剥离（Linux 下头行一般无 BOM，防御式处理）
8static std::string strip_utf8_bom(const std::string& s) {
9    if (s.size() >= 3 && static_cast<unsigned char>(s[0]) == 0xEF &&
10        static_cast<unsigned char>(s[1]) == 0xBB &&
11        static_cast<unsigned char>(s[2]) == 0xBF)
12        return s.substr(3);
13    return s;
14}

18void RankAssembler::load_item_lookup(const std::string& path) {
19    std::ifstream in(path);
20    if (!in) throw std::runtime_error("rank item lookup 文件不可读: " + path);
21    std::string line;
22    bool first = true;
23    int64_t rows = 0;
24    while (std::getline(in, line)) {
25        if (line.empty()) continue;
26        if (first) {  // 表头
27            first = false;
28            continue;
29        }
30        std::istringstream ss = std::istringstream(strip_utf8_bom(line));
31        std::string tok;
32        // item_id \t artist_ids \t album_ids \t d1..d15
33        std::vector<std::string> cols;
34        while (std::getline(ss, tok, '\t')) cols.push_back(tok);
35        if (cols.size() < 3) continue;  // 坏行跳过
36        int64_t item_id = std::strtoll(cols[0].c_str(), nullptr, 10);
37        int64_t cat = std::strtoll(cols[1].c_str(), nullptr, 10);
38        ItemRank ir;
39        ir.artist_ids.push_back(cat);
40        ir.album_ids.push_back(cat);  // artist/album 复用 video_category
41        ir.dense.reserve(static_cast<size_t>(n_cand_dense_));
42        for (int i = 3; i < static_cast<int>(cols.size()) && i < 3 + n_cand_dense_; ++i)
43            ir.dense.push_back(static_cast<float>(std::strtod(cols[i].c_str(), nullptr)));
44        while (static_cast<int>(ir.dense.size()) < n_cand_dense_) ir.dense.push_back(0.0f);
45        items_[item_id] = std::move(ir);
46        ++rows;
47    }
48    if (rows == 0) throw std::runtime_error("rank item lookup 无有效数据行: " + path);
49}

51void RankAssembler::load_user_features(const std::string& path) {
52    std::ifstream in(path);
53    if (!in) throw std::runtime_error("rank user features 文件不可读: " + path);
54    std::string line;
55    bool first = true;
56    int64_t rows = 0;
57    while (std::getline(in, line)) {
58        if (line.empty()) continue;
59        if (first) {  // 表头
60            first = false;
61            continue;
62        }
63        std::istringstream ss = std::istringstream(strip_utf8_bom(line));
64        std::string tok;
65        std::vector<std::string> cols;
66        while (std::getline(ss, tok, '\t')) cols.push_back(tok);
67        if (cols.size() < 3) continue;  // 需 user_id/gender/age
68        int64_t uid = std::strtoll(cols[0].c_str(), nullptr, 10);
69        UserFeat uf;
70        uf.gender = static_cast<float>(std::strtod(cols[1].c_str(), nullptr));
71        uf.age = static_cast<float>(std::strtod(cols[2].c_str(), nullptr));
72        users_[uid] = uf;
73        ++rows;
74    }
75    if (rows == 0) throw std::runtime_error("rank user features 无有效数据行: " + path);
76}

78ScoreInput RankAssembler::assemble(const std::string& user_id,
79                                   const std::vector<int64_t>& item_ids) const {
80    ScoreInput in;
81    in.user_id = user_id;
82    in.uid_sparse = std::strtoll(user_id.c_str(), nullptr, 10);  // Tenrec 原生 user_id
83
84    // user_dense：gender/age + 补 0 至 n_user_dense 列
85    in.user_dense.assign(static_cast<size_t>(n_user_dense_), 0.0f);
86    int64_t uid = std::strtoll(user_id.c_str(), nullptr, 10);
87    auto uit = users_.find(uid);
88    if (uit != users_.end()) {
89        in.user_dense[0] = uit->second.gender;
90        if (n_user_dense_ >= 2) in.user_dense[1] = uit->second.age;
91    }
92
93    in.candidates.reserve(item_ids.size());
94    for (int64_t id : item_ids) {
95        CandidateInput cand;
96        cand.item_id = id;
97        auto it = items_.find(id);
98        if (it != items_.end()) {
99            cand.artist_ids = it->second.artist_ids;
100            cand.album_ids = it->second.album_ids;
101            cand.dense = it->second.dense;
102        } else {
103            cand.dense.assign(static_cast<size_t>(n_cand_dense_), 0.0f);
104        }
105        in.candidates.push_back(std::move(cand));
106    }
107    return in;
108}
