39void Metrics::observe(const std::string& name, double value_us) {
40    const auto& bounds = hist_bounds();
41    const double v = value_us / 1e6;  // 微秒 → 秒（桶界已按秒对齐 PaiRec）
42    std::lock_guard<std::mutex> lk(mu_);
43    auto& h = hists_[name];
44    if (h.bucket_counts.size() != bounds.size() + 1)
45        h.bucket_counts.assign(bounds.size() + 1, 0.0);
46    h.sum += v;
47    h.count += 1.0;
48    size_t idx = bounds.size();  // 默认落到 +Inf 桶
49    for (size_t i = 0; i < bounds.size(); ++i) {
50        if (v <= bounds[i]) {
51            idx = i;
52            break;
53        }
54    }
55    h.bucket_counts[idx] += 1.0;
56}

62Metrics::Timer::~Timer() {
63    auto us = std::chrono::duration_cast<std::chrono::microseconds>(
64                  std::chrono::steady_clock::now() - t0_)
65                  .count();
66    m_.count(name_ + "_us_total", static_cast<double>(us));
67    m_.count(name_ + "_n", 1.0);

71std::string Metrics::dump() const {
72    std::unordered_map<std::string, double> snap;
73    std::unordered_map<std::string, Histogram> hist_snap;
74    {
75        std::lock_guard<std::mutex> lk(mu_);
76        snap = values_;
77        hist_snap = hists_;
78    }
79    const auto& bounds = hist_bounds();
80    std::ostringstream os;
81    for (const auto& [k, v] : snap) os << k << " " << v << "\n";
82    // 直方图：桶分布（Prometheus 风格，固定 le 标签维，label 基数值可控）
83    for (const auto& [k, h] : hist_snap) {
84        if (h.bucket_counts.size() != bounds.size() + 1) continue;
85        for (size_t i = 0; i < bounds.size(); ++i)
86            os << k << "_seconds_bucket{le=\"" << bounds[i] << "\"} "
87               << h.bucket_counts[i] << "\n";
88        os << k << "_seconds_bucket{le=\"+Inf\"} " << h.bucket_counts.back() << "\n";
89        os << k << "_seconds_sum " << h.sum << "\n";
90        os << k << "_seconds_count " << h.count << "\n";
91    }
92    return os.str();
93}

98IngestResult NearlineWorker::ingest(const IngestInput& in) {
99    IngestResult res;
100    const auto t0 = std::chrono::steady_clock::now();
101    try {
102        Metrics::Timer timer(metrics_, "nearline.ingest");
103        auto [s_emb, s_mask] = frontend_.encode_s(in.item_ids, lookup_);
104        UserKV kv = runner_.encode_s(s_emb, s_mask);
105        std::string payload = kv_serialize(kv);
106
107        UserKVRecord rec;
108        rec.key = KVKey{model_version_, in.user_id};
109        rec.s_len = kv.s_len;
110        rec.per_layer_len = kv.per_layer_len;
111        rec.seq_ts_last = in.timestamps.empty() ? 0 : in.timestamps.back();
112        rec.created_at = now_unix();
113        rec.payload = std::move(payload);
114        res.checksum = rec.checksum();
115        res.accepted = store_.put(rec);
116
117        metrics_.count("nearline.events_ingested", static_cast<double>(in.item_ids.size()));
118        metrics_.gauge("kv.obj_cnt", static_cast<double>(store_.size()));
119    } catch (const std::exception& e) {
120        res.accepted = false;
121        res.reason = e.what();
122    }
123    res.ingest_us = std::chrono::duration_cast<std::chrono::microseconds>(
124                        std::chrono::steady_clock::now() - t0).count();
125    return res;
126}

131ScoreOutcome OnlineWorker::score(const ScoreInput& in) {
132    Metrics::Timer timer(metrics_, "online.score");
133    Tensor ns_emb = frontend_.encode_ns(in, lookup_);
134    const int64_t M = ns_emb.shape[0];
135    const int64_t T = runner_.model().head_w.shape[0];
136
137    auto rec = store_.get(KVKey{model_version_, in.user_id});
138    if (!rec) {
139        // KV miss（冷启动/过期/搬迁）：全零 legal logits 降级（对齐 G8 语义）
140        metrics_.count("kv.miss");
141        ScoreOutcome out;
142        out.logits = Tensor::zeros({M, T});
143        out.kv_hit = false;
144        return out;
145    }
146    metrics_.count("kv.hit");
147
148    UserKV kv = kv_deserialize(rec->payload);
149    ScoreOutcome out;
150    out.logits = runner_.score_ns(kv, ns_emb);
151    metrics_.count("online.qps");
152    metrics_.count("online.candidate_throughput", static_cast<double>(M));
153    return out;
154}

156std::vector<ScoreOutcome> OnlineWorker::score_batch(const std::vector<ScoreInput>& batch) {
157    Metrics::Timer timer(metrics_, "online.score_batch");
158    if (batch.empty()) return {};
159
160    // 1. 逐请求前处理
161    std::vector<Tensor> ns_embs;
162    ns_embs.reserve(batch.size());
163    for (const auto& in : batch) ns_embs.push_back(frontend_.encode_ns(in, lookup_));
164
165    // 2. 一次 mget
166    std::vector<KVKey> keys;
167    keys.reserve(batch.size());
168    for (const auto& in : batch) keys.push_back(KVKey{model_version_, in.user_id});
169    auto recs = store_.mget(keys);
170
171    // 3. 命中请求：K/V 反序列化 + 候选展平；miss 请求记录位置
172    std::vector<std::unique_ptr<UserKV>> kvs;
173    std::vector<Tensor> hit_embs;
174    std::vector<int> hit_of_req(batch.size(), -1);  // 请求 → 命中展平行
175    int64_t total = 0;
176    for (size_t r = 0; r < batch.size(); ++r) {
177        const int64_t M = ns_embs[r].shape[0];
178        if (recs[r]) {
179            metrics_.count("kv.hit");
180            kvs.push_back(std::make_unique<UserKV>(kv_deserialize(recs[r]->payload)));
181            hit_embs.push_back(ns_embs[r]);
182            hit_of_req[r] = static_cast<int>(hit_embs.size()) - 1;
183        } else {
184            metrics_.count("kv.miss");
185        }
186        total += M;
187    }
188
189    // 4. 一次 score_ns_batch：每候选一行（同用户重复指向其 KV）
190    const int64_t T = runner_.model().head_w.shape[0];
191    Tensor hit_logits;
192    if (!kvs.empty()) {
193        int64_t rows = 0;
194        for (const auto& e : hit_embs) rows += e.shape[0];
195        Tensor ns_cat({rows, ns_embs[0].shape[1], ns_embs[0].shape[2]});
196        std::vector<const UserKV*> kv_rows;
197        kv_rows.reserve(static_cast<size_t>(rows));
198        int64_t off = 0;
199        for (size_t h = 0; h < hit_embs.size(); ++h) {
200            const auto& e = hit_embs[h];
201            std::copy(e.data.begin(), e.data.end(), ns_cat.data.begin() + off);
202            off += e.data.size();
203            for (int64_t m = 0; m < e.shape[0]; ++m) kv_rows.push_back(kvs[h].get());
204        }
205        hit_logits = runner_.score_ns_batch(kv_rows, ns_cat);
206    }
207
208    // 5. 回填：miss 全零行，顺序与 batch 一致
209    std::vector<ScoreOutcome> out(batch.size());
210    for (size_t r = 0; r < batch.size(); ++r) {
211        const int64_t M = ns_embs[r].shape[0];
212        out[r].kv_hit = hit_of_req[r] >= 0;
213        out[r].logits = Tensor::zeros({M, T});
214        if (hit_of_req[r] >= 0) {
215            int64_t row0 = 0;
216            for (int h = 0; h < hit_of_req[r]; ++h) row0 += hit_embs[h].shape[0];
217            std::copy(hit_logits.data.begin() + row0 * T, hit_logits.data.begin() + (row0 + M) * T,
218                      out[r].logits.data.begin());
219        }
220    }
221    metrics_.count("online.qps", static_cast<double>(batch.size()));
222    metrics_.count("online.candidate_throughput", static_cast<double>(total));
223    metrics_.gauge("online.batch_size", static_cast<double>(total));
224    return out;
225}

230void BatchScheduler::submit(ScoreJob job) {
231    {
232        std::lock_guard<std::mutex> lk(mu_);
233        queue_.push_back(std::move(job));
234    }
235    cv_.notify_one();
236}

238std::vector<ScoreJob> BatchScheduler::next_batch() {
239    std::unique_lock<std::mutex> lk(mu_);
240    cv_.wait(lk, [&] { return !queue_.empty() || shutdown_; });
241    if (queue_.empty()) return {};  // shutdown 唤醒
242    auto deadline = std::chrono::steady_clock::now() + max_wait_;
243    while (queue_.size() < max_batch_) {
244        if (cv_.wait_until(lk, deadline) == std::cv_status::timeout) break;
245    }
246    std::vector<ScoreJob> batch;
247    batch.reserve(std::min(max_batch_, queue_.size()));
248    while (!queue_.empty() && batch.size() < max_batch_) {
249        batch.push_back(std::move(queue_.front()));
250        queue_.pop_front();
251    }
252    return batch;
253}
254
255void BatchScheduler::shutdown() {
256    {
257        std::lock_guard<std::mutex> lk(mu_);
258        shutdown_ = true;
259    }
260    cv_.notify_all();
261}
262
263size_t BatchScheduler::pending() const {
264    std::lock_guard<std::mutex> lk(mu_);
265    return queue_.size();
266}

279std::future<ScoreOutcome> Dispatcher::submit(ScoreInput input) {
280    auto promise = std::make_shared<std::promise<ScoreOutcome>>();
281    auto fut = promise->get_future();
282    sched_.submit(ScoreJob{std::move(input), promise});
283    return fut;

286void Dispatcher::start() {
287    if (running_.exchange(true)) return;
288    for (int i = 0; i < num_threads_; ++i) threads_.emplace_back([this] { worker_loop(); });
289}
290
291void Dispatcher::stop() {
292    if (!running_.exchange(false)) return;
293    sched_.shutdown();
294    for (auto& t : threads_) {
295        if (t.joinable()) t.join();
296    }
297    threads_.clear();
298}
299
300void Dispatcher::worker_loop() {
301    while (running_) {
302        auto batch = sched_.next_batch();
303        if (batch.empty()) break;  // shutdown
304        std::vector<ScoreInput> inputs;
305        inputs.reserve(batch.size());
306        for (auto& job : batch) inputs.push_back(std::move(job.input));
307        try {
308            auto outcomes = worker_.score_batch(inputs);
309            for (size_t i = 0; i < batch.size(); ++i)
310                batch[i].promise->set_value(std::move(outcomes[i]));
311        } catch (const std::exception& e) {
312            for (auto& job : batch)
313                job.promise->set_exception(std::make_exception_ptr(std::runtime_error(e.what())));
314        }
315    }
316}
