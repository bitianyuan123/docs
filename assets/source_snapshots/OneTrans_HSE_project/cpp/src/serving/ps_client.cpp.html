14std::string ps_table_key(const std::string& prefix, const std::string& table) {
15    return prefix.empty() ? table : prefix + "/" + table;
16}

18struct PsLookupClient::Impl {
19    std::string prefix_;
20    brpc::Channel channel_;
21    std::unique_ptr<onetrans::ps::EmbeddingService_Stub> stub_;
22    bool ok_ = false;
23
24    Impl(std::string server, std::string prefix)
25        : prefix_(std::move(prefix)) {
26        brpc::ChannelOptions opt;
27        opt.timeout_ms = 5000;
28        opt.connect_timeout_ms = 1000;
29        if (channel_.Init(server.c_str(), &opt) == 0) {
30            stub_ = std::make_unique<onetrans::ps::EmbeddingService_Stub>(&channel_);
31            ok_ = true;
32        }
33    }
34};

36PsLookupClient::PsLookupClient(std::string server, std::string table_prefix)

39bool PsLookupClient::available() const noexcept { return impl_ && impl_->ok_; }

41std::vector<float> PsLookupClient::lookup(const std::string& table,
42                                          const std::vector<int64_t>& ids) const {
43    if (!impl_ || !impl_->ok_)
44        throw std::runtime_error("ps client 未就绪（channel 初始化失败）");
45    onetrans::ps::EmbeddingLookupRequest req;
46    onetrans::ps::EmbeddingLookupResponse resp;
47    req.set_table(ps_table_key(impl_->prefix_, table));
48    for (int64_t id : ids) req.add_ids(id);
49    brpc::Controller cntl;
50    impl_->stub_->Lookup(&cntl, &req, &resp, nullptr);
51    if (cntl.Failed())
52        throw std::runtime_error("PS Lookup 失败: " + cntl.ErrorText());
53    return std::vector<float>(resp.weights().begin(), resp.weights().end());
54}
