22bool PythonComputeBridge::init_python(const std::string& module_dir,
23                                      const std::string& weights_dir) {
24    // Py_Initialize 线程安全且幂等；首次调用线程自此持有 GIL。初始化全程在
25    // 本线程执行；首轮结束时 PyEval_SaveThread 释放 GIL 并注销线程状态——
26    // 否则本（detached）线程退出后 GIL 仍被持有，run 线程的 Ensure 将永久
27    // 阻塞。重试轮（GIL 已释放）走 Ensure/Release 配对。
28    Py_Initialize();
29    const bool first_round = !py_thread_saved_;
30    PyGILState_STATE gil = PyGILState_STATE(0);
31    if (!first_round) gil = PyGILState_Ensure();  // 重试轮：GIL 已释放，需显式取
32    bool ok = false;
33    do {
34        // 归一化绝对路径（相对路径会破坏向上探测）
35        char abs_buf[PATH_MAX];
36        std::string module_abs =
37            ::realpath(module_dir.c_str(), abs_buf) ? abs_buf : module_dir;
38        // sys.path 注入：bridge_score.py 目录 + 向上探测仓库根（onetrans 包所在）
39        std::string repo_root = module_abs;
40        for (int i = 0; i < 6 && !repo_root.empty(); ++i) {
41            if (::access((repo_root + "/onetrans/__init__.py").c_str(), F_OK) == 0) break;
42            size_t slash = repo_root.find_last_of('/');
43            if (slash == std::string::npos || slash == 0) {
44                repo_root = slash == 0 ? "/" : "";
45                if (::access((repo_root + "onetrans/__init__.py").c_str(), F_OK) != 0)
46                    repo_root.clear();
47                break;
48            }
49            repo_root = repo_root.substr(0, slash);
50        }
51        std::string add_path = "import sys; sys.path.insert(0, r'" + module_abs + "')";
52        if (!repo_root.empty())
53            add_path += "; sys.path.insert(0, r'" + repo_root + "')";
54        if (PyRun_SimpleString(add_path.c_str()) != 0) {
55            last_error_ = "sys.path 注入失败";
56            break;
57        }
58        PyObject* mod = PyImport_ImportModule("bridge_score");
59        if (!mod) {
60            PyErr_Print();
61            last_error_ = "import bridge_score 失败（检查 module_dir / onetrans 包）";
62            break;
63        }
64        PyObject* init_fn = PyObject_GetAttrString(mod, "init");
65        PyObject* score_fn = PyObject_GetAttrString(mod, "score_batch");
66        PyObject* meta_fn = PyObject_GetAttrString(mod, "meta");
67        if (!init_fn || !score_fn || !meta_fn) {
68            last_error_ = "bridge_score 缺少 init/score_batch/meta";
69            Py_XDECREF(init_fn);
70            Py_XDECREF(score_fn);
71            Py_XDECREF(meta_fn);
72            Py_DECREF(mod);
73            break;
74        }
75        // init(weights_dir)：import torch + 构建 OneTrans + 权重上 GPU（耗时，持 GIL）
76        PyObject* args = Py_BuildValue("(s)", weights_dir.c_str());
77        PyObject* meta = PyObject_CallObject(init_fn, args);
78        Py_XDECREF(args);
79        Py_DECREF(init_fn);
80        if (!meta) {
81            PyErr_Print();
82            last_error_ = "bridge_score.init 失败（torch/权重加载）";
83            Py_DECREF(score_fn);
84            Py_DECREF(meta_fn);
85            Py_DECREF(mod);
86            break;
87        }
88        Py_DECREF(meta);
89        Py_DECREF(mod);
90
91        py_module_ = nullptr;  // mod 不长期持有（函数已 incref）
92        py_score_fn_ = score_fn;
93        py_meta_fn_ = meta_fn;
94        ok = true;
95    } while (false);
96    if (first_round) {
97        // 首轮：释放 GIL 并注销本线程的线程状态（此后本线程不再触碰 Python C API）
98        PyEval_SaveThread();
99        py_thread_saved_ = true;
100    } else {
101        PyGILState_Release(gil);
102    }
103    return ok;
104}

106bool PythonComputeBridge::start(const std::string& module_dir, const std::string& weights_dir,
107                                int init_timeout_s) {
108    if (ready_.load(std::memory_order_acquire)) return true;
109    {
110        std::lock_guard<std::mutex> lk(mu_);
111        if (!stopped_) return false;  // 已在启动中
112        stopped_ = false;
113    }
114    // 初始化在独立线程执行（import torch 可达数十秒，不阻塞调用线程）
115    std::thread init_thread([this, module_dir, weights_dir] {
116        if (init_python(module_dir, weights_dir)) {
117            ready_.store(true, std::memory_order_release);
118        } else {
119            std::lock_guard<std::mutex> lk(mu_);
120            stopped_ = true;
121        }
122        cv_.notify_all();
123    });
124    init_thread.detach();
125
126    // 等待初始化完成或超时（超时降级：ready_ 仍可能稍后置位，无害）
127    auto deadline = std::chrono::steady_clock::now() +
128                    std::chrono::seconds(init_timeout_s > 0 ? init_timeout_s : 120);
129    std::unique_lock<std::mutex> lk(mu_);
130    while (!cv_.wait_until(lk, deadline,
131                           [this] { return stopped_ || ready_.load(std::memory_order_acquire); })) {
132        if (std::chrono::steady_clock::now() >= deadline) {
133            last_error_ = "bridge 初始化超时（import torch / 权重上载）";
134            return false;
135        }
136    }
137    if (!ready_.load(std::memory_order_acquire)) return false;  // init 失败
138    lk.unlock();
139
140    // 桥工作线程（唯一持 GIL 的计算线程）
141    thread_ = std::thread([this] { run(); });
142    return true;
143}

145void PythonComputeBridge::stop() {
146    {
147        std::lock_guard<std::mutex> lk(mu_);
148        if (stopped_ && !thread_.joinable()) return;
149        stopped_ = true;
150    }
151    cv_.notify_all();
152    if (thread_.joinable()) thread_.join();
153    // 释放 Python 函数引用（桥线程已 join，GIL 安全）
154#if defined(ONETRANS_WITH_PYTHON)
155    if (py_score_fn_ || py_meta_fn_) {
156        PyGILState_STATE gil = PyGILState_Ensure();
157        Py_XDECREF(reinterpret_cast<PyObject*>(py_score_fn_));
158        Py_XDECREF(reinterpret_cast<PyObject*>(py_meta_fn_));
159        py_score_fn_ = nullptr;
160        py_meta_fn_ = nullptr;
161        PyGILState_Release(gil);
162    }
163#endif
164}

166bool PythonComputeBridge::submit(BridgeBatch batch) {
167    {
168        std::lock_guard<std::mutex> lk(mu_);
169        if (stopped_ || !ready_.load(std::memory_order_acquire)) return false;
170        if (q_.size() >= cap_) return false;
171        q_.push_back(std::move(batch));
172    }
173    cv_.notify_one();
174    return true;
175}

177size_t PythonComputeBridge::queued() const {
178    std::lock_guard<std::mutex> lk(mu_);
179    return q_.size();

182// --------------------------------------------------------------------------- //
183// 桥线程主循环：C++ 队列 pop（不持 GIL）→ Ensure → 前向 → Release → 回调
184// --------------------------------------------------------------------------- //
185void PythonComputeBridge::run() {
186    while (true) {
187        BridgeBatch batch;
188        {
189            std::unique_lock<std::mutex> lk(mu_);
190            cv_.wait(lk, [this] { return stopped_ || !q_.empty(); });
191            if (q_.empty()) return;  // stop 且 drain 完
192            batch = std::move(q_.front());
193            q_.pop_front();
194        }
195        try {
196            std::string logits = call_score(batch);
197            if (batch.on_ok) batch.on_ok(std::move(logits));
198        } catch (const std::exception& e) {
199            if (batch.on_fail) batch.on_fail(e.what());
200        } catch (...) {
201            if (batch.on_fail) batch.on_fail("bridge: 未知异常");
202        }
203    }
204}

206std::string PythonComputeBridge::call_score(BridgeBatch& b) {
207    PyGILState_STATE gil = PyGILState_Ensure();
208    std::string out;
209    do {
210        PyObject* fn = reinterpret_cast<PyObject*>(py_score_fn_);
211        if (!fn) throw std::runtime_error("bridge: score_batch 未初始化");
212
213        // 参数：kv_blobs: list[bytes], row_map: list[int], ns_blob: bytes,
214        //       n_rows: int, ns: int, d: int
215        PyObject* kv_list = PyList_New(static_cast<Py_ssize_t>(b.kv_payloads.size()));
216        for (size_t i = 0; i < b.kv_payloads.size(); ++i) {
217            PyObject* bytes = PyBytes_FromStringAndSize(
218                b.kv_payloads[i].data(), static_cast<Py_ssize_t>(b.kv_payloads[i].size()));
219            PyList_SetItem(kv_list, static_cast<Py_ssize_t>(i), bytes);  // 窃取引用
220        }
221        PyObject* row_list = PyList_New(static_cast<Py_ssize_t>(b.row_kv_idx.size()));
222        for (size_t i = 0; i < b.row_kv_idx.size(); ++i)
223            PyList_SetItem(row_list, static_cast<Py_ssize_t>(i),
224                           PyLong_FromLong(b.row_kv_idx[i]));
225
226        PyObject* ns_bytes =
227            PyBytes_FromStringAndSize(b.ns_blob.data(), static_cast<Py_ssize_t>(b.ns_blob.size()));
228        PyObject* args = Py_BuildValue("(OOOiii)", kv_list, row_list, ns_bytes,
229                                        static_cast<int>(b.n_rows), static_cast<int>(b.ns),
230                                        static_cast<int>(b.d_model));
231        Py_DECREF(kv_list);
232        Py_DECREF(row_list);
233        Py_DECREF(ns_bytes);
234
235        PyObject* res = PyObject_CallObject(fn, args);
236        Py_DECREF(args);
237        if (!res) {
238            PyErr_Print();
239            throw std::runtime_error("bridge: score_batch 调用失败");
240        }
241        char* buf = nullptr;
242        Py_ssize_t len = 0;
243        if (PyBytes_AsStringAndSize(res, &buf, &len) != 0) {
244            Py_DECREF(res);
245            throw std::runtime_error("bridge: score_batch 返回类型非 bytes");
246        }
247        out.assign(buf, static_cast<size_t>(len));
248        Py_DECREF(res);
249    } while (false);
250    PyGILState_Release(gil);
251    return out;
252}
