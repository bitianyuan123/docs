23const char* status_text(int code) {
24    switch (code) {
25        case 200: return "OK";
26        case 400: return "Bad Request";
27        case 404: return "Not Found";
28        case 405: return "Method Not Allowed";
29        case 413: return "Payload Too Large";
30        case 500: return "Internal Server Error";
31        default: return "Unknown";
32    }
33}

35// 简单fd队列：accept 线程投递，worker 线程消费
36class FdQueue {
37public:
38    void push(int fd) {
39        {
40            std::lock_guard<std::mutex> lk(mu_);
41            q_.push(fd);
42        }
43        cv_.notify_one();
44    }
45
46    int pop() {  // 阻塞；stop 后返回 -1
47        std::unique_lock<std::mutex> lk(mu_);
48        cv_.wait(lk, [&] { return !q_.empty() || stopped_; });
49        if (q_.empty()) return -1;
50        int fd = q_.front();
51        q_.pop();
52        return fd;
53    }
54
55    void stop() {
56        {
57            std::lock_guard<std::mutex> lk(mu_);
58            stopped_ = true;
59        }
60        cv_.notify_all();
61    }
62
63private:
64    std::mutex mu_;
65    std::condition_variable cv_;
66    std::queue<int> q_;
67    bool stopped_ = false;
68};

75std::map<std::string, std::string> HttpRequest::query_params() const {
76    std::map<std::string, std::string> out;
77    size_t pos = 0;
78    while (pos < query.size()) {
79        size_t amp = query.find('&', pos);
80        if (amp == std::string::npos) amp = query.size();
81        if (amp > pos) {
82            std::string kv = query.substr(pos, amp - pos);
83            size_t eq = kv.find('=');
84            if (eq == std::string::npos)
85                out[kv] = "";
86            else
87                out[kv.substr(0, eq)] = kv.substr(eq + 1);
88        }
89        pos = amp + 1;
90    }
91    return out;
92}

127void HttpServer::run() {
128    listen_fd_ = ::socket(AF_INET, SOCK_STREAM, 0);
129    if (listen_fd_ < 0) throw std::runtime_error("socket() 失败");
130
131    int opt = 1;
132    ::setsockopt(listen_fd_, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
133
134    sockaddr_in addr{};
135    addr.sin_family = AF_INET;
136    addr.sin_port = htons(static_cast<uint16_t>(port_));
137    if (::inet_pton(AF_INET, host_.c_str(), &addr.sin_addr) != 1)
138        throw std::runtime_error("非法监听地址: " + host_);
139    if (::bind(listen_fd_, reinterpret_cast<sockaddr*>(&addr), sizeof(addr)) < 0)
140        throw std::runtime_error("bind 失败: " + host_ + ":" + std::to_string(port_));
141    if (::listen(listen_fd_, 512) < 0) throw std::runtime_error("listen 失败");
142
143    // 本进程单 server 实例：accept 线程 → worker 池（static 存储期，无需捕获）
144    static FdQueue queue;
145    running_ = true;
146    for (int i = 0; i < num_threads_; ++i) {
147        threads_.emplace_back([this] {
148            while (running_) {
149                int fd = queue.pop();
150                if (fd < 0) break;  // stop
151                handle_connection(fd);
152            }
153        });
154    }
155
156    // accept 循环（run 调用线程执行；返回即停止）
157    while (running_) {
158        sockaddr_in peer{};
159        socklen_t plen = sizeof(peer);
160        int fd = ::accept(listen_fd_, reinterpret_cast<sockaddr*>(&peer), &plen);
161        if (fd < 0) {
162            if (errno == EINTR) continue;
163            break;
164        }
165        int one = 1;
166        ::setsockopt(fd, IPPROTO_TCP, TCP_NODELAY, &one, sizeof(one));
167        queue.push(fd);
168    }
169    queue.stop();
170}

172void HttpServer::stop() {
173    if (!running_.exchange(false)) return;
174    if (listen_fd_ >= 0) {
175        ::shutdown(listen_fd_, SHUT_RDWR);
176        ::close(listen_fd_);
177        listen_fd_ = -1;
178    }
179    for (auto& t : threads_) {
180        if (t.joinable()) t.join();
181    }
182    threads_.clear();
183    // drain 在途异步请求（完成线程会写回并关闭 fd；上限 10s 防御性退出）
184    for (int i = 0; i < 10000 && pending_async_.load() > 0; ++i)
185        std::this_thread::sleep_for(std::chrono::milliseconds(1));
186}

188// 读到 "\r\n\r\n" 为止（请求头），继续读 Content-Length 字节 body
189bool HttpServer::read_request(int fd, HttpRequest& req, std::string& raw_head) {
190    std::string buf;
191    buf.reserve(2048);
192    char chunk[4096];
193    size_t head_end = std::string::npos;
194    while (buf.size() < kMaxHead) {
195        ssize_t n = ::recv(fd, chunk, sizeof(chunk), 0);
196        if (n <= 0) return false;
197        buf.append(chunk, static_cast<size_t>(n));
198        head_end = buf.find("\r\n\r\n");
199        if (head_end != std::string::npos) break;
200    }
201    if (head_end == std::string::npos) return false;
202    raw_head = buf.substr(0, head_end);
203    std::string rest = buf.substr(head_end + 4);
204
205    // request line: METHOD /path?query HTTP/1.1
206    size_t line_end = raw_head.find("\r\n");
207    std::string reqline = raw_head.substr(0, line_end);
208    size_t sp1 = reqline.find(' ');
209    size_t sp2 = reqline.rfind(' ');
210    if (sp1 == std::string::npos || sp2 == sp1) return false;
211    req.method = reqline.substr(0, sp1);
212    std::string target = reqline.substr(sp1 + 1, sp2 - sp1 - 1);
213    size_t q = target.find('?');
214    if (q == std::string::npos) {
215        req.path = target;
216    } else {
217        req.path = target.substr(0, q);
218        req.query = target.substr(q + 1);
219    }
220
221    // headers：Content-Length
222    size_t content_len = 0;
223    size_t pos = line_end + 2;
224    while (pos < raw_head.size()) {
225        size_t he = raw_head.find("\r\n", pos);
226        if (he == std::string::npos) he = raw_head.size();
227        std::string line = raw_head.substr(pos, he - pos);
228        pos = he + 2;
229        size_t colon = line.find(':');
230        if (colon == std::string::npos) continue;
231        std::string key = line.substr(0, colon);
232        std::string val = line.substr(colon + 1);
233        while (!val.empty() && (val.front() == ' ' || val.front() == '\t')) val.erase(0, 1);
234        for (auto& c : key) c = static_cast<char>(::tolower(static_cast<unsigned char>(c)));
235        if (key == "content-length") content_len = static_cast<size_t>(::atoll(val.c_str()));
236    }
237    if (content_len > kMaxBody) return false;
238
239    // body：rest 已读部分 + 剩余
240    req.body.reserve(content_len);
241    req.body = rest.substr(0, std::min(rest.size(), content_len));
242    while (req.body.size() < content_len) {
243        ssize_t n = ::recv(fd, chunk, sizeof(chunk), 0);
244        if (n <= 0) return false;
245        size_t take = std::min(static_cast<size_t>(n), content_len - req.body.size());
246        req.body.append(chunk, take);
247    }
248    return true;
249}

251void HttpServer::write_response(int fd, const HttpResponse& resp) {
252    std::string head = "HTTP/1.1 " + std::to_string(resp.status) + " " + status_text(resp.status) +
253                       "\r\nContent-Type: " + resp.content_type +
254                       "\r\nContent-Length: " + std::to_string(resp.body.size()) +
255                       "\r\nConnection: close\r\n\r\n";
256    ::send(fd, head.data(), head.size(), MSG_NOSIGNAL);
257    if (!resp.body.empty()) ::send(fd, resp.body.data(), resp.body.size(), MSG_NOSIGNAL);
258    ::shutdown(fd, SHUT_RDWR);
259    ::close(fd);
260}

262void HttpServer::handle_connection(int fd) {
263    HttpRequest req;
264    std::string raw_head;
265    if (!read_request(fd, req, raw_head)) {
266        ::close(fd);
267        return;
268    }
269
270    const std::string key = req.method + " " + req.path;
271
272    // 异步路由：接入线程提交后立即返回（不占 worker 等待计算，§7.4.3 接入边界）；
273    // 响应由流水线完成线程经 done 写回。done 单次生效（原子标记防重复写）。
274    auto ait = async_routes_.find(key);
275    if (ait != async_routes_.end()) {
276        pending_async_.fetch_add(1);
277        auto done = [this, fd, called = std::make_shared<std::atomic<bool>>(false)](
278                        HttpResponse resp) {
279            if (called->exchange(true)) return;  // 幂等：只写一次
280            write_response(fd, resp);
281            pending_async_.fetch_sub(1);
282        };
283        try {
284            ait->second(req, done);
285        } catch (const std::exception& e) {
286            done(HttpResponse::json(500, std::string("{\"error\":\"") + e.what() + "\"}"));
287        } catch (...) {
288            done(HttpResponse::json(500, "{\"error\":\"unknown\"}"));
289        }
290        return;  // worker 立即处理下一连接
291    }
292
293    HttpResponse resp;
294    auto it = routes_.find(key);
295    if (it != routes_.end()) {
296        try {
297            resp = it->second(req);
298        } catch (const std::exception& e) {
299            resp = HttpResponse::json(500, std::string("{\"error\":\"") + e.what() + "\"}");
300        } catch (...) {
301            resp = HttpResponse::json(500, "{\"error\":\"unknown\"}");
302        }
303    } else {
304        // 存在路径但方法不符 → 405（同步与异步路由都算路径存在）
305        bool path_exists = false;
306        for (const auto& [k, h] : routes_) {
307            if (k.substr(k.find(' ') + 1) == req.path) path_exists = true;
308        }
309        for (const auto& [k, h] : async_routes_) {
310            if (k.substr(k.find(' ') + 1) == req.path) path_exists = true;
311        }
312        resp = path_exists ? HttpResponse::json(405, "{\"error\":\"method not allowed\"}")
313                           : HttpResponse::json(404, "{\"error\":\"not found\"}");
314    }
315
316    write_response(fd, resp);
317}
