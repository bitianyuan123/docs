7Executor::Executor(std::string name, int num_threads, size_t queue_capacity, bool io)
8    : name_(std::move(name)),
9      cap_(queue_capacity == 0 ? 1 : queue_capacity),
10      io_(io) {
11    const size_t n = num_threads > 0 ? static_cast<size_t>(num_threads) : 1;
12    if (io_) {
13        // IO 池（EmbedLookup / KV I/O）：每线程独立 event base，承接阻塞式网络/外部调用。
14        io_exec_ = std::make_shared<folly::IOThreadPoolExecutor>(
15            n, std::make_shared<folly::NamedThreadFactory>(name_));
16    } else {
17        // CPU 池（Frontend Encode / Compute）：folly 绑核线程 + 无界队列，
18        // 有界性由 add() 侧的 pending 快照软判定兜底（避免阻塞式满队列死锁）。
19        cpu_ = std::make_shared<folly::CPUThreadPoolExecutor>(
20            n, std::make_shared<folly::NamedThreadFactory>(name_));
21    }
22}

26void Executor::add(std::function<void()> fn) {
27    size_t pending = 0;
28    {
29        std::lock_guard<std::mutex> lk(mu_);
30        if (stopped_) throw std::logic_error("executor '" + name_ + "' stopped");
31        // 快照式软背压：判定与入队非原子；瞬时超限由接入层降级吸收即可。
32        pending = cpu_ ? cpu_->getPendingTaskCount() : io_exec_->getPendingTaskCount();
33        if (pending >= cap_) throw ExecutorOverloaded(name_);
34    }
35    if (cpu_) {
36        cpu_->add(std::move(fn));
37    } else {
38        io_exec_->add(std::move(fn));
39    }
40}

42void Executor::stop() {
43    {
44        std::lock_guard<std::mutex> lk(mu_);
45        if (stopped_) return;
46        stopped_ = true;
47    }
48    // join：folly 在退出时排空已入队任务并回收线程（与自研池的 drain+join 语义对齐）。
49    if (cpu_) cpu_->join();
50    if (io_exec_) io_exec_->join();
51}

53std::shared_ptr<Executor> ExecutorSet::make(const std::string& name, int num_threads,
54                                            size_t queue_capacity, bool io) {
55    auto pool = std::make_shared<Executor>(name, num_threads, queue_capacity, io);
56    std::lock_guard<std::mutex> lk(mu_);
57    pools_.push_back(pool);
58    return pool;
59}

61void ExecutorSet::stop_all() {
62    std::lock_guard<std::mutex> lk(mu_);
63    for (auto& p : pools_) p->stop();
64    pools_.clear();
