24std::vector<uint8_t> sha256(const void* data, size_t len) {
25    uint32_t h[8] = {0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a,
26                     0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19};
27    const auto* bytes = static_cast<const uint8_t*>(data);
28
29    // padding：0x80 + 0..63 个 0 + 8 字节大端 bit 长度
30    std::vector<uint8_t> buf(bytes, bytes + len);
31    size_t total = len + 1 + 8;
32    size_t padded = ((total + 63) / 64) * 64;
33    buf.resize(padded, 0);
34    buf[len] = 0x80;
35    uint64_t bits = static_cast<uint64_t>(len) * 8;
36    for (int i = 0; i < 8; ++i) buf[padded - 1 - i] = static_cast<uint8_t>((bits >> (8 * i)) & 0xFF);
37
38    uint32_t w[64];
39    for (size_t off = 0; off < padded; off += 64) {
40        for (int i = 0; i < 16; ++i) {
41            w[i] = (static_cast<uint32_t>(buf[off + 4 * i]) << 24) |
42                   (static_cast<uint32_t>(buf[off + 4 * i + 1]) << 16) |
43                   (static_cast<uint32_t>(buf[off + 4 * i + 2]) << 8) |
44                   static_cast<uint32_t>(buf[off + 4 * i + 3]);
45        }
46        for (int i = 16; i < 64; ++i) {
47            uint32_t s0 = rotr(w[i - 15], 7) ^ rotr(w[i - 15], 18) ^ (w[i - 15] >> 3);
48            uint32_t s1 = rotr(w[i - 2], 17) ^ rotr(w[i - 2], 19) ^ (w[i - 2] >> 10);
49            w[i] = w[i - 16] + s0 + w[i - 7] + s1;
50        }
51        uint32_t a = h[0], b = h[1], c = h[2], d = h[3];
52        uint32_t e = h[4], f = h[5], g = h[6], hh = h[7];
53        for (int i = 0; i < 64; ++i) {
54            uint32_t s1 = rotr(e, 6) ^ rotr(e, 11) ^ rotr(e, 25);
55            uint32_t ch = (e & f) ^ (~e & g);
56            uint32_t t1 = hh + s1 + ch + kK[i] + w[i];
57            uint32_t s0 = rotr(a, 2) ^ rotr(a, 13) ^ rotr(a, 22);
58            uint32_t maj = (a & b) ^ (a & c) ^ (b & c);
59            uint32_t t2 = s0 + maj;
60            hh = g; g = f; f = e; e = d + t1;
61            d = c; c = b; b = a; a = t1 + t2;
62        }
63        h[0] += a; h[1] += b; h[2] += c; h[3] += d;
64        h[4] += e; h[5] += f; h[6] += g; h[7] += hh;
65    }
66
67    std::vector<uint8_t> out(32);
68    for (int i = 0; i < 8; ++i) {
69        out[4 * i] = static_cast<uint8_t>(h[i] >> 24);
70        out[4 * i + 1] = static_cast<uint8_t>(h[i] >> 16);
71        out[4 * i + 2] = static_cast<uint8_t>(h[i] >> 8);
72        out[4 * i + 3] = static_cast<uint8_t>(h[i]);
73    }
74    return out;
75}

81std::string sha256_hex(const void* data, size_t len) {
82    auto digest = sha256(data, len);

95uint64_t stable_hash64(const std::string& key) {
96    auto digest = sha256(key);
97    uint64_t v = 0;
98    for (int i = 0; i < 8; ++i) v = (v << 8) | digest[static_cast<size_t>(i)];
99    return v;
