17struct Tensor {
18    std::vector<int64_t> shape;
19    std::vector<float> data;
20
21    Tensor() = default;
22    explicit Tensor(std::vector<int64_t> shp) : shape(std::move(shp)) {
23        int64_t n = 1;
24        for (auto s : shape) n *= s;
25        data.assign(static_cast<size_t>(n), 0.0f);
26    }
27    static Tensor zeros(std::vector<int64_t> shp) { return Tensor(std::move(shp)); }
28
29    int64_t numel() const {
30        int64_t n = 1;
31        for (auto s : shape) n *= s;
32        return n;
33    }
34    float& at(int64_t i) { return data[static_cast<size_t>(i)]; }
35    const float& at(int64_t i) const { return data[static_cast<size_t>(i)]; }
36};

38// 线性代数原语（无 bias；PyTorch nn.Linear 语义 y = x @ W^T）
39// x: [n, in]，W: [out, in]，y: [n, out]
40void matmul_nt(const float* x, const float* w, float* y, int64_t n, int64_t in, int64_t out);
41
42// RMSNorm：out = x * rsqrt(mean(x^2) + eps) * g（逐行）
43void rms_norm(const float* x, const float* g, float* out, int64_t rows, int64_t dim, float eps);

45// 精确 GELU：0.5x(1 + erf(x / sqrt(2)))，对齐 torch.nn.GELU 默认
46inline float gelu(float v) {
47    return 0.5f * v * (1.0f + std::erf(v * 0.70710678118654752440f));
48}
