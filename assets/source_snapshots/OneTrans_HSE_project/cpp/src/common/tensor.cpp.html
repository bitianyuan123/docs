11void matmul_nt(const float* x, const float* w, float* y, int64_t n, int64_t in, int64_t out) {
12    for (int64_t r = 0; r < n; ++r) {
13        const float* xr = x + r * in;
14        float* yr = y + r * out;
15        for (int64_t o = 0; o < out; ++o) {
16            const float* wo = w + o * in;
17            float acc = 0.0f;
18            for (int64_t i = 0; i < in; ++i) acc += xr[i] * wo[i];
19            yr[o] = acc;
20        }
21    }
22}

24void rms_norm(const float* x, const float* g, float* out, int64_t rows, int64_t dim, float eps) {
25    for (int64_t r = 0; r < rows; ++r) {
26        const float* xr = x + r * dim;
27        float* orow = out + r * dim;
28        float ms = 0.0f;
29        for (int64_t i = 0; i < dim; ++i) ms += xr[i] * xr[i];
30        ms /= static_cast<float>(dim);
31        float inv = 1.0f / std::sqrt(ms + eps);
32        for (int64_t i = 0; i < dim; ++i) orow[i] = xr[i] * inv * g[i];
33    }
34}
