9struct LayerQkv {
10    // NS 段 Q/K/V：[B, Ns, H*d]（bshd 去 batch 维）
11    std::vector<float> q, k, v;
12};
13
14// NS 逐 token 投影：h [B, Ns, D] → q/k/v [B, Ns, H*d]
15// 语义：token i 的所有候选行经 W_ns[i] 投影（Python _project_ns）。
16LayerQkv project_ns(const AttnWeights& attn, const float* h, int64_t B, int64_t Ns, int64_t D,
17                    int H, int64_t d) {
18    LayerQkv out;
19    out.q.resize(static_cast<size_t>(B * Ns * D));
20    out.k.resize(static_cast<size_t>(B * Ns * D));
21    out.v.resize(static_cast<size_t>(B * Ns * D));
22    std::vector<float> qkv3(static_cast<size_t>(3 * D));
23    for (int64_t b = 0; b < B; ++b) {
24        for (int64_t i = 0; i < Ns; ++i) {
25            // token i 的候选 b 输入行
26            const float* x = h + (b * Ns + i) * D;
27            linear_rows(x, attn.w_ns[i].data.data(), qkv3.data(), 1, D, 3 * D);
28            float* q = out.q.data() + (b * Ns + i) * D;
29            float* k = out.k.data() + (b * Ns + i) * D;
30            float* v = out.v.data() + (b * Ns + i) * D;
31            std::copy_n(qkv3.begin(), D, q);
32            std::copy_n(qkv3.begin() + D, D, k);
33            std::copy_n(qkv3.begin() + 2 * D, D, v);
34        }
35    }
36    return out;
37}

39// NS 段逐 token FFN：h [B, Ns, D] → out [B, Ns, D]
40void apply_ns_ffn(const std::vector<FfnWeights>& ffn_ns, const float* h, float* out, int64_t B,
41                  int64_t Ns, int64_t D) {
42    for (int64_t b = 0; b < B; ++b)
43        for (int64_t i = 0; i < Ns; ++i)
44            ffn_forward(ffn_ns[i], h + (b * Ns + i) * D, out + (b * Ns + i) * D, 1, D);
45}

53UserKV TwoStageRunner::encode_s(const Tensor& s_emb, const Tensor& s_mask) const {
54    if (s_emb.shape[0] != 1) throw std::runtime_error("encode_s 只支持 B=1");
55    const int64_t D = m_.cfg.d_model;
56    const int H = m_.cfg.num_heads, d = D / H;
57    const float eps = static_cast<float>(m_.cfg.rms_eps);
58
59    int64_t S = s_emb.shape[1];
60    std::vector<float> s(s_emb.data);
61    std::vector<uint8_t> smask(S);
62    int64_t s_len = 0;
63    for (int64_t j = 0; j < S; ++j) {
64        smask[j] = s_mask.at(j) != 0.0f ? 1 : 0;
65        s_len += smask[j];
66    }
67
68    UserKV kv;
69    kv.s_len = s_len;
70
71    for (int l = 0; l < m_.cfg.num_blocks; ++l) {
72        const BlockWeights& bw = m_.blocks[l];
73        int64_t valid = 0;
74        for (auto v : smask) valid += v;
75        kv.per_layer_len.push_back(valid);
76
77        std::vector<float> h(static_cast<size_t>(S * D));
78        rms_norm_rows(s.data(), bw.norm_g.data.data(), h.data(), S, D, eps);
79
80        // W_s 投影 → q/k/v [S, H, d]
81        std::vector<float> qkv(static_cast<size_t>(S * 3 * D));
82        linear_rows(h.data(), bw.attn.w_s.data.data(), qkv.data(), S, D, 3 * D);
83        Tensor k_t({1, S, H, d}), v_t({1, S, H, d});
84        std::vector<float> q(static_cast<size_t>(S * D));
85        for (int64_t r = 0; r < S; ++r) {
86            std::copy_n(qkv.begin() + r * 3 * D, D, q.begin() + r * D);
87            std::copy_n(qkv.begin() + r * 3 * D + D, D, k_t.data.begin() + r * D);
88            std::copy_n(qkv.begin() + r * 3 * D + 2 * D, D, v_t.data.begin() + r * D);
89        }
90        kv.k.push_back(std::move(k_t));
91        kv.v.push_back(std::move(v_t));
92
93        // S 自注意力：padding 列掩 + causal
94        std::vector<uint8_t> allow(static_cast<size_t>(S * S), 1);
95        for (int64_t i = 0; i < S; ++i)
96            for (int64_t j = 0; j < S; ++j)
97                if (j > i || !smask[j]) allow[static_cast<size_t>(i * S + j)] = 0;
98
99        std::vector<float> attn(static_cast<size_t>(S * D));
100        sdpa_masked(q.data(), kv.k[l].data.data(), kv.v[l].data.data(), attn.data(), S, S, H, d,
101                    allow.data());
102        std::vector<float> proj(static_cast<size_t>(S * D));
103        linear_rows(attn.data(), bw.attn.final_proj.data.data(), proj.data(), S, D, D);
104
105        std::vector<float> z(static_cast<size_t>(S * D));
106        for (size_t i = 0; i < z.size(); ++i) z[i] = proj[i] + s[i];
107
108        std::vector<float> h2(static_cast<size_t>(S * D));
109        rms_norm_rows(z.data(), bw.norm_g.data.data(), h2.data(), S, D, eps);
110        std::vector<float> f(static_cast<size_t>(S * D));
111        ffn_forward(bw.ffn_s, h2.data(), f.data(), S, D);
112        for (size_t i = 0; i < z.size(); ++i) z[i] += f[i];
113
114        // pyramid 尾部裁剪（token 与 mask 同步）
115        int64_t keep = bw.out_seq_num;
116        s.assign(z.begin() + static_cast<long>((S - keep) * D), z.end());
117        smask.assign(smask.begin() + (S - keep), smask.end());
118        S = keep;
119    }
120    return kv;
121}

129Tensor score_ns_impl(const OneTransModel& m, const std::vector<const UserKV*>& kvs,
130                     const Tensor& ns_emb) {
131    const int64_t B = ns_emb.shape[0];
132    if (static_cast<int64_t>(kvs.size()) != B)
133        throw std::runtime_error("kvs 数量须等于 ns_emb batch");
134    const int64_t Ns = ns_emb.shape[1];
135    const int64_t D = ns_emb.shape[2];
136    const int H = m.cfg.num_heads, d = D / H;
137    const float eps = static_cast<float>(m.cfg.rms_eps);
138
139    std::vector<float> ns(ns_emb.data);
140    const int64_t T = m.head_w.shape[0];
141
142    for (int l = 0; l < m.cfg.num_blocks; ++l) {
143        const BlockWeights& bw = m.blocks[l];
144        const int64_t S_l = kvs[0]->k[l].shape[1];  // pyramid 宽度恒定
145
146        std::vector<float> h(static_cast<size_t>(B * Ns * D));
147        rms_norm_rows(ns.data(), bw.norm_g.data.data(), h.data(), B * Ns, D, eps);
148        LayerQkv nsq = project_ns(bw.attn, h.data(), B, Ns, D, H, d);
149
150        // K/V 拼接：每候选 b：[k_s^l ∥ k_ns(b)] → [B, S_l+Ns, H, d]
151        std::vector<float> kk(static_cast<size_t>(B * (S_l + Ns) * D));
152        std::vector<float> vv(static_cast<size_t>(B * (S_l + Ns) * D));
153        for (int64_t b = 0; b < B; ++b) {
154            const UserKV* kv = kvs[b];
155            std::copy(kv->k[l].data.begin(), kv->k[l].data.end(),
156                      kk.begin() + b * (S_l + Ns) * D);
157            std::copy(nsq.k.begin() + b * Ns * D, nsq.k.begin() + (b + 1) * Ns * D,
158                      kk.begin() + b * (S_l + Ns) * D + S_l * D);
159            std::copy(kv->v[l].data.begin(), kv->v[l].data.end(),
160                      vv.begin() + b * (S_l + Ns) * D);
161            std::copy(nsq.v.begin() + b * Ns * D, nsq.v.begin() + (b + 1) * Ns * D,
162                      vv.begin() + b * (S_l + Ns) * D + S_l * D);
163        }
164
165        // 逐候选左 padding 有效掩码 + NS causal
166        std::vector<uint8_t> allow(static_cast<size_t>(B * Ns * (S_l + Ns)), 1);
167        for (int64_t b = 0; b < B; ++b) {
168            int64_t valid = kvs[b]->per_layer_len[l];
169            for (int64_t i = 0; i < Ns; ++i) {
170                uint8_t* row = allow.data() + (b * Ns + i) * (S_l + Ns);
171                for (int64_t j = 0; j < S_l; ++j)
172                    if (j < S_l - valid) row[j] = 0;  // 左 padding 列
173                for (int64_t j = S_l; j < S_l + Ns; ++j)
174                    if (j - S_l > i) row[j] = 0;  // NS causal
175            }
176        }
177
178        // 分候选 SDPA（Q 布局 [b, Ns, H, d] → 每候选一个 [Ns, H, d] 块）
179        std::vector<float> attn(static_cast<size_t>(B * Ns * D));
180        for (int64_t b = 0; b < B; ++b) {
181            sdpa_masked(nsq.q.data() + b * Ns * D, kk.data() + b * (S_l + Ns) * D,
182                        vv.data() + b * (S_l + Ns) * D, attn.data() + b * Ns * D, Ns, S_l + Ns,
183                        H, d, allow.data() + b * Ns * (S_l + Ns));
184        }
185        std::vector<float> proj(static_cast<size_t>(B * Ns * D));
186        linear_rows(attn.data(), bw.attn.final_proj.data.data(), proj.data(), B * Ns, D, D);
187
188        std::vector<float> z(static_cast<size_t>(B * Ns * D));
189        for (size_t i = 0; i < z.size(); ++i) z[i] = proj[i] + ns[i];
190
191        std::vector<float> h2(static_cast<size_t>(B * Ns * D));
192        rms_norm_rows(z.data(), bw.norm_g.data.data(), h2.data(), B * Ns, D, eps);
193        std::vector<float> f(static_cast<size_t>(B * Ns * D));
194        apply_ns_ffn(bw.ffn_ns, h2.data(), f.data(), B, Ns, D);
195        for (size_t i = 0; i < z.size(); ++i) z[i] += f[i];
196        ns = std::move(z);
197    }
198
199    // head：末 Ns token 均值 → linear（含 bias）
200    Tensor logits({B, T});
201    std::vector<float> pooled(static_cast<size_t>(D));
202    for (int64_t b = 0; b < B; ++b) {
203        std::fill(pooled.begin(), pooled.end(), 0.0f);
204        for (int64_t i = 0; i < Ns; ++i)
205            for (int64_t t = 0; t < D; ++t) pooled[t] += ns[(b * Ns + i) * D + t];
206        for (int64_t t = 0; t < D; ++t) pooled[t] /= static_cast<float>(Ns);
207        linear_rows(pooled.data(), m.head_w.data.data(), logits.data.data() + b * T, 1, D, T);
208        for (int64_t t = 0; t < T; ++t) logits.data.data()[b * T + t] += m.head_b.at(t);
209    }
210    return logits;
211}

215Tensor TwoStageRunner::score_ns(const UserKV& kv, const Tensor& ns_emb) const {
216    const int64_t B = ns_emb.shape[0];
217    std::vector<const UserKV*> kvs(static_cast<size_t>(B), &kv);
218    return score_ns_impl(m_, kvs, ns_emb);
219}

221Tensor TwoStageRunner::score_ns_batch(const std::vector<const UserKV*>& kvs,
