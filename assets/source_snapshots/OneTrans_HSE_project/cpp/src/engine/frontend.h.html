1// serving 前处理（YambdaEmbedder + OneTransTokenizer 的 C++ 等价实现）。
2//
3// 与 cpp/tools/gen_golden.py 的 make_s_input / make_ns_input 严格同语义：
4// - S 侧：截尾保留最新 max_seq_len 条 → 左 padding（有效靠尾）→ item 查表 →
5//   STokenizer（mlp×mask + type_emb + mask + pos_emb + RMSNorm）。
6// - NS 侧：dense=[用户级 ∥ 候选级] → piecewise 分箱；uid/item 查表；
7//   artist/album mean-bag → NSGroupWiseTokenizer（组顺序 dense/uid/item/artist/album）
8//   → RMSNorm。
9//
10// 稀疏查表经 LookupFn 注入（本地直查 / PS HTTP 客户端均可），保持数据面解耦。

22struct CandidateInput {
23    int64_t item_id = 0;
24    std::vector<int64_t> artist_ids;
25    std::vector<int64_t> album_ids;
26    std::vector<float> dense;  // 候选级 dense（n_cand_dense 列）
27};
28
29struct ScoreInput {
30    std::string user_id;
31    int64_t uid_sparse = 0;
32    std::vector<float> user_dense;       // 用户级 dense（n_user_dense 列）
33    std::vector<CandidateInput> candidates;
34};

36struct IngestInput {
37    std::string user_id;
38    std::vector<int64_t> item_ids;     // 时间升序（最旧在前、最新在尾）
39    std::vector<int64_t> timestamps;
40};

42// 查表函数：表名（item/user/artist/album）+ id 列表 → fp32 平铺 [N * d_model]
43using LookupFn = std::function<std::vector<float>(const std::string&, const std::vector<int64_t>&)>;
44
45// NS 侧查表预取结果（EmbedLookup Pool 产出 → Frontend Encode Pool 消费的契约，
46// §7.4.3 阶段边界数据结构；bag 聚合属查表侧，在 lookup 阶段完成）
47struct NsEmbeddings {
48    std::vector<float> uid;     // [D]（单行，encode 时广播 M 份）
49    std::vector<float> items;   // [M * D]
50    std::vector<float> artists;  // [M * D]（mean-bag 已聚合）
51    std::vector<float> albums;   // [M * D]（mean-bag 已聚合）
52};
53
54class EmbeddingFrontend {
55public:
56    static EmbeddingFrontend load(const ArtifactStore& store);
57
58    // → (s_emb [1, max_seq_len, D], s_mask [1, max_seq_len] 0/1)
59    std::pair<Tensor, Tensor> encode_s(const std::vector<int64_t>& item_ids,
60                                       const LookupFn& lookup) const;
61    // → ns_emb [M, Ns, D]（lookup + encode 一步式：nearline/兼容路径）
62    Tensor encode_ns(const ScoreInput& in, const LookupFn& lookup) const;
63
64    // 两步式（SEDA，§7.4.3）：
65    // 阶段 1（EmbedLookup Pool）：查表 + mean-bag 聚合
66    NsEmbeddings lookup_ns(const ScoreInput& in, const LookupFn& lookup) const;
67    // 阶段 2（Frontend Encode Pool）：piecewise + 组 MLP + RMSNorm → ns_emb [M, Ns, D]
68    Tensor encode_ns_with(const ScoreInput& in, const NsEmbeddings& emb) const;
69
70    const ArtifactConfig& config() const { return cfg_; }
71
72private:
73    ArtifactConfig cfg_;
74    // tokenizer
75    Tensor s_mlp_w1, s_mlp_w2;   // [D, seq_in_dim] / [D, D]
76    Tensor s_type_emb;            // [1, D]
77    Tensor pos_emb;               // [max_seq_len, D]
78    Tensor tok_rms_g;             // [D]
79    std::vector<Tensor> ns_w1, ns_w2;  // 每组 [D, in] / [D, D]
80    // piecewise
81    Tensor pw_w, pw_b;            // [n_features, max_n_bins]
82};
