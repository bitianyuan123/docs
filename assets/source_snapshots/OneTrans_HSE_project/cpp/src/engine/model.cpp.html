22void ffn_forward(const FfnWeights& w, const float* x, float* y, int64_t n, int64_t dim) {
23    int64_t hidden = w.w1.shape[0];
24    std::vector<float> h(static_cast<size_t>(n * hidden));
25    linear_rows(x, w.w1.data.data(), h.data(), n, dim, hidden);
26    for (auto& v : h) v = gelu(v);
27    linear_rows(h.data(), w.w2.data.data(), y, n, hidden, dim);

30void sdpa_masked(const float* q, const float* k, const float* v, float* out,
31                 int64_t q_rows, int64_t k_rows, int H, int64_t d, const uint8_t* allow) {
32    const float scale = 1.0f / std::sqrt(static_cast<float>(d));
33    const int64_t qd = H * d, kd = H * d;
34    std::vector<float> scores(static_cast<size_t>(k_rows));
35
36    for (int64_t i = 0; i < q_rows; ++i) {
37        const uint8_t* allow_row = allow + i * k_rows;
38        for (int h = 0; h < H; ++h) {
39            const float* qi = q + i * qd + h * d;
40            float m = -std::numeric_limits<float>::infinity();
41            bool any = false;
42            for (int64_t j = 0; j < k_rows; ++j) {
43                if (!allow_row[j]) {
44                    scores[j] = -std::numeric_limits<float>::infinity();
45                    continue;
46                }
47                const float* kj = k + j * kd + h * d;
48                float acc = 0.0f;
49                for (int64_t t = 0; t < d; ++t) acc += qi[t] * kj[t];
50                float s = acc * scale;
51                scores[j] = s;
52                if (s > m) m = s;
53                any = true;
54            }
55            float* oi = out + i * qd + h * d;
56            if (!any) {
57                // 全掩行：输出 0（对齐 PyTorch SDPA，避免 0/0 NaN）
58                std::fill(oi, oi + d, 0.0f);
59                continue;
60            }
61            float denom = 0.0f;
62            for (int64_t j = 0; j < k_rows; ++j) {
63                float e = (scores[j] == -std::numeric_limits<float>::infinity())
64                             ? 0.0f
65                             : std::exp(scores[j] - m);
66                scores[j] = e;
67                denom += e;
68            }
69            float inv = 1.0f / denom;
70            std::fill(oi, oi + d, 0.0f);
71            for (int64_t j = 0; j < k_rows; ++j) {
72                float p = scores[j] * inv;
73                if (p == 0.0f) continue;
74                const float* vj = v + j * kd + h * d;
75                for (int64_t t = 0; t < d; ++t) oi[t] += p * vj[t];
76            }
77        }
78    }
79}

85OneTransModel OneTransModel::load(const ArtifactStore& store) {
86    OneTransModel m;
87    m.cfg = store.config();
88    const int L = m.cfg.num_blocks;
89    const int D = m.cfg.d_model;
90    const int Ns = m.cfg.ns_tokens_num;
91
92    m.blocks.resize(L);
93    for (int l = 0; l < L; ++l) {
94        BlockWeights& bw = m.blocks[l];
95        bw.out_seq_num = m.cfg.dims[l + 1];
96        bw.norm_g = store.get("backbone.blocks." + std::to_string(l) + ".norm.weight");
97        bw.attn.w_s = store.get("backbone.blocks." + std::to_string(l) + ".mixed_attn.W_s.weight");
98        bw.attn.w_ns.resize(Ns);
99        bw.ffn_ns.resize(Ns);
100        for (int i = 0; i < Ns; ++i) {
101            bw.attn.w_ns[i] = store.get("backbone.blocks." + std::to_string(l) +
102                                        ".mixed_attn.W_ns_list." + std::to_string(i) + ".weight");
103            bw.ffn_ns[i].w1 = store.get("backbone.blocks." + std::to_string(l) +
104                                        ".mixed_ffn.networks_ns_list." + std::to_string(i) + ".0.weight");
105            bw.ffn_ns[i].w2 = store.get("backbone.blocks." + std::to_string(l) +
106                                        ".mixed_ffn.networks_ns_list." + std::to_string(i) + ".3.weight");
107        }
108        bw.attn.final_proj =
109            store.get("backbone.blocks." + std::to_string(l) + ".mixed_attn.final_proj.weight");
110        bw.ffn_s.w1 =
111            store.get("backbone.blocks." + std::to_string(l) + ".mixed_ffn.network_s.0.weight");
112        bw.ffn_s.w2 =
113            store.get("backbone.blocks." + std::to_string(l) + ".mixed_ffn.network_s.3.weight");
114    }
115    m.head_w = store.get("backbone.linear.weight");
116    m.head_b = store.get("backbone.linear.bias");
117    return m;
118}

126struct Qkv {
127    std::vector<float> q, k, v;  // [rows, H, d]
128};
129
130// W_s 投影 S 段：x [S, D] → q/k/v [S, H, d]（布局 [S][3][H][d] 切段）
131Qkv project_s(const Tensor& w_s, const float* x, int64_t S, int64_t D, int H, int64_t d) {
132    Qkv out;
133    std::vector<float> qkv(static_cast<size_t>(S * 3 * D));
134    linear_rows(x, w_s.data.data(), qkv.data(), S, D, 3 * D);
135    out.q.resize(static_cast<size_t>(S * D));
136    out.k.resize(static_cast<size_t>(S * D));
137    out.v.resize(static_cast<size_t>(S * D));
138    for (int64_t s = 0; s < S; ++s) {
139        std::copy_n(qkv.begin() + s * 3 * D, D, out.q.begin() + s * D);
140        std::copy_n(qkv.begin() + s * 3 * D + D, D, out.k.begin() + s * D);
141        std::copy_n(qkv.begin() + s * 3 * D + 2 * D, D, out.v.begin() + s * D);
142    }
143    return out;
144}

148Tensor OneTransModel::forward(const Tensor& tokens, const Tensor& mask) const {
149    const int64_t B = tokens.shape[0], L = tokens.shape[1], D = tokens.shape[2];
150    const int H = cfg.num_heads, d = D / H;
151    const int Ns = cfg.ns_tokens_num;
152    const int64_t S = L - Ns;
153    const float eps = static_cast<float>(cfg.rms_eps);
154
155    Tensor logits({B, head_w.shape[0]});
156
157    for (int64_t b = 0; b < B; ++b) {
158        std::vector<float> x(tokens.data.begin() + b * L * D,
159                             tokens.data.begin() + (b + 1) * L * D);
160        std::vector<uint8_t> mrow(L);
161        for (int64_t j = 0; j < L; ++j) mrow[j] = mask.at(b * L + j) != 0.0f ? 1 : 0;
162
163        int64_t cur_s = S, cur_l = L;
164        for (int l = 0; l < cfg.num_blocks; ++l) {
165            const BlockWeights& bw = blocks[l];
166            // attention
167            std::vector<float> h(static_cast<size_t>(cur_l * D));
168            rms_norm_rows(x.data(), bw.norm_g.data.data(), h.data(), cur_l, D, eps);
169
170            Qkv sq = project_s(bw.attn.w_s, h.data(), cur_s, D, H, d);
171            // NS 逐 token 投影（每 token 一个 [B=1, D] 输入）
172            Qkv ns;
173            ns.q.resize(static_cast<size_t>(Ns * D));
174            ns.k.resize(static_cast<size_t>(Ns * D));
175            ns.v.resize(static_cast<size_t>(Ns * D));
176            for (int i = 0; i < Ns; ++i) {
177                std::vector<float> qkv3(static_cast<size_t>(3 * D));
178                linear_rows(h.data() + (cur_s + i) * D, bw.attn.w_ns[i].data.data(), qkv3.data(),
179                            1, D, 3 * D);
180                std::copy_n(qkv3.begin(), D, ns.q.begin() + i * D);
181                std::copy_n(qkv3.begin() + D, D, ns.k.begin() + i * D);
182                std::copy_n(qkv3.begin() + 2 * D, D, ns.v.begin() + i * D);
183            }
184            // 拼接 K/V：[L, H, d]
185            std::vector<float> kk(static_cast<size_t>(cur_l * D)), vv(static_cast<size_t>(cur_l * D));
186            std::copy(sq.k.begin(), sq.k.end(), kk.begin());
187            std::copy(ns.k.begin(), ns.k.end(), kk.begin() + cur_s * D);
188            std::copy(sq.v.begin(), sq.v.end(), vv.begin());
189            std::copy(ns.v.begin(), ns.v.end(), vv.begin() + cur_s * D);
190
191            // 掩码：padding 列 + causal
192            std::vector<uint8_t> allow(static_cast<size_t>(cur_l * cur_l), 1);
193            for (int64_t i = 0; i < cur_l; ++i)
194                for (int64_t j = 0; j < cur_l; ++j)
195                    if (j > i || !mrow[j]) allow[static_cast<size_t>(i * cur_l + j)] = 0;
196
197            // 拼接 Q：[L, H, d]
198            std::vector<float> qq(static_cast<size_t>(cur_l * D));
199            std::copy(sq.q.begin(), sq.q.end(), qq.begin());
200            std::copy(ns.q.begin(), ns.q.end(), qq.begin() + cur_s * D);
201
202            std::vector<float> attn(static_cast<size_t>(cur_l * D));
203            sdpa_masked(qq.data(), kk.data(), vv.data(), attn.data(), cur_l, cur_l, H, d,
204                        allow.data());
205            std::vector<float> proj(static_cast<size_t>(cur_l * D));
206            linear_rows(attn.data(), bw.attn.final_proj.data.data(), proj.data(), cur_l, D, D);
207            std::vector<float> z(static_cast<size_t>(cur_l * D));
208            for (size_t i = 0; i < z.size(); ++i) z[i] = proj[i] + x[i];
209
210            // FFN（S 段共享 / NS 段逐 token）——norm 共享同一权重
211            std::vector<float> h2(static_cast<size_t>(cur_l * D));
212            rms_norm_rows(z.data(), bw.norm_g.data.data(), h2.data(), cur_l, D, eps);
213            std::vector<float> f(static_cast<size_t>(cur_l * D));
214            ffn_forward(bw.ffn_s, h2.data(), f.data(), cur_s, D);
215            for (int i = 0; i < Ns; ++i) {
216                ffn_forward(bw.ffn_ns[i], h2.data() + (cur_s + i) * D, f.data() + (cur_s + i) * D,
217                            1, D);
218            }
219            for (size_t i = 0; i < z.size(); ++i) z[i] += f[i];
220
221            // pyramid：保留尾部 out_seq_num 个 S token + 全部 NS
222            int64_t keep = bw.out_seq_num;
223            int64_t new_s = keep;
224            std::vector<float> nx(static_cast<size_t>((new_s + Ns) * D));
225            std::vector<uint8_t> nm(static_cast<size_t>(new_s + Ns));
226            std::copy_n(z.begin() + (cur_s - keep) * D, (new_s + Ns) * D, nx.begin());
227            std::copy_n(mrow.begin() + (cur_s - keep), new_s + Ns, nm.begin());
228            x = std::move(nx);
229            mrow = std::move(nm);
230            cur_s = new_s;
231            cur_l = new_s + Ns;
232        }
233
234        // head：末 Ns token 均值 → [D] → linear
235        std::vector<float> pooled(static_cast<size_t>(D), 0.0f);
236        for (int i = 0; i < Ns; ++i)
237            for (int64_t t = 0; t < D; ++t) pooled[t] += x[(cur_s + i) * D + t];
238        for (int64_t t = 0; t < D; ++t) pooled[t] /= static_cast<float>(Ns);
239        std::vector<float> lg(static_cast<size_t>(head_w.shape[0]));
240        linear_rows(pooled.data(), head_w.data.data(), lg.data(), 1, D, head_w.shape[0]);
241        for (int64_t t = 0; t < head_w.shape[0]; ++t)
242            logits.at(b * head_w.shape[0] + t) = lg[t] + head_b.at(t);
243    }
244    return logits;
245}
