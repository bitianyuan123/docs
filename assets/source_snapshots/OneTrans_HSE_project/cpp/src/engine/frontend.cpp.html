10// _mlp：Linear(in→out) → GELU → Linear(out→out)（均无 bias）
11struct Mlp {
12    Tensor w1, w2;
13    void apply(const float* x, float* y, int64_t rows, int64_t in, int64_t out) const {
14        std::vector<float> h(static_cast<size_t>(rows * out));
15        linear_rows(x, w1.data.data(), h.data(), rows, in, out);
16        for (auto& v : h) v = gelu(v);
17        linear_rows(h.data(), w2.data.data(), y, rows, out, out);
18    }
19};

21// piecewise 分箱编码（对齐 onetrans/nn/encoders/piecewise.py forward）
22// x [M, n_features] → out [M, sum(n_bins)]（行主序：feature 主、bin 次）
23Tensor piecewise_forward(const Tensor& w, const Tensor& b, const std::vector<int64_t>& n_bins,
24                         const float* x, int64_t M, int n_features) {
25    const int64_t max_bins = w.shape[1];
26    int64_t out_dim = 0;
27    for (auto nb : n_bins) out_dim += nb;
28    Tensor out({M, out_dim});
29    for (int64_t m = 0; m < M; ++m) {
30        const float* xm = x + m * n_features;
31        float* om = out.data.data() + m * out_dim;
32        int64_t o = 0;
33        for (int i = 0; i < n_features; ++i) {
34            const int64_t nb = n_bins[i];
35            for (int64_t j = 0; j < nb; ++j, ++o) {
36                float v = b.at(i * max_bins + j) + w.at(i * max_bins + j) * xm[i];
37                if (nb == 1) {
38                    // single bin：x >= bias ? 1 : 0
39                    v = (xm[i] >= b.at(i * max_bins)) ? 1.0f : 0.0f;
40                } else if (j == 0) {
41                    v = std::min(v, 1.0f);
42                } else if (j == nb - 1) {
43                    v = std::max(v, 0.0f);
44                } else {
45                    v = std::min(std::max(v, 0.0f), 1.0f);
46                }
47                om[o] = v;
48            }
49        }
50    }
51    return out;
52}

56EmbeddingFrontend EmbeddingFrontend::load(const ArtifactStore& store) {
57    EmbeddingFrontend f;
58    f.cfg_ = store.config();
59    const int D = f.cfg_.d_model;
60
61    f.s_mlp_w1 = store.get("tokenizer.s_tokenizer.mlps.0.0.weight");
62    f.s_mlp_w2 = store.get("tokenizer.s_tokenizer.mlps.0.2.weight");
63    f.s_type_emb = store.get("tokenizer.s_tokenizer.type_embeddings.weight");
64    f.pos_emb = store.get("tokenizer.pos_embedding.weight");
65    f.tok_rms_g = store.get("tokenizer.rms_norm.weight");
66
67    f.ns_w1.resize(f.cfg_.ns_group_dims.size());
68    f.ns_w2.resize(f.cfg_.ns_group_dims.size());
69    for (size_t g = 0; g < f.cfg_.ns_group_dims.size(); ++g) {
70        f.ns_w1[g] = store.get("tokenizer.ns_tokenizer.mlps." + std::to_string(g) + ".0.weight");
71        f.ns_w2[g] = store.get("tokenizer.ns_tokenizer.mlps." + std::to_string(g) + ".2.weight");
72    }
73    f.pw_w = store.get("embedder.piecewise_encoder.weight");
74    f.pw_b = store.get("embedder.piecewise_encoder.bias");
75    return f;
76}

78std::pair<Tensor, Tensor> EmbeddingFrontend::encode_s(const std::vector<int64_t>& item_ids,
79                                                      const LookupFn& lookup) const {
80    const int S = cfg_.max_seq_len;
81    const int D = cfg_.d_model;
82    const float eps = static_cast<float>(cfg_.rms_eps);
83
84    // 截尾保留最新 + 左 padding
85    size_t valid = item_ids.size() > static_cast<size_t>(S) ? static_cast<size_t>(S) : item_ids.size();
86    std::vector<int64_t> ids(valid);
87    if (valid) ids.assign(item_ids.end() - static_cast<long>(valid), item_ids.end());
88    const size_t pad = static_cast<size_t>(S) - valid;
89
90    std::vector<int64_t> padded(static_cast<size_t>(S), 0);
91    std::copy(ids.begin(), ids.end(), padded.begin() + static_cast<long>(pad));
92    std::vector<uint8_t> mask(static_cast<size_t>(S), 0);
93    std::fill(mask.begin() + static_cast<long>(pad), mask.end(), 1);
94
95    // item 查表 → [S, D]，padding 位置（id=0 → 表 0 行）再乘 mask
96    std::vector<float> seq_emb = lookup("item", padded);
97    for (size_t i = 0; i < static_cast<size_t>(S); ++i)
98        if (!mask[i]) std::fill_n(seq_emb.begin() + static_cast<long>(i * D), D, 0.0f);
99
100    // STokenizer：mlp × mask（timestamp_agnostic 合并不加 type_emb、无 sep token）
101    Mlp mlp;
102    mlp.w1 = s_mlp_w1;
103    mlp.w2 = s_mlp_w2;
104    Tensor s_tokens({1, S, D});
105    mlp.apply(seq_emb.data(), s_tokens.data.data(), S, cfg_.seq_in_dim, D);
106    for (int i = 0; i < S; ++i)
107        if (!mask[i]) std::fill_n(s_tokens.data.begin() + i * D, D, 0.0f);
108
109    // + pos_emb → RMSNorm（OneTransTokenizer.encode_s）
110    Tensor s_emb({1, S, D});
111    std::vector<float> tmp(static_cast<size_t>(S * D));
112    for (int i = 0; i < S; ++i)
113        for (int t = 0; t < D; ++t) tmp[i * D + t] = s_tokens.at(i * D + t) + pos_emb.at(i * D + t);
114    rms_norm_rows(tmp.data(), tok_rms_g.data.data(), s_emb.data.data(), S, D, eps);
115
116    Tensor s_mask({1, S});
117    for (int i = 0; i < S; ++i) s_mask.at(i) = mask[i] ? 1.0f : 0.0f;
118    return {s_emb, s_mask};
119}

121// ---- 两步式（SEDA，§7.4.3）----------------------------------------------- //
122// 阶段 1（EmbedLookup Pool）：查表 + mean-bag 聚合（阻塞 I/O 归 IO 池）
123NsEmbeddings EmbeddingFrontend::lookup_ns(const ScoreInput& in, const LookupFn& lookup) const {
124    if (in.candidates.empty()) throw std::runtime_error("score: candidates 为空");
125    const int M = static_cast<int>(in.candidates.size());
126    const int D = cfg_.d_model;
127
128    NsEmbeddings emb;
129    emb.uid = lookup("user", {in.uid_sparse});
130
131    std::vector<int64_t> item_ids;
132    item_ids.reserve(in.candidates.size());
133    for (const auto& c : in.candidates) item_ids.push_back(c.item_id);
134    emb.items = lookup("item", item_ids);
135
136    // mean-bag（artist/album）：多值查表求均值，空 bag → 0
137    auto bag = [&](const std::string& table,
138                   const std::vector<std::vector<int64_t>>& ids_per) -> std::vector<float> {
139        std::vector<int64_t> flat;
140        for (const auto& v : ids_per) flat.insert(flat.end(), v.begin(), v.end());
141        std::vector<float> rows =
142            flat.empty() ? std::vector<float>(D, 0.0f) : lookup(table, flat);
143        std::vector<float> out(static_cast<size_t>(M) * D, 0.0f);
144        size_t off = 0;
145        for (int m = 0; m < M; ++m) {
146            size_t n = ids_per[static_cast<size_t>(m)].size();
147            if (n == 0) continue;
148            for (int t = 0; t < D; ++t) {
149                float acc = 0.0f;
150                for (size_t r = 0; r < n; ++r) acc += rows[(off + r) * D + t];
151                out[static_cast<size_t>(m) * D + t] = acc / static_cast<float>(n);
152            }
153            off += n;
154        }
155        return out;
156    };
157    std::vector<std::vector<int64_t>> artist_ids, album_ids;
158    for (const auto& c : in.candidates) {
159        artist_ids.push_back(c.artist_ids);
160        album_ids.push_back(c.album_ids);
161    }
162    emb.artists = bag("artist", artist_ids);
163    emb.albums = bag("album", album_ids);
164    return emb;
165}

167// 阶段 2（Frontend Encode Pool）：piecewise + 组 MLP + RMSNorm（纯 CPU 计算）
168Tensor EmbeddingFrontend::encode_ns_with(const ScoreInput& in, const NsEmbeddings& emb) const {
169    const int M = static_cast<int>(in.candidates.size());
170    const int D = cfg_.d_model;
171    const int Ns = static_cast<int>(cfg_.ns_group_dims.size());
172    const float eps = static_cast<float>(cfg_.rms_eps);
173
174    // dense：用户级 ∥ 候选级 → piecewise
175    const int nf = cfg_.piecewise_n_features;
176    std::vector<float> dense(static_cast<size_t>(M * nf));
177    for (int m = 0; m < M; ++m) {
178        const auto& c = in.candidates[static_cast<size_t>(m)];
179        if (in.user_dense.size() != static_cast<size_t>(cfg_.n_user_dense) ||
180            c.dense.size() != static_cast<size_t>(cfg_.n_cand_dense))
181            throw std::runtime_error("dense 列数不匹配");
182        std::copy(in.user_dense.begin(), in.user_dense.end(),
183                  dense.begin() + static_cast<long>(m * nf));
184        std::copy(c.dense.begin(), c.dense.end(),
185                  dense.begin() + static_cast<long>(m * nf + cfg_.n_user_dense));
186    }
187    Tensor dense_enc = piecewise_forward(pw_w, pw_b, cfg_.piecewise_n_bins, dense.data(), M, nf);
188
189    // uid 广播 M 行
190    if (emb.uid.size() != static_cast<size_t>(D)) throw std::runtime_error("uid 表维度不匹配");
191    std::vector<float> uid_emb(static_cast<size_t>(M) * D);
192    for (int m = 0; m < M; ++m)
193        std::copy(emb.uid.begin(), emb.uid.end(), uid_emb.begin() + static_cast<long>(m * D));
194
195    // NSGroupWise：各组 mlp → stack [M, Ns, D] → RMSNorm
196    Tensor ns_emb({M, Ns, D});
197    std::vector<const std::vector<float>*> groups = {&dense_enc.data, &uid_emb, &emb.items,
198                                                      &emb.artists, &emb.albums};
199    std::vector<float> stacked(static_cast<size_t>(M * Ns * D));
200    std::vector<float> g_tmp(static_cast<size_t>(M) * D);
201    for (size_t g = 0; g < cfg_.ns_group_dims.size(); ++g) {
202        Mlp mlp;
203        mlp.w1 = ns_w1[g];
204        mlp.w2 = ns_w2[g];
205        // 组 g：[M, in] → [M, D] → 散射到 [M, Ns, D] 的 token g 槽位
206        mlp.apply(groups[g]->data(), g_tmp.data(), M, cfg_.ns_group_dims[g], D);
207        for (int m = 0; m < M; ++m)
208            std::copy_n(g_tmp.begin() + static_cast<long>(m * D), D,
209                        stacked.begin() + static_cast<long>((m * Ns + static_cast<int>(g)) * D));
210    }
211    rms_norm_rows(stacked.data(), tok_rms_g.data.data(), ns_emb.data.data(), M * Ns, D, eps);
212    return ns_emb;
213}

215// 一步式 = 两步式组合（兼容路径）
216Tensor EmbeddingFrontend::encode_ns(const ScoreInput& in, const LookupFn& lookup) const {
217    NsEmbeddings emb = lookup_ns(in, lookup);
218    return encode_ns_with(in, emb);
219}
