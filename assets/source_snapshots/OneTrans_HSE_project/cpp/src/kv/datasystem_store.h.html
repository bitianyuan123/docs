1// Datasystem 后端（yuanrong-datasystem C++ SDK）——实现 KVStore 逻辑接口。
2//
3// 通过 datasystem::KVClient 直接把 UserKV payload 落到分布式 KV（set/get），
4// 与 onetrans/serving/datasystem_adapter.py 的存储语义一致：
5//   - key = KVKey::str()（kv:{b64url(mv)}:{b64url(uid)}，字符集受限）；
6//   - value = UserKVRecord.payload（kv_serialize 二进制，自含 s_len/per_layer_len 元数据）；
7//   - TTL 经 SetParam.ttlSecond / Expire 落地，由 datasystem 服务端回收。
8//
9// 仅在链接了 datasystem C++ SDK（CMake: ONETRANS_DATASYSTEM=ON 且 find_package(Datasystem)）
10// 时才编译本实现；SDK 缺失时热路径回退 LocalKVStore（见 kv/store.h）。
11#pragma once
12
13#include "kv/store.h"
14
15#include <memory>
16
17namespace datasystem {
18class KVClient;
19}
20
21namespace onetrans {
22
23class DatasystemKVStore : public KVStore {
24public:
25    struct Options {
26        std::string host = "127.0.0.1";
27        int port = 9088;              // datasystem worker 默认端口（部署时覆盖）
28        int64_t default_ttl_seconds = 0;  // 0 = 永不过期
29    };
30
31    explicit DatasystemKVStore(const Options& options);
32    ~DatasystemKVStore() override;
33
34    bool put(const UserKVRecord& rec) override;
35    std::shared_ptr<const UserKVRecord> get(const KVKey& key) override;
36    std::vector<std::shared_ptr<const UserKVRecord>> mget(
37        const std::vector<KVKey>& keys) override;
38    int del(const std::vector<KVKey>& keys) override;
39    void ttl(const KVKey& key, int64_t seconds) override;
40    size_t size() override { return 0; }  // datasystem 无全量计数；healthz 显示 0
41    int sweep() override { return 0; }    // TTL 由服务端惰性回收，客户端无主动 sweep
42    const char* backend() const override { return "datasystem"; }
43
44private:
45    std::unique_ptr<datasystem::KVClient> client_;
46    int64_t default_ttl_;
47};
48
49}  // namespace onetrans
