12constexpr char kMagic[] = "ONETRANSKV\x01";  // 与 Python _MAGIC 严格一致：11 字节（不含 \0）
13constexpr size_t kMagicLen = 11;
14
15struct ParsedHeader {
16    JsonValue header;
17    size_t body_offset;  // raw bytes 起始偏移
18};

20size_t tensor_bytes(const std::vector<int64_t>& shape) {
21    int64_t n = 1;
22    for (auto s : shape) n *= s;
23    return static_cast<size_t>(n) * sizeof(float);
24}

26ParsedHeader parse_header(const std::string& payload) {
27    if (payload.size() < kMagicLen || std::memcmp(payload.data(), kMagic, kMagicLen) != 0)
28        throw std::runtime_error("未知 payload 魔数");
29    uint32_t hlen;
30    std::memcpy(&hlen, payload.data() + kMagicLen, 4);  // 小端
31    if (payload.size() < kMagicLen + 4 + hlen)
32        throw std::runtime_error("payload header 截断");
33    ParsedHeader out;
34    out.header = json_parse(payload.substr(kMagicLen + 4, hlen));
35    out.body_offset = kMagicLen + 4 + hlen;
36    return out;
37}

41std::string kv_serialize(const UserKV& kv) {
42    JsonValue header = JsonValue::object();
43    header.set("dtype", JsonValue::str("float32"));
44    header.set("n_layers", JsonValue::number(static_cast<double>(kv.k.size())));
45    header.set("s_len", JsonValue::number(static_cast<double>(kv.s_len)));
46    JsonValue layers = JsonValue::array({});
47    size_t total = 0;
48    for (size_t l = 0; l < kv.k.size(); ++l) {
49        JsonValue m = JsonValue::object();
50        m.set("l", JsonValue::number(static_cast<double>(l)));
51        JsonValue ks = JsonValue::array({});
52        for (auto s : kv.k[l].shape) ks.push_back(JsonValue::number(static_cast<double>(s)));
53        JsonValue vs = JsonValue::array({});
54        for (auto s : kv.v[l].shape) vs.push_back(JsonValue::number(static_cast<double>(s)));
55        m.set("k_shape", ks);
56        m.set("v_shape", vs);
57        m.set("len", JsonValue::number(static_cast<double>(kv.per_layer_len[l])));
58        layers.push_back(std::move(m));
59        total += kv.k[l].data.size() + kv.v[l].data.size();
60    }
61    header.set("layers", std::move(layers));
62    std::string header_json = header.dump();
63
64    std::string payload;
65    payload.reserve(kMagicLen + 4 + header_json.size() + total * sizeof(float));
66    payload.append(kMagic, kMagicLen);
67    uint32_t hlen = static_cast<uint32_t>(header_json.size());
68    payload.append(reinterpret_cast<const char*>(&hlen), 4);
69    payload.append(header_json);
70    for (size_t l = 0; l < kv.k.size(); ++l) {
71        payload.append(reinterpret_cast<const char*>(kv.k[l].data.data()),
72                       kv.k[l].data.size() * sizeof(float));
73        payload.append(reinterpret_cast<const char*>(kv.v[l].data.data()),
74                       kv.v[l].data.size() * sizeof(float));
75    }
76    return payload;
77}

79KvPayload kv_read_header(const std::string& payload) {
80    ParsedHeader p = parse_header(payload);
81    KvPayload out;
82    out.dtype = p.header.at("dtype").as_string();
83    out.s_len = p.header.at("s_len").as_int();
84    out.n_layers = static_cast<int>(p.header.at("n_layers").as_int());
85    for (const auto& m : p.header.at("layers").as_array())
86        out.per_layer_len.push_back(m.at("len").as_int());
87    return out;
88}

90UserKV kv_deserialize(const std::string& payload) {
91    ParsedHeader p = parse_header(payload);
92    if (p.header.at("dtype").as_string() != "float32")
93        throw std::runtime_error("仅支持 float32 payload");
94
95    UserKV kv;
96    kv.s_len = p.header.at("s_len").as_int();
97    size_t off = p.body_offset;
98    for (const auto& m : p.header.at("layers").as_array()) {
99        Tensor k({1, 1, 1, 1}), v({1, 1, 1, 1});
100        k.shape.clear();
101        v.shape.clear();
102        for (const auto& s : m.at("k_shape").as_array()) k.shape.push_back(s.as_int());
103        for (const auto& s : m.at("v_shape").as_array()) v.shape.push_back(s.as_int());
104        k.data.resize(tensor_bytes(k.shape) / sizeof(float));
105        v.data.resize(tensor_bytes(v.shape) / sizeof(float));
106        if (off + k.data.size() * 4 > payload.size() || off + v.data.size() * 4 > payload.size())
107            throw std::runtime_error("payload body 截断");
108        std::memcpy(k.data.data(), payload.data() + off, k.data.size() * sizeof(float));
109        off += k.data.size() * 4;
110        std::memcpy(v.data.data(), payload.data() + off, v.data.size() * sizeof(float));
111        off += v.data.size() * 4;
112        kv.per_layer_len.push_back(m.at("len").as_int());
113        kv.k.push_back(std::move(k));
114        kv.v.push_back(std::move(v));
115    }
116    return kv;
117}

119std::string kv_concat_payload(const std::string& base, const std::string& delta) {
120    ParsedHeader b = parse_header(base);
121    ParsedHeader d = parse_header(delta);
122    if (b.header.at("n_layers").as_int() != d.header.at("n_layers").as_int())
123        throw std::runtime_error("layer_mismatch");
124
125    // 逐层拼接 K/V 数值（S_l += delta_len），重建 header 与 body
126    const auto& bl = b.header.at("layers").as_array();
127    const auto& dl = d.header.at("layers").as_array();
128    JsonValue layers = JsonValue::array({});
129    std::string body;
130    size_t boff = b.body_offset, doff = d.body_offset;
131    for (size_t l = 0; l < bl.size(); ++l) {
132        // K
133        size_t bkb = tensor_bytes({});
134        {
135            std::vector<int64_t> shape;
136            for (const auto& s : bl[l].at("k_shape").as_array()) shape.push_back(s.as_int());
137            bkb = tensor_bytes(shape);
138        }
139        size_t dkb = 0;
140        {
141            std::vector<int64_t> shape;
142            for (const auto& s : dl[l].at("k_shape").as_array()) shape.push_back(s.as_int());
143            dkb = tensor_bytes(shape);
144        }
145        JsonValue m = JsonValue::object();
146        JsonValue ks = JsonValue::array({});
147        for (const auto& s : bl[l].at("k_shape").as_array())
148            ks.push_back(JsonValue::number(s.as_number() + dl[l].at("k_shape").as_array()[0].as_number()));
149        // 仅第 1 维（序列长度）变化
150        JsonValue vs = JsonValue::array({});
151        for (const auto& s : bl[l].at("v_shape").as_array())
152            vs.push_back(JsonValue::number(s.as_number() + dl[l].at("v_shape").as_array()[0].as_number()));
153        m.set("l", JsonValue::number(static_cast<double>(l)));
154        m.set("k_shape", ks);
155        m.set("v_shape", vs);
156        m.set("len", JsonValue::number(bl[l].at("len").as_number() + dl[l].at("len").as_number()));
157        layers.push_back(std::move(m));
158
159        body.append(base, boff, bkb);
160        body.append(delta, doff, dkb);
161        boff += bkb;
162        doff += dkb;
163        // V
164        size_t bvb = 0, dvb = 0;
165        {
166            std::vector<int64_t> shape;
167            for (const auto& s : bl[l].at("v_shape").as_array()) shape.push_back(s.as_int());
168            bvb = tensor_bytes(shape);
169            shape.clear();
170            for (const auto& s : dl[l].at("v_shape").as_array()) shape.push_back(s.as_int());
171            dvb = tensor_bytes(shape);
172        }
173        body.append(base, boff, bvb);
174        body.append(delta, doff, dvb);
175        boff += bvb;
176        doff += dvb;
177    }
178
179    JsonValue header = JsonValue::object();
180    header.set("dtype", JsonValue::str("float32"));
181    header.set("n_layers", b.header.at("n_layers"));
182    header.set("s_len", JsonValue::number(b.header.at("s_len").as_number() +
183                                           d.header.at("s_len").as_number()));
184    header.set("layers", std::move(layers));
185    std::string header_json = header.dump();
186
187    std::string payload;
188    payload.append(kMagic, kMagicLen);
189    uint32_t hlen = static_cast<uint32_t>(header_json.size());
190    payload.append(reinterpret_cast<const char*>(&hlen), 4);
191    payload.append(header_json);
192    payload.append(body);
193    return payload;
194}
