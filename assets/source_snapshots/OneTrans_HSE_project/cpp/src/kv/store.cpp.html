11// url-safe base64（无 padding），对齐 Python base64.urlsafe_b64encode(...).rstrip("=")
12std::string b64url(const std::string& in) {
13    static const char kTab[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
14    std::string out;
15    out.reserve((in.size() + 2) / 3 * 4);
16    size_t i = 0;
17    while (i + 2 < in.size()) {
18        uint32_t v = (static_cast<uint8_t>(in[i]) << 16) | (static_cast<uint8_t>(in[i + 1]) << 8) |
19                     static_cast<uint8_t>(in[i + 2]);
20        out += kTab[(v >> 18) & 63];
21        out += kTab[(v >> 12) & 63];
22        out += kTab[(v >> 6) & 63];
23        out += kTab[v & 63];
24        i += 3;
25    }
26    if (i + 1 == in.size()) {
27        uint32_t v = static_cast<uint8_t>(in[i]) << 16;
28        out += kTab[(v >> 18) & 63];
29        out += kTab[(v >> 12) & 63];
30    } else if (i + 2 == in.size()) {
31        uint32_t v = (static_cast<uint8_t>(in[i]) << 16) | (static_cast<uint8_t>(in[i + 1]) << 8);
32        out += kTab[(v >> 18) & 63];
33        out += kTab[(v >> 12) & 63];
34        out += kTab[(v >> 6) & 63];
35    }
36    return out;
37}

49bool LocalKVStore::put(const UserKVRecord& rec) {
50    Entry e;
51    e.rec = std::make_shared<UserKVRecord>(rec);
52    e.expire_at = default_ttl_ > 0 ? now_unix() + default_ttl_ : 0;
53    std::lock_guard<std::mutex> lk(mu_);
54    map_[rec.key.str()] = std::move(e);
55    return true;
56}

58std::shared_ptr<const UserKVRecord> LocalKVStore::get(const KVKey& key) {
59    std::lock_guard<std::mutex> lk(mu_);
60    auto it = map_.find(key.str());
61    if (it == map_.end()) return nullptr;
62    if (it->second.expire_at > 0 && it->second.expire_at <= now_unix()) {
63        map_.erase(it);
64        return nullptr;
65    }
66    return it->second.rec;
67}

69std::vector<std::shared_ptr<const UserKVRecord>> LocalKVStore::mget(
70    const std::vector<KVKey>& keys) {
71    std::vector<std::shared_ptr<const UserKVRecord>> out;
72    out.reserve(keys.size());
73    std::lock_guard<std::mutex> lk(mu_);
74    for (const auto& k : keys) {
75        auto it = map_.find(k.str());
76        if (it == map_.end() ||
77            (it->second.expire_at > 0 && it->second.expire_at <= now_unix())) {
78            out.push_back(nullptr);
79        } else {
80            out.push_back(it->second.rec);
81        }
82    }
83    return out;
84}

86int LocalKVStore::del(const std::vector<KVKey>& keys) {
87    std::lock_guard<std::mutex> lk(mu_);
88    int n = 0;
89    for (const auto& k : keys) n += static_cast<int>(map_.erase(k.str()));

93void LocalKVStore::ttl(const KVKey& key, int64_t seconds) {
94    std::lock_guard<std::mutex> lk(mu_);
95    auto it = map_.find(key.str());
96    if (it == map_.end()) return;
97    it->second.expire_at = seconds > 0 ? now_unix() + seconds : 0;
98}

105int LocalKVStore::sweep() {
106    std::lock_guard<std::mutex> lk(mu_);
107    int64_t now = now_unix();
108    int n = 0;
109    for (auto it = map_.begin(); it != map_.end();) {
110        if (it->second.expire_at > 0 && it->second.expire_at <= now) {
111            it = map_.erase(it);
112            ++n;
113        } else {
114            ++it;
115        }
116    }
117    return n;
118}
