11// 从 datasystem 取回的 payload 重建 UserKVRecord（元数据回读 header）。
12std::shared_ptr<const UserKVRecord> build_record(const KVKey& key, std::string payload) {
13    if (payload.empty()) return nullptr;
14    try {
15        auto rec = std::make_shared<UserKVRecord>();
16        rec->key = key;
17        rec->payload = std::move(payload);
18        KvPayload hdr = kv_read_header(rec->payload);
19        rec->s_len = hdr.s_len;
20        rec->per_layer_len = hdr.per_layer_len;
21        rec->dtype = hdr.dtype;
22        rec->created_at = 0;  // datasystem 未单独记录，置 0（计算只依赖 payload）
23        return rec;
24    } catch (const std::exception&) {
25        return nullptr;  // 损坏/旧格式 → 按 miss 降级
26    }
27}

31DatasystemKVStore::DatasystemKVStore(const Options& options)
32    : default_ttl_(options.default_ttl_seconds) {
33    datasystem::ConnectOptions co;
34    co.host = options.host;
35    co.port = options.port;
36    client_ = std::make_unique<datasystem::KVClient>(co);
37    // Init 建立与 worker 的连接（空配置：走环境变量/默认）；失败不抛，
38    // 后续 put/get 返回错误并由 G8 miss 语义统一降级。
39    (void)client_->Init();
40}

46bool DatasystemKVStore::put(const UserKVRecord& rec) {
47    datasystem::SetParam param;
48    if (default_ttl_ > 0) param.ttlSecond = static_cast<uint32_t>(default_ttl_);
49    datasystem::Status s = client_->Set(rec.key.str(), rec.payload, param);
50    return s.IsOk();
51}

53std::shared_ptr<const UserKVRecord> DatasystemKVStore::get(const KVKey& key) {
54    std::string val;
55    datasystem::Status s = client_->Get(key.str(), val);
56    // 未命中 / 传输错误统一降级为 miss（与 LocalKVStore 的 not-found 语义一致）。
57    if (s.GetCode() == datasystem::K_NOT_FOUND || s.IsError()) return nullptr;
58    return build_record(key, std::move(val));
59}

61std::vector<std::shared_ptr<const UserKVRecord>> DatasystemKVStore::mget(
62    const std::vector<KVKey>& keys) {
63    std::vector<std::shared_ptr<const UserKVRecord>> out(keys.size());
64    // SEDA 热路径按请求逐 key get（KV I/O 池并发）；mget 仅用于测试/旧同步路径，
65    // 逐 key 保证顺序与 miss 语义严格一致。
66    for (size_t i = 0; i < keys.size(); ++i) out[i] = get(keys[i]);
67    return out;
68}

70int DatasystemKVStore::del(const std::vector<KVKey>& keys) {
71    if (keys.empty()) return 0;
72    std::vector<std::string> kstr;
73    kstr.reserve(keys.size());
74    for (const auto& k : keys) kstr.push_back(k.str());
75    std::vector<std::string> failed;
76    datasystem::Status s = client_->Del(kstr, failed);
77    if (s.IsError()) return 0;
78    return static_cast<int>(keys.size() - failed.size());
79}

81void DatasystemKVStore::ttl(const KVKey& key, int64_t seconds) {
82    std::vector<std::string> failed;
83    // seconds<=0 → Expire 传 0，表示取消 TTL（改由显式 Del 管理生命周期）。
84    uint32_t ttl = seconds > 0 ? static_cast<uint32_t>(seconds) : 0;
85    (void)client_->Expire({key.str()}, ttl, failed);
86}
