1// OneTrans 精排服务入口：接入层 HTTP + SEDA 编排层 + 数据面 + 混合引擎装配。
2//
3// 启动：
4//   ./onetrans_server --weights cpp/artifacts/weights --port 8080 [--compute-backend auto]
5//
6// 接口：
7//   GET  /healthz                     存活探测（含实际生效计算后端）
8//   POST /ingest                      Stage I：行为序列 → UserKV（nearline，全 C++ CPU）
9//   POST /score                       Stage II：候选交叉打分（SEDA 流水线，异步路由）
10//   POST /rank                        PaiRec 精排：自装配 → Stage II → sigmoid 回填
11//   GET  /metrics                     Prometheus 风格指标
12//
13// 计算后端（§7.4.1 混合架构）：
14//   python  Stage II 前向经 PythonComputeBridge 下发 PyTorch 算子（CUDA/CPU）
15//   cpp     Stage II 前向走 C++ CPU score_ns_batch（数值等价的降级路径）
16//   auto    优先 python（torch 可用即启用），失败/队列满自动降级 cpp

52struct ServerConfig {
53    std::string weights_dir = "cpp/artifacts/weights";
54    std::string host = "0.0.0.0";
55    int port = 8080;
56    int http_threads = 8;
57    // SEDA 池（§7.4.2）
58    int lookup_threads = 4;
59    int encode_threads = 2;
60    int kv_threads = 4;
61    int compute_threads = 0;  // ≤0 → 硬件核数
62    size_t queue_cap = 1024;
63    size_t max_batch = 32;
64    int max_wait_ms = 5;
65    int nearline_threads = 2;  // Stage I Compute Pool（nearline prefill）
66    std::string model_version = "v42";
67    int shards = 1;
68    // PaiRec 精排契约：响应 model_role 必须等于 PaiRec 配置 required_model_role
69    std::string model_role = "engineering";
70    // /rank 自装配特征源（dense 是"数值"，不进 PS，从本地文件装）
71    std::string rank_item_lookup = "artifacts/domain_features/item_rank_lookup.tsv";
72    std::string rank_user_features = "artifacts/domain_features/user_features.tsv";
73    int64_t kv_ttl_seconds = 0;
74    std::string kv_backend = "local";       // local | datasystem
75    std::string datasystem_host = "127.0.0.1";
76    int datasystem_port = 9088;
77    std::string embedding_source = "local"; // local | ps（稀疏查表数据面，D2）
78    std::string ps_server = "127.0.0.1:8000";
79    std::string compute_backend = "auto";  // auto | python | cpp
80    std::string bridge_module_dir = "cpp/tools";
81    int bridge_init_timeout_s = 120;
82    int log_level = onetrans::L_INFO;  // --log-level: debug|info|warn|error
83};

85ServerConfig parse_args(int argc, char** argv) {
86    ServerConfig c;
87    for (int i = 1; i < argc; ++i) {
88        std::string a = argv[i];
89        auto next = [&]() -> std::string {
90            if (i + 1 >= argc) throw std::runtime_error("参数缺值: " + a);
91            return argv[++i];
92        };
93        if (a == "--weights") c.weights_dir = next();
94        else if (a == "--host") c.host = next();
95        else if (a == "--port") c.port = std::atoi(next().c_str());
96        else if (a == "--http-threads") c.http_threads = std::atoi(next().c_str());
97        else if (a == "--lookup-threads") c.lookup_threads = std::atoi(next().c_str());
98        else if (a == "--encode-threads") c.encode_threads = std::atoi(next().c_str());
99        else if (a == "--kv-threads") c.kv_threads = std::atoi(next().c_str());
100        else if (a == "--compute-threads") c.compute_threads = std::atoi(next().c_str());
101        else if (a == "--nearline-threads") c.nearline_threads = std::atoi(next().c_str());
102        else if (a == "--queue-cap") c.queue_cap = static_cast<size_t>(std::atoi(next().c_str()));
103        else if (a == "--max-batch") c.max_batch = static_cast<size_t>(std::atoi(next().c_str()));
104        else if (a == "--max-wait-ms") c.max_wait_ms = std::atoi(next().c_str());
105        else if (a == "--model-version") c.model_version = next();
106        else if (a == "--model-role") c.model_role = next();
107        else if (a == "--rank-item-lookup") c.rank_item_lookup = next();
108        else if (a == "--rank-user-features") c.rank_user_features = next();
109        else if (a == "--shards") c.shards = std::atoi(next().c_str());
110        else if (a == "--kv-ttl-seconds") c.kv_ttl_seconds = std::atoll(next().c_str());
111        else if (a == "--kv-backend") c.kv_backend = next();
112        else if (a == "--datasystem-host") c.datasystem_host = next();
113        else if (a == "--datasystem-port") c.datasystem_port = std::atoi(next().c_str());
114        else if (a == "--embedding-source") c.embedding_source = next();
115        else if (a == "--ps-server") c.ps_server = next();
116        else if (a == "--compute-backend") c.compute_backend = next();
117        else if (a == "--bridge-module-dir") c.bridge_module_dir = next();
118        else if (a == "--bridge-init-timeout") c.bridge_init_timeout_s = std::atoi(next().c_str());
119        else if (a == "--log-level")
120            c.log_level = onetrans::parse_level(next().c_str());
121        else throw std::runtime_error("未知参数: " + a);
122    }
123    return c;
124}

128int main(int argc, char** argv) {
129    ServerConfig conf;
130    try {
131        conf = parse_args(argc, argv);
132    } catch (const std::exception& e) {
133        std::cerr << "参数错误: " << e.what() << "\n";
134        return 2;
135    }
136
137    try {
138        onetrans::set_log_level(conf.log_level);
139        ONETRANS_INFO("log level=%d", conf.log_level);
140        // ---- 装配：权重 → 引擎 → 数据面 → 桥 → SEDA 编排 → 接入 ---- //
141        auto store = std::make_shared<ArtifactStore>();
142        store->load(conf.weights_dir, "weights.bin");
143        auto model = std::make_shared<OneTransModel>(OneTransModel::load(*store));
144        auto frontend = std::make_shared<EmbeddingFrontend>(EmbeddingFrontend::load(*store));
145        auto tables = std::make_shared<EmbeddingTables>(EmbeddingTables::load(*store));
146        // 稀疏查表数据面（D2）：默认进程内直查；--embedding-source ps 切 PS(brpc) 拉表。
147        LookupFn lookup_fn = tables->lookup_fn();
148#ifdef ONETRANS_WITH_PS
149        std::shared_ptr<PsLookupClient> ps_client;
150        if (conf.embedding_source == "ps") {
151            ps_client = std::make_shared<PsLookupClient>(conf.ps_server, conf.model_version);
152            if (!ps_client->available())
153                throw std::runtime_error("PS 客户端初始化失败: " + conf.ps_server);
154            lookup_fn = ps_client->lookup_fn();
155            std::cout << "[onetrans] embedding source = PS(" << conf.ps_server
156                      << ", table prefix=" << conf.model_version << ")\n";
157        }
158#endif
159        // KV 后端选择（§1.4/§1.5）：datasystem（C++ SDK）或进程内 local 回退。
160        std::shared_ptr<KVStore> kv_store;
161#ifdef ONETRANS_WITH_DATASYSTEM
162        if (conf.kv_backend == "datasystem") {
163            DatasystemKVStore::Options ds_opt;
164            ds_opt.host = conf.datasystem_host;
165            ds_opt.port = conf.datasystem_port;
166            ds_opt.default_ttl_seconds = conf.kv_ttl_seconds;
167            kv_store = std::make_shared<DatasystemKVStore>(ds_opt);
168        }
169#endif
170        if (!kv_store) {
171            kv_store = std::make_shared<LocalKVStore>(conf.kv_ttl_seconds);
172        }
173        auto metrics = std::make_shared<Metrics>();
174        auto runner = std::make_shared<TwoStageRunner>(*model);
175        Router router(conf.shards);
176
177        // Python 计算桥（Stage II 算子下发；cpp 后端时跳过启动）
178        auto bridge = std::make_shared<PythonComputeBridge>();
179        bool want_python = conf.compute_backend == "python" || conf.compute_backend == "auto";
180        if (want_python) {
181            if (bridge->start(conf.bridge_module_dir, conf.weights_dir,
182                              conf.bridge_init_timeout_s)) {
183                std::cout << "[onetrans] python bridge ready (Stage II → PyTorch)\n";
184            } else if (conf.compute_backend == "python") {
185                std::cerr << "[onetrans] python bridge 启动失败: " << bridge->last_error()
186                          << "\n";
187                return 1;
188            } else {
189                std::cerr << "[onetrans] python bridge 不可用（降级 C++ CPU）: "
190                          << bridge->last_error() << "\n";
191            }
192        }
193
194        // SEDA 流水线（§7.4.3：lookup → encode → kv → batch → python/cpp 计算）
195        ScoreFlow::Config fc;
196        fc.lookup_threads = conf.lookup_threads;
197        fc.encode_threads = conf.encode_threads;
198        fc.kv_threads = conf.kv_threads;
199        fc.compute_threads = conf.compute_threads;
200        fc.queue_cap = conf.queue_cap;
201        fc.max_batch = conf.max_batch;
202        fc.batch_wait_ms = conf.max_wait_ms;
203        ScoreFlow flow(*frontend, *runner, *kv_store, lookup_fn, *metrics,
204                      want_python ? bridge.get() : nullptr, conf.model_version, fc);
205        flow.start();
206
207        // Nearline（§7.4.4：全 C++ CPU，Stage I Compute Pool 异步执行）
208        ExecutorSet nearline_execs;
209        auto nearline_pool =
210            nearline_execs.make("stage1_compute", conf.nearline_threads, conf.queue_cap, false);
211        NearlineWorker nearline(*frontend, *runner, *kv_store, *metrics, lookup_fn,
212                                conf.model_version);
213
214        // PaiRec 精排 /rank 自装配：PaiRec 只给 user_id + item_id[]，dense/id 从这里装
215        RankAssembler assembler(frontend->config().n_user_dense, frontend->config().n_cand_dense);
216        assembler.load_item_lookup(conf.rank_item_lookup);
217        assembler.load_user_features(conf.rank_user_features);
218        std::cout << "[onetrans] rank assembler loaded: items=" << assembler.item_size()
219                  << " users=" << assembler.user_size()
220                  << " model_role=" << conf.model_role << "\n";
221
222        HttpServer server(conf.host, conf.port, conf.http_threads);
223
224        server.route("GET", "/healthz", [&](const HttpRequest&) {
225            JsonValue j = JsonValue::object();
226            j.set("status", JsonValue::str("ok"));
227            j.set("model_version", JsonValue::str(conf.model_version));
228            j.set("compute_backend", JsonValue::str(flow.backend()));
229            j.set("kv_objects", JsonValue::number(static_cast<double>(kv_store->size())));
230            return HttpResponse::json(200, j.dump());
231        });
232
233        // 异步路由：接入线程提交后立即释放（响应由完成线程写回，§7.4.3）
234        server.route_async("POST", "/ingest", [&](const HttpRequest& req, auto done) {
235            try {
236                IngestInput in = parse_ingest_input(req.body);
237                int shard = router.route(in.user_id);
238                metrics->count("ingest.qps");
239                nearline_pool->add([&nearline, in = std::move(in), shard, done]() mutable {
240                    IngestResult res = nearline.ingest(in);
241                    JsonValue j = JsonValue::object();
242                    j.set("accepted", JsonValue::boolean(res.accepted));
243                    j.set("shard", JsonValue::number(static_cast<double>(shard)));
244                    j.set("checksum", JsonValue::str(res.checksum));
245                    j.set("reason", JsonValue::str(res.reason));
246                    j.set("ingest_us", JsonValue::number(double(res.ingest_us)));
247                    done(HttpResponse::json(res.accepted ? 200 : 400, j.dump()));
248                    ONETRANS_DEBUG("ingest user=%s accepted=%d shard=%d ingest_us=%ld reason=%s",
249                                   in.user_id.c_str(), (int)res.accepted, shard,
250                                   (long)res.ingest_us, res.reason.c_str());
251                });
252            } catch (const std::exception& e) {
253                done(HttpResponse::json(400, std::string("{\"error\":\"") + e.what() + "\"}"));
254            }
255        });
256
257        server.route_async("POST", "/score", [&](const HttpRequest& req, auto done) {
258            try {
259                ScoreInput in = parse_score_input(req.body);
260                std::string user_id = in.user_id;
261                int shard = router.route(user_id);
262                // 回调式：流水线完成线程直接驱动响应写回（不占接入线程）
263                flow.submit(
264                    std::move(in),
265                    [done, user_id = std::move(user_id), shard](ScoreOutcome out,
266                                                                const std::string& err) mutable {
267                        if (!err.empty()) {
268                            done(HttpResponse::json(
269                                503, std::string("{\"error\":\"") + err + "\"}"));
270                            return;
271                        }
272                        JsonValue j = JsonValue::object();
273                        j.set("user_id", JsonValue::str(user_id));
274                        j.set("shard", JsonValue::number(static_cast<double>(shard)));
275                        j.set("kv_hit", JsonValue::boolean(out.kv_hit));
276                        j.set("logits", json_parse(logits_to_json(out.logits)));
277                        done(HttpResponse::json(200, j.dump()));
278                    });
279            } catch (const std::exception& e) {
280                done(HttpResponse::json(400, std::string("{\"error\":\"") + e.what() + "\"}"));
281            }
282        });
283
284        // PaiRec 精排 /rank（契约见 docs/pairec_rank_integration_plan.md §3.2）：
285        // 请求 {request_id, user_id, items:[{item_id}]}，自装配 → Stage II → sigmoid → 回填。
286        server.route_async("POST", "/rank", [&](const HttpRequest& req, auto done) {
287            try {
288                PaiRecRankRequest rank = parse_rank_request(req.body);
289                std::string user_id = rank.user_id;
290                int shard = router.route(user_id);
291                ScoreInput in = assembler.assemble(user_id, rank.item_ids);
292                flow.submit(
293                    std::move(in),
294                    [done, rank = std::move(rank), user_id = std::move(user_id), shard,
295                     model_version = conf.model_version, model_role = conf.model_role](
296                        ScoreOutcome out, const std::string& err) mutable {
297                        if (!err.empty()) {
298                            done(HttpResponse::json(
299                                503, std::string("{\"code\":503,\"error\":\"") + err + "\"}"));
300                            return;
301                        }
302                        const int64_t M = out.logits.shape[0];
303                        const int64_t T = out.logits.shape[1] > 0 ? out.logits.shape[1] : 1;
304                        JsonValue items = JsonValue::array({});
305                        std::size_t distinct = 0;
306                        float score_prev = 0.0f;
307                        for (int64_t m = 0; m < M; ++m) {
308                            float logit = out.logits.at(m * T);  // 首类 logit 作排序分数
309                            float score = 1.0f / (1.0f + std::exp(-logit));
310                            if (m == 0 || score != score_prev) ++distinct;
311                            score_prev = score;
312                            JsonValue it = JsonValue::object();
313                            it.set("item_id", JsonValue::str(std::to_string(rank.item_ids[m])));
314                            it.set("score", JsonValue::number(score));
315                            items.push_back(it);
316                        }
317                        JsonValue resp = JsonValue::object();
318                        resp.set("code", JsonValue::number(200));
319                        resp.set("msg", JsonValue::str("ok"));
320                        resp.set("request_id", JsonValue::str(rank.request_id));
321                        resp.set("model_version", JsonValue::str(model_version));
322                        resp.set("model_role", JsonValue::str(model_role));
323                        resp.set("items", items);
324                        JsonValue trace = JsonValue::object();
325                        trace.set("kv_hit", JsonValue::boolean(out.kv_hit));
326                        trace.set("score_unique_count", JsonValue::number(double(distinct)));
327                        trace.set("feature_us", JsonValue::number(double(out.feature_us)));
328                        trace.set("compute_us", JsonValue::number(double(out.compute_us)));
329                        trace.set("backend_rpc_us", JsonValue::number(double(out.backend_rpc_us)));
330                        trace.set("lookup_us", JsonValue::number(double(out.lookup_us)));
331                        trace.set("encode_us", JsonValue::number(double(out.encode_us)));
332                        trace.set("kv_us", JsonValue::number(double(out.kv_us)));
333                        trace.set("batch_wait_us", JsonValue::number(double(out.batch_wait_us)));
334                        trace.set("batch_compute_us", JsonValue::number(double(out.batch_compute_us)));
335                        trace.set("batch_size", JsonValue::number(double(out.batch_size)));
336                        trace.set("n_candidates",
337                                  JsonValue::number(double(rank.item_ids.size())));
338                        resp.set("trace", trace);
339                        done(HttpResponse::json(200, resp.dump()));
340                        ONETRANS_DEBUG(
341                            "rank user=%s items=%zu kv_hit=%d distinct=%zu "
342                            "feature_us=%ld compute_us=%ld backend_rpc_us=%ld batch=%d",
343                            user_id.c_str(), rank.item_ids.size(), (int)out.kv_hit, distinct,
344                            (long)out.feature_us, (long)out.compute_us, (long)out.backend_rpc_us,
345                            (int)out.batch_size);
346                    });
347            } catch (const std::exception& e) {
348                done(HttpResponse::json(400, std::string("{\"code\":400,\"error\":\"") + e.what() +
349                                               "\"}"));
350            }
351        });
352
353        server.route("GET", "/metrics", [&](const HttpRequest&) {
354            return HttpResponse::text(200, metrics->dump());
355        });
356
357        std::cout << "[onetrans] listening on " << conf.host << ":" << conf.port
358                  << " model=" << conf.model_version << " shards=" << conf.shards
359                  << " batch<=" << conf.max_batch << " wait=" << conf.max_wait_ms << "ms"
360                  << " backend=" << (want_python ? "auto→" : "cpp") << flow.backend() << "\n";
361        server.run();
362        flow.stop();
363        nearline_execs.stop_all();
364        bridge->stop();
365    } catch (const std::exception& e) {
366        std::cerr << "[onetrans] 启动失败: " << e.what() << "\n";
367        return 1;
368    }
369    return 0;
