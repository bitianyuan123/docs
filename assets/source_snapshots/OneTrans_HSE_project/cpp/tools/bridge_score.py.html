1"""Stage II 计算桥（PyTorch 算子下发，§7.4.5）。
2
3由 C++ PythonComputeBridge 进程内嵌入调用（嵌入式解释器，非子进程）：
4
5- ``init(weights_dir)``：解析 weights.bin（与 C++ ArtifactStore 同一格式）→ 构建
6  OneTrans backbone + TwoStageRunner，权重常驻 GPU（无 CUDA 时 CPU）。
7- ``score_batch(kv_blobs, row_map, ns_blob, n_rows, ns, d)``：批量交叉注意力前向
8  （C++ 已完成 lookup/encode_ns/KV mget/攒批，本函数只做最终计算下发）。
9
10数值契约：与 C++ ``TwoStageRunner::score_ns_batch`` 逐位对拍（e2e_test 守护，
11golden 阈值 1e-5）。UserKV payload 为 C++/Python 二进制兼容格式（serialize.py）。
12"""
13
14from __future__ import annotations
15
16import json
17from pathlib import Path
18
19import numpy as np
20import torch
21
22from onetrans.models.one_trans import OneTrans
23from onetrans.serving.serialize import deserialize_with_meta
24from onetrans.serving.two_stage import TwoStageRunner, UserKV
25
26_DEVICE = "cuda" if torch.cuda.is_available() else "cpu"
27_runner: TwoStageRunner | None = None
28_META: dict = {}
29
30
31def _load_state_dict(weights_dir: Path) -> dict[str, torch.Tensor]:
32    manifest = json.loads((weights_dir / "manifest.json").read_text(encoding="utf-8"))
33    blob = np.fromfile(weights_dir / "weights.bin", dtype="<f4")
34    sd: dict[str, torch.Tensor] = {}
35    for name, meta in manifest["tensors"].items():
36        n = 1
37        for s in meta["shape"]:
38            n *= s
39        flat = blob[meta["offset"] : meta["offset"] + n].copy()
40        sd[name] = torch.from_numpy(flat).view(*meta["shape"])
41    return sd
42
43
44def init(weights_dir: str) -> dict:
45    """加载权重并构建 runner（幂等；返回运行元数据供 C++ 侧校验）。"""
46    global _runner, _META
47
48    wdir = Path(weights_dir)
49    cfg = json.loads((wdir / "manifest.json").read_text(encoding="utf-8"))["config"]
50    sd = _load_state_dict(wdir)
51
52    backbone = OneTrans(
53        d_model=cfg["d_model"],
54        num_blocks=cfg["num_blocks"],
55        num_heads=cfg["num_heads"],
56        max_seq_len=cfg["max_seq_len"],
57        min_seq_len=cfg["min_seq_len"],
58        ns_tokens_num=cfg["ns_tokens_num"],
59        dropout=0.0,
60        use_cls_token=cfg["use_cls_token"],
61    )
62    # manifest 中 backbone.* 前缀即本模块 state_dict 键（dims 为普通属性不进 state_dict）
63    backbone_sd = {k[len("backbone.") :]: v for k, v in sd.items() if k.startswith("backbone.")}
64    missing, unexpected = backbone.load_state_dict(backbone_sd, strict=False)
65    if unexpected or missing:
66        raise RuntimeError(f"bridge: 权重不完整 missing={missing} unexpected={unexpected}")
67    backbone = backbone.to(_DEVICE).eval()
68
69    _runner = TwoStageRunner(backbone)
70    _META = {
71        "device": _DEVICE,
72        "d_model": cfg["d_model"],
73        "num_blocks": cfg["num_blocks"],
74        "ns_tokens_num": cfg["ns_tokens_num"],
75        "dims": cfg["dims"],
76    }
77    return dict(_META)
78
79
80def meta() -> dict:
81    return dict(_META)

84def _kv_from_payload(payload: bytes) -> UserKV:
85    per_layer, s_len, per_layer_len = deserialize_with_meta(payload)
86    return UserKV(
87        per_layer=[(k.to(_DEVICE), v.to(_DEVICE)) for k, v in per_layer],
88        per_layer_len=per_layer_len,
89        s_len=s_len,
90    )
91

93def score_batch(
94    kv_blobs: list[bytes],
95    row_map: list[int],
96    ns_blob: bytes,
97    n_rows: int,
98    ns: int,
99    d: int,
100) -> bytes:
101    """批量 Stage II 前向。
102
103    :param kv_blobs: 去重后的 UserKV payload 列表（C++ kv_serialize 格式）
104    :param row_map: 每候选行 → kv_blobs 下标（长度 == n_rows；同用户多候选共享 KV）
105    :param ns_blob: fp32 小端连续字节，[n_rows, ns, d]（C++ Frontend Encode Pool 产出）
106    :return: logits fp32 小端字节 [n_rows, T]
107    """
108    if _runner is None:
109        raise RuntimeError("bridge: 未初始化（先调 init）")
110
111    ns_emb = (
112        torch.frombuffer(bytearray(ns_blob), dtype=torch.float32)
113        .reshape(n_rows, ns, d)
114        .to(_DEVICE)
115    )
116
117    cache: dict[int, UserKV] = {}
118    kvs: list[UserKV] = []
119    for idx in row_map:
120        kv = cache.get(idx)
121        if kv is None:
122            kv = _kv_from_payload(kv_blobs[idx])
123            cache[idx] = kv
124        kvs.append(kv)
125
126    with torch.no_grad():
127        logits = _runner.score_ns_batch(kvs, ns_emb)
128    return logits.detach().to("cpu").contiguous().numpy().tobytes()
