49# ---- KV 后端：可选 datasystem C++ SDK（§1.5）---- #
50# ONETRANS_WITH_DATASYSTEM=ON 且 find_package(Datasystem) 命中 SDK 时才编译
51# DatasystemKVStore（调 datasystem::KVClient set/get），否则热路径回退 LocalKVStore。
52option(ONETRANS_WITH_DATASYSTEM "datasystem C++ SDK 后端（KV set/get）" OFF)
53option(ONETRANS_DATASYSTEM_MOCK
54    "构建 DataSystem mock SDK + mock server（无真实 wheel 时本地验证接入链路；隐含 ONETRANS_WITH_DATASYSTEM=ON）" OFF)
55if(ONETRANS_DATASYSTEM_MOCK)
56    set(ONETRANS_WITH_DATASYSTEM ON)
57endif()
58if(ONETRANS_WITH_DATASYSTEM)
59    if(ONETRANS_DATASYSTEM_MOCK)
60        # mock SDK（API 兼容子集）+ datasystem_mock_server：先跑通装配/运行/验证链路，
61        # 真实 wheel 到位后去掉本开关重编即可。
62        add_subdirectory(mocks/datasystem)
63        set(Datasystem_FOUND TRUE)
64        set(Datasystem_INCLUDE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/mocks/datasystem/include)
65        set(Datasystem_LIBRARIES datasystem)  # CMake target（产物在 <build>/datasystem-mock/lib）
66        set(Datasystem_RUNTIME_LIB_DIR ${CMAKE_BINARY_DIR}/datasystem-mock/lib)
67        message(STATUS "datasystem: 使用 mock SDK（配合 ./datasystem_mock_server 验证接入链路）")
68    else()
69        # FindDatasystem.cmake（本仓库 cpp/cmake/），支持 -DDATASYSTEM_ROOT=<SDK根目录>
70        list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)
71        find_package(Datasystem QUIET)
72        if(NOT Datasystem_FOUND)
73            message(WARNING "datasystem: ONETRANS_WITH_DATASYSTEM=ON 但未找到 SDK，回退 LocalKVStore")
74        else()
75            # protobuf 单一化：SDK 自带 libprotobuf.so.25.5.0（全版本 SONAME）与主链
76            # 系统库 libprotobuf.so.25 若同时装载，同进程双运行时会导致 descriptor
77            # 重复注册 FATAL（File already exists in database）。
78            # 且 25.5.0 是 25.0.0 的符号超集（Any::MergeImpl 等新符号系统库缺失），
79            # 故统一方向：全部软链 SDK 库，并把 libprotobuf.so.25 也指向 SDK 的
80            # 25.5.0 —— 两个 NEEDED 解析到同一 inode，进程内只装一份 25.5.0；
81            # brpc/ps-gen（按 25.0 头编译）调用的旧符号在 25.5.0 中均存在。
82            set(_DS_COMPAT ${CMAKE_BINARY_DIR}/datasystem-compat)
83            file(MAKE_DIRECTORY ${_DS_COMPAT})
84            file(GLOB _DS_LIB_FILES "${Datasystem_RUNTIME_LIB_DIR}/*")
85            foreach(_f ${_DS_LIB_FILES})
86                get_filename_component(_n "${_f}" NAME)
87                file(REMOVE "${_DS_COMPAT}/${_n}")
88                file(CREATE_LINK "${_f}" "${_DS_COMPAT}/${_n}" SYMBOLIC)
89            endforeach()
90            if(EXISTS "${Datasystem_RUNTIME_LIB_DIR}/libprotobuf.so.25.5.0")
91                file(REMOVE "${_DS_COMPAT}/libprotobuf.so.25")
92                file(CREATE_LINK "${Datasystem_RUNTIME_LIB_DIR}/libprotobuf.so.25.5.0"
93                    "${_DS_COMPAT}/libprotobuf.so.25" SYMBOLIC)
94                set(Datasystem_RUNTIME_LIB_DIR ${_DS_COMPAT})
95                message(STATUS "datasystem: protobuf 统一为 SDK 25.5.0（compat=${_DS_COMPAT}）")
96            else()
97                message(WARNING "datasystem: SDK 无 libprotobuf.so.25.5.0，跳过去重")
98            endif()
99        endif()
100    endif()
101    if(Datasystem_FOUND)
102        list(APPEND CORE_SOURCES src/kv/datasystem_store.cpp)
103        if(NOT ONETRANS_DATASYSTEM_MOCK)
104            message(STATUS "datasystem: SDK 命中，启用 DatasystemKVStore")
105        endif()
106    endif()
107endif()

109# ---- PS 稀疏数据面（D2）：可选接入参数服务器 ---- #
110# ONETRANS_WITH_PS=ON 且 pkg-config 命中 brpc 时，启用 PsLookupClient；
111# server_main 以 --embedding-source ps 切换。否则编译为降级（不链 brpc，
112# 热路径走进程内 EmbeddingTables 直查）。brpc 为静态库，依赖见 libbrpc.a 的 brpc.pc。
113option(ONETRANS_WITH_PS "PS 稀疏参数服务器（brpc）查表数据面" OFF)
114set(BRPC_PKG_CONFIG_DIR "" CACHE PATH "brpc 的 pkg-config 目录（含 brpc.pc）")
115if(ONETRANS_WITH_PS)
116    find_package(PkgConfig REQUIRED)
117    set(_old_pc "$ENV{PKG_CONFIG_PATH}")
118    if(BRPC_PKG_CONFIG_DIR)
119        list(PREPEND CMAKE_PREFIX_PATH "${BRPC_PKG_CONFIG_DIR}/../../")
120        set(ENV{PKG_CONFIG_PATH} "${BRPC_PKG_CONFIG_DIR}:$ENV{PKG_CONFIG_PATH}")
121    endif()
122    pkg_check_modules(BRPC QUIET IMPORTED_TARGET brpc)
123    find_package(Protobuf REQUIRED)
124    set(_ps_gen "${CMAKE_BINARY_DIR}/ps-gen")
125    file(MAKE_DIRECTORY "${_ps_gen}")
126    protobuf_generate_cpp(_PS_PB_SRC _PS_PB_HDR
127        APPEND_PATH
128        PROTOC_OUT_DIR "${_ps_gen}"
129        PROTOS "${CMAKE_CURRENT_SOURCE_DIR}/../deploy/ps/embedding_service.proto")
130    set(ENV{PKG_CONFIG_PATH} "${_old_pc}")
131    if(BRPC_FOUND)
132        list(APPEND CORE_SOURCES src/serving/ps_client.cpp ${_PS_PB_SRC})
133        message(STATUS "PS: brpc 命中，启用 PsLookupClient（--embedding-source ps）")
134    else()
135        message(WARNING "PS: ONETRANS_WITH_PS=ON 但未找到 brpc，禁用 PS 数据面")
136    endif()
137endif()

143# ---- PS 链接（仅 conf.embedding_source=ps 路径需要；不链外部 glog，见 deploy/ps）---- #
144if(ONETRANS_WITH_PS AND BRPC_FOUND)
145    target_include_directories(onetrans_core PUBLIC
146        "${_ps_gen}" ${BRPC_INCLUDE_DIRS} ${Protobuf_INCLUDE_DIRS})
147    target_compile_definitions(onetrans_core PUBLIC ONETRANS_WITH_PS=1)
148    target_link_directories(onetrans_core PUBLIC ${BRPC_LIBRARY_DIRS})
149    # 注：不链 protoc（protobuf 编译器库）——pb 生成代码只用 libprotobuf 运行时；
150    # 且系统 libprotoc.so.25 依赖 SDK 25.5.0 已移除的 StringPiece::npos 符号，
151    # 与 SDK 自带 protobuf 单一化（上方 compat 软链）冲突。
152    #
153    # abseil：本机 libprotobuf(4.25)/glog 均基于 abseil-lts 构建，而 brpc.pc 的
154    # Libs.private 未列出 abseil。仅按名链 protobuf 时不会带入 abseil，导致
155    # ps_client.cpp.o 引用 absl::log_internal::MakeCheckOpString 时报
156    # "libabsl_log_internal_*: DSO missing from command line"。
157    # 解法：用 pkg-config --static 展开 protobuf 的 Requires.private（其中列出 abseil
158    # 各组件）拼出 abseil 链接闭包，避免手写版本相关的 -labsl_*。
159    # （不能用 brpc 自身 STATIC 展开，那会带入我们刻意排除的 -lprotoc）
160    # 注：pkg_check_modules 的 STATIC 关键字在 pkgconf 环境下不会返回私库依赖
161    # （实测 <prefix>_FOUND 为空），故改走 execute_process 直接抓 pkg-config 输出。
162    set(_absl_link "")
163    if(PKG_CONFIG_EXECUTABLE)
164        execute_process(
165            COMMAND ${PKG_CONFIG_EXECUTABLE} --static --libs protobuf
166            RESULT_VARIABLE _absl_rc
167            OUTPUT_VARIABLE _absl_raw
168            OUTPUT_STRIP_TRAILING_WHITESPACE)
169        if(_absl_rc EQUAL 0 AND _absl_raw)
170            separate_arguments(_absl_link UNIX_COMMAND "${_absl_raw}")
171            message(STATUS "PS: abseil 闭包已纳入链接（pkg-config --static --libs protobuf）")
172        endif()
173    endif()
174    if(NOT _absl_link)
175        message(WARNING "PS: 无法解析 abseil 链接闭包，若链接报 absl 缺符号请检查 protobuf.pc")
176    endif()
177    target_link_libraries(onetrans_core PUBLIC
178        ${BRPC_LIBRARIES} glog protobuf gflags leveldb ssl crypto dl z rt pthread
179        ${_absl_link})
180endif()

197# ---- Python 计算桥（可选，§7.4.5）：嵌入解释器下发 PyTorch 算子 ---- #
198# ONETRANS_PYTHON_BRIDGE=ON 且找到 Python3 Development 时启用；
199# 否则编译为降级桩（available()==false，Stage II 走 C++ CPU）。
200option(ONETRANS_PYTHON_BRIDGE "嵌入 Python 计算桥（Stage II → PyTorch）" ON)
201if(ONETRANS_PYTHON_BRIDGE)
202    # 优先用当前 PATH 里的 python3（与 torch 安装解释器一致），
203    # 可 -DPython3_EXECUTABLE=... 显式覆盖
204    if(NOT Python3_EXECUTABLE)
205        execute_process(COMMAND python3 -c "import sys; print(sys.executable)"
206                        OUTPUT_VARIABLE _PY_EXE OUTPUT_STRIP_TRAILING_WHITESPACE)
207        if(_PY_EXE)
208            set(Python3_EXECUTABLE ${_PY_EXE})
209        endif()
210    endif()
211    find_package(Python3 COMPONENTS Interpreter Development)
212    if(Python3_Development_FOUND)
213        message(STATUS "python bridge: ${Python3_EXECUTABLE} (${Python3_VERSION})")
214        target_compile_definitions(onetrans_core PUBLIC ONETRANS_WITH_PYTHON=1)
215        target_link_libraries(onetrans_core PUBLIC Python3::Python)
216    else()
217        message(STATUS "python bridge: 未找到 Python3 Development（降级 C++ CPU）")
218    endif()
219endif()
