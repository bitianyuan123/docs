88/* This function links the client to the global linked list of clients.
89 * unlinkClient() does the opposite, among other things. */
90void linkClient(client *c) {
91    listAddNodeTail(server.clients,c);
92    /* Note that we remember the linked list node where the client is stored,
93     * this way removing the client in unlinkClient() will not require
94     * a linear scan, but just a constant time operation. */
95    c->client_list_node = listLast(server.clients);
96    uint64_t id = htonu64(c->id);
97    raxInsert(server.clients_index,(unsigned char*)&id,sizeof(id),c,NULL);
98}

220void installClientWriteHandler(client *c) {
221    int ae_barrier = 0;
222    /* For the fsync=always policy, we want that a given FD is never
223     * served for reading and writing in the same event loop iteration,
224     * so that in the middle of receiving the query, and serving it
225     * to the client, we'll call beforeSleep() that will do the
226     * actual fsync of AOF to disk. the write barrier ensures that. */
227    if (server.aof_state == AOF_ON &&
228        server.aof_fsync == AOF_FSYNC_ALWAYS)
229    {
230        ae_barrier = 1;
231    }
232    if (connSetWriteHandlerWithBarrier(c->conn, sendReplyToClient, ae_barrier) == C_ERR) {
233        freeClientAsync(c);
234    }
235}

237/* This function puts the client in the queue of clients that should write
238 * their output buffers to the socket. Note that it does not *yet* install
239 * the write handler, to start clients are put in a queue of clients that need
240 * to write, so we try to do that before returning in the event loop (see the
241 * handleClientsWithPendingWrites() function).
242 * If we fail and there is more data to write, compared to what the socket
243 * buffers can hold, then we'll really install the handler. */
244void putClientInPendingWriteQueue(client *c) {
245    /* Schedule the client to write the output buffers to the socket only
246     * if not already done and, for slaves, if the slave can actually receive
247     * writes at this stage. */
248    if (!(c->flags & CLIENT_PENDING_WRITE) &&
249        (c->replstate == REPL_STATE_NONE ||
250         (c->replstate == SLAVE_STATE_ONLINE && !c->repl_start_cmd_stream_on_ack)))
251    {
252        /* Here instead of installing the write handler, we just flag the
253         * client and put it into a list of clients that have something
254         * to write to the socket. This way before re-entering the event
255         * loop, we can try to directly write to the client sockets avoiding
256         * a system call. We'll only really install the write handler if
257         * we'll not be able to write the whole reply at once. */
258        c->flags |= CLIENT_PENDING_WRITE;
259        listLinkNodeHead(server.clients_pending_write, &c->clients_pending_write_node);
260    }
261}

263/* This function is called every time we are going to transmit new data
264 * to the client. The behavior is the following:
265 *
266 * If the client should receive new data (normal clients will) the function
267 * returns C_OK, and make sure to install the write handler in our event
268 * loop so that when the socket is writable new data gets written.
269 *
270 * If the client should not receive new data, because it is a fake client
271 * (used to load AOF in memory), a master or because the setup of the write
272 * handler failed, the function returns C_ERR.
273 *
274 * The function may return C_OK without actually installing the write
275 * event handler in the following cases:
276 *
277 * 1) The event handler should already be installed since the output buffer
278 *    already contains something.
279 * 2) The client is a slave but not yet online, so we want to just accumulate
280 *    writes in the buffer but not actually sending them yet.
281 *
282 * Typically gets called every time a reply is built, before adding more
283 * data to the clients output buffers. If the function returns C_ERR no
284 * data should be appended to the output buffers. */
285int prepareClientToWrite(client *c) {
286    /* If it's the Lua client we always return ok without installing any
287     * handler since there is no socket at all. */
288    if (c->flags & (CLIENT_SCRIPT|CLIENT_MODULE)) return C_OK;
289
290    /* If CLIENT_CLOSE_ASAP flag is set, we need not write anything. */
291    if (c->flags & CLIENT_CLOSE_ASAP) return C_ERR;
292
293    /* CLIENT REPLY OFF / SKIP handling: don't send replies.
294     * CLIENT_PUSHING handling: disables the reply silencing flags. */
295    if ((c->flags & (CLIENT_REPLY_OFF|CLIENT_REPLY_SKIP)) &&
296        !(c->flags & CLIENT_PUSHING)) return C_ERR;
297
298    /* Masters don't receive replies, unless CLIENT_MASTER_FORCE_REPLY flag
299     * is set. */
300    if ((c->flags & CLIENT_MASTER) &&
301        !(c->flags & CLIENT_MASTER_FORCE_REPLY)) return C_ERR;
302
303    if (!c->conn) return C_ERR; /* Fake client for AOF loading. */
304
305    /* Schedule the client to write the output buffers to the socket, unless
306     * it should already be setup to do so (it has already pending data).
307     *
308     * If CLIENT_PENDING_READ is set, we're in an IO thread and should
309     * not put the client in pending write queue. Instead, it will be
310     * done by handleClientsWithPendingReadsUsingThreads() upon return.
311     */
312    if (!clientHasPendingReplies(c) && io_threads_op == IO_THREADS_OP_IDLE)
313        putClientInPendingWriteQueue(c);
314
315    /* Authorize the caller to queue in the output buffer of this client. */
316    return C_OK;
317}

323/* Attempts to add the reply to the static buffer in the client struct.
324 * Returns the length of data that is added to the reply buffer.
325 *
326 * Sanitizer suppression: client->buf_usable_size determined by
327 * zmalloc_usable_size() call. Writing beyond client->buf boundaries confuses
328 * sanitizer and generates a false positive out-of-bounds error */
329REDIS_NO_SANITIZE("bounds")
330size_t _addReplyToBuffer(client *c, const char *s, size_t len) {
331    size_t available = c->buf_usable_size - c->bufpos;
332
333    /* If there already are entries in the reply list, we cannot
334     * add anything more to the static buffer. */
335    if (listLength(c->reply) > 0) return 0;
336
337    size_t reply_len = len > available ? available : len;
338    memcpy(c->buf+c->bufpos,s,reply_len);
339    c->bufpos+=reply_len;
340    /* We update the buffer peak after appending the reply to the buffer */
341    if(c->buf_peak < (size_t)c->bufpos)
342        c->buf_peak = (size_t)c->bufpos;
343    return reply_len;
344}

346/* Adds the reply to the reply linked list.
347 * Note: some edits to this function need to be relayed to AddReplyFromClient. */
348void _addReplyProtoToList(client *c, list *reply_list, const char *s, size_t len) {
349    listNode *ln = listLast(reply_list);
350    clientReplyBlock *tail = ln? listNodeValue(ln): NULL;
351
352    /* Note that 'tail' may be NULL even if we have a tail node, because when
353     * addReplyDeferredLen() is used, it sets a dummy node to NULL just
354     * to fill it later, when the size of the bulk length is set. */
355
356    /* Append to tail string when possible. */
357    if (tail) {
358        /* Copy the part we can fit into the tail, and leave the rest for a
359         * new node */
360        size_t avail = tail->size - tail->used;
361        size_t copy = avail >= len? len: avail;
362        memcpy(tail->buf + tail->used, s, copy);
363        tail->used += copy;
364        s += copy;
365        len -= copy;
366    }
367    if (len) {
368        /* Create a new node, make sure it is allocated to at
369         * least PROTO_REPLY_CHUNK_BYTES */
370        size_t usable_size;
371        size_t size = len < PROTO_REPLY_CHUNK_BYTES? PROTO_REPLY_CHUNK_BYTES: len;
372        tail = zmalloc_usable(size + sizeof(clientReplyBlock), &usable_size);
373        /* take over the allocation's internal fragmentation */
374        tail->size = usable_size - sizeof(clientReplyBlock);
375        tail->used = len;
376        memcpy(tail->buf, s, len);
377        listAddNodeTail(reply_list, tail);
378        c->reply_bytes += tail->size;
379
380        closeClientOnOutputBufferLimitReached(c, 1);
381    }
382}

384/* The subscribe / unsubscribe command family has a push as a reply,
385 * or in other words, it responds with a push (or several of them
386 * depending on how many arguments it got), and has no reply. */
387int cmdHasPushAsReply(struct redisCommand *cmd) {
388    if (!cmd) return 0;
389    return cmd->proc == subscribeCommand  || cmd->proc == unsubscribeCommand ||
390           cmd->proc == psubscribeCommand || cmd->proc == punsubscribeCommand ||
391           cmd->proc == ssubscribeCommand || cmd->proc == sunsubscribeCommand;
392}

394void _addReplyToBufferOrList(client *c, const char *s, size_t len) {
395    if (c->flags & CLIENT_CLOSE_AFTER_REPLY) return;
396
397    /* Replicas should normally not cause any writes to the reply buffer. In case a rogue replica sent a command on the
398     * replication link that caused a reply to be generated we'll simply disconnect it.
399     * Note this is the simplest way to check a command added a response. Replication links are used to write data but
400     * not for responses, so we should normally never get here on a replica client. */
401    if (getClientType(c) == CLIENT_TYPE_SLAVE) {
402        sds cmdname = c->lastcmd ? c->lastcmd->fullname : NULL;
403        logInvalidUseAndFreeClientAsync(c, "Replica generated a reply to command '%s'",
404                                        cmdname ? cmdname : "<unknown>");
405        return;
406    }
407
408    /* We call it here because this function may affect the reply
409     * buffer offset (see function comment) */
410    reqresSaveClientReplyOffset(c);
411
412    /* If we're processing a push message into the current client (i.e. executing PUBLISH
413     * to a channel which we are subscribed to, then we wanna postpone that message to be added
414     * after the command's reply (specifically important during multi-exec). the exception is
415     * the SUBSCRIBE command family, which (currently) have a push message instead of a proper reply.
416     * The check for executing_client also avoids affecting push messages that are part of eviction. */
417    if (c == server.current_client && (c->flags & CLIENT_PUSHING) &&
418        server.executing_client && !cmdHasPushAsReply(server.executing_client->cmd))
419    {
420        _addReplyProtoToList(c,server.pending_push_messages,s,len);
421        return;
422    }
423
424    size_t reply_len = _addReplyToBuffer(c,s,len);
425    if (len > reply_len) _addReplyProtoToList(c,c->reply,s+reply_len,len-reply_len);
426}

433/* Add the object 'obj' string representation to the client output buffer. */
434void addReply(client *c, robj *obj) {
435    if (prepareClientToWrite(c) != C_OK) return;
436
437    if (sdsEncodedObject(obj)) {
438        _addReplyToBufferOrList(c,obj->ptr,sdslen(obj->ptr));
439    } else if (obj->encoding == OBJ_ENCODING_INT) {
440        /* For integer encoded strings we just convert it into a string
441         * using our optimized function, and attach the resulting string
442         * to the output buffer. */
443        char buf[32];
444        size_t len = ll2string(buf,sizeof(buf),(long)obj->ptr);
445        _addReplyToBufferOrList(c,buf,len);
446    } else {
447        serverPanic("Wrong obj->encoding in addReply()");
448    }
449}

476/* Low level function called by the addReplyError...() functions.
477 * It emits the protocol for a Redis error, in the form:
478 *
479 * -ERRORCODE Error Message<CR><LF>
480 *
481 * If the error code is already passed in the string 's', the error
482 * code provided is used, otherwise the string "-ERR " for the generic
483 * error code is automatically added.
484 * Note that 's' must NOT end with \r\n. */
485void addReplyErrorLength(client *c, const char *s, size_t len) {
486    /* If the string already starts with "-..." then the error code
487     * is provided by the caller. Otherwise we use "-ERR". */
488    if (!len || s[0] != '-') addReplyProto(c,"-ERR ",5);
489    addReplyProto(c,s,len);
490    addReplyProto(c,"\r\n",2);
491}

493/* Do some actions after an error reply was sent (Log if needed, updates stats, etc.)
494 * Possible flags:
495 * * ERR_REPLY_FLAG_NO_STATS_UPDATE - indicate not to update any error stats. */
496void afterErrorReply(client *c, const char *s, size_t len, int flags) {
497    /* Module clients fall into two categories:
498     * Calls to RM_Call, in which case the error isn't being returned to a client, so should not be counted.
499     * Module thread safe context calls to RM_ReplyWithError, which will be added to a real client by the main thread later. */
500    if (c->flags & CLIENT_MODULE) {
501        if (!c->deferred_reply_errors) {
502            c->deferred_reply_errors = listCreate();
503            listSetFreeMethod(c->deferred_reply_errors, (void (*)(void*))sdsfree);
504        }
505        listAddNodeTail(c->deferred_reply_errors, sdsnewlen(s, len));
506        return;
507    }
508
509    if (!(flags & ERR_REPLY_FLAG_NO_STATS_UPDATE)) {
510        /* Increment the global error counter */
511        server.stat_total_error_replies++;
512        /* Increment the error stats
513         * If the string already starts with "-..." then the error prefix
514         * is provided by the caller ( we limit the search to 32 chars). Otherwise we use "-ERR". */
515        if (s[0] != '-') {
516            incrementErrorCount("ERR", 3);
517        } else {
518            char *spaceloc = memchr(s, ' ', len < 32 ? len : 32);
519            if (spaceloc) {
520                const size_t errEndPos = (size_t)(spaceloc - s);
521                incrementErrorCount(s+1, errEndPos-1);
522            } else {
523                /* Fallback to ERR if we can't retrieve the error prefix */
524                incrementErrorCount("ERR", 3);
525            }
526        }
527    } else {
528        /* stat_total_error_replies will not be updated, which means that
529         * the cmd stats will not be updated as well, we still want this command
530         * to be counted as failed so we update it here. We update c->realcmd in
531         * case c->cmd was changed (like in GEOADD). */
532        c->realcmd->failed_calls++;
533    }
534
535    /* Sometimes it could be normal that a slave replies to a master with
536     * an error and this function gets called. Actually the error will never
537     * be sent because addReply*() against master clients has no effect...
538     * A notable example is:
539     *
540     *    EVAL 'redis.call("incr",KEYS[1]); redis.call("nonexisting")' 1 x
541     *
542     * Where the master must propagate the first change even if the second
543     * will produce an error. However it is useful to log such events since
544     * they are rare and may hint at errors in a script or a bug in Redis. */
545    int ctype = getClientType(c);
546    if (ctype == CLIENT_TYPE_MASTER || ctype == CLIENT_TYPE_SLAVE || c->id == CLIENT_ID_AOF) {
547        char *to, *from;
548
549        if (c->id == CLIENT_ID_AOF) {
550            to = "AOF-loading-client";
551            from = "server";
552        } else if (ctype == CLIENT_TYPE_MASTER) {
553            to = "master";
554            from = "replica";
555        } else {
556            to = "replica";
557            from = "master";
558        }
559
560        if (len > 4096) len = 4096;
561        sds cmdname = c->lastcmd ? c->lastcmd->fullname : NULL;
562        serverLog(LL_WARNING,"== CRITICAL == This %s is sending an error "
563                             "to its %s: '%.*s' after processing the command "
564                             "'%s'", from, to, (int)len, s, cmdname ? cmdname : "<unknown>");
565        if (ctype == CLIENT_TYPE_MASTER && server.repl_backlog &&
566            server.repl_backlog->histlen > 0)
567        {
568            showLatestBacklog();
569        }
570        server.stat_unexpected_error_replies++;
571
572        /* Based off the propagation error behavior, check if we need to panic here. There
573         * are currently two checked cases:
574         * * If this command was from our master and we are not a writable replica.
575         * * We are reading from an AOF file. */
576        int panic_in_replicas = (ctype == CLIENT_TYPE_MASTER && server.repl_slave_ro)
577            && (server.propagation_error_behavior == PROPAGATION_ERR_BEHAVIOR_PANIC ||
578            server.propagation_error_behavior == PROPAGATION_ERR_BEHAVIOR_PANIC_ON_REPLICAS);
579        int panic_in_aof = c->id == CLIENT_ID_AOF 
580            && server.propagation_error_behavior == PROPAGATION_ERR_BEHAVIOR_PANIC;
581        if (panic_in_replicas || panic_in_aof) {
582            serverPanic("This %s panicked sending an error to its %s"
583                " after processing the command '%s'",
584                from, to, cmdname ? cmdname : "<unknown>");
585        }
586    }
587}

701/* Sometimes we are forced to create a new reply node, and we can't append to
702 * the previous one, when that happens, we wanna try to trim the unused space
703 * at the end of the last reply node which we won't use anymore. */
704void trimReplyUnusedTailSpace(client *c) {
705    listNode *ln = listLast(c->reply);
706    clientReplyBlock *tail = ln? listNodeValue(ln): NULL;
707
708    /* Note that 'tail' may be NULL even if we have a tail node, because when
709     * addReplyDeferredLen() is used */
710    if (!tail) return;
711
712    /* We only try to trim the space is relatively high (more than a 1/4 of the
713     * allocation), otherwise there's a high chance realloc will NOP.
714     * Also, to avoid large memmove which happens as part of realloc, we only do
715     * that if the used part is small.  */
716    if (tail->size - tail->used > tail->size / 4 &&
717        tail->used < PROTO_REPLY_CHUNK_BYTES)
718    {
719        size_t usable_size;
720        size_t old_size = tail->size;
721        tail = zrealloc_usable(tail, tail->used + sizeof(clientReplyBlock), &usable_size);
722        /* take over the allocation's internal fragmentation (at least for
723         * memory usage tracking) */
724        tail->size = usable_size - sizeof(clientReplyBlock);
725        c->reply_bytes = c->reply_bytes + tail->size - old_size;
726        listNodeValue(ln) = tail;
727    }
728}

730/* Adds an empty object to the reply list that will contain the multi bulk
731 * length, which is not known when this function is called. */
732void *addReplyDeferredLen(client *c) {
733    /* Note that we install the write event here even if the object is not
734     * ready to be sent, since we are sure that before returning to the
735     * event loop setDeferredAggregateLen() will be called. */
736    if (prepareClientToWrite(c) != C_OK) return NULL;
737
738    /* Replicas should normally not cause any writes to the reply buffer. In case a rogue replica sent a command on the
739     * replication link that caused a reply to be generated we'll simply disconnect it.
740     * Note this is the simplest way to check a command added a response. Replication links are used to write data but
741     * not for responses, so we should normally never get here on a replica client. */
742    if (getClientType(c) == CLIENT_TYPE_SLAVE) {
743        sds cmdname = c->lastcmd ? c->lastcmd->fullname : NULL;
744        logInvalidUseAndFreeClientAsync(c, "Replica generated a reply to command '%s'",
745                                        cmdname ? cmdname : "<unknown>");
746        return NULL;
747    }
748
749    /* We call it here because this function conceptually affects the reply
750     * buffer offset (see function comment) */
751    reqresSaveClientReplyOffset(c);
752
753    trimReplyUnusedTailSpace(c);
754    listAddNodeTail(c->reply,NULL); /* NULL is our placeholder. */
755    return listLast(c->reply);
756}

758void setDeferredReply(client *c, void *node, const char *s, size_t length) {
759    listNode *ln = (listNode*)node;
760    clientReplyBlock *next, *prev;
761
762    /* Abort when *node is NULL: when the client should not accept writes
763     * we return NULL in addReplyDeferredLen() */
764    if (node == NULL) return;
765    serverAssert(!listNodeValue(ln));
766
767    /* Normally we fill this dummy NULL node, added by addReplyDeferredLen(),
768     * with a new buffer structure containing the protocol needed to specify
769     * the length of the array following. However sometimes there might be room
770     * in the previous/next node so we can instead remove this NULL node, and
771     * suffix/prefix our data in the node immediately before/after it, in order
772     * to save a write(2) syscall later. Conditions needed to do it:
773     *
774     * - The prev node is non-NULL and has space in it or
775     * - The next node is non-NULL,
776     * - It has enough room already allocated
777     * - And not too large (avoid large memmove) */
778    if (ln->prev != NULL && (prev = listNodeValue(ln->prev)) &&
779        prev->size - prev->used > 0)
780    {
781        size_t len_to_copy = prev->size - prev->used;
782        if (len_to_copy > length)
783            len_to_copy = length;
784        memcpy(prev->buf + prev->used, s, len_to_copy);
785        prev->used += len_to_copy;
786        length -= len_to_copy;
787        if (length == 0) {
788            listDelNode(c->reply, ln);
789            return;
790        }
791        s += len_to_copy;
792    }
793
794    if (ln->next != NULL && (next = listNodeValue(ln->next)) &&
795        next->size - next->used >= length &&
796        next->used < PROTO_REPLY_CHUNK_BYTES * 4)
797    {
798        memmove(next->buf + length, next->buf, next->used);
799        memcpy(next->buf, s, length);
800        next->used += length;
801        listDelNode(c->reply,ln);
802    } else {
803        /* Create a new node */
804        size_t usable_size;
805        clientReplyBlock *buf = zmalloc_usable(length + sizeof(clientReplyBlock), &usable_size);
806        /* Take over the allocation's internal fragmentation */
807        buf->size = usable_size - sizeof(clientReplyBlock);
808        buf->used = length;
809        memcpy(buf->buf, s, length);
810        listNodeValue(ln) = buf;
811        c->reply_bytes += buf->size;
812
813        closeClientOnOutputBufferLimitReached(c, 1);
814    }
815}

817/* Populate the length object and try gluing it to the next chunk. */
818void setDeferredAggregateLen(client *c, void *node, long length, char prefix) {
819    serverAssert(length >= 0);
820
821    /* Abort when *node is NULL: when the client should not accept writes
822     * we return NULL in addReplyDeferredLen() */
823    if (node == NULL) return;
824
825    /* Things like *2\r\n, %3\r\n or ~4\r\n are emitted very often by the protocol
826     * so we have a few shared objects to use if the integer is small
827     * like it is most of the times. */
828    const size_t hdr_len = OBJ_SHARED_HDR_STRLEN(length);
829    const int opt_hdr = length < OBJ_SHARED_BULKHDR_LEN;
830    if (prefix == '*' && opt_hdr) {
831        setDeferredReply(c, node, shared.mbulkhdr[length]->ptr, hdr_len);
832        return;
833    }
834    if (prefix == '%' && opt_hdr) {
835        setDeferredReply(c, node, shared.maphdr[length]->ptr, hdr_len);
836        return;
837    }
838    if (prefix == '~' && opt_hdr) {
839        setDeferredReply(c, node, shared.sethdr[length]->ptr, hdr_len);
840        return;
841    }
842
843    char lenstr[128];
844    size_t lenstr_len = snprintf(lenstr, sizeof(lenstr), "%c%ld\r\n", prefix, length);
845    setDeferredReply(c, node, lenstr, lenstr_len);
846}

873/* Add a double as a bulk reply */
874void addReplyDouble(client *c, double d) {
875    if (c->resp == 3) {
876        char dbuf[MAX_D2STRING_CHARS+3];
877        dbuf[0] = ',';
878        const int dlen = d2string(dbuf+1,sizeof(dbuf)-1,d);
879        dbuf[dlen+1] = '\r';
880        dbuf[dlen+2] = '\n';
881        dbuf[dlen+3] = '\0';
882        addReplyProto(c,dbuf,dlen+3);
883    } else {
884        char dbuf[MAX_LONG_DOUBLE_CHARS+32];
885        /* In order to prepend the string length before the formatted number,
886         * but still avoid an extra memcpy of the whole number, we reserve space
887         * for maximum header `$0000\r\n`, print double, add the resp header in
888         * front of it, and then send the buffer with the right `start` offset. */
889        const int dlen = d2string(dbuf+7,sizeof(dbuf)-7,d);
890        int digits = digits10(dlen);
891        int start = 4 - digits;
892        serverAssert(start >= 0);
893        dbuf[start] = '$';
894
895        /* Convert `dlen` to string, putting it's digits after '$' and before the
896            * formatted double string. */
897        for(int i = digits, val = dlen; val && i > 0 ; --i, val /= 10) {
898            dbuf[start + i] = "0123456789"[val % 10];
899        }
900        dbuf[5] = '\r';
901        dbuf[6] = '\n';
902        dbuf[dlen+7] = '\r';
903        dbuf[dlen+8] = '\n';
904        dbuf[dlen+9] = '\0';
905        addReplyProto(c,dbuf+start,dlen+9-start);
906    }
907}

1155/* Append 'src' client output buffers into 'dst' client output buffers.
1156 * This function clears the output buffers of 'src' */
1157void AddReplyFromClient(client *dst, client *src) {
1158    /* If the source client contains a partial response due to client output
1159     * buffer limits, propagate that to the dest rather than copy a partial
1160     * reply. We don't wanna run the risk of copying partial response in case
1161     * for some reason the output limits don't reach the same decision (maybe
1162     * they changed) */
1163    if (src->flags & CLIENT_CLOSE_ASAP) {
1164        sds client = catClientInfoString(sdsempty(),dst);
1165        freeClientAsync(dst);
1166        serverLog(LL_WARNING,"Client %s scheduled to be closed ASAP for overcoming of output buffer limits.", client);
1167        sdsfree(client);
1168        return;
1169    }
1170
1171    /* First add the static buffer (either into the static buffer or reply list) */
1172    addReplyProto(dst,src->buf, src->bufpos);
1173
1174    /* We need to check with prepareClientToWrite again (after addReplyProto)
1175     * since addReplyProto may have changed something (like CLIENT_CLOSE_ASAP) */
1176    if (prepareClientToWrite(dst) != C_OK)
1177        return;
1178
1179    /* We're bypassing _addReplyProtoToList, so we need to add the pre/post
1180     * checks in it. */
1181    if (dst->flags & CLIENT_CLOSE_AFTER_REPLY) return;
1182
1183    /* Concatenate the reply list into the dest */
1184    if (listLength(src->reply))
1185        listJoin(dst->reply,src->reply);
1186    dst->reply_bytes += src->reply_bytes;
1187    src->reply_bytes = 0;
1188    src->bufpos = 0;
1189
1190    if (src->deferred_reply_errors) {
1191        deferredAfterErrorReply(dst, src->deferred_reply_errors);
1192        listRelease(src->deferred_reply_errors);
1193        src->deferred_reply_errors = NULL;
1194    }
1195
1196    /* Check output buffer limits */
1197    closeClientOnOutputBufferLimitReached(dst, 1);
1198}

1299void acceptCommonHandler(connection *conn, int flags, char *ip) {
1300    client *c;
1301    UNUSED(ip);
1302
1303    if (connGetState(conn) != CONN_STATE_ACCEPTING) {
1304        char addr[NET_ADDR_STR_LEN] = {0};
1305        char laddr[NET_ADDR_STR_LEN] = {0};
1306        connFormatAddr(conn, addr, sizeof(addr), 1);
1307        connFormatAddr(conn, laddr, sizeof(addr), 0);
1308        serverLog(LL_VERBOSE,
1309                  "Accepted client connection in error state: %s (addr=%s laddr=%s)",
1310                  connGetLastError(conn), addr, laddr);
1311        connClose(conn);
1312        return;
1313    }
1314
1315    /* Limit the number of connections we take at the same time.
1316     *
1317     * Admission control will happen before a client is created and connAccept()
1318     * called, because we don't want to even start transport-level negotiation
1319     * if rejected. */
1320    if (listLength(server.clients) + getClusterConnectionsCount()
1321        >= server.maxclients)
1322    {
1323        char *err;
1324        if (server.cluster_enabled)
1325            err = "-ERR max number of clients + cluster "
1326                  "connections reached\r\n";
1327        else
1328            err = "-ERR max number of clients reached\r\n";
1329
1330        /* That's a best effort error message, don't check write errors.
1331         * Note that for TLS connections, no handshake was done yet so nothing
1332         * is written and the connection will just drop. */
1333        if (connWrite(conn,err,strlen(err)) == -1) {
1334            /* Nothing to do, Just to avoid the warning... */
1335        }
1336        server.stat_rejected_conn++;
1337        connClose(conn);
1338        return;
1339    }
1340
1341    /* Create connection and client */
1342    if ((c = createClient(conn)) == NULL) {
1343        char addr[NET_ADDR_STR_LEN] = {0};
1344        char laddr[NET_ADDR_STR_LEN] = {0};
1345        connFormatAddr(conn, addr, sizeof(addr), 1);
1346        connFormatAddr(conn, laddr, sizeof(addr), 0);
1347        serverLog(LL_WARNING,
1348                  "Error registering fd event for the new client connection: %s (addr=%s laddr=%s)",
1349                  connGetLastError(conn), addr, laddr);
1350        connClose(conn); /* May be already closed, just ignore errors */
1351        return;
1352    }
1353
1354    /* Last chance to keep flags */
1355    c->flags |= flags;
1356
1357    /* Initiate accept.
1358     *
1359     * Note that connAccept() is free to do two things here:
1360     * 1. Call clientAcceptHandler() immediately;
1361     * 2. Schedule a future call to clientAcceptHandler().
1362     *
1363     * Because of that, we must do nothing else afterwards.
1364     */
1365    if (connAccept(conn, clientAcceptHandler) == C_ERR) {
1366        if (connGetState(conn) == CONN_STATE_ERROR)
1367            serverLog(LL_WARNING,
1368                      "Error accepting a client connection: %s (addr=%s laddr=%s)",
1369                      connGetLastError(conn), getClientPeerId(c), getClientSockname(c));
1370        freeClient(connGetPrivateData(conn));
1371        return;
1372    }
1373}

1429/* Remove the specified client from global lists where the client could
1430 * be referenced, not including the Pub/Sub channels.
1431 * This is used by freeClient() and replicationCacheMaster(). */
1432void unlinkClient(client *c) {
1433    listNode *ln;
1434
1435    /* If this is marked as current client unset it. */
1436    if (server.current_client == c) server.current_client = NULL;
1437
1438    /* Certain operations must be done only if the client has an active connection.
1439     * If the client was already unlinked or if it's a "fake client" the
1440     * conn is already set to NULL. */
1441    if (c->conn) {
1442        /* Remove from the list of active clients. */
1443        if (c->client_list_node) {
1444            uint64_t id = htonu64(c->id);
1445            raxRemove(server.clients_index,(unsigned char*)&id,sizeof(id),NULL);
1446            listDelNode(server.clients,c->client_list_node);
1447            c->client_list_node = NULL;
1448        }
1449
1450        /* Check if this is a replica waiting for diskless replication (rdb pipe),
1451         * in which case it needs to be cleaned from that list */
1452        if (c->flags & CLIENT_SLAVE &&
1453            c->replstate == SLAVE_STATE_WAIT_BGSAVE_END &&
1454            server.rdb_pipe_conns)
1455        {
1456            int i;
1457            for (i=0; i < server.rdb_pipe_numconns; i++) {
1458                if (server.rdb_pipe_conns[i] == c->conn) {
1459                    rdbPipeWriteHandlerConnRemoved(c->conn);
1460                    server.rdb_pipe_conns[i] = NULL;
1461                    break;
1462                }
1463            }
1464        }
1465        /* Only use shutdown when the fork is active and we are the parent. */
1466        if (server.child_type) connShutdown(c->conn);
1467        connClose(c->conn);
1468        c->conn = NULL;
1469    }
1470
1471    /* Remove from the list of pending writes if needed. */
1472    if (c->flags & CLIENT_PENDING_WRITE) {
1473        serverAssert(&c->clients_pending_write_node.next != NULL || 
1474                     &c->clients_pending_write_node.prev != NULL);
1475        listUnlinkNode(server.clients_pending_write, &c->clients_pending_write_node);
1476        c->flags &= ~CLIENT_PENDING_WRITE;
1477    }
1478
1479    /* Remove from the list of pending reads if needed. */
1480    serverAssert(io_threads_op == IO_THREADS_OP_IDLE);
1481    if (c->pending_read_list_node != NULL) {
1482        listDelNode(server.clients_pending_read,c->pending_read_list_node);
1483        c->pending_read_list_node = NULL;
1484    }
1485
1486
1487    /* When client was just unblocked because of a blocking operation,
1488     * remove it from the list of unblocked clients. */
1489    if (c->flags & CLIENT_UNBLOCKED) {
1490        ln = listSearchKey(server.unblocked_clients,c);
1491        serverAssert(ln != NULL);
1492        listDelNode(server.unblocked_clients,ln);
1493        c->flags &= ~CLIENT_UNBLOCKED;
1494    }
1495
1496    /* Clear the tracking status. */
1497    if (c->flags & CLIENT_TRACKING) disableTracking(c);
1498}

1546void freeClient(client *c) {
1547    listNode *ln;
1548
1549    /* If a client is protected, yet we need to free it right now, make sure
1550     * to at least use asynchronous freeing. */
1551    if (c->flags & CLIENT_PROTECTED) {
1552        freeClientAsync(c);
1553        return;
1554    }
1555
1556    /* For connected clients, call the disconnection event of modules hooks. */
1557    if (c->conn) {
1558        moduleFireServerEvent(REDISMODULE_EVENT_CLIENT_CHANGE,
1559                              REDISMODULE_SUBEVENT_CLIENT_CHANGE_DISCONNECTED,
1560                              c);
1561    }
1562
1563    /* Notify module system that this client auth status changed. */
1564    moduleNotifyUserChanged(c);
1565
1566    /* Free the RedisModuleBlockedClient held onto for reprocessing if not already freed. */
1567    zfree(c->module_blocked_client);
1568
1569    /* If this client was scheduled for async freeing we need to remove it
1570     * from the queue. Note that we need to do this here, because later
1571     * we may call replicationCacheMaster() and the client should already
1572     * be removed from the list of clients to free. */
1573    if (c->flags & CLIENT_CLOSE_ASAP) {
1574        ln = listSearchKey(server.clients_to_close,c);
1575        serverAssert(ln != NULL);
1576        listDelNode(server.clients_to_close,ln);
1577    }
1578
1579    /* If it is our master that's being disconnected we should make sure
1580     * to cache the state to try a partial resynchronization later.
1581     *
1582     * Note that before doing this we make sure that the client is not in
1583     * some unexpected state, by checking its flags. */
1584    if (server.master && c->flags & CLIENT_MASTER) {
1585        serverLog(LL_NOTICE,"Connection with master lost.");
1586        if (!(c->flags & (CLIENT_PROTOCOL_ERROR|CLIENT_BLOCKED))) {
1587            c->flags &= ~(CLIENT_CLOSE_ASAP|CLIENT_CLOSE_AFTER_REPLY);
1588            replicationCacheMaster(c);
1589            return;
1590        }
1591    }
1592
1593    /* Log link disconnection with slave */
1594    if (getClientType(c) == CLIENT_TYPE_SLAVE) {
1595        serverLog(LL_NOTICE,"Connection with replica %s lost.",
1596            replicationGetSlaveName(c));
1597    }
1598
1599    /* Free the query buffer */
1600    sdsfree(c->querybuf);
1601    c->querybuf = NULL;
1602
1603    /* Deallocate structures used to block on blocking ops. */
1604    /* If there is any in-flight command, we don't record their duration. */
1605    c->duration = 0;
1606    if (c->flags & CLIENT_BLOCKED) unblockClient(c, 1);
1607    dictRelease(c->bstate.keys);
1608
1609    /* UNWATCH all the keys */
1610    unwatchAllKeys(c);
1611    listRelease(c->watched_keys);
1612
1613    /* Unsubscribe from all the pubsub channels */
1614    pubsubUnsubscribeAllChannels(c,0);
1615    pubsubUnsubscribeShardAllChannels(c, 0);
1616    pubsubUnsubscribeAllPatterns(c,0);
1617    dictRelease(c->pubsub_channels);
1618    dictRelease(c->pubsub_patterns);
1619    dictRelease(c->pubsubshard_channels);
1620
1621    /* Free data structures. */
1622    listRelease(c->reply);
1623    zfree(c->buf);
1624    freeReplicaReferencedReplBuffer(c);
1625    freeClientArgv(c);
1626    freeClientOriginalArgv(c);
1627    if (c->deferred_reply_errors)
1628        listRelease(c->deferred_reply_errors);
1629#ifdef LOG_REQ_RES
1630    reqresReset(c, 1);
1631#endif
1632
1633    /* Unlink the client: this will close the socket, remove the I/O
1634     * handlers, and remove references of the client from different
1635     * places where active clients may be referenced. */
1636    unlinkClient(c);
1637
1638    /* Master/slave cleanup Case 1:
1639     * we lost the connection with a slave. */
1640    if (c->flags & CLIENT_SLAVE) {
1641        /* If there is no any other slave waiting dumping RDB finished, the
1642         * current child process need not continue to dump RDB, then we kill it.
1643         * So child process won't use more memory, and we also can fork a new
1644         * child process asap to dump rdb for next full synchronization or bgsave.
1645         * But we also need to check if users enable 'save' RDB, if enable, we
1646         * should not remove directly since that means RDB is important for users
1647         * to keep data safe and we may delay configured 'save' for full sync. */
1648        if (server.saveparamslen == 0 &&
1649            c->replstate == SLAVE_STATE_WAIT_BGSAVE_END &&
1650            server.child_type == CHILD_TYPE_RDB &&
1651            server.rdb_child_type == RDB_CHILD_TYPE_DISK &&
1652            anyOtherSlaveWaitRdb(c) == 0)
1653        {
1654            killRDBChild();
1655        }
1656        if (c->replstate == SLAVE_STATE_SEND_BULK) {
1657            if (c->repldbfd != -1) close(c->repldbfd);
1658            if (c->replpreamble) sdsfree(c->replpreamble);
1659        }
1660        list *l = (c->flags & CLIENT_MONITOR) ? server.monitors : server.slaves;
1661        ln = listSearchKey(l,c);
1662        serverAssert(ln != NULL);
1663        listDelNode(l,ln);
1664        /* We need to remember the time when we started to have zero
1665         * attached slaves, as after some time we'll free the replication
1666         * backlog. */
1667        if (getClientType(c) == CLIENT_TYPE_SLAVE && listLength(server.slaves) == 0)
1668            server.repl_no_slaves_since = server.unixtime;
1669        refreshGoodSlavesCount();
1670        /* Fire the replica change modules event. */
1671        if (c->replstate == SLAVE_STATE_ONLINE)
1672            moduleFireServerEvent(REDISMODULE_EVENT_REPLICA_CHANGE,
1673                                  REDISMODULE_SUBEVENT_REPLICA_CHANGE_OFFLINE,
1674                                  NULL);
1675    }
1676
1677    /* Master/slave cleanup Case 2:
1678     * we lost the connection with the master. */
1679    if (c->flags & CLIENT_MASTER) replicationHandleMasterDisconnection();
1680
1681    /* Remove the contribution that this client gave to our
1682     * incrementally computed memory usage. */
1683    server.stat_clients_type_memory[c->last_memory_type] -=
1684        c->last_memory_usage;
1685    /* Remove client from memory usage buckets */
1686    if (c->mem_usage_bucket) {
1687        c->mem_usage_bucket->mem_usage_sum -= c->last_memory_usage;
1688        listDelNode(c->mem_usage_bucket->clients, c->mem_usage_bucket_node);
1689    }
1690
1691    /* Release other dynamically allocated client structure fields,
1692     * and finally release the client structure itself. */
1693    if (c->name) decrRefCount(c->name);
1694    if (c->lib_name) decrRefCount(c->lib_name);
1695    if (c->lib_ver) decrRefCount(c->lib_ver);

1703/* Schedule a client to free it at a safe time in the serverCron() function.
1704 * This function is useful when we need to terminate a client but we are in
1705 * a context where calling freeClient() is not possible, because the client
1706 * should be valid for the continuation of the flow of the program. */
1707void freeClientAsync(client *c) {
1708    /* We need to handle concurrent access to the server.clients_to_close list
1709     * only in the freeClientAsync() function, since it's the only function that
1710     * may access the list while Redis uses I/O threads. All the other accesses
1711     * are in the context of the main thread while the other threads are
1712     * idle. */
1713    if (c->flags & CLIENT_CLOSE_ASAP || c->flags & CLIENT_SCRIPT) return;
1714    c->flags |= CLIENT_CLOSE_ASAP;
1715    if (server.io_threads_num == 1) {
1716        /* no need to bother with locking if there's just one thread (the main thread) */
1717        listAddNodeTail(server.clients_to_close,c);
1718        return;
1719    }
1720    static pthread_mutex_t async_free_queue_mutex = PTHREAD_MUTEX_INITIALIZER;
1721    pthread_mutex_lock(&async_free_queue_mutex);
1722    listAddNodeTail(server.clients_to_close,c);
1723    pthread_mutex_unlock(&async_free_queue_mutex);
1724}

1753

1763    if (c && (c->flags & CLIENT_CLOSE_ASAP)) {

1800/* This function should be called from _writeToClient when the reply list is not empty,
1801 * it gathers the scattered buffers from reply list and sends them away with connWritev.
1802 * If we write successfully, it returns C_OK, otherwise, C_ERR is returned,
1803 * and 'nwritten' is an output parameter, it means how many bytes server write
1804 * to client. */
1805static int _writevToClient(client *c, ssize_t *nwritten) {
1806    int iovcnt = 0;
1807    int iovmax = min(IOV_MAX, c->conn->iovcnt);
1808    struct iovec iov[iovmax];
1809    size_t iov_bytes_len = 0;
1810    /* If the static reply buffer is not empty, 
1811     * add it to the iov array for writev() as well. */
1812    if (c->bufpos > 0) {
1813        iov[iovcnt].iov_base = c->buf + c->sentlen;
1814        iov[iovcnt].iov_len = c->bufpos - c->sentlen;
1815        iov_bytes_len += iov[iovcnt++].iov_len;
1816    }
1817    /* The first node of reply list might be incomplete from the last call,
1818     * thus it needs to be calibrated to get the actual data address and length. */
1819    size_t offset = c->bufpos > 0 ? 0 : c->sentlen;
1820    listIter iter;
1821    listNode *next;
1822    clientReplyBlock *o;
1823    listRewind(c->reply, &iter);
1824    while ((next = listNext(&iter)) && iovcnt < iovmax && iov_bytes_len < NET_MAX_WRITES_PER_EVENT) {
1825        o = listNodeValue(next);
1826        if (o->used == 0) { /* empty node, just release it and skip. */
1827            c->reply_bytes -= o->size;
1828            listDelNode(c->reply, next);
1829            offset = 0;
1830            continue;
1831        }
1832
1833        iov[iovcnt].iov_base = o->buf + offset;
1834        iov[iovcnt].iov_len = o->used - offset;
1835        iov_bytes_len += iov[iovcnt++].iov_len;
1836        offset = 0;
1837    }
1838    if (iovcnt == 0) return C_OK;
1839    *nwritten = connWritev(c->conn, iov, iovcnt);
1840    if (*nwritten <= 0) return C_ERR;
1841
1842    /* Locate the new node which has leftover data and
1843     * release all nodes in front of it. */
1844    ssize_t remaining = *nwritten;
1845    if (c->bufpos > 0) { /* deal with static reply buffer first. */
1846        int buf_len = c->bufpos - c->sentlen;
1847        c->sentlen += remaining;
1848        /* If the buffer was sent, set bufpos to zero to continue with
1849         * the remainder of the reply. */
1850        if (remaining >= buf_len) {
1851            c->bufpos = 0;
1852            c->sentlen = 0;
1853        }
1854        remaining -= buf_len;
1855    }
1856    listRewind(c->reply, &iter);
1857    while (remaining > 0) {
1858        next = listNext(&iter);
1859        o = listNodeValue(next);
1860        if (remaining < (ssize_t)(o->used - c->sentlen)) {
1861            c->sentlen += remaining;
1862            break;
1863        }
1864        remaining -= (ssize_t)(o->used - c->sentlen);
1865        c->reply_bytes -= o->size;
1866        listDelNode(c->reply, next);
1867        c->sentlen = 0;
1868    }
1869
1870    return C_OK;
1871}

1873/* This function does actual writing output buffers to different types of
1874 * clients, it is called by writeToClient.
1875 * If we write successfully, it returns C_OK, otherwise, C_ERR is returned,
1876 * and 'nwritten' is an output parameter, it means how many bytes server write
1877 * to client. */
1878int _writeToClient(client *c, ssize_t *nwritten) {
1879    *nwritten = 0;
1880    if (getClientType(c) == CLIENT_TYPE_SLAVE) {
1881        serverAssert(c->bufpos == 0 && listLength(c->reply) == 0);
1882
1883        replBufBlock *o = listNodeValue(c->ref_repl_buf_node);
1884        serverAssert(o->used >= c->ref_block_pos);
1885        /* Send current block if it is not fully sent. */
1886        if (o->used > c->ref_block_pos) {
1887            *nwritten = connWrite(c->conn, o->buf+c->ref_block_pos,
1888                                  o->used-c->ref_block_pos);
1889            if (*nwritten <= 0) return C_ERR;
1890            c->ref_block_pos += *nwritten;
1891        }
1892
1893        /* If we fully sent the object on head, go to the next one. */
1894        listNode *next = listNextNode(c->ref_repl_buf_node);
1895        if (next && c->ref_block_pos == o->used) {
1896            o->refcount--;
1897            ((replBufBlock *)(listNodeValue(next)))->refcount++;
1898            c->ref_repl_buf_node = next;
1899            c->ref_block_pos = 0;
1900            incrementalTrimReplicationBacklog(REPL_BACKLOG_TRIM_BLOCKS_PER_CALL);
1901        }
1902        return C_OK;
1903    }
1904
1905    /* When the reply list is not empty, it's better to use writev to save us some
1906     * system calls and TCP packets. */
1907    if (listLength(c->reply) > 0) {
1908        int ret = _writevToClient(c, nwritten);
1909        if (ret != C_OK) return ret;
1910
1911        /* If there are no longer objects in the list, we expect
1912         * the count of reply bytes to be exactly zero. */
1913        if (listLength(c->reply) == 0)
1914            serverAssert(c->reply_bytes == 0);
1915    } else if (c->bufpos > 0) {
1916        *nwritten = connWrite(c->conn, c->buf + c->sentlen, c->bufpos - c->sentlen);
1917        if (*nwritten <= 0) return C_ERR;
1918        c->sentlen += *nwritten;
1919
1920        /* If the buffer was sent, set bufpos to zero to continue with
1921         * the remainder of the reply. */
1922        if ((int)c->sentlen == c->bufpos) {
1923            c->bufpos = 0;
1924            c->sentlen = 0;
1925        }
1926    } 
1927
1928    return C_OK;
1929}

1931/* Write data in output buffers to client. Return C_OK if the client
1932 * is still valid after the call, C_ERR if it was freed because of some
1933 * error.  If handler_installed is set, it will attempt to clear the
1934 * write event.
1935 *
1936 * This function is called by threads, but always with handler_installed
1937 * set to 0. So when handler_installed is set to 0 the function must be
1938 * thread safe. */
1939int writeToClient(client *c, int handler_installed) {
1940    /* Update total number of writes on server */
1941    atomicIncr(server.stat_total_writes_processed, 1);
1942
1943    ssize_t nwritten = 0, totwritten = 0;
1944
1945    while(clientHasPendingReplies(c)) {
1946        int ret = _writeToClient(c, &nwritten);
1947        if (ret == C_ERR) break;
1948        totwritten += nwritten;
1949        /* Note that we avoid to send more than NET_MAX_WRITES_PER_EVENT
1950         * bytes, in a single threaded server it's a good idea to serve
1951         * other clients as well, even if a very large request comes from
1952         * super fast link that is always able to accept data (in real world
1953         * scenario think about 'KEYS *' against the loopback interface).
1954         *
1955         * However if we are over the maxmemory limit we ignore that and
1956         * just deliver as much data as it is possible to deliver.
1957         *
1958         * Moreover, we also send as much as possible if the client is
1959         * a slave or a monitor (otherwise, on high-speed traffic, the
1960         * replication/output buffer will grow indefinitely) */
1961        if (totwritten > NET_MAX_WRITES_PER_EVENT &&
1962            (server.maxmemory == 0 ||
1963             zmalloc_used_memory() < server.maxmemory) &&
1964            !(c->flags & CLIENT_SLAVE)) break;
1965    }
1966
1967    if (getClientType(c) == CLIENT_TYPE_SLAVE) {
1968        atomicIncr(server.stat_net_repl_output_bytes, totwritten);
1969    } else {
1970        atomicIncr(server.stat_net_output_bytes, totwritten);
1971    }
1972
1973    if (nwritten == -1) {
1974        if (connGetState(c->conn) != CONN_STATE_CONNECTED) {
1975            serverLog(LL_VERBOSE,
1976                "Error writing to client: %s", connGetLastError(c->conn));
1977            freeClientAsync(c);
1978            return C_ERR;
1979        }
1980    }
1981    if (totwritten > 0) {
1982        /* For clients representing masters we don't count sending data
1983         * as an interaction, since we always send REPLCONF ACK commands
1984         * that take some time to just fill the socket output buffer.
1985         * We just rely on data / pings received for timeout detection. */
1986        if (!(c->flags & CLIENT_MASTER)) c->lastinteraction = server.unixtime;
1987    }
1988    if (!clientHasPendingReplies(c)) {
1989        c->sentlen = 0;
1990        /* Note that writeToClient() is called in a threaded way, but
1991         * aeDeleteFileEvent() is not thread safe: however writeToClient()
1992         * is always called with handler_installed set to 0 from threads
1993         * so we are fine. */
1994        if (handler_installed) {
1995            serverAssert(io_threads_op == IO_THREADS_OP_IDLE);
1996            connSetWriteHandler(c->conn, NULL);
1997        }
1998
1999        /* Close connection after entire reply has been sent. */
2000        if (c->flags & CLIENT_CLOSE_AFTER_REPLY) {
2001            freeClientAsync(c);
2002            return C_ERR;
2003        }
2004    }
2005    /* Update client's memory usage after writing.
2006     * Since this isn't thread safe we do this conditionally. In case of threaded writes this is done in
2007     * handleClientsWithPendingWritesUsingThreads(). */
2008    if (io_threads_op == IO_THREADS_OP_IDLE)
2009        updateClientMemUsageAndBucket(c);
2010    return C_OK;
2011}

2020 * we can just write the replies to the client output buffer without any
2021 * need to use a syscall in order to install the writable event handler,
2022 * get it called, and so forth. */

2031        c->flags &= ~CLIENT_PENDING_WRITE;
2032        listUnlinkNode(server.clients_pending_write,ln);
2033
2034        /* If a client is protected, don't do anything,
2035         * that may trigger write error or recreate handler. */
2036        if (c->flags & CLIENT_PROTECTED) continue;

2128/* Like processMultibulkBuffer(), but for the inline protocol instead of RESP,
2129 * this function consumes the client query buffer and creates a command ready
2130 * to be executed inside the client structure. Returns C_OK if the command
2131 * is ready to be executed, or C_ERR if there is still protocol to read to
2132 * have a well formed command. The function also returns C_ERR when there is
2133 * a protocol error: in such a case the client structure is setup to reply
2134 * with the error and close the connection. */
2135int processInlineBuffer(client *c) {
2136    char *newline;
2137    int argc, j, linefeed_chars = 1;
2138    sds *argv, aux;
2139    size_t querylen;
2140
2141    /* Search for end of line */
2142    newline = strchr(c->querybuf+c->qb_pos,'\n');
2143
2144    /* Nothing to do without a \r\n */
2145    if (newline == NULL) {
2146        if (sdslen(c->querybuf)-c->qb_pos > PROTO_INLINE_MAX_SIZE) {
2147            addReplyError(c,"Protocol error: too big inline request");
2148            setProtocolError("too big inline request",c);
2149        }
2150        return C_ERR;
2151    }
2152
2153    /* Handle the \r\n case. */
2154    if (newline != c->querybuf+c->qb_pos && *(newline-1) == '\r')
2155        newline--, linefeed_chars++;
2156
2157    /* Split the input buffer up to the \r\n */
2158    querylen = newline-(c->querybuf+c->qb_pos);
2159    aux = sdsnewlen(c->querybuf+c->qb_pos,querylen);
2160    argv = sdssplitargs(aux,&argc);
2161    sdsfree(aux);
2162    if (argv == NULL) {
2163        addReplyError(c,"Protocol error: unbalanced quotes in request");
2164        setProtocolError("unbalanced quotes in inline request",c);
2165        return C_ERR;
2166    }
2167
2168    /* Newline from slaves can be used to refresh the last ACK time.
2169     * This is useful for a slave to ping back while loading a big
2170     * RDB file. */
2171    if (querylen == 0 && getClientType(c) == CLIENT_TYPE_SLAVE)
2172        c->repl_ack_time = server.unixtime;
2173
2174    /* Masters should never send us inline protocol to run actual
2175     * commands. If this happens, it is likely due to a bug in Redis where
2176     * we got some desynchronization in the protocol, for example
2177     * because of a PSYNC gone bad.
2178     *
2179     * However there is an exception: masters may send us just a newline
2180     * to keep the connection active. */
2181    if (querylen != 0 && c->flags & CLIENT_MASTER) {
2182        sdsfreesplitres(argv,argc);
2183        serverLog(LL_WARNING,"WARNING: Receiving inline protocol from master, master stream corruption? Closing the master connection and discarding the cached master.");
2184        setProtocolError("Master using the inline protocol. Desync?",c);
2185        return C_ERR;
2186    }
2187
2188    /* Move querybuffer position to the next query in the buffer. */
2189    c->qb_pos += querylen+linefeed_chars;
2190
2191    /* Setup argv array on client structure */
2192    if (argc) {
2193        if (c->argv) zfree(c->argv);
2194        c->argv_len = argc;
2195        c->argv = zmalloc(sizeof(robj*)*c->argv_len);
2196        c->argv_len_sum = 0;
2197    }
2198
2199    /* Create redis objects for all arguments. */
2200    for (c->argc = 0, j = 0; j < argc; j++) {
2201        c->argv[c->argc] = createObject(OBJ_STRING,argv[j]);
2202        c->argc++;
2203        c->argv_len_sum += sdslen(argv[j]);
2204    }
2205    zfree(argv);
2206    return C_OK;
2207}

2242/* Process the query buffer for client 'c', setting up the client argument
2243 * vector for command execution. Returns C_OK if after running the function
2244 * the client has a well-formed ready to be processed command, otherwise
2245 * C_ERR if there is still to read more buffer to get the full command.
2246 * The function also returns C_ERR when there is a protocol error: in such a
2247 * case the client structure is setup to reply with the error and close
2248 * the connection.
2249 *
2250 * This function is called if processInputBuffer() detects that the next
2251 * command is in RESP format, so the first byte in the command is found
2252 * to be '*'. Otherwise for inline commands processInlineBuffer() is called. */
2253int processMultibulkBuffer(client *c) {
2254    char *newline = NULL;
2255    int ok;
2256    long long ll;
2257
2258    if (c->multibulklen == 0) {
2259        /* The client should have been reset */
2260        serverAssertWithInfo(c,NULL,c->argc == 0);
2261
2262        /* Multi bulk length cannot be read without a \r\n */
2263        newline = strchr(c->querybuf+c->qb_pos,'\r');
2264        if (newline == NULL) {
2265            if (sdslen(c->querybuf)-c->qb_pos > PROTO_INLINE_MAX_SIZE) {
2266                addReplyError(c,"Protocol error: too big mbulk count string");
2267                setProtocolError("too big mbulk count string",c);
2268            }
2269            return C_ERR;
2270        }
2271
2272        /* Buffer should also contain \n */
2273        if (newline-(c->querybuf+c->qb_pos) > (ssize_t)(sdslen(c->querybuf)-c->qb_pos-2))
2274            return C_ERR;
2275
2276        /* We know for sure there is a whole line since newline != NULL,
2277         * so go ahead and find out the multi bulk length. */
2278        serverAssertWithInfo(c,NULL,c->querybuf[c->qb_pos] == '*');
2279        ok = string2ll(c->querybuf+1+c->qb_pos,newline-(c->querybuf+1+c->qb_pos),&ll);
2280        if (!ok || ll > INT_MAX) {
2281            addReplyError(c,"Protocol error: invalid multibulk length");
2282            setProtocolError("invalid mbulk count",c);
2283            return C_ERR;
2284        } else if (ll > 10 && authRequired(c)) {
2285            addReplyError(c, "Protocol error: unauthenticated multibulk length");
2286            setProtocolError("unauth mbulk count", c);
2287            return C_ERR;
2288        }
2289
2290        c->qb_pos = (newline-c->querybuf)+2;
2291
2292        if (ll <= 0) return C_OK;
2293
2294        c->multibulklen = ll;
2295
2296        /* Setup argv array on client structure */
2297        if (c->argv) zfree(c->argv);
2298        c->argv_len = min(c->multibulklen, 1024);
2299        c->argv = zmalloc(sizeof(robj*)*c->argv_len);
2300        c->argv_len_sum = 0;
2301    }
2302
2303    serverAssertWithInfo(c,NULL,c->multibulklen > 0);
2304    while(c->multibulklen) {
2305        /* Read bulk length if unknown */
2306        if (c->bulklen == -1) {
2307            newline = strchr(c->querybuf+c->qb_pos,'\r');
2308            if (newline == NULL) {
2309                if (sdslen(c->querybuf)-c->qb_pos > PROTO_INLINE_MAX_SIZE) {
2310                    addReplyError(c,
2311                        "Protocol error: too big bulk count string");
2312                    setProtocolError("too big bulk count string",c);
2313                    return C_ERR;
2314                }
2315                break;
2316            }
2317
2318            /* Buffer should also contain \n */
2319            if (newline-(c->querybuf+c->qb_pos) > (ssize_t)(sdslen(c->querybuf)-c->qb_pos-2))
2320                break;
2321
2322            if (c->querybuf[c->qb_pos] != '$') {
2323                addReplyErrorFormat(c,
2324                    "Protocol error: expected '$', got '%c'",
2325                    c->querybuf[c->qb_pos]);
2326                setProtocolError("expected $ but got something else",c);
2327                return C_ERR;
2328            }
2329
2330            ok = string2ll(c->querybuf+c->qb_pos+1,newline-(c->querybuf+c->qb_pos+1),&ll);
2331            if (!ok || ll < 0 ||
2332                (!(c->flags & CLIENT_MASTER) && ll > server.proto_max_bulk_len)) {
2333                addReplyError(c,"Protocol error: invalid bulk length");
2334                setProtocolError("invalid bulk length",c);
2335                return C_ERR;
2336            } else if (ll > 16384 && authRequired(c)) {
2337                addReplyError(c, "Protocol error: unauthenticated bulk length");
2338                setProtocolError("unauth bulk length", c);
2339                return C_ERR;
2340            }
2341
2342            c->qb_pos = newline-c->querybuf+2;
2343            if (!(c->flags & CLIENT_MASTER) && ll >= PROTO_MBULK_BIG_ARG) {
2344                /* When the client is not a master client (because master
2345                 * client's querybuf can only be trimmed after data applied
2346                 * and sent to replicas).
2347                 *
2348                 * If we are going to read a large object from network
2349                 * try to make it likely that it will start at c->querybuf
2350                 * boundary so that we can optimize object creation
2351                 * avoiding a large copy of data.
2352                 *
2353                 * But only when the data we have not parsed is less than
2354                 * or equal to ll+2. If the data length is greater than
2355                 * ll+2, trimming querybuf is just a waste of time, because
2356                 * at this time the querybuf contains not only our bulk. */
2357                if (sdslen(c->querybuf)-c->qb_pos <= (size_t)ll+2) {
2358                    sdsrange(c->querybuf,c->qb_pos,-1);
2359                    c->qb_pos = 0;
2360                    /* Hint the sds library about the amount of bytes this string is
2361                     * going to contain. */
2362                    c->querybuf = sdsMakeRoomForNonGreedy(c->querybuf,ll+2-sdslen(c->querybuf));
2363                    /* We later set the peak to the used portion of the buffer, but here we over
2364                     * allocated because we know what we need, make sure it'll not be shrunk before used. */
2365                    if (c->querybuf_peak < (size_t)ll + 2) c->querybuf_peak = ll + 2;
2366                }
2367            }
2368            c->bulklen = ll;
2369        }
2370
2371        /* Read bulk argument */
2372        if (sdslen(c->querybuf)-c->qb_pos < (size_t)(c->bulklen+2)) {
2373            /* Not enough data (+2 == trailing \r\n) */
2374            break;
2375        } else {
2376            /* Check if we have space in argv, grow if needed */
2377            if (c->argc >= c->argv_len) {
2378                c->argv_len = min(c->argv_len < INT_MAX/2 ? c->argv_len*2 : INT_MAX, c->argc+c->multibulklen);
2379                c->argv = zrealloc(c->argv, sizeof(robj*)*c->argv_len);
2380            }
2381
2382            /* Optimization: if a non-master client's buffer contains JUST our bulk element
2383             * instead of creating a new object by *copying* the sds we
2384             * just use the current sds string. */
2385            if (!(c->flags & CLIENT_MASTER) &&
2386                c->qb_pos == 0 &&
2387                c->bulklen >= PROTO_MBULK_BIG_ARG &&
2388                sdslen(c->querybuf) == (size_t)(c->bulklen+2))
2389            {
2390                c->argv[c->argc++] = createObject(OBJ_STRING,c->querybuf);
2391                c->argv_len_sum += c->bulklen;
2392                sdsIncrLen(c->querybuf,-2); /* remove CRLF */
2393                /* Assume that if we saw a fat argument we'll see another one
2394                 * likely... */
2395                c->querybuf = sdsnewlen(SDS_NOINIT,c->bulklen+2);
2396                sdsclear(c->querybuf);
2397            } else {
2398                c->argv[c->argc++] =
2399                    createStringObject(c->querybuf+c->qb_pos,c->bulklen);
2400                c->argv_len_sum += c->bulklen;
2401                c->qb_pos += c->bulklen+2;
2402            }
2403            c->bulklen = -1;
2404            c->multibulklen--;
2405        }
2406    }
2407
2408    /* We're done when c->multibulk == 0 */
2409    if (c->multibulklen == 0) return C_OK;
2410
2411    /* Still not ready to process the command */
2412    return C_ERR;
2413}

2415/* Perform necessary tasks after a command was executed:
2416 *
2417 * 1. The client is reset unless there are reasons to avoid doing it.
2418 * 2. In the case of master clients, the replication offset is updated.
2419 * 3. Propagate commands we got from our master to replicas down the line. */
2420void commandProcessed(client *c) {
2421    /* If client is blocked(including paused), just return avoid reset and replicate.
2422     *
2423     * 1. Don't reset the client structure for blocked clients, so that the reply
2424     *    callback will still be able to access the client argv and argc fields.
2425     *    The client will be reset in unblockClient().
2426     * 2. Don't update replication offset or propagate commands to replicas,
2427     *    since we have not applied the command. */
2428    if (c->flags & CLIENT_BLOCKED) return;
2429
2430    reqresAppendResponse(c);
2431    resetClient(c);
2432
2433    long long prev_offset = c->reploff;
2434    if (c->flags & CLIENT_MASTER && !(c->flags & CLIENT_MULTI)) {
2435        /* Update the applied replication offset of our master. */
2436        c->reploff = c->read_reploff - sdslen(c->querybuf) + c->qb_pos;
2437    }
2438
2439    /* If the client is a master we need to compute the difference
2440     * between the applied offset before and after processing the buffer,
2441     * to understand how much of the replication stream was actually
2442     * applied to the master state: this quantity, and its corresponding
2443     * part of the replication stream, will be propagated to the
2444     * sub-replicas and to the replication backlog. */
2445    if (c->flags & CLIENT_MASTER) {
2446        long long applied = c->reploff - prev_offset;
2447        if (applied) {
2448            replicationFeedStreamFromMasterStream(c->querybuf+c->repl_applied,applied);
2449            c->repl_applied += applied;
2450        }
2451    }
2452}

2454/* This function calls processCommand(), but also performs a few sub tasks
2455 * for the client that are useful in that context:
2456 *
2457 * 1. It sets the current client to the client 'c'.
2458 * 2. calls commandProcessed() if the command was handled.
2459 *
2460 * The function returns C_ERR in case the client was freed as a side effect
2461 * of processing the command, otherwise C_OK is returned. */
2462int processCommandAndResetClient(client *c) {
2463    int deadclient = 0;
2464    client *old_client = server.current_client;
2465    server.current_client = c;
2466    if (processCommand(c) == C_OK) {
2467        commandProcessed(c);
2468        /* Update the client's memory to include output buffer growth following the
2469         * processed command. */
2470        updateClientMemUsageAndBucket(c);
2471    }
2472
2473    if (server.current_client == NULL) deadclient = 1;
2474    /*
2475     * Restore the old client, this is needed because when a script
2476     * times out, we will get into this code from processEventsWhileBlocked.
2477     * Which will cause to set the server.current_client. If not restored
2478     * we will return 1 to our caller which will falsely indicate the client
2479     * is dead and will stop reading from its buffer.
2480     */
2481    server.current_client = old_client;
2482    /* performEvictions may flush slave output buffers. This may
2483     * result in a slave, that may be the active client, to be
2484     * freed. */
2485    return deadclient ? C_ERR : C_OK;
2486}

2489/* This function will execute any fully parsed commands pending on
2490 * the client. Returns C_ERR if the client is no longer valid after executing
2491 * the command, and C_OK for all other cases. */
2492int processPendingCommandAndInputBuffer(client *c) {
2493    /* Notice, this code is also called from 'processUnblockedClients'.
2494     * But in case of a module blocked client (see RM_Call 'K' flag) we do not reach this code path.
2495     * So whenever we change the code here we need to consider if we need this change on module
2496     * blocked client as well */
2497    if (c->flags & CLIENT_PENDING_COMMAND) {
2498        c->flags &= ~CLIENT_PENDING_COMMAND;
2499        if (processCommandAndResetClient(c) == C_ERR) {
2500            return C_ERR;
2501        }
2502    }
2503
2504    /* Now process client if it has more data in it's buffer.
2505     *
2506     * Note: when a master client steps into this function,
2507     * it can always satisfy this condition, because its querybuf
2508     * contains data not applied. */
2509    if (c->querybuf && sdslen(c->querybuf) > 0) {
2510        return processInputBuffer(c);
2511    }
2512    return C_OK;
2513}

2515/* This function is called every time, in the client structure 'c', there is
2516 * more query buffer to process, because we read more data from the socket
2517 * or because a client was blocked and later reactivated, so there could be
2518 * pending query buffer, already representing a full command, to process.
2519 * return C_ERR in case the client was freed during the processing */
2520int processInputBuffer(client *c) {
2521    /* Keep processing while there is something in the input buffer */
2522    while(c->qb_pos < sdslen(c->querybuf)) {
2523        /* Immediately abort if the client is in the middle of something. */
2524        if (c->flags & CLIENT_BLOCKED) break;
2525
2526        /* Don't process more buffers from clients that have already pending
2527         * commands to execute in c->argv. */
2528        if (c->flags & CLIENT_PENDING_COMMAND) break;
2529
2530        /* Don't process input from the master while there is a busy script
2531         * condition on the slave. We want just to accumulate the replication
2532         * stream (instead of replying -BUSY like we do with other clients) and
2533         * later resume the processing. */
2534        if (isInsideYieldingLongCommand() && c->flags & CLIENT_MASTER) break;
2535
2536        /* CLIENT_CLOSE_AFTER_REPLY closes the connection once the reply is
2537         * written to the client. Make sure to not let the reply grow after
2538         * this flag has been set (i.e. don't process more commands).
2539         *
2540         * The same applies for clients we want to terminate ASAP. */
2541        if (c->flags & (CLIENT_CLOSE_AFTER_REPLY|CLIENT_CLOSE_ASAP)) break;
2542
2543        /* Determine request type when unknown. */
2544        if (!c->reqtype) {
2545            if (c->querybuf[c->qb_pos] == '*') {
2546                c->reqtype = PROTO_REQ_MULTIBULK;
2547            } else {
2548                c->reqtype = PROTO_REQ_INLINE;
2549            }
2550        }
2551
2552        if (c->reqtype == PROTO_REQ_INLINE) {
2553            if (processInlineBuffer(c) != C_OK) break;
2554        } else if (c->reqtype == PROTO_REQ_MULTIBULK) {
2555            if (processMultibulkBuffer(c) != C_OK) break;
2556        } else {
2557            serverPanic("Unknown request type");
2558        }
2559
2560        /* Multibulk processing could see a <= 0 length. */
2561        if (c->argc == 0) {
2562            resetClient(c);
2563        } else {
2564            /* If we are in the context of an I/O thread, we can't really
2565             * execute the command here. All we can do is to flag the client
2566             * as one that needs to process the command. */
2567            if (io_threads_op != IO_THREADS_OP_IDLE) {
2568                serverAssert(io_threads_op == IO_THREADS_OP_READ);
2569                c->flags |= CLIENT_PENDING_COMMAND;
2570                break;
2571            }
2572
2573            /* We are finally ready to execute the command. */
2574            if (processCommandAndResetClient(c) == C_ERR) {
2575                /* If the client is no longer valid, we avoid exiting this
2576                 * loop and trimming the client buffer later. So we return
2577                 * ASAP in that case. */
2578                return C_ERR;
2579            }
2580        }
2581    }
2582
2583    if (c->flags & CLIENT_MASTER) {
2584        /* If the client is a master, trim the querybuf to repl_applied,
2585         * since master client is very special, its querybuf not only
2586         * used to parse command, but also proxy to sub-replicas.
2587         *
2588         * Here are some scenarios we cannot trim to qb_pos:
2589         * 1. we don't receive complete command from master
2590         * 2. master client blocked cause of client pause
2591         * 3. io threads operate read, master client flagged with CLIENT_PENDING_COMMAND
2592         *
2593         * In these scenarios, qb_pos points to the part of the current command
2594         * or the beginning of next command, and the current command is not applied yet,
2595         * so the repl_applied is not equal to qb_pos. */
2596        if (c->repl_applied) {
2597            sdsrange(c->querybuf,c->repl_applied,-1);
2598            c->qb_pos -= c->repl_applied;
2599            c->repl_applied = 0;
2600        }
2601    } else if (c->qb_pos) {
2602        /* Trim to pos */
2603        sdsrange(c->querybuf,c->qb_pos,-1);
2604        c->qb_pos = 0;
2605    }
2606
2607    /* Update client memory usage after processing the query buffer, this is
2608     * important in case the query buffer is big and wasn't drained during
2609     * the above loop (because of partially sent big commands). */
2610    if (io_threads_op == IO_THREADS_OP_IDLE)
2611        updateClientMemUsageAndBucket(c);
2612
2613    return C_OK;
2614}

2616void readQueryFromClient(connection *conn) {
2617    client *c = connGetPrivateData(conn);
2618    int nread, big_arg = 0;
2619    size_t qblen, readlen;
2620
2621    /* Check if we want to read from the client later when exiting from
2622     * the event loop. This is the case if threaded I/O is enabled. */
2623    if (postponeClientRead(c)) return;
2624
2625    /* Update total number of reads on server */
2626    atomicIncr(server.stat_total_reads_processed, 1);
2627
2628    readlen = PROTO_IOBUF_LEN;
2629    /* If this is a multi bulk request, and we are processing a bulk reply
2630     * that is large enough, try to maximize the probability that the query
2631     * buffer contains exactly the SDS string representing the object, even
2632     * at the risk of requiring more read(2) calls. This way the function
2633     * processMultiBulkBuffer() can avoid copying buffers to create the
2634     * Redis Object representing the argument. */
2635    if (c->reqtype == PROTO_REQ_MULTIBULK && c->multibulklen && c->bulklen != -1
2636        && c->bulklen >= PROTO_MBULK_BIG_ARG)
2637    {
2638        ssize_t remaining = (size_t)(c->bulklen+2)-(sdslen(c->querybuf)-c->qb_pos);
2639        big_arg = 1;
2640
2641        /* Note that the 'remaining' variable may be zero in some edge case,
2642         * for example once we resume a blocked client after CLIENT PAUSE. */
2643        if (remaining > 0) readlen = remaining;
2644
2645        /* Master client needs expand the readlen when meet BIG_ARG(see #9100),
2646         * but doesn't need align to the next arg, we can read more data. */
2647        if (c->flags & CLIENT_MASTER && readlen < PROTO_IOBUF_LEN)
2648            readlen = PROTO_IOBUF_LEN;
2649    }
2650
2651    qblen = sdslen(c->querybuf);
2652    if (!(c->flags & CLIENT_MASTER) && // master client's querybuf can grow greedy.
2653        (big_arg || sdsalloc(c->querybuf) < PROTO_IOBUF_LEN)) {
2654        /* When reading a BIG_ARG we won't be reading more than that one arg
2655         * into the query buffer, so we don't need to pre-allocate more than we
2656         * need, so using the non-greedy growing. For an initial allocation of
2657         * the query buffer, we also don't wanna use the greedy growth, in order
2658         * to avoid collision with the RESIZE_THRESHOLD mechanism. */
2659        c->querybuf = sdsMakeRoomForNonGreedy(c->querybuf, readlen);
2660        /* We later set the peak to the used portion of the buffer, but here we over
2661         * allocated because we know what we need, make sure it'll not be shrunk before used. */
2662        if (c->querybuf_peak < qblen + readlen) c->querybuf_peak = qblen + readlen;
2663    } else {
2664        c->querybuf = sdsMakeRoomFor(c->querybuf, readlen);
2665
2666        /* Read as much as possible from the socket to save read(2) system calls. */
2667        readlen = sdsavail(c->querybuf);
2668    }
2669    nread = connRead(c->conn, c->querybuf+qblen, readlen);
2670    if (nread == -1) {
2671        if (connGetState(conn) == CONN_STATE_CONNECTED) {
2672            return;
2673        } else {
2674            serverLog(LL_VERBOSE, "Reading from client: %s",connGetLastError(c->conn));
2675            freeClientAsync(c);
2676            goto done;
2677        }
2678    } else if (nread == 0) {
2679        if (server.verbosity <= LL_VERBOSE) {
2680            sds info = catClientInfoString(sdsempty(), c);
2681            serverLog(LL_VERBOSE, "Client closed connection %s", info);
2682            sdsfree(info);
2683        }
2684        freeClientAsync(c);
2685        goto done;
2686    }
2687
2688    sdsIncrLen(c->querybuf,nread);
2689    qblen = sdslen(c->querybuf);
2690    if (c->querybuf_peak < qblen) c->querybuf_peak = qblen;
2691
2692    c->lastinteraction = server.unixtime;
2693    if (c->flags & CLIENT_MASTER) {
2694        c->read_reploff += nread;
2695        atomicIncr(server.stat_net_repl_input_bytes, nread);
2696    } else {
2697        atomicIncr(server.stat_net_input_bytes, nread);
2698    }
2699
2700    if (!(c->flags & CLIENT_MASTER) && sdslen(c->querybuf) > server.client_max_querybuf_len) {
2701        sds ci = catClientInfoString(sdsempty(),c), bytes = sdsempty();
2702
2703        bytes = sdscatrepr(bytes,c->querybuf,64);
2704        serverLog(LL_WARNING,"Closing client that reached max query buffer length: %s (qbuf initial bytes: %s)", ci, bytes);
2705        sdsfree(ci);
2706        sdsfree(bytes);
2707        freeClientAsync(c);
2708        goto done;
2709    }
2710
2711    /* There is more data in the client input buffer, continue parsing it
2712     * and check if there is a full command to execute. */
2713    if (processInputBuffer(c) == C_ERR)
2714         c = NULL;
2715
2716done:
2717    beforeNextClient(c);
2718}

2720/* A Redis "Address String" is a colon separated ip:port pair.
2721 * For IPv4 it's in the form x.y.z.k:port, example: "127.0.0.1:1234".
2722 * For IPv6 addresses we use [] around the IP part, like in "[::1]:1234".
2723 * For Unix sockets we use path:0, like in "/tmp/redis:0".
2724 *
2725 * An Address String always fits inside a buffer of NET_ADDR_STR_LEN bytes,
2726 * including the null term.
2727 *
2728 * On failure the function still populates 'addr' with the "?:0" string in case
2729 * you want to relax error checking or need to display something anyway (see
2730 * anetFdToString implementation for more info). */
2731void genClientAddrString(client *client, char *addr,
2732                         size_t addr_len, int remote) {
2733    if (client->flags & CLIENT_UNIX_SOCKET) {
2734        /* Unix socket client. */
2735        snprintf(addr,addr_len,"%s:0",server.unixsocket);
2736    } else {
2737        /* TCP client. */
2738        connFormatAddr(client->conn,addr,addr_len,remote);
2739    }
2740}

3654/* Keep track of the original command arguments so that we can generate
3655 * an accurate slowlog entry after the command has been executed. */
3656static void retainOriginalCommandVector(client *c) {
3657    /* We already rewrote this command, so don't rewrite it again */
3658    if (c->original_argv) return;

3660    c->original_argv = zmalloc(sizeof(robj*)*(c->argc));
3661    for (int j = 0; j < c->argc; j++) {
3662        c->original_argv[j] = c->argv[j];

3670void redactClientCommandArgument(client *c, int argc) {

3680/* Rewrite the command vector of the client. All the new objects ref count
3681 * is incremented. The old command vector is freed, and the old objects
3682 * ref count is decremented. */

3716/* Rewrite a single item in the command vector.
3717 * The new val ref count is incremented, and the old decremented.
3718 *
3719 * It is possible to specify an argument over the current size of the
3720 * argument vector: in this case the array of objects gets reallocated
3721 * and c->argc set to the max value. However it's up to the caller to
3722 *
3723 * 1. Make sure there are no "holes" and all the arguments are set.
3724 * 2. If the original argument vector was longer than the one we
3725 *    want to end with, it's up to the caller to set c->argc and
3726 *    free the no longer used objects on c->argv. */
3727void rewriteClientCommandArgument(client *c, int i, robj *newval) {
3728    robj *oldval;
3729    retainOriginalCommandVector(c);
3730
3731    /* We need to handle both extending beyond argc (just update it and
3732     * initialize the new element) or beyond argv_len (realloc is needed).
3733     */
3734    if (i >= c->argc) {
3735        if (i >= c->argv_len) {
3736            c->argv = zrealloc(c->argv,sizeof(robj*)*(i+1));
3737            c->argv_len = i+1;
3738        }
3739        c->argc = i+1;
3740        c->argv[i] = NULL;
3741    }
3742    oldval = c->argv[i];
3743    if (oldval) c->argv_len_sum -= getStringObjectLen(oldval);
3744    if (newval) c->argv_len_sum += getStringObjectLen(newval);
3745    c->argv[i] = newval;
3746    incrRefCount(newval);
3747    if (oldval) decrRefCount(oldval);
3748
3749    /* If this is the command name make sure to fix c->cmd. */
3750    if (i == 0) {
3751        c->cmd = lookupCommandOrOriginal(c->argv,c->argc);
3752        serverAssertWithInfo(c,NULL,c->cmd != NULL);
3753    }
3754}

3756/* This function returns the number of bytes that Redis is
3757 * using to store the reply still not read by the client.
3758 *
3759 * Note: this function is very fast so can be called as many time as
3760 * the caller wishes. The main usage of this function currently is
3761 * enforcing the client output length limits. */
3762size_t getClientOutputBufferMemoryUsage(client *c) {
3763    if (getClientType(c) == CLIENT_TYPE_SLAVE) {
3764        size_t repl_buf_size = 0;
3765        size_t repl_node_num = 0;
3766        size_t repl_node_size = sizeof(listNode) + sizeof(replBufBlock);
3767        if (c->ref_repl_buf_node) {
3768            replBufBlock *last = listNodeValue(listLast(server.repl_buffer_blocks));
3769            replBufBlock *cur = listNodeValue(c->ref_repl_buf_node);
3770            repl_buf_size = last->repl_offset + last->size - cur->repl_offset;
3771            repl_node_num = last->id - cur->id + 1;
3772        }
3773        return repl_buf_size + (repl_node_size*repl_node_num);
3774    } else { 
3775        size_t list_item_size = sizeof(listNode) + sizeof(clientReplyBlock);
3776        return c->reply_bytes + (list_item_size*listLength(c->reply));
3777    }
3778}

3780/* Returns the total client's memory usage.
3781 * Optionally, if output_buffer_mem_usage is not NULL, it fills it with
3782 * the client output buffer memory usage portion of the total. */
3783size_t getClientMemoryUsage(client *c, size_t *output_buffer_mem_usage) {
3784    size_t mem = getClientOutputBufferMemoryUsage(c);
3785    if (output_buffer_mem_usage != NULL)
3786        *output_buffer_mem_usage = mem;
3787    mem += sdsZmallocSize(c->querybuf);
3788    mem += zmalloc_size(c);
3789    mem += c->buf_usable_size;
3790    /* For efficiency (less work keeping track of the argv memory), it doesn't include the used memory
3791     * i.e. unused sds space and internal fragmentation, just the string length. but this is enough to
3792     * spot problematic clients. */
3793    mem += c->argv_len_sum + sizeof(robj*)*c->argc;
3794    mem += multiStateMemOverhead(c);
3795
3796    /* Add memory overhead of pubsub channels and patterns. Note: this is just the overhead of the robj pointers
3797     * to the strings themselves because they aren't stored per client. */
3798    mem += pubsubMemOverhead(c);
3799
3800    /* Add memory overhead of the tracking prefixes, this is an underestimation so we don't need to traverse the entire rax */
3801    if (c->client_tracking_prefixes)
3802        mem += c->client_tracking_prefixes->numnodes * (sizeof(raxNode) * sizeof(raxNode*));
3803
3804    return mem;
3805}

3845/* The function checks if the client reached output buffer soft or hard
3846 * limit, and also update the state needed to check the soft limit as
3847 * a side effect.
3848 *
3849 * Return value: non-zero if the client reached the soft or the hard limit.
3850 *               Otherwise zero is returned. */
3851int checkClientOutputBufferLimits(client *c) {
3852    int soft = 0, hard = 0, class;
3853    unsigned long used_mem = getClientOutputBufferMemoryUsage(c);
3854
3855    class = getClientType(c);
3856    /* For the purpose of output buffer limiting, masters are handled
3857     * like normal clients. */
3858    if (class == CLIENT_TYPE_MASTER) class = CLIENT_TYPE_NORMAL;
3859
3860    /* Note that it doesn't make sense to set the replica clients output buffer
3861     * limit lower than the repl-backlog-size config (partial sync will succeed
3862     * and then replica will get disconnected).
3863     * Such a configuration is ignored (the size of repl-backlog-size will be used).
3864     * This doesn't have memory consumption implications since the replica client
3865     * will share the backlog buffers memory. */
3866    size_t hard_limit_bytes = server.client_obuf_limits[class].hard_limit_bytes;
3867    if (class == CLIENT_TYPE_SLAVE && hard_limit_bytes &&
3868        (long long)hard_limit_bytes < server.repl_backlog_size)
3869        hard_limit_bytes = server.repl_backlog_size;
3870    if (server.client_obuf_limits[class].hard_limit_bytes &&
3871        used_mem >= hard_limit_bytes)
3872        hard = 1;
3873    if (server.client_obuf_limits[class].soft_limit_bytes &&
3874        used_mem >= server.client_obuf_limits[class].soft_limit_bytes)
3875        soft = 1;
3876
3877    /* We need to check if the soft limit is reached continuously for the
3878     * specified amount of seconds. */
3879    if (soft) {
3880        if (c->obuf_soft_limit_reached_time == 0) {
3881            c->obuf_soft_limit_reached_time = server.unixtime;
3882            soft = 0; /* First time we see the soft limit reached */
3883        } else {
3884            time_t elapsed = server.unixtime - c->obuf_soft_limit_reached_time;
3885
3886            if (elapsed <=
3887                server.client_obuf_limits[class].soft_limit_seconds) {
3888                soft = 0; /* The client still did not reached the max number of
3889                             seconds for the soft limit to be considered
3890                             reached. */
3891            }
3892        }
3893    } else {
3894        c->obuf_soft_limit_reached_time = 0;
3895    }
3896    return soft || hard;
3897}

3899/* Asynchronously close a client if soft or hard limit is reached on the
3900 * output buffer size. The caller can check if the client will be closed
3901 * checking if the client CLIENT_CLOSE_ASAP flag is set.
3902 *
3903 * Note: we need to close the client asynchronously because this function is
3904 * called from contexts where the client can't be freed safely, i.e. from the
3905 * lower level functions pushing data inside the client output buffers.
3906 * When `async` is set to 0, we close the client immediately, this is
3907 * useful when called from cron.
3908 *
3909 * Returns 1 if client was (flagged) closed. */
3910int closeClientOnOutputBufferLimitReached(client *c, int async) {
3911    if (!c->conn) return 0; /* It is unsafe to free fake clients. */
3912    serverAssert(c->reply_bytes < SIZE_MAX-(1024*64));
3913    /* Note that c->reply_bytes is irrelevant for replica clients
3914     * (they use the global repl buffers). */
3915    if ((c->reply_bytes == 0 && getClientType(c) != CLIENT_TYPE_SLAVE) ||
3916        c->flags & CLIENT_CLOSE_ASAP) return 0;
3917    if (checkClientOutputBufferLimits(c)) {

3920        if (async) {
3921            freeClientAsync(c);

3923                      "Client %s scheduled to be closed ASAP for overcoming of output buffer limits.",
3924                      client);

3927            serverLog(LL_WARNING,
3928                      "Client %s closed for overcoming of output buffer limits.",
3929                      client);
3930        }

4082/* This function is called by Redis in order to process a few events from
4083 * time to time while blocked into some not interruptible operation.
4084 * This allows to reply to clients with the -LOADING error while loading the
4085 * data set at startup or after a full resynchronization with the master
4086 * and so forth.
4087 *
4088 * It calls the event loop in order to process a few events. Specifically we
4089 * try to call the event loop 4 times as long as we receive acknowledge that
4090 * some event was processed, in order to go forward with the accept, read,
4091 * write, close sequence needed to serve a client.
4092 *
4093 * The function returns the total number of events processed. */
4094void processEventsWhileBlocked(void) {
4095    int iterations = 4; /* See the function top-comment. */
4096
4097    /* Update our cached time since it is used to create and update the last
4098     * interaction time with clients and for other important things. */
4099    updateCachedTime(0);
4100
4101    /* For the few commands that are allowed during busy scripts, we rather
4102     * provide a fresher time than the one from when the script started (they
4103     * still won't get it from the call due to execution_nesting. For commands
4104     * during loading this doesn't matter. */
4105    mstime_t prev_cmd_time_snapshot = server.cmd_time_snapshot;
4106    server.cmd_time_snapshot = server.mstime;
4107
4108    /* Note: when we are processing events while blocked (for instance during
4109     * busy Lua scripts), we set a global flag. When such flag is set, we
4110     * avoid handling the read part of clients using threaded I/O.
4111     * See https://github.com/redis/redis/issues/6988 for more info.
4112     * Note that there could be cases of nested calls to this function,
4113     * specifically on a busy script during async_loading rdb, and scripts
4114     * that came from AOF. */
4115    ProcessingEventsWhileBlocked++;
4116    while (iterations--) {
4117        long long startval = server.events_processed_while_blocked;
4118        long long ae_events = aeProcessEvents(server.el,
4119            AE_FILE_EVENTS|AE_DONT_WAIT|
4120            AE_CALL_BEFORE_SLEEP|AE_CALL_AFTER_SLEEP);
4121        /* Note that server.events_processed_while_blocked will also get
4122         * incremented by callbacks called by the event loop handlers. */
4123        server.events_processed_while_blocked += ae_events;
4124        long long events = server.events_processed_while_blocked - startval;
4125        if (!events) break;
4126    }
4127
4128    whileBlockedCron();
4129
4130    ProcessingEventsWhileBlocked--;
4131    serverAssert(ProcessingEventsWhileBlocked >= 0);
4132
4133    server.cmd_time_snapshot = prev_cmd_time_snapshot;
4134}

4173void *IOThreadMain(void *myid) {
4174    /* The ID is the thread number (from 0 to server.io_threads_num-1), and is
4175     * used by the thread to just manipulate a single sub-array of clients. */
4176    long id = (unsigned long)myid;
4177    char thdname[16];
4178
4179    snprintf(thdname, sizeof(thdname), "io_thd_%ld", id);
4180    redis_set_thread_title(thdname);
4181    redisSetCpuAffinity(server.server_cpulist);
4182    makeThreadKillable();
4183
4184    while(1) {
4185        /* Wait for start */
4186        for (int j = 0; j < 1000000; j++) {
4187            if (getIOPendingCount(id) != 0) break;
4188        }
4189
4190        /* Give the main thread a chance to stop this thread. */
4191        if (getIOPendingCount(id) == 0) {
4192            pthread_mutex_lock(&io_threads_mutex[id]);
4193            pthread_mutex_unlock(&io_threads_mutex[id]);
4194            continue;
4195        }
4196
4197        serverAssert(getIOPendingCount(id) != 0);
4198
4199        /* Process: note that the main thread will never touch our list
4200         * before we drop the pending count to 0. */
4201        listIter li;
4202        listNode *ln;
4203        listRewind(io_threads_list[id],&li);
4204        while((ln = listNext(&li))) {
4205            client *c = listNodeValue(ln);
4206            if (io_threads_op == IO_THREADS_OP_WRITE) {
4207                writeToClient(c,0);
4208            } else if (io_threads_op == IO_THREADS_OP_READ) {
4209                readQueryFromClient(c->conn);
4210            } else {
4211                serverPanic("io_threads_op value is unknown");
4212            }
4213        }
4214        listEmpty(io_threads_list[id]);
4215        setIOPendingCount(id, 0);
4216    }
4217}

4219/* Initialize the data structures needed for threaded I/O. */
4220void initThreadedIO(void) {
4221    server.io_threads_active = 0; /* We start with threads not active. */
4222
4223    /* Indicate that io-threads are currently idle */
4224    io_threads_op = IO_THREADS_OP_IDLE;
4225
4226    /* Don't spawn any thread if the user selected a single thread:
4227     * we'll handle I/O directly from the main thread. */
4228    if (server.io_threads_num == 1) return;
4229
4230    if (server.io_threads_num > IO_THREADS_MAX_NUM) {
4231        serverLog(LL_WARNING,"Fatal: too many I/O threads configured. "
4232                             "The maximum number is %d.", IO_THREADS_MAX_NUM);
4233        exit(1);
4234    }
4235
4236    /* Spawn and initialize the I/O threads. */
4237    for (int i = 0; i < server.io_threads_num; i++) {
4238        /* Things we do for all the threads including the main thread. */
4239        io_threads_list[i] = listCreate();
4240        if (i == 0) continue; /* Thread 0 is the main thread. */
4241
4242        /* Things we do only for the additional threads. */
4243        pthread_t tid;
4244        pthread_mutex_init(&io_threads_mutex[i],NULL);
4245        setIOPendingCount(i, 0);
4246        pthread_mutex_lock(&io_threads_mutex[i]); /* Thread will be stopped. */
4247        if (pthread_create(&tid,NULL,IOThreadMain,(void*)(long)i) != 0) {
4248            serverLog(LL_WARNING,"Fatal: Can't initialize IO thread.");
4249            exit(1);
4250        }
4251        io_threads[i] = tid;
4252    }
4253}

4289/* This function checks if there are not enough pending clients to justify
4290 * taking the I/O threads active: in that case I/O threads are stopped if
4291 * currently active. We track the pending writes as a measure of clients
4292 * we need to handle in parallel, however the I/O threading is disabled
4293 * globally for reads as well if we have too little pending clients.
4294 *
4295 * The function returns 0 if the I/O threading should be used because there
4296 * are enough active threads, otherwise 1 is returned and the I/O threads
4297 * could be possibly stopped (if already active) as a side effect. */
4298int stopThreadedIOIfNeeded(void) {
4299    int pending = listLength(server.clients_pending_write);
4300
4301    /* Return ASAP if IO threads are disabled (single threaded mode). */
4302    if (server.io_threads_num == 1) return 1;
4303
4304    if (pending < (server.io_threads_num*2)) {
4305        if (server.io_threads_active) stopThreadedIO();
4306        return 1;
4307    } else {
4308        return 0;
4309    }

4312/* This function achieves thread safety using a fan-out -> fan-in paradigm:
4313 * Fan out: The main thread fans out work to the io-threads which block until
4314 * setIOPendingCount() is called with a value larger than 0 by the main thread.
4315 * Fan in: The main thread waits until getIOPendingCount() returns 0. Then
4316 * it can safely perform post-processing and return to normal synchronous
4317 * work. */
4318int handleClientsWithPendingWritesUsingThreads(void) {
4319    int processed = listLength(server.clients_pending_write);
4320    if (processed == 0) return 0; /* Return ASAP if there are no clients. */
4321
4322    /* If I/O threads are disabled or we have few clients to serve, don't
4323     * use I/O threads, but the boring synchronous code. */
4324    if (server.io_threads_num == 1 || stopThreadedIOIfNeeded()) {
4325        return handleClientsWithPendingWrites();
4326    }
4327
4328    /* Start threads if needed. */
4329    if (!server.io_threads_active) startThreadedIO();
4330
4331    /* Distribute the clients across N different lists. */
4332    listIter li;
4333    listNode *ln;
4334    listRewind(server.clients_pending_write,&li);
4335    int item_id = 0;
4336    while((ln = listNext(&li))) {
4337        client *c = listNodeValue(ln);
4338        c->flags &= ~CLIENT_PENDING_WRITE;
4339
4340        /* Remove clients from the list of pending writes since
4341         * they are going to be closed ASAP. */
4342        if (c->flags & CLIENT_CLOSE_ASAP) {
4343            listUnlinkNode(server.clients_pending_write, ln);
4344            continue;
4345        }
4346
4347        /* Since all replicas and replication backlog use global replication
4348         * buffer, to guarantee data accessing thread safe, we must put all
4349         * replicas client into io_threads_list[0] i.e. main thread handles
4350         * sending the output buffer of all replicas. */
4351        if (getClientType(c) == CLIENT_TYPE_SLAVE) {
4352            listAddNodeTail(io_threads_list[0],c);
4353            continue;
4354        }
4355
4356        int target_id = item_id % server.io_threads_num;
4357        listAddNodeTail(io_threads_list[target_id],c);
4358        item_id++;
4359    }
4360
4361    /* Give the start condition to the waiting threads, by setting the
4362     * start condition atomic var. */
4363    io_threads_op = IO_THREADS_OP_WRITE;
4364    for (int j = 1; j < server.io_threads_num; j++) {
4365        int count = listLength(io_threads_list[j]);
4366        setIOPendingCount(j, count);
4367    }
4368
4369    /* Also use the main thread to process a slice of clients. */
4370    listRewind(io_threads_list[0],&li);
4371    while((ln = listNext(&li))) {
4372        client *c = listNodeValue(ln);
4373        writeToClient(c,0);
4374    }
4375    listEmpty(io_threads_list[0]);
4376
4377    /* Wait for all the other threads to end their work. */
4378    while(1) {
4379        unsigned long pending = 0;
4380        for (int j = 1; j < server.io_threads_num; j++)
4381            pending += getIOPendingCount(j);
4382        if (pending == 0) break;
4383    }
4384
4385    io_threads_op = IO_THREADS_OP_IDLE;
4386
4387    /* Run the list of clients again to install the write handler where
4388     * needed. */
4389    listRewind(server.clients_pending_write,&li);
4390    while((ln = listNext(&li))) {
4391        client *c = listNodeValue(ln);
4392
4393        /* Update the client in the mem usage after we're done processing it in the io-threads */
4394        updateClientMemUsageAndBucket(c);
4395
4396        /* Install the write handler if there are pending writes in some
4397         * of the clients. */
4398        if (clientHasPendingReplies(c)) {
4399            installClientWriteHandler(c);
4400        }
4401    }
4402    while(listLength(server.clients_pending_write) > 0) {
4403        listUnlinkNode(server.clients_pending_write, server.clients_pending_write->head);
4404    }
4405
4406    /* Update processed count on server */
4407    server.stat_io_writes_processed += processed;
4408
4409    return processed;
4410}

4432 * readable handler will just put normal clients into a queue of clients to
4433 * process (instead of serving them synchronously). This function runs
4434 * the queue using the I/O threads, and process them in order to accumulate
4435 * the reads in the buffers, and also parse the first command available
4436 * rendering it in the client structures.
4437 * This function achieves thread safety using a fan-out -> fan-in paradigm:
4438 * Fan out: The main thread fans out work to the io-threads which block until
4439 * setIOPendingCount() is called with a value larger than 0 by the main thread.
4440 * Fan in: The main thread waits until getIOPendingCount() returns 0. Then
4441 * it can safely perform post-processing and return to normal synchronous
4442 * work. */
4443int handleClientsWithPendingReadsUsingThreads(void) {
4444    if (!server.io_threads_active || !server.io_threads_do_reads) return 0;
4445    int processed = listLength(server.clients_pending_read);
4446    if (processed == 0) return 0;
4447
4448    /* Distribute the clients across N different lists. */
4449    listIter li;
4450    listNode *ln;
4451    listRewind(server.clients_pending_read,&li);
4452    int item_id = 0;
4453    while((ln = listNext(&li))) {
4454        client *c = listNodeValue(ln);
4455        int target_id = item_id % server.io_threads_num;
4456        listAddNodeTail(io_threads_list[target_id],c);
4457        item_id++;
4458    }
4459
4460    /* Give the start condition to the waiting threads, by setting the
4461     * start condition atomic var. */
4462    io_threads_op = IO_THREADS_OP_READ;
4463    for (int j = 1; j < server.io_threads_num; j++) {
4464        int count = listLength(io_threads_list[j]);
4465        setIOPendingCount(j, count);
4466    }
4467
4468    /* Also use the main thread to process a slice of clients. */
4469    listRewind(io_threads_list[0],&li);
4470    while((ln = listNext(&li))) {
4471        client *c = listNodeValue(ln);
4472        readQueryFromClient(c->conn);
4473    }
4474    listEmpty(io_threads_list[0]);
4475
4476    /* Wait for all the other threads to end their work. */
4477    while(1) {
4478        unsigned long pending = 0;
4479        for (int j = 1; j < server.io_threads_num; j++)
4480            pending += getIOPendingCount(j);
4481        if (pending == 0) break;
4482    }
4483
4484    io_threads_op = IO_THREADS_OP_IDLE;
4485
4486    /* Run the list of clients again to process the new buffers. */
4487    while(listLength(server.clients_pending_read)) {
4488        ln = listFirst(server.clients_pending_read);
4489        client *c = listNodeValue(ln);
4490        listDelNode(server.clients_pending_read,ln);
4491        c->pending_read_list_node = NULL;
4492
4493        serverAssert(!(c->flags & CLIENT_BLOCKED));
4494
4495        if (beforeNextClient(c) == C_ERR) {
4496            /* If the client is no longer valid, we avoid
4497             * processing the client later. So we just go
4498             * to the next. */
4499            continue;
4500        }
4501
4502        /* Once io-threads are idle we can update the client in the mem usage */
4503        updateClientMemUsageAndBucket(c);
4504
4505        if (processPendingCommandAndInputBuffer(c) == C_ERR) {
4506            /* If the client is no longer valid, we avoid
4507             * processing the client later. So we just go
4508             * to the next. */
4509            continue;
4510        }
4511
4512        /* We may have pending replies if a thread readQueryFromClient() produced
4513         * replies and did not put the client in pending write queue (it can't).
4514         */
4515        if (!(c->flags & CLIENT_PENDING_WRITE) && clientHasPendingReplies(c))
4516            putClientInPendingWriteQueue(c);
4517    }
4518
4519    /* Update processed count on server */
4520    server.stat_io_reads_processed += processed;
4521
4522    return processed;
4523}

4525/* Returns the actual client eviction limit based on current configuration or
4526 * 0 if no limit. */
4527size_t getClientEvictionLimit(void) {
4528    size_t maxmemory_clients_actual = SIZE_MAX;
4529
4530    /* Handle percentage of maxmemory*/
4531    if (server.maxmemory_clients < 0 && server.maxmemory > 0) {
4532        unsigned long long maxmemory_clients_bytes = (unsigned long long)((double)server.maxmemory * -(double) server.maxmemory_clients / 100);
4533        if (maxmemory_clients_bytes <= SIZE_MAX)
4534            maxmemory_clients_actual = maxmemory_clients_bytes;
4535    }
4536    else if (server.maxmemory_clients > 0)
4537        maxmemory_clients_actual = server.maxmemory_clients;
4538    else
4539        return 0;
4540
4541    /* Don't allow a too small maxmemory-clients to avoid cases where we can't communicate
4542     * at all with the server because of bad configuration */
4543    if (maxmemory_clients_actual < 1024*128)
4544        maxmemory_clients_actual = 1024*128;
4545
4546    return maxmemory_clients_actual;
4547}

4549void evictClients(void) {
4550    if (!server.client_mem_usage_buckets)
4551        return;
4552    /* Start eviction from topmost bucket (largest clients) */
4553    int curr_bucket = CLIENT_MEM_USAGE_BUCKETS-1;
4554    listIter bucket_iter;
4555    listRewind(server.client_mem_usage_buckets[curr_bucket].clients, &bucket_iter);
4556    size_t client_eviction_limit = getClientEvictionLimit();
4557    if (client_eviction_limit == 0)
4558        return;
4559    while (server.stat_clients_type_memory[CLIENT_TYPE_NORMAL] +
4560           server.stat_clients_type_memory[CLIENT_TYPE_PUBSUB] >= client_eviction_limit) {
4561        listNode *ln = listNext(&bucket_iter);
4562        if (ln) {
4563            client *c = ln->value;
4564            sds ci = catClientInfoString(sdsempty(),c);
4565            serverLog(LL_NOTICE, "Evicting client: %s", ci);
4566            freeClient(c);
4567            sdsfree(ci);
4568            server.stat_evictedclients++;
4569        } else {
4570            curr_bucket--;
4571            if (curr_bucket < 0) {
4572                serverLog(LL_WARNING, "Over client maxmemory after evicting all evictable clients");
4573                break;
4574            }
4575            listRewind(server.client_mem_usage_buckets[curr_bucket].clients, &bucket_iter);
4576        }
4577    }
