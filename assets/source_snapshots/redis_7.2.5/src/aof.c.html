54/* ----------------------------------------------------------------------------
55 * AOF Manifest file implementation.
56 *
57 * The following code implements the read/write logic of AOF manifest file, which
58 * is used to track and manage all AOF files.
59 *
60 * Append-only files consist of three types:
61 *
62 * BASE: Represents a Redis snapshot from the time of last AOF rewrite. The manifest
63 * file contains at most a single BASE file, which will always be the first file in the
64 * list.
65 *
66 * INCR: Represents all write commands executed by Redis following the last successful
67 * AOF rewrite. In some cases it is possible to have several ordered INCR files. For
68 * example:
69 *   - During an on-going AOF rewrite
70 *   - After an AOF rewrite was aborted/failed, and before the next one succeeded.
71 *
72 * HISTORY: After a successful rewrite, the previous BASE and INCR become HISTORY files.
73 * They will be automatically removed unless garbage collection is disabled.
74 *
75 * The following is a possible AOF manifest file content:
76 *
77 * file appendonly.aof.2.base.rdb seq 2 type b
78 * file appendonly.aof.1.incr.aof seq 1 type h
79 * file appendonly.aof.2.incr.aof seq 2 type h
80 * file appendonly.aof.3.incr.aof seq 3 type h
81 * file appendonly.aof.4.incr.aof seq 4 type i
82 * file appendonly.aof.5.incr.aof seq 5 type i
83 * ------------------------------------------------------------------------- */

120/* Format aofInfo as a string and it will be a line in the manifest.
121 *
122 * When update this format, make sure to update redis-check-aof as well. */
123sds aofInfoFormat(sds buf, aofInfo *ai) {
124    sds filename_repr = NULL;
125
126    if (sdsneedsrepr(ai->file_name))
127        filename_repr = sdscatrepr(sdsempty(), ai->file_name, sdslen(ai->file_name));
128
129    sds ret = sdscatprintf(buf, "%s %s %s %lld %s %c\n",
130        AOF_MANIFEST_KEY_FILE_NAME, filename_repr ? filename_repr : ai->file_name,
131        AOF_MANIFEST_KEY_FILE_SEQ, ai->file_seq,
132        AOF_MANIFEST_KEY_FILE_TYPE, ai->file_type);
133    sdsfree(filename_repr);
134
135    return ret;
136}

179/* Returns the string representation of aofManifest pointed to by am.
180 *
181 * The string is multiple lines separated by '\n', and each line represents
182 * an AOF file.
183 *
184 * Each line is space delimited and contains 6 fields, as follows:
185 * "file" [filename] "seq" [sequence] "type" [type]
186 *
187 * Where "file", "seq" and "type" are keywords that describe the next value,
188 * [filename] and [sequence] describe file name and order, and [type] is one
189 * of 'b' (base), 'h' (history) or 'i' (incr).
190 *
191 * The base file, if exists, will always be first, followed by history files,
192 * and incremental files.
193 */
194sds getAofManifestAsString(aofManifest *am) {
195    serverAssert(am != NULL);
196
197    sds buf = sdsempty();
198    listNode *ln;
199    listIter li;
200
201    /* 1. Add BASE File information, it is always at the beginning
202     * of the manifest file. */
203    if (am->base_aof_info) {
204        buf = aofInfoFormat(buf, am->base_aof_info);
205    }
206
207    /* 2. Add HISTORY type AOF information. */
208    listRewind(am->history_aof_list, &li);
209    while ((ln = listNext(&li)) != NULL) {
210        aofInfo *ai = (aofInfo*)ln->value;
211        buf = aofInfoFormat(buf, ai);
212    }
213
214    /* 3. Add INCR type AOF information. */
215    listRewind(am->incr_aof_list, &li);
216    while ((ln = listNext(&li)) != NULL) {
217        aofInfo *ai = (aofInfo*)ln->value;
218        buf = aofInfoFormat(buf, ai);
219    }
220
221    return buf;
222}

224/* Load the manifest information from the disk to `server.aof_manifest`
225 * when the Redis server start.
226 *
227 * During loading, this function does strict error checking and will abort
228 * the entire Redis server process on error (I/O error, invalid format, etc.)
229 *
230 * If the AOF directory or manifest file do not exist, this will be ignored
231 * in order to support seamless upgrades from previous versions which did not
232 * use them.
233 */
234void aofLoadManifestFromDisk(void) {
235    server.aof_manifest = aofManifestCreate();
236    if (!dirExists(server.aof_dirname)) {
237        serverLog(LL_DEBUG, "The AOF directory %s doesn't exist", server.aof_dirname);
238        return;
239    }
240
241    sds am_name = getAofManifestFileName();
242    sds am_filepath = makePath(server.aof_dirname, am_name);
243    if (!fileExist(am_filepath)) {
244        serverLog(LL_DEBUG, "The AOF manifest file %s doesn't exist", am_name);
245        sdsfree(am_name);
246        sdsfree(am_filepath);
247        return;
248    }
249
250    aofManifest *am = aofLoadManifestFromFile(am_filepath);
251    if (am) aofManifestFreeAndUpdate(am);
252    sdsfree(am_name);
253    sdsfree(am_filepath);
254}

256/* Generic manifest loading function, used in `aofLoadManifestFromDisk` and redis-check-aof tool. */
257#define MANIFEST_MAX_LINE 1024
258aofManifest *aofLoadManifestFromFile(sds am_filepath) {
259    const char *err = NULL;
260    long long maxseq = 0;
261
262    aofManifest *am = aofManifestCreate();
263    FILE *fp = fopen(am_filepath, "r");
264    if (fp == NULL) {
265        serverLog(LL_WARNING, "Fatal error: can't open the AOF manifest "
266            "file %s for reading: %s", am_filepath, strerror(errno));
267        exit(1);
268    }
269
270    char buf[MANIFEST_MAX_LINE+1];
271    sds *argv = NULL;
272    int argc;
273    aofInfo *ai = NULL;
274
275    sds line = NULL;
276    int linenum = 0;
277
278    while (1) {
279        if (fgets(buf, MANIFEST_MAX_LINE+1, fp) == NULL) {
280            if (feof(fp)) {
281                if (linenum == 0) {
282                    err = "Found an empty AOF manifest";
283                    goto loaderr;
284                } else {
285                    break;
286                }
287            } else {
288                err = "Read AOF manifest failed";
289                goto loaderr;
290            }
291        }
292
293        linenum++;
294
295        /* Skip comments lines */
296        if (buf[0] == '#') continue;
297
298        if (strchr(buf, '\n') == NULL) {
299            err = "The AOF manifest file contains too long line";
300            goto loaderr;
301        }
302
303        line = sdstrim(sdsnew(buf), " \t\r\n");
304        if (!sdslen(line)) {
305            err = "Invalid AOF manifest file format";
306            goto loaderr;
307        }
308
309        argv = sdssplitargs(line, &argc);
310        /* 'argc < 6' was done for forward compatibility. */
311        if (argv == NULL || argc < 6 || (argc % 2)) {
312            err = "Invalid AOF manifest file format";
313            goto loaderr;
314        }
315
316        ai = aofInfoCreate();
317        for (int i = 0; i < argc; i += 2) {
318            if (!strcasecmp(argv[i], AOF_MANIFEST_KEY_FILE_NAME)) {
319                ai->file_name = sdsnew(argv[i+1]);
320                if (!pathIsBaseName(ai->file_name)) {
321                    err = "File can't be a path, just a filename";
322                    goto loaderr;
323                }
324            } else if (!strcasecmp(argv[i], AOF_MANIFEST_KEY_FILE_SEQ)) {
325                ai->file_seq = atoll(argv[i+1]);
326            } else if (!strcasecmp(argv[i], AOF_MANIFEST_KEY_FILE_TYPE)) {
327                ai->file_type = (argv[i+1])[0];
328            }
329            /* else if (!strcasecmp(argv[i], AOF_MANIFEST_KEY_OTHER)) {} */
330        }
331
332        /* We have to make sure we load all the information. */
333        if (!ai->file_name || !ai->file_seq || !ai->file_type) {
334            err = "Invalid AOF manifest file format";
335            goto loaderr;
336        }
337
338        sdsfreesplitres(argv, argc);
339        argv = NULL;
340
341        if (ai->file_type == AOF_FILE_TYPE_BASE) {
342            if (am->base_aof_info) {
343                err = "Found duplicate base file information";
344                goto loaderr;
345            }
346            am->base_aof_info = ai;
347            am->curr_base_file_seq = ai->file_seq;
348        } else if (ai->file_type == AOF_FILE_TYPE_HIST) {
349            listAddNodeTail(am->history_aof_list, ai);
350        } else if (ai->file_type == AOF_FILE_TYPE_INCR) {
351            if (ai->file_seq <= maxseq) {
352                err = "Found a non-monotonic sequence number";
353                goto loaderr;
354            }
355            listAddNodeTail(am->incr_aof_list, ai);
356            am->curr_incr_file_seq = ai->file_seq;
357            maxseq = ai->file_seq;
358        } else {
359            err = "Unknown AOF file type";
360            goto loaderr;
361        }
362
363        sdsfree(line);
364        line = NULL;
365        ai = NULL;
366    }
367
368    fclose(fp);
369    return am;
370
371loaderr:
372    /* Sanitizer suppression: may report a false positive if we goto loaderr
373     * and exit(1) without freeing these allocations. */
374    if (argv) sdsfreesplitres(argv, argc);
375    if (ai) aofInfoFree(ai);
376
377    serverLog(LL_WARNING, "\n*** FATAL AOF MANIFEST FILE ERROR ***\n");
378    if (line) {
379        serverLog(LL_WARNING, "Reading the manifest file, at line %d\n", linenum);
380        serverLog(LL_WARNING, ">>> '%s'\n", line);
381    }
382    serverLog(LL_WARNING, "%s\n", err);
383    exit(1);
384}

386/* Deep copy an aofManifest from orig.
387 *
388 * In `backgroundRewriteDoneHandler` and `openNewIncrAofForAppend`, we will
389 * first deep copy a temporary AOF manifest from the `server.aof_manifest` and
390 * try to modify it. Once everything is modified, we will atomically make the
391 * `server.aof_manifest` point to this temporary aof_manifest.
392 */
393aofManifest *aofManifestDup(aofManifest *orig) {
394    serverAssert(orig != NULL);
395    aofManifest *am = zcalloc(sizeof(aofManifest));
396
397    am->curr_base_file_seq = orig->curr_base_file_seq;
398    am->curr_incr_file_seq = orig->curr_incr_file_seq;
399    am->dirty = orig->dirty;
400
401    if (orig->base_aof_info) {
402        am->base_aof_info = aofInfoDup(orig->base_aof_info);
403    }
404
405    am->incr_aof_list = listDup(orig->incr_aof_list);
406    am->history_aof_list = listDup(orig->history_aof_list);
407    serverAssert(am->incr_aof_list != NULL);
408    serverAssert(am->history_aof_list != NULL);
409    return am;
410}

420/* Called in `backgroundRewriteDoneHandler` to get a new BASE file
421 * name, and mark the previous (if we have) BASE file as HISTORY type.
422 *
423 * BASE file naming rules: `server.aof_filename`.seq.base.format
424 *
425 * for example:
426 *  appendonly.aof.1.base.aof  (server.aof_use_rdb_preamble is no)
427 *  appendonly.aof.1.base.rdb  (server.aof_use_rdb_preamble is yes)
428 */
429sds getNewBaseFileNameAndMarkPreAsHistory(aofManifest *am) {
430    serverAssert(am != NULL);
431    if (am->base_aof_info) {
432        serverAssert(am->base_aof_info->file_type == AOF_FILE_TYPE_BASE);
433        am->base_aof_info->file_type = AOF_FILE_TYPE_HIST;
434        listAddNodeHead(am->history_aof_list, am->base_aof_info);
435    }
436
437    char *format_suffix = server.aof_use_rdb_preamble ?
438        RDB_FORMAT_SUFFIX:AOF_FORMAT_SUFFIX;
439
440    aofInfo *ai = aofInfoCreate();
441    ai->file_name = sdscatprintf(sdsempty(), "%s.%lld%s%s", server.aof_filename,
442                        ++am->curr_base_file_seq, BASE_FILE_SUFFIX, format_suffix);
443    ai->file_seq = am->curr_base_file_seq;
444    ai->file_type = AOF_FILE_TYPE_BASE;
445    am->base_aof_info = ai;
446    am->dirty = 1;
447    return am->base_aof_info->file_name;
448}

489/* Called in `backgroundRewriteDoneHandler`. when AOFRW success, This
490 * function will change the AOF file type in 'incr_aof_list' from
491 * AOF_FILE_TYPE_INCR to AOF_FILE_TYPE_HIST, and move them to the
492 * 'history_aof_list'.
493 */
494void markRewrittenIncrAofAsHistory(aofManifest *am) {
495    serverAssert(am != NULL);
496    if (!listLength(am->incr_aof_list)) {
497        return;
498    }
499
500    listNode *ln;
501    listIter li;
502
503    listRewindTail(am->incr_aof_list, &li);
504
505    /* "server.aof_fd != -1" means AOF enabled, then we must skip the
506     * last AOF, because this file is our currently writing. */
507    if (server.aof_fd != -1) {
508        ln = listNext(&li);
509        serverAssert(ln != NULL);
510    }
511
512    /* Move aofInfo from 'incr_aof_list' to 'history_aof_list'. */
513    while ((ln = listNext(&li)) != NULL) {
514        aofInfo *ai = (aofInfo*)ln->value;
515        serverAssert(ai->file_type == AOF_FILE_TYPE_INCR);
516
517        aofInfo *hai = aofInfoDup(ai);
518        hai->file_type = AOF_FILE_TYPE_HIST;
519        listAddNodeHead(am->history_aof_list, hai);
520        listDelNode(am->incr_aof_list, ln);
521    }
522
523    am->dirty = 1;
524}

526/* Write the formatted manifest string to disk. */
527int writeAofManifestFile(sds buf) {
528    int ret = C_OK;
529    ssize_t nwritten;
530    int len;
531
532    sds am_name = getAofManifestFileName();
533    sds am_filepath = makePath(server.aof_dirname, am_name);
534    sds tmp_am_name = getTempAofManifestFileName();
535    sds tmp_am_filepath = makePath(server.aof_dirname, tmp_am_name);
536
537    int fd = open(tmp_am_filepath, O_WRONLY|O_TRUNC|O_CREAT, 0644);
538    if (fd == -1) {
539        serverLog(LL_WARNING, "Can't open the AOF manifest file %s: %s",
540            tmp_am_name, strerror(errno));
541
542        ret = C_ERR;
543        goto cleanup;
544    }
545
546    len = sdslen(buf);
547    while(len) {
548        nwritten = write(fd, buf, len);
549
550        if (nwritten < 0) {
551            if (errno == EINTR) continue;
552
553            serverLog(LL_WARNING, "Error trying to write the temporary AOF manifest file %s: %s",
554                tmp_am_name, strerror(errno));
555
556            ret = C_ERR;
557            goto cleanup;
558        }
559
560        len -= nwritten;
561        buf += nwritten;
562    }
563
564    if (redis_fsync(fd) == -1) {
565        serverLog(LL_WARNING, "Fail to fsync the temp AOF file %s: %s.",
566            tmp_am_name, strerror(errno));
567
568        ret = C_ERR;
569        goto cleanup;
570    }
571
572    if (rename(tmp_am_filepath, am_filepath) != 0) {
573        serverLog(LL_WARNING,
574            "Error trying to rename the temporary AOF manifest file %s into %s: %s",
575            tmp_am_name, am_name, strerror(errno));
576
577        ret = C_ERR;
578        goto cleanup;
579    }
580
581    /* Also sync the AOF directory as new AOF files may be added in the directory */
582    if (fsyncFileDir(am_filepath) == -1) {
583        serverLog(LL_WARNING, "Fail to fsync AOF directory %s: %s.",
584            am_filepath, strerror(errno));
585
586        ret = C_ERR;
587        goto cleanup;
588    }
589
590cleanup:
591    if (fd != -1) close(fd);
592    sdsfree(am_name);
593    sdsfree(am_filepath);
594    sdsfree(tmp_am_name);
595    sdsfree(tmp_am_filepath);
596    return ret;
597}

612/* Called in `loadAppendOnlyFiles` when we upgrade from a old version redis.
613 *
614 * 1) Create AOF directory use 'server.aof_dirname' as the name.
615 * 2) Use 'server.aof_filename' to construct a BASE type aofInfo and add it to
616 *    aofManifest, then persist the manifest file to AOF directory.
617 * 3) Move the old AOF file (server.aof_filename) to AOF directory.
618 *
619 * If any of the above steps fails or crash occurs, this will not cause any
620 * problems, and redis will retry the upgrade process when it restarts.
621 */
622void aofUpgradePrepare(aofManifest *am) {
623    serverAssert(!aofFileExist(server.aof_filename));
624
625    /* Create AOF directory use 'server.aof_dirname' as the name. */
626    if (dirCreateIfMissing(server.aof_dirname) == -1) {
627        serverLog(LL_WARNING, "Can't open or create append-only dir %s: %s",
628            server.aof_dirname, strerror(errno));
629        exit(1);
630    }
631
632    /* Manually construct a BASE type aofInfo and add it to aofManifest. */
633    if (am->base_aof_info) aofInfoFree(am->base_aof_info);
634    aofInfo *ai = aofInfoCreate();
635    ai->file_name = sdsnew(server.aof_filename);
636    ai->file_seq = 1;
637    ai->file_type = AOF_FILE_TYPE_BASE;
638    am->base_aof_info = ai;
639    am->curr_base_file_seq = 1;
640    am->dirty = 1;
641
642    /* Persist the manifest file to AOF directory. */
643    if (persistAofManifest(am) != C_OK) {
644        exit(1);
645    }
646
647    /* Move the old AOF file to AOF directory. */
648    sds aof_filepath = makePath(server.aof_dirname, server.aof_filename);
649    if (rename(server.aof_filename, aof_filepath) == -1) {

656        exit(1);
657    }
658    sdsfree(aof_filepath);
659
660    serverLog(LL_NOTICE, "Successfully migrated an old-style AOF file (%s) into the AOF directory (%s).",
661        server.aof_filename, server.aof_dirname);
662}

707/* Called after `loadDataFromDisk` when redis start. If `server.aof_state` is
708 * 'AOF_ON', It will do three things:
709 * 1. Force create a BASE file when redis starts with an empty dataset
710 * 2. Open the last opened INCR type AOF for writing, If not, create a new one
711 * 3. Synchronously update the manifest file to the disk
712 *
713 * If any of the above steps fails, the redis process will exit.
714 */
715void aofOpenIfNeededOnServerStart(void) {
716    if (server.aof_state != AOF_ON) {
717        return;
718    }
719
720    serverAssert(server.aof_manifest != NULL);
721    serverAssert(server.aof_fd == -1);
722
723    if (dirCreateIfMissing(server.aof_dirname) == -1) {
724        serverLog(LL_WARNING, "Can't open or create append-only dir %s: %s",
725            server.aof_dirname, strerror(errno));
726        exit(1);
727    }
728
729    /* If we start with an empty dataset, we will force create a BASE file. */
730    size_t incr_aof_len = listLength(server.aof_manifest->incr_aof_list);
731    if (!server.aof_manifest->base_aof_info && !incr_aof_len) {
732        sds base_name = getNewBaseFileNameAndMarkPreAsHistory(server.aof_manifest);
733        sds base_filepath = makePath(server.aof_dirname, base_name);
734        if (rewriteAppendOnlyFile(base_filepath) != C_OK) {
735            exit(1);
736        }
737        sdsfree(base_filepath);
738        serverLog(LL_NOTICE, "Creating AOF base file %s on server start",
739            base_name);
740    }
741
742    /* Because we will 'exit(1)' if open AOF or persistent manifest fails, so
743     * we don't need atomic modification here. */
744    sds aof_name = getLastIncrAofName(server.aof_manifest);
745
746    /* Here we should use 'O_APPEND' flag. */
747    sds aof_filepath = makePath(server.aof_dirname, aof_name);
748    server.aof_fd = open(aof_filepath, O_WRONLY|O_APPEND|O_CREAT, 0644);
749    sdsfree(aof_filepath);
750    if (server.aof_fd == -1) {
751        serverLog(LL_WARNING, "Can't open the append-only file %s: %s",
752            aof_name, strerror(errno));
753        exit(1);
754    }
755
756    /* Persist our changes. */
757    int ret = persistAofManifest(server.aof_manifest);
758    if (ret != C_OK) {
759        exit(1);
760    }
761
762    server.aof_last_incr_size = getAppendOnlyFileSize(aof_name, NULL);
763    server.aof_last_incr_fsync_offset = server.aof_last_incr_size;
764
765    if (incr_aof_len) {
766        serverLog(LL_NOTICE, "Opening AOF incr file %s on server start", aof_name);
767    } else {
768        serverLog(LL_NOTICE, "Creating AOF incr file %s on server start", aof_name);
769    }
770}

779/* Called in `rewriteAppendOnlyFileBackground`. If `server.aof_state`
780 * is 'AOF_ON', It will do two things:
781 * 1. Open a new INCR type AOF for writing
782 * 2. Synchronously update the manifest file to the disk
783 *
784 * The above two steps of modification are atomic, that is, if
785 * any step fails, the entire operation will rollback and returns
786 * C_ERR, and if all succeeds, it returns C_OK.
787 * 
788 * If `server.aof_state` is 'AOF_WAIT_REWRITE', It will open a temporary INCR AOF 
789 * file to accumulate data during AOF_WAIT_REWRITE, and it will eventually be 
790 * renamed in the `backgroundRewriteDoneHandler` and written to the manifest file.
791 * */
792int openNewIncrAofForAppend(void) {
793    serverAssert(server.aof_manifest != NULL);
794    int newfd = -1;
795    aofManifest *temp_am = NULL;
796    sds new_aof_name = NULL;
797
798    /* Only open new INCR AOF when AOF enabled. */
799    if (server.aof_state == AOF_OFF) return C_OK;
800
801    /* Open new AOF. */
802    if (server.aof_state == AOF_WAIT_REWRITE) {
803        /* Use a temporary INCR AOF file to accumulate data during AOF_WAIT_REWRITE. */
804        new_aof_name = getTempIncrAofName();
805    } else {
806        /* Dup a temp aof_manifest to modify. */
807        temp_am = aofManifestDup(server.aof_manifest);
808        new_aof_name = sdsdup(getNewIncrAofName(temp_am));
809    }
810    sds new_aof_filepath = makePath(server.aof_dirname, new_aof_name);
811    newfd = open(new_aof_filepath, O_WRONLY|O_TRUNC|O_CREAT, 0644);
812    sdsfree(new_aof_filepath);
813    if (newfd == -1) {
814        serverLog(LL_WARNING, "Can't open the append-only file %s: %s",
815            new_aof_name, strerror(errno));
816        goto cleanup;
817    }
818
819    if (temp_am) {
820        /* Persist AOF Manifest. */
821        if (persistAofManifest(temp_am) == C_ERR) {
822            goto cleanup;
823        }
824    }
825
826    serverLog(LL_NOTICE, "Creating AOF incr file %s on background rewrite",
827            new_aof_name);
828    sdsfree(new_aof_name);
829
830    /* If reaches here, we can safely modify the `server.aof_manifest`
831     * and `server.aof_fd`. */
832
833    /* fsync and close old aof_fd if needed. In fsync everysec it's ok to delay
834     * the fsync as long as we grantee it happens, and in fsync always the file
835     * is already synced at this point so fsync doesn't matter. */
836    if (server.aof_fd != -1) {
837        aof_background_fsync_and_close(server.aof_fd);
838        server.aof_last_fsync = server.unixtime;
839    }
840    server.aof_fd = newfd;
841
842    /* Reset the aof_last_incr_size. */
843    server.aof_last_incr_size = 0;
844    /* Reset the aof_last_incr_fsync_offset. */
845    server.aof_last_incr_fsync_offset = 0;
846    /* Update `server.aof_manifest`. */
847    if (temp_am) aofManifestFreeAndUpdate(temp_am);
848    return C_OK;
849
850cleanup:
851    if (new_aof_name) sdsfree(new_aof_name);
852    if (newfd != -1) close(newfd);
853    if (temp_am) aofManifestFree(temp_am);
854    return C_ERR;
855}

857/* Whether to limit the execution of Background AOF rewrite.
858 *
859 * At present, if AOFRW fails, redis will automatically retry. If it continues
860 * to fail, we may get a lot of very small INCR files. so we need an AOFRW
861 * limiting measure.
862 *
863 * We can't directly use `server.aof_current_size` and `server.aof_last_incr_size`,
864 * because there may be no new writes after AOFRW fails.
865 *
866 * So, we use time delay to achieve our goal. When AOFRW fails, we delay the execution
867 * of the next AOFRW by 1 minute. If the next AOFRW also fails, it will be delayed by 2
868 * minutes. The next is 4, 8, 16, the maximum delay is 60 minutes (1 hour).
869 *
870 * During the limit period, we can still use the 'bgrewriteaof' command to execute AOFRW
871 * immediately.
872 *
873 * Return 1 means that AOFRW is limited and cannot be executed. 0 means that we can execute
874 * AOFRW, which may be that we have reached the 'next_rewrite_time' or the number of INCR
875 * AOFs has not reached the limit threshold.
876 * */
877#define AOF_REWRITE_LIMITE_THRESHOLD    3
878#define AOF_REWRITE_LIMITE_MAX_MINUTES  60 /* 1 hour */
879int aofRewriteLimited(void) {
880    static int next_delay_minutes = 0;
881    static time_t next_rewrite_time = 0;
882
883    if (server.stat_aofrw_consecutive_failures < AOF_REWRITE_LIMITE_THRESHOLD) {
884        /* We may be recovering from limited state, so reset all states. */
885        next_delay_minutes = 0;
886        next_rewrite_time = 0;
887        return 0;
888    }
889
890    /* if it is in the limiting state, then check if the next_rewrite_time is reached */
891    if (next_rewrite_time != 0) {
892        if (server.unixtime < next_rewrite_time) {
893            return 1;
894        } else {
895            next_rewrite_time = 0;
896            return 0;
897        }
898    }
899
900    next_delay_minutes = (next_delay_minutes == 0) ? 1 : (next_delay_minutes * 2);
901    if (next_delay_minutes > AOF_REWRITE_LIMITE_MAX_MINUTES) {
902        next_delay_minutes = AOF_REWRITE_LIMITE_MAX_MINUTES;
903    }
904
905    next_rewrite_time = server.unixtime + next_delay_minutes * 60;
906    serverLog(LL_WARNING,
907        "Background AOF rewrite has repeatedly failed and triggered the limit, will retry in %d minutes", next_delay_minutes);
908    return 1;
909}

977 * at runtime using the CONFIG command. */

979    serverAssert(server.aof_state == AOF_OFF);

982    if (hasActiveChildProcess() && server.child_type != CHILD_TYPE_AOF) {
983        server.aof_rewrite_scheduled = 1;

985    } else if (server.in_exec){

988    } else {

990         * start a new one: the old one cannot be reused because it is not
991         * accumulating the AOF buffer. */
992        if (server.child_type == CHILD_TYPE_AOF) {

1002    }

1010        atomicSet(server.aof_bio_fsync_status,C_OK);
1011    }
1012

1014    if (server.aof_last_write_status == C_ERR) {
1015        serverLog(LL_WARNING,"AOF reopen, just ignore the last error.");
1016        server.aof_last_write_status = C_OK;
1017    }
1018    return C_OK;
1019}

1021/* This is a wrapper to the write syscall in order to retry on short writes
1022 * or if the syscall gets interrupted. It could look strange that we retry
1023 * on short writes given that we are writing to a block device: normally if
1024 * the first call is short, there is a end-of-space condition, so the next
1025 * is likely to fail. However apparently in modern systems this is no longer
1026 * true, and in general it looks just more resilient to retry the write. If
1027 * there is an actual error condition we'll get it at the next try. */
1028ssize_t aofWrite(int fd, const char *buf, size_t len) {
1029    ssize_t nwritten = 0, totwritten = 0;
1030
1031    while(len) {
1032        nwritten = write(fd, buf, len);
1033
1034        if (nwritten < 0) {
1035            if (errno == EINTR) continue;
1036            return totwritten ? totwritten : -1;
1037        }
1038
1039        len -= nwritten;
1040        buf += nwritten;
1041        totwritten += nwritten;
1042    }
1043
1044    return totwritten;
1045}

1047/* Write the append only file buffer on disk.
1048 *
1049 * Since we are required to write the AOF before replying to the client,
1050 * and the only way the client socket can get a write is entering when
1051 * the event loop, we accumulate all the AOF writes in a memory
1052 * buffer and write it on disk using this function just before entering
1053 * the event loop again.
1054 *
1055 * About the 'force' argument:
1056 *
1057 * When the fsync policy is set to 'everysec' we may delay the flush if there
1058 * is still an fsync() going on in the background thread, since for instance
1059 * on Linux write(2) will be blocked by the background fsync anyway.
1060 * When this happens we remember that there is some aof buffer to be
1061 * flushed ASAP, and will try to do that in the serverCron() function.
1062 *
1063 * However if force is set to 1 we'll write regardless of the background
1064 * fsync. */
1065#define AOF_WRITE_LOG_ERROR_RATE 30 /* Seconds between errors logging. */
1066void flushAppendOnlyFile(int force) {
1067    ssize_t nwritten;
1068    int sync_in_progress = 0;
1069    mstime_t latency;
1070
1071    if (sdslen(server.aof_buf) == 0) {
1072        /* Check if we need to do fsync even the aof buffer is empty,
1073         * because previously in AOF_FSYNC_EVERYSEC mode, fsync is
1074         * called only when aof buffer is not empty, so if users
1075         * stop write commands before fsync called in one second,
1076         * the data in page cache cannot be flushed in time. */
1077        if (server.aof_fsync == AOF_FSYNC_EVERYSEC &&
1078            server.aof_last_incr_fsync_offset != server.aof_last_incr_size &&
1079            server.unixtime > server.aof_last_fsync &&
1080            !(sync_in_progress = aofFsyncInProgress())) {
1081            goto try_fsync;
1082
1083        /* Check if we need to do fsync even the aof buffer is empty,
1084         * the reason is described in the previous AOF_FSYNC_EVERYSEC block,
1085         * and AOF_FSYNC_ALWAYS is also checked here to handle a case where
1086         * aof_fsync is changed from everysec to always. */
1087        } else if (server.aof_fsync == AOF_FSYNC_ALWAYS &&
1088                   server.aof_last_incr_fsync_offset != server.aof_last_incr_size)
1089        {
1090            goto try_fsync;
1091        } else {
1092            return;
1093        }
1094    }
1095
1096    if (server.aof_fsync == AOF_FSYNC_EVERYSEC)
1097        sync_in_progress = aofFsyncInProgress();
1098
1099    if (server.aof_fsync == AOF_FSYNC_EVERYSEC && !force) {
1100        /* With this append fsync policy we do background fsyncing.
1101         * If the fsync is still in progress we can try to delay
1102         * the write for a couple of seconds. */
1103        if (sync_in_progress) {
1104            if (server.aof_flush_postponed_start == 0) {
1105                /* No previous write postponing, remember that we are
1106                 * postponing the flush and return. */
1107                server.aof_flush_postponed_start = server.unixtime;
1108                return;
1109            } else if (server.unixtime - server.aof_flush_postponed_start < 2) {
1110                /* We were already waiting for fsync to finish, but for less
1111                 * than two seconds this is still ok. Postpone again. */
1112                return;
1113            }
1114            /* Otherwise fall through, and go write since we can't wait
1115             * over two seconds. */
1116            server.aof_delayed_fsync++;
1117            serverLog(LL_NOTICE,"Asynchronous AOF fsync is taking too long (disk is busy?). Writing the AOF buffer without waiting for fsync to complete, this may slow down Redis.");
1118        }
1119    }
1120    /* We want to perform a single write. This should be guaranteed atomic
1121     * at least if the filesystem we are writing is a real physical one.
1122     * While this will save us against the server being killed I don't think
1123     * there is much to do about the whole server stopping for power problems
1124     * or alike */
1125
1126    if (server.aof_flush_sleep && sdslen(server.aof_buf)) {
1127        usleep(server.aof_flush_sleep);
1128    }
1129
1130    latencyStartMonitor(latency);
1131    nwritten = aofWrite(server.aof_fd,server.aof_buf,sdslen(server.aof_buf));
1132    latencyEndMonitor(latency);
1133    /* We want to capture different events for delayed writes:
1134     * when the delay happens with a pending fsync, or with a saving child
1135     * active, and when the above two conditions are missing.
1136     * We also use an additional event name to save all samples which is
1137     * useful for graphing / monitoring purposes. */
1138    if (sync_in_progress) {
1139        latencyAddSampleIfNeeded("aof-write-pending-fsync",latency);
1140    } else if (hasActiveChildProcess()) {
1141        latencyAddSampleIfNeeded("aof-write-active-child",latency);
1142    } else {
1143        latencyAddSampleIfNeeded("aof-write-alone",latency);
1144    }
1145    latencyAddSampleIfNeeded("aof-write",latency);
1146
1147    /* We performed the write so reset the postponed flush sentinel to zero. */
1148    server.aof_flush_postponed_start = 0;
1149
1150    if (nwritten != (ssize_t)sdslen(server.aof_buf)) {
1151        static time_t last_write_error_log = 0;
1152        int can_log = 0;
1153
1154        /* Limit logging rate to 1 line per AOF_WRITE_LOG_ERROR_RATE seconds. */
1155        if ((server.unixtime - last_write_error_log) > AOF_WRITE_LOG_ERROR_RATE) {
1156            can_log = 1;
1157            last_write_error_log = server.unixtime;
1158        }
1159
1160        /* Log the AOF write error and record the error code. */
1161        if (nwritten == -1) {
1162            if (can_log) {
1163                serverLog(LL_WARNING,"Error writing to the AOF file: %s",
1164                    strerror(errno));
1165            }
1166            server.aof_last_write_errno = errno;
1167        } else {
1168            if (can_log) {
1169                serverLog(LL_WARNING,"Short write while writing to "
1170                                       "the AOF file: (nwritten=%lld, "
1171                                       "expected=%lld)",
1172                                       (long long)nwritten,
1173                                       (long long)sdslen(server.aof_buf));
1174            }
1175
1176            if (ftruncate(server.aof_fd, server.aof_last_incr_size) == -1) {
1177                if (can_log) {
1178                    serverLog(LL_WARNING, "Could not remove short write "
1179                             "from the append-only file.  Redis may refuse "
1180                             "to load the AOF the next time it starts.  "
1181                             "ftruncate: %s", strerror(errno));
1182                }
1183            } else {
1184                /* If the ftruncate() succeeded we can set nwritten to
1185                 * -1 since there is no longer partial data into the AOF. */
1186                nwritten = -1;
1187            }
1188            server.aof_last_write_errno = ENOSPC;
1189        }
1190
1191        /* Handle the AOF write error. */
1192        if (server.aof_fsync == AOF_FSYNC_ALWAYS) {
1193            /* We can't recover when the fsync policy is ALWAYS since the reply
1194             * for the client is already in the output buffers (both writes and
1195             * reads), and the changes to the db can't be rolled back. Since we
1196             * have a contract with the user that on acknowledged or observed
1197             * writes are is synced on disk, we must exit. */
1198            serverLog(LL_WARNING,"Can't recover from AOF write error when the AOF fsync policy is 'always'. Exiting...");
1199            exit(1);
1200        } else {
1201            /* Recover from failed write leaving data into the buffer. However
1202             * set an error to stop accepting writes as long as the error
1203             * condition is not cleared. */
1204            server.aof_last_write_status = C_ERR;
1205
1206            /* Trim the sds buffer if there was a partial write, and there
1207             * was no way to undo it with ftruncate(2). */
1208            if (nwritten > 0) {
1209                server.aof_current_size += nwritten;
1210                server.aof_last_incr_size += nwritten;
1211                sdsrange(server.aof_buf,nwritten,-1);
1212            }
1213            return; /* We'll try again on the next call... */
1214        }
1215    } else {
1216        /* Successful write(2). If AOF was in error state, restore the
1217         * OK state and log the event. */
1218        if (server.aof_last_write_status == C_ERR) {
1219            serverLog(LL_NOTICE,
1220                "AOF write error looks solved, Redis can write again.");
1221            server.aof_last_write_status = C_OK;
1222        }
1223    }
1224    server.aof_current_size += nwritten;
1225    server.aof_last_incr_size += nwritten;
1226
1227    /* Re-use AOF buffer when it is small enough. The maximum comes from the
1228     * arena size of 4k minus some overhead (but is otherwise arbitrary). */
1229    if ((sdslen(server.aof_buf)+sdsavail(server.aof_buf)) < 4000) {
1230        sdsclear(server.aof_buf);
1231    } else {
1232        sdsfree(server.aof_buf);
1233        server.aof_buf = sdsempty();
1234    }
1235
1236try_fsync:
1237    /* Don't fsync if no-appendfsync-on-rewrite is set to yes and there are
1238     * children doing I/O in the background. */
1239    if (server.aof_no_fsync_on_rewrite && hasActiveChildProcess())
1240        return;
1241
1242    /* Perform the fsync if needed. */
1243    if (server.aof_fsync == AOF_FSYNC_ALWAYS) {
1244        /* redis_fsync is defined as fdatasync() for Linux in order to avoid
1245         * flushing metadata. */
1246        latencyStartMonitor(latency);
1247        /* Let's try to get this data on the disk. To guarantee data safe when
1248         * the AOF fsync policy is 'always', we should exit if failed to fsync
1249         * AOF (see comment next to the exit(1) after write error above). */
1250        if (redis_fsync(server.aof_fd) == -1) {
1251            serverLog(LL_WARNING,"Can't persist AOF for fsync error when the "
1252              "AOF fsync policy is 'always': %s. Exiting...", strerror(errno));
1253            exit(1);
1254        }
1255        latencyEndMonitor(latency);
1256        latencyAddSampleIfNeeded("aof-fsync-always",latency);
1257        server.aof_last_incr_fsync_offset = server.aof_last_incr_size;
1258        server.aof_last_fsync = server.unixtime;
1259        atomicSet(server.fsynced_reploff_pending, server.master_repl_offset);
1260    } else if (server.aof_fsync == AOF_FSYNC_EVERYSEC &&
1261               server.unixtime > server.aof_last_fsync) {
1262        if (!sync_in_progress) {
1263            aof_background_fsync(server.aof_fd);
1264            server.aof_last_incr_fsync_offset = server.aof_last_incr_size;
1265        }
1266        server.aof_last_fsync = server.unixtime;
1267    }
1268}

1270sds catAppendOnlyGenericCommand(sds dst, int argc, robj **argv) {
1271    char buf[32];
1272    int len, j;
1273    robj *o;
1274
1275    buf[0] = '*';
1276    len = 1+ll2string(buf+1,sizeof(buf)-1,argc);
1277    buf[len++] = '\r';
1278    buf[len++] = '\n';
1279    dst = sdscatlen(dst,buf,len);
1280
1281    for (j = 0; j < argc; j++) {
1282        o = getDecodedObject(argv[j]);
1283        buf[0] = '$';
1284        len = 1+ll2string(buf+1,sizeof(buf)-1,sdslen(o->ptr));
1285        buf[len++] = '\r';
1286        buf[len++] = '\n';
1287        dst = sdscatlen(dst,buf,len);
1288        dst = sdscatlen(dst,o->ptr,sdslen(o->ptr));
1289        dst = sdscatlen(dst,"\r\n",2);
1290        decrRefCount(o);
1291    }
1292    return dst;

1295/* Generate a piece of timestamp annotation for AOF if current record timestamp
1296 * in AOF is not equal server unix time. If we specify 'force' argument to 1,
1297 * we would generate one without check, currently, it is useful in AOF rewriting
1298 * child process which always needs to record one timestamp at the beginning of
1299 * rewriting AOF.
1300 *
1301 * Timestamp annotation format is "#TS:${timestamp}\r\n". "TS" is short of
1302 * timestamp and this method could save extra bytes in AOF. */
1303sds genAofTimestampAnnotationIfNeeded(int force) {
1304    sds ts = NULL;
1305
1306    if (force || server.aof_cur_timestamp < server.unixtime) {
1307        server.aof_cur_timestamp = force ? time(NULL) : server.unixtime;
1308        ts = sdscatfmt(sdsempty(), "#TS:%I\r\n", server.aof_cur_timestamp);
1309        serverAssert(sdslen(ts) <= AOF_ANNOTATION_LINE_MAX_LEN);
1310    }
1311    return ts;
1312}

1314/* Write the given command to the aof file.
1315 * dictid - dictionary id the command should be applied to,
1316 *          this is used in order to decide if a `select` command
1317 *          should also be written to the aof. Value of -1 means
1318 *          to avoid writing `select` command in any case.
1319 * argv   - The command to write to the aof.
1320 * argc   - Number of values in argv
1321 */
1322void feedAppendOnlyFile(int dictid, robj **argv, int argc) {
1323    sds buf = sdsempty();
1324
1325    serverAssert(dictid == -1 || (dictid >= 0 && dictid < server.dbnum));
1326
1327    /* Feed timestamp if needed */
1328    if (server.aof_timestamp_enabled) {
1329        sds ts = genAofTimestampAnnotationIfNeeded(0);
1330        if (ts != NULL) {
1331            buf = sdscatsds(buf, ts);
1332            sdsfree(ts);
1333        }
1334    }
1335
1336    /* The DB this command was targeting is not the same as the last command
1337     * we appended. To issue a SELECT command is needed. */
1338    if (dictid != -1 && dictid != server.aof_selected_db) {
1339        char seldb[64];
1340
1341        snprintf(seldb,sizeof(seldb),"%d",dictid);
1342        buf = sdscatprintf(buf,"*2\r\n$6\r\nSELECT\r\n$%lu\r\n%s\r\n",
1343            (unsigned long)strlen(seldb),seldb);
1344        server.aof_selected_db = dictid;
1345    }
1346
1347    /* All commands should be propagated the same way in AOF as in replication.
1348     * No need for AOF-specific translation. */
1349    buf = catAppendOnlyGenericCommand(buf,argc,argv);
1350
1351    /* Append to the AOF buffer. This will be flushed on disk just before
1352     * of re-entering the event loop, so before the client will get a
1353     * positive reply about the operation performed. */
1354    if (server.aof_state == AOF_ON ||
1355        (server.aof_state == AOF_WAIT_REWRITE && server.child_type == CHILD_TYPE_AOF))
1356    {
1357        server.aof_buf = sdscatlen(server.aof_buf, buf, sdslen(buf));
1358    }
1359
1360    sdsfree(buf);
1361}

1367/* In Redis commands are always executed in the context of a client, so in
1368 * order to load the append only file we need to create a fake client. */
1369struct client *createAOFClient(void) {
1370    struct client *c = createClient(NULL);
1371
1372    c->id = CLIENT_ID_AOF; /* So modules can identify it's the AOF client. */
1373
1374    /*
1375     * The AOF client should never be blocked (unlike master
1376     * replication connection).
1377     * This is because blocking the AOF client might cause
1378     * deadlock (because potentially no one will unblock it).
1379     * Also, if the AOF client will be blocked just for
1380     * background processing there is a chance that the
1381     * command execution order will be violated.
1382     */
1383    c->flags = CLIENT_DENY_BLOCKING;
1384
1385    /* We set the fake client as a slave waiting for the synchronization
1386     * so that Redis will not try to send replies to this client. */
1387    c->replstate = SLAVE_STATE_WAIT_BGSAVE_START;
1388    return c;
1389}

1391/* Replay an append log file. On success AOF_OK or AOF_TRUNCATED is returned,
1392 * otherwise, one of the following is returned:
1393 * AOF_OPEN_ERR: Failed to open the AOF file.
1394 * AOF_NOT_EXIST: AOF file doesn't exist.
1395 * AOF_EMPTY: The AOF file is empty (nothing to load).
1396 * AOF_FAILED: Failed to load the AOF file. */
1397int loadSingleAppendOnlyFile(char *filename) {
1398    struct client *fakeClient;
1399    struct redis_stat sb;
1400    int old_aof_state = server.aof_state;
1401    long loops = 0;
1402    off_t valid_up_to = 0; /* Offset of latest well-formed command loaded. */
1403    off_t valid_before_multi = 0; /* Offset before MULTI command loaded. */
1404    off_t last_progress_report_size = 0;
1405    int ret = AOF_OK;
1406
1407    sds aof_filepath = makePath(server.aof_dirname, filename);
1408    FILE *fp = fopen(aof_filepath, "r");
1409    if (fp == NULL) {
1410        int en = errno;
1411        if (redis_stat(aof_filepath, &sb) == 0 || errno != ENOENT) {
1412            serverLog(LL_WARNING,"Fatal error: can't open the append log file %s for reading: %s", filename, strerror(en));
1413            sdsfree(aof_filepath);
1414            return AOF_OPEN_ERR;
1415        } else {
1416            serverLog(LL_WARNING,"The append log file %s doesn't exist: %s", filename, strerror(errno));
1417            sdsfree(aof_filepath);
1418            return AOF_NOT_EXIST;
1419        }
1420    }
1421
1422    if (fp && redis_fstat(fileno(fp),&sb) != -1 && sb.st_size == 0) {
1423        fclose(fp);
1424        sdsfree(aof_filepath);
1425        return AOF_EMPTY;
1426    }
1427
1428    /* Temporarily disable AOF, to prevent EXEC from feeding a MULTI
1429     * to the same file we're about to read. */
1430    server.aof_state = AOF_OFF;
1431
1432    client *old_cur_client = server.current_client;
1433    client *old_exec_client = server.executing_client;
1434    fakeClient = createAOFClient();
1435    server.current_client = server.executing_client = fakeClient;
1436
1437    /* Check if the AOF file is in RDB format (it may be RDB encoded base AOF
1438     * or old style RDB-preamble AOF). In that case we need to load the RDB file 
1439     * and later continue loading the AOF tail if it is an old style RDB-preamble AOF. */
1440    char sig[5]; /* "REDIS" */
1441    if (fread(sig,1,5,fp) != 5 || memcmp(sig,"REDIS",5) != 0) {
1442        /* Not in RDB format, seek back at 0 offset. */
1443        if (fseek(fp,0,SEEK_SET) == -1) goto readerr;
1444    } else {
1445        /* RDB format. Pass loading the RDB functions. */
1446        rio rdb;
1447        int old_style = !strcmp(filename, server.aof_filename);
1448        if (old_style)
1449            serverLog(LL_NOTICE, "Reading RDB preamble from AOF file...");
1450        else 
1451            serverLog(LL_NOTICE, "Reading RDB base file on AOF loading..."); 
1452
1453        if (fseek(fp,0,SEEK_SET) == -1) goto readerr;
1454        rioInitWithFile(&rdb,fp);
1455        if (rdbLoadRio(&rdb,RDBFLAGS_AOF_PREAMBLE,NULL) != C_OK) {
1456            if (old_style)
1457                serverLog(LL_WARNING, "Error reading the RDB preamble of the AOF file %s, AOF loading aborted", filename);
1458            else
1459                serverLog(LL_WARNING, "Error reading the RDB base file %s, AOF loading aborted", filename);
1460
1461            ret = AOF_FAILED;
1462            goto cleanup;
1463        } else {
1464            loadingAbsProgress(ftello(fp));
1465            last_progress_report_size = ftello(fp);
1466            if (old_style) serverLog(LL_NOTICE, "Reading the remaining AOF tail...");
1467        }
1468    }
1469
1470    /* Read the actual AOF file, in REPL format, command by command. */
1471    while(1) {
1472        int argc, j;
1473        unsigned long len;
1474        robj **argv;
1475        char buf[AOF_ANNOTATION_LINE_MAX_LEN];
1476        sds argsds;
1477        struct redisCommand *cmd;
1478
1479        /* Serve the clients from time to time */
1480        if (!(loops++ % 1024)) {
1481            off_t progress_delta = ftello(fp) - last_progress_report_size;
1482            loadingIncrProgress(progress_delta);
1483            last_progress_report_size += progress_delta;
1484            processEventsWhileBlocked();
1485            processModuleLoadingProgressEvent(1);
1486        }
1487        if (fgets(buf,sizeof(buf),fp) == NULL) {
1488            if (feof(fp)) {
1489                break;
1490            } else {
1491                goto readerr;
1492            }
1493        }
1494        if (buf[0] == '#') continue; /* Skip annotations */
1495        if (buf[0] != '*') goto fmterr;
1496        if (buf[1] == '\0') goto readerr;
1497        argc = atoi(buf+1);
1498        if (argc < 1) goto fmterr;
1499        if ((size_t)argc > SIZE_MAX / sizeof(robj*)) goto fmterr;
1500
1501        /* Load the next command in the AOF as our fake client
1502         * argv. */
1503        argv = zmalloc(sizeof(robj*)*argc);
1504        fakeClient->argc = argc;
1505        fakeClient->argv = argv;
1506        fakeClient->argv_len = argc;
1507
1508        for (j = 0; j < argc; j++) {
1509            /* Parse the argument len. */
1510            char *readres = fgets(buf,sizeof(buf),fp);
1511            if (readres == NULL || buf[0] != '$') {
1512                fakeClient->argc = j; /* Free up to j-1. */
1513                freeClientArgv(fakeClient);
1514                if (readres == NULL)
1515                    goto readerr;
1516                else
1517                    goto fmterr;
1518            }
1519            len = strtol(buf+1,NULL,10);
1520
1521            /* Read it into a string object. */
1522            argsds = sdsnewlen(SDS_NOINIT,len);
1523            if (len && fread(argsds,len,1,fp) == 0) {
1524                sdsfree(argsds);
1525                fakeClient->argc = j; /* Free up to j-1. */
1526                freeClientArgv(fakeClient);
1527                goto readerr;
1528            }
1529            argv[j] = createObject(OBJ_STRING,argsds);
1530
1531            /* Discard CRLF. */
1532            if (fread(buf,2,1,fp) == 0) {
1533                fakeClient->argc = j+1; /* Free up to j. */
1534                freeClientArgv(fakeClient);
1535                goto readerr;
1536            }
1537        }
1538
1539        /* Command lookup */
1540        cmd = lookupCommand(argv,argc);
1541        if (!cmd) {
1542            serverLog(LL_WARNING,
1543                "Unknown command '%s' reading the append only file %s",
1544                (char*)argv[0]->ptr, filename);
1545            freeClientArgv(fakeClient);
1546            ret = AOF_FAILED;
1547            goto cleanup;
1548        }
1549
1550        if (cmd->proc == multiCommand) valid_before_multi = valid_up_to;
1551
1552        /* Run the command in the context of a fake client */
1553        fakeClient->cmd = fakeClient->lastcmd = cmd;
1554        if (fakeClient->flags & CLIENT_MULTI &&
1555            fakeClient->cmd->proc != execCommand)
1556        {
1557            /* Note: we don't have to attempt calling evalGetCommandFlags,
1558             * since this is AOF, the checks in processCommand are not made
1559             * anyway.*/
1560            queueMultiCommand(fakeClient, cmd->flags);
1561        } else {
1562            cmd->proc(fakeClient);
1563        }
1564
1565        /* The fake client should not have a reply */
1566        serverAssert(fakeClient->bufpos == 0 &&
1567                     listLength(fakeClient->reply) == 0);
1568
1569        /* The fake client should never get blocked */
1570        serverAssert((fakeClient->flags & CLIENT_BLOCKED) == 0);
1571
1572        /* Clean up. Command code may have changed argv/argc so we use the
1573         * argv/argc of the client instead of the local variables. */
1574        freeClientArgv(fakeClient);
1575        if (server.aof_load_truncated) valid_up_to = ftello(fp);
1576        if (server.key_load_delay)
1577            debugDelay(server.key_load_delay);
1578    }
1579
1580    /* This point can only be reached when EOF is reached without errors.
1581     * If the client is in the middle of a MULTI/EXEC, handle it as it was
1582     * a short read, even if technically the protocol is correct: we want
1583     * to remove the unprocessed tail and continue. */
1584    if (fakeClient->flags & CLIENT_MULTI) {
1585        serverLog(LL_WARNING,
1586            "Revert incomplete MULTI/EXEC transaction in AOF file %s", filename);
1587        valid_up_to = valid_before_multi;
1588        goto uxeof;
1589    }
1590
1591loaded_ok: /* DB loaded, cleanup and return success (AOF_OK or AOF_TRUNCATED). */
1592    loadingIncrProgress(ftello(fp) - last_progress_report_size);
1593    server.aof_state = old_aof_state;
1594    goto cleanup;
1595
1596readerr: /* Read error. If feof(fp) is true, fall through to unexpected EOF. */
1597    if (!feof(fp)) {
1598        serverLog(LL_WARNING,"Unrecoverable error reading the append only file %s: %s", filename, strerror(errno));
1599        ret = AOF_FAILED;
1600        goto cleanup;
1601    }
1602
1603uxeof: /* Unexpected AOF end of file. */
1604    if (server.aof_load_truncated) {
1605        serverLog(LL_WARNING,"!!! Warning: short read while loading the AOF file %s!!!", filename);
1606        serverLog(LL_WARNING,"!!! Truncating the AOF %s at offset %llu !!!",
1607            filename, (unsigned long long) valid_up_to);
1608        if (valid_up_to == -1 || truncate(aof_filepath,valid_up_to) == -1) {
1609            if (valid_up_to == -1) {
1610                serverLog(LL_WARNING,"Last valid command offset is invalid");
1611            } else {
1612                serverLog(LL_WARNING,"Error truncating the AOF file %s: %s",
1613                    filename, strerror(errno));
1614            }
1615        } else {
1616            /* Make sure the AOF file descriptor points to the end of the
1617             * file after the truncate call. */
1618            if (server.aof_fd != -1 && lseek(server.aof_fd,0,SEEK_END) == -1) {
1619                serverLog(LL_WARNING,"Can't seek the end of the AOF file %s: %s",
1620                    filename, strerror(errno));
1621            } else {
1622                serverLog(LL_WARNING,
1623                    "AOF %s loaded anyway because aof-load-truncated is enabled", filename);
1624                ret = AOF_TRUNCATED;
1625                goto loaded_ok;
1626            }
1627        }
1628    }
1629    serverLog(LL_WARNING, "Unexpected end of file reading the append only file %s. You can: "
1630        "1) Make a backup of your AOF file, then use ./redis-check-aof --fix <filename.manifest>. "
1631        "2) Alternatively you can set the 'aof-load-truncated' configuration option to yes and restart the server.", filename);
1632    ret = AOF_FAILED;
1633    goto cleanup;
1634
1635fmterr: /* Format error. */
1636    serverLog(LL_WARNING, "Bad file format reading the append only file %s: "
1637        "make a backup of your AOF file, then use ./redis-check-aof --fix <filename.manifest>", filename);
1638    ret = AOF_FAILED;
1639    /* fall through to cleanup. */
1640
1641cleanup:
1642    if (fakeClient) freeClient(fakeClient);
1643    server.current_client = old_cur_client;
1644    server.executing_client = old_exec_client;
1645    fclose(fp);
1646    sdsfree(aof_filepath);
1647    return ret;
1648}

1650/* Load the AOF files according the aofManifest pointed by am. */
1651int loadAppendOnlyFiles(aofManifest *am) {
1652    serverAssert(am != NULL);
1653    int status, ret = AOF_OK;
1654    long long start;
1655    off_t total_size = 0, base_size = 0;
1656    sds aof_name;
1657    int total_num, aof_num = 0, last_file;
1658
1659    /* If the 'server.aof_filename' file exists in dir, we may be starting
1660     * from an old redis version. We will use enter upgrade mode in three situations.
1661     *
1662     * 1. If the 'server.aof_dirname' directory not exist
1663     * 2. If the 'server.aof_dirname' directory exists but the manifest file is missing
1664     * 3. If the 'server.aof_dirname' directory exists and the manifest file it contains
1665     *    has only one base AOF record, and the file name of this base AOF is 'server.aof_filename',
1666     *    and the 'server.aof_filename' file not exist in 'server.aof_dirname' directory
1667     * */
1668    if (fileExist(server.aof_filename)) {
1669        if (!dirExists(server.aof_dirname) ||
1670            (am->base_aof_info == NULL && listLength(am->incr_aof_list) == 0) ||
1671            (am->base_aof_info != NULL && listLength(am->incr_aof_list) == 0 &&
1672             !strcmp(am->base_aof_info->file_name, server.aof_filename) && !aofFileExist(server.aof_filename)))
1673        {
1674            aofUpgradePrepare(am);
1675        }
1676    }
1677
1678    if (am->base_aof_info == NULL && listLength(am->incr_aof_list) == 0) {
1679        return AOF_NOT_EXIST;
1680    }
1681
1682    total_num = getBaseAndIncrAppendOnlyFilesNum(am);
1683    serverAssert(total_num > 0);
1684
1685    /* Here we calculate the total size of all BASE and INCR files in
1686     * advance, it will be set to `server.loading_total_bytes`. */
1687    total_size = getBaseAndIncrAppendOnlyFilesSize(am, &status);
1688    if (status != AOF_OK) {
1689        /* If an AOF exists in the manifest but not on the disk, we consider this to be a fatal error. */
1690        if (status == AOF_NOT_EXIST) status = AOF_FAILED;
1691
1692        return status;
1693    } else if (total_size == 0) {
1694        return AOF_EMPTY;
1695    }
1696
1697    startLoading(total_size, RDBFLAGS_AOF_PREAMBLE, 0);
1698
1699    /* Load BASE AOF if needed. */
1700    if (am->base_aof_info) {
1701        serverAssert(am->base_aof_info->file_type == AOF_FILE_TYPE_BASE);
1702        aof_name = (char*)am->base_aof_info->file_name;
1703        updateLoadingFileName(aof_name);
1704        base_size = getAppendOnlyFileSize(aof_name, NULL);
1705        last_file = ++aof_num == total_num;
1706        start = ustime();
1707        ret = loadSingleAppendOnlyFile(aof_name);
1708        if (ret == AOF_OK || (ret == AOF_TRUNCATED && last_file)) {
1709            serverLog(LL_NOTICE, "DB loaded from base file %s: %.3f seconds",
1710                aof_name, (float)(ustime()-start)/1000000);
1711        }
1712
1713        /* If the truncated file is not the last file, we consider this to be a fatal error. */
1714        if (ret == AOF_TRUNCATED && !last_file) {
1715            ret = AOF_FAILED;
1716            serverLog(LL_WARNING, "Fatal error: the truncated file is not the last file");
1717        }
1718
1719        if (ret == AOF_OPEN_ERR || ret == AOF_FAILED) {
1720            goto cleanup;
1721        }
1722    }
1723
1724    /* Load INCR AOFs if needed. */
1725    if (listLength(am->incr_aof_list)) {
1726        listNode *ln;
1727        listIter li;
1728
1729        listRewind(am->incr_aof_list, &li);
1730        while ((ln = listNext(&li)) != NULL) {
1731            aofInfo *ai = (aofInfo*)ln->value;
1732            serverAssert(ai->file_type == AOF_FILE_TYPE_INCR);
1733            aof_name = (char*)ai->file_name;
1734            updateLoadingFileName(aof_name);
1735            last_file = ++aof_num == total_num;
1736            start = ustime();
1737            ret = loadSingleAppendOnlyFile(aof_name);
1738            if (ret == AOF_OK || (ret == AOF_TRUNCATED && last_file)) {
1739                serverLog(LL_NOTICE, "DB loaded from incr file %s: %.3f seconds",
1740                    aof_name, (float)(ustime()-start)/1000000);
1741            }
1742
1743            /* We know that (at least) one of the AOF files has data (total_size > 0),
1744             * so empty incr AOF file doesn't count as a AOF_EMPTY result */
1745            if (ret == AOF_EMPTY) ret = AOF_OK;
1746
1747            /* If the truncated file is not the last file, we consider this to be a fatal error. */
1748            if (ret == AOF_TRUNCATED && !last_file) {
1749                ret = AOF_FAILED;
1750                serverLog(LL_WARNING, "Fatal error: the truncated file is not the last file");
1751            }
1752
1753            if (ret == AOF_OPEN_ERR || ret == AOF_FAILED) {
1754                goto cleanup;
1755            }
1756        }
1757    }
1758
1759    server.aof_current_size = total_size;
1760    /* Ideally, the aof_rewrite_base_size variable should hold the size of the
1761     * AOF when the last rewrite ended, this should include the size of the
1762     * incremental file that was created during the rewrite since otherwise we
1763     * risk the next automatic rewrite to happen too soon (or immediately if
1764     * auto-aof-rewrite-percentage is low). However, since we do not persist
1765     * aof_rewrite_base_size information anywhere, we initialize it on restart
1766     * to the size of BASE AOF file. This might cause the first AOFRW to be
1767     * executed early, but that shouldn't be a problem since everything will be
1768     * fine after the first AOFRW. */
1769    server.aof_rewrite_base_size = base_size;
1770
1771cleanup:
1772    stopLoading(ret == AOF_OK || ret == AOF_TRUNCATED);
1773    return ret;
1774}

1800    listTypeEntry entry;

1806                !rioWriteBulkString(r,"RPUSH",5) ||

1820                listTypeReleaseIterator(li);

1869 * The function returns 0 on error, 1 on success. */
1870int rewriteSortedSetObject(rio *r, robj *key, robj *o) {
1871    long long count = 0, items = zsetLength(o);
1872
1873    if (o->encoding == OBJ_ENCODING_LISTPACK) {
1874        unsigned char *zl = o->ptr;
1875        unsigned char *eptr, *sptr;
1876        unsigned char *vstr;
1877        unsigned int vlen;
1878        long long vll;
1879        double score;
1880
1881        eptr = lpSeek(zl,0);
1882        serverAssert(eptr != NULL);
1883        sptr = lpNext(zl,eptr);
1884        serverAssert(sptr != NULL);
1885
1886        while (eptr != NULL) {
1887            vstr = lpGetValue(eptr,&vlen,&vll);
1888            score = zzlGetScore(sptr);
1889
1890            if (count == 0) {
1891                int cmd_items = (items > AOF_REWRITE_ITEMS_PER_CMD) ?
1892                    AOF_REWRITE_ITEMS_PER_CMD : items;
1893
1894                if (!rioWriteBulkCount(r,'*',2+cmd_items*2) ||
1895                    !rioWriteBulkString(r,"ZADD",4) ||
1896                    !rioWriteBulkObject(r,key)) 
1897                {
1898                    return 0;
1899                }
1900            }
1901            if (!rioWriteBulkDouble(r,score)) return 0;
1902            if (vstr != NULL) {
1903                if (!rioWriteBulkString(r,(char*)vstr,vlen)) return 0;
1904            } else {
1905                if (!rioWriteBulkLongLong(r,vll)) return 0;
1906            }
1907            zzlNext(zl,&eptr,&sptr);
1908            if (++count == AOF_REWRITE_ITEMS_PER_CMD) count = 0;
1909            items--;
1910        }
1911    } else if (o->encoding == OBJ_ENCODING_SKIPLIST) {
1912        zset *zs = o->ptr;
1913        dictIterator *di = dictGetIterator(zs->dict);
1914        dictEntry *de;
1915
1916        while((de = dictNext(di)) != NULL) {
1917            sds ele = dictGetKey(de);
1918            double *score = dictGetVal(de);
1919
1920            if (count == 0) {
1921                int cmd_items = (items > AOF_REWRITE_ITEMS_PER_CMD) ?
1922                    AOF_REWRITE_ITEMS_PER_CMD : items;
1923
1924                if (!rioWriteBulkCount(r,'*',2+cmd_items*2) ||
1925                    !rioWriteBulkString(r,"ZADD",4) ||
1926                    !rioWriteBulkObject(r,key)) 
1927                {
1928                    dictReleaseIterator(di);
1929                    return 0;
1930                }
1931            }
1932            if (!rioWriteBulkDouble(r,*score) ||
1933                !rioWriteBulkString(r,ele,sdslen(ele)))
1934            {
1935                dictReleaseIterator(di);
1936                return 0;
1937            }
1938            if (++count == AOF_REWRITE_ITEMS_PER_CMD) count = 0;
1939            items--;
1940        }
1941        dictReleaseIterator(di);
1942    } else {
1943        serverPanic("Unknown sorted zset encoding");
1944    }
1945    return 1;

1948/* Write either the key or the value of the currently selected item of a hash.
1949 * The 'hi' argument passes a valid Redis hash iterator.
1950 * The 'what' filed specifies if to write a key or a value and can be
1951 * either OBJ_HASH_KEY or OBJ_HASH_VALUE.
1952 *
1953 * The function returns 0 on error, non-zero on success. */
1954static int rioWriteHashIteratorCursor(rio *r, hashTypeIterator *hi, int what) {
1955    if (hi->encoding == OBJ_ENCODING_LISTPACK) {
1956        unsigned char *vstr = NULL;
1957        unsigned int vlen = UINT_MAX;
1958        long long vll = LLONG_MAX;
1959
1960        hashTypeCurrentFromListpack(hi, what, &vstr, &vlen, &vll);
1961        if (vstr)
1962            return rioWriteBulkString(r, (char*)vstr, vlen);
1963        else
1964            return rioWriteBulkLongLong(r, vll);
1965    } else if (hi->encoding == OBJ_ENCODING_HT) {
1966        sds value = hashTypeCurrentFromHashTable(hi, what);
1967        return rioWriteBulkString(r, value, sdslen(value));
1968    }
1969
1970    serverPanic("Unknown hash encoding");
1971    return 0;
1972}

2003    }
2004

2021/* Helper for rewriteStreamObject(): emit the XCLAIM needed in order to
2022 * add the message described by 'nack' having the id 'rawid', into the pending
2023 * list of the specified consumer. All this in the context of the specified
2024 * key and group. */
2025int rioWriteStreamPendingEntry(rio *r, robj *key, const char *groupname, size_t groupname_len, streamConsumer *consumer, unsigned char *rawid, streamNACK *nack) {
2026     /* XCLAIM <key> <group> <consumer> 0 <id> TIME <milliseconds-unix-time>
2027               RETRYCOUNT <count> JUSTID FORCE. */
2028    streamID id;
2029    streamDecodeID(rawid,&id);
2030    if (rioWriteBulkCount(r,'*',12) == 0) return 0;
2031    if (rioWriteBulkString(r,"XCLAIM",6) == 0) return 0;
2032    if (rioWriteBulkObject(r,key) == 0) return 0;
2033    if (rioWriteBulkString(r,groupname,groupname_len) == 0) return 0;
2034    if (rioWriteBulkString(r,consumer->name,sdslen(consumer->name)) == 0) return 0;
2035    if (rioWriteBulkString(r,"0",1) == 0) return 0;
2036    if (rioWriteBulkStreamID(r,&id) == 0) return 0;
2037    if (rioWriteBulkString(r,"TIME",4) == 0) return 0;
2038    if (rioWriteBulkLongLong(r,nack->delivery_time) == 0) return 0;
2039    if (rioWriteBulkString(r,"RETRYCOUNT",10) == 0) return 0;
2040    if (rioWriteBulkLongLong(r,nack->delivery_count) == 0) return 0;
2041    if (rioWriteBulkString(r,"JUSTID",6) == 0) return 0;
2042    if (rioWriteBulkString(r,"FORCE",5) == 0) return 0;
2043    return 1;
2044}

2060/* Emit the commands needed to rebuild a stream object.
2061 * The function returns 0 on error, 1 on success. */
2062int rewriteStreamObject(rio *r, robj *key, robj *o) {
2063    stream *s = o->ptr;
2064    streamIterator si;
2065    streamIteratorStart(&si,s,NULL,NULL,0);
2066    streamID id;
2067    int64_t numfields;
2068
2069    if (s->length) {
2070        /* Reconstruct the stream data using XADD commands. */
2071        while(streamIteratorGetID(&si,&id,&numfields)) {
2072            /* Emit a two elements array for each item. The first is
2073             * the ID, the second is an array of field-value pairs. */
2074
2075            /* Emit the XADD <key> <id> ...fields... command. */
2076            if (!rioWriteBulkCount(r,'*',3+numfields*2) || 
2077                !rioWriteBulkString(r,"XADD",4) ||
2078                !rioWriteBulkObject(r,key) ||
2079                !rioWriteBulkStreamID(r,&id)) 
2080            {
2081                streamIteratorStop(&si);
2082                return 0;
2083            }
2084            while(numfields--) {
2085                unsigned char *field, *value;
2086                int64_t field_len, value_len;
2087                streamIteratorGetField(&si,&field,&value,&field_len,&value_len);
2088                if (!rioWriteBulkString(r,(char*)field,field_len) ||
2089                    !rioWriteBulkString(r,(char*)value,value_len)) 
2090                {
2091                    streamIteratorStop(&si);
2092                    return 0;                  
2093                }
2094            }
2095        }
2096    } else {
2097        /* Use the XADD MAXLEN 0 trick to generate an empty stream if
2098         * the key we are serializing is an empty string, which is possible
2099         * for the Stream type. */
2100        id.ms = 0; id.seq = 1; 
2101        if (!rioWriteBulkCount(r,'*',7) ||
2102            !rioWriteBulkString(r,"XADD",4) ||
2103            !rioWriteBulkObject(r,key) ||
2104            !rioWriteBulkString(r,"MAXLEN",6) ||
2105            !rioWriteBulkString(r,"0",1) ||
2106            !rioWriteBulkStreamID(r,&id) ||
2107            !rioWriteBulkString(r,"x",1) ||
2108            !rioWriteBulkString(r,"y",1))
2109        {
2110            streamIteratorStop(&si);
2111            return 0;     
2112        }
2113    }
2114
2115    /* Append XSETID after XADD, make sure lastid is correct,
2116     * in case of XDEL lastid. */
2117    if (!rioWriteBulkCount(r,'*',7) ||
2118        !rioWriteBulkString(r,"XSETID",6) ||
2119        !rioWriteBulkObject(r,key) ||
2120        !rioWriteBulkStreamID(r,&s->last_id) ||
2121        !rioWriteBulkString(r,"ENTRIESADDED",12) ||
2122        !rioWriteBulkLongLong(r,s->entries_added) ||
2123        !rioWriteBulkString(r,"MAXDELETEDID",12) ||
2124        !rioWriteBulkStreamID(r,&s->max_deleted_entry_id)) 
2125    {
2126        streamIteratorStop(&si);
2127        return 0; 
2128    }
2129
2130
2131    /* Create all the stream consumer groups. */
2132    if (s->cgroups) {
2133        raxIterator ri;
2134        raxStart(&ri,s->cgroups);
2135        raxSeek(&ri,"^",NULL,0);
2136        while(raxNext(&ri)) {
2137            streamCG *group = ri.data;
2138            /* Emit the XGROUP CREATE in order to create the group. */
2139            if (!rioWriteBulkCount(r,'*',7) ||
2140                !rioWriteBulkString(r,"XGROUP",6) ||
2141                !rioWriteBulkString(r,"CREATE",6) ||
2142                !rioWriteBulkObject(r,key) ||
2143                !rioWriteBulkString(r,(char*)ri.key,ri.key_len) ||
2144                !rioWriteBulkStreamID(r,&group->last_id) ||
2145                !rioWriteBulkString(r,"ENTRIESREAD",11) ||
2146                !rioWriteBulkLongLong(r,group->entries_read))
2147            {
2148                raxStop(&ri);
2149                streamIteratorStop(&si);
2150                return 0;
2151            }
2152
2153            /* Generate XCLAIMs for each consumer that happens to
2154             * have pending entries. Empty consumers would be generated with
2155             * XGROUP CREATECONSUMER. */
2156            raxIterator ri_cons;
2157            raxStart(&ri_cons,group->consumers);
2158            raxSeek(&ri_cons,"^",NULL,0);
2159            while(raxNext(&ri_cons)) {
2160                streamConsumer *consumer = ri_cons.data;
2161                /* If there are no pending entries, just emit XGROUP CREATECONSUMER */
2162                if (raxSize(consumer->pel) == 0) {
2163                    if (rioWriteStreamEmptyConsumer(r,key,(char*)ri.key,
2164                                                    ri.key_len,consumer) == 0)
2165                    {
2166                        raxStop(&ri_cons);
2167                        raxStop(&ri);
2168                        streamIteratorStop(&si);
2169                        return 0;
2170                    }
2171                    continue;
2172                }
2173                /* For the current consumer, iterate all the PEL entries
2174                 * to emit the XCLAIM protocol. */
2175                raxIterator ri_pel;
2176                raxStart(&ri_pel,consumer->pel);
2177                raxSeek(&ri_pel,"^",NULL,0);
2178                while(raxNext(&ri_pel)) {
2179                    streamNACK *nack = ri_pel.data;
2180                    if (rioWriteStreamPendingEntry(r,key,(char*)ri.key,
2181                                                   ri.key_len,consumer,
2182                                                   ri_pel.key,nack) == 0)
2183                    {
2184                        raxStop(&ri_pel);
2185                        raxStop(&ri_cons);
2186                        raxStop(&ri);
2187                        streamIteratorStop(&si);
2188                        return 0;
2189                    }
2190                }
2191                raxStop(&ri_pel);
2192            }
2193            raxStop(&ri_cons);
2194        }
2195        raxStop(&ri);
2196    }
2197
2198    streamIteratorStop(&si);
2199    return 1;
2200}

2237int rewriteAppendOnlyFileRio(rio *aof) {
2238    dictIterator *di = NULL;
2239    dictEntry *de;
2240    int j;
2241    long key_count = 0;
2242    long long updated_time = 0;
2243
2244    /* Record timestamp at the beginning of rewriting AOF. */
2245    if (server.aof_timestamp_enabled) {
2246        sds ts = genAofTimestampAnnotationIfNeeded(1);
2247        if (rioWrite(aof,ts,sdslen(ts)) == 0) { sdsfree(ts); goto werr; }
2248        sdsfree(ts);
2249    }
2250
2251    if (rewriteFunctions(aof) == 0) goto werr;
2252
2253    for (j = 0; j < server.dbnum; j++) {
2254        char selectcmd[] = "*2\r\n$6\r\nSELECT\r\n";
2255        redisDb *db = server.db+j;
2256        dict *d = db->dict;
2257        if (dictSize(d) == 0) continue;
2258        di = dictGetSafeIterator(d);
2259
2260        /* SELECT the new DB */
2261        if (rioWrite(aof,selectcmd,sizeof(selectcmd)-1) == 0) goto werr;
2262        if (rioWriteBulkLongLong(aof,j) == 0) goto werr;
2263
2264        /* Iterate this DB writing every entry */
2265        while((de = dictNext(di)) != NULL) {
2266            sds keystr;
2267            robj key, *o;
2268            long long expiretime;
2269            size_t aof_bytes_before_key = aof->processed_bytes;
2270
2271            keystr = dictGetKey(de);
2272            o = dictGetVal(de);
2273            initStaticStringObject(key,keystr);
2274
2275            expiretime = getExpire(db,&key);
2276
2277            /* Save the key and associated value */
2278            if (o->type == OBJ_STRING) {
2279                /* Emit a SET command */
2280                char cmd[]="*3\r\n$3\r\nSET\r\n";
2281                if (rioWrite(aof,cmd,sizeof(cmd)-1) == 0) goto werr;
2282                /* Key and value */
2283                if (rioWriteBulkObject(aof,&key) == 0) goto werr;
2284                if (rioWriteBulkObject(aof,o) == 0) goto werr;
2285            } else if (o->type == OBJ_LIST) {
2286                if (rewriteListObject(aof,&key,o) == 0) goto werr;
2287            } else if (o->type == OBJ_SET) {
2288                if (rewriteSetObject(aof,&key,o) == 0) goto werr;
2289            } else if (o->type == OBJ_ZSET) {
2290                if (rewriteSortedSetObject(aof,&key,o) == 0) goto werr;
2291            } else if (o->type == OBJ_HASH) {
2292                if (rewriteHashObject(aof,&key,o) == 0) goto werr;
2293            } else if (o->type == OBJ_STREAM) {
2294                if (rewriteStreamObject(aof,&key,o) == 0) goto werr;
2295            } else if (o->type == OBJ_MODULE) {
2296                if (rewriteModuleObject(aof,&key,o,j) == 0) goto werr;
2297            } else {
2298                serverPanic("Unknown object type");
2299            }
2300
2301            /* In fork child process, we can try to release memory back to the
2302             * OS and possibly avoid or decrease COW. We give the dismiss
2303             * mechanism a hint about an estimated size of the object we stored. */
2304            size_t dump_size = aof->processed_bytes - aof_bytes_before_key;
2305            if (server.in_fork_child) dismissObject(o, dump_size);
2306
2307            /* Save the expire time */
2308            if (expiretime != -1) {
2309                char cmd[]="*3\r\n$9\r\nPEXPIREAT\r\n";
2310                if (rioWrite(aof,cmd,sizeof(cmd)-1) == 0) goto werr;
2311                if (rioWriteBulkObject(aof,&key) == 0) goto werr;
2312                if (rioWriteBulkLongLong(aof,expiretime) == 0) goto werr;
2313            }
2314
2315            /* Update info every 1 second (approximately).
2316             * in order to avoid calling mstime() on each iteration, we will
2317             * check the diff every 1024 keys */
2318            if ((key_count++ & 1023) == 0) {
2319                long long now = mstime();
2320                if (now - updated_time >= 1000) {
2321                    sendChildInfo(CHILD_INFO_TYPE_CURRENT_INFO, key_count, "AOF rewrite");
2322                    updated_time = now;
2323                }
2324            }
2325
2326            /* Delay before next key if required (for testing) */
2327            if (server.rdb_key_save_delay)
2328                debugDelay(server.rdb_key_save_delay);
2329        }
2330        dictReleaseIterator(di);
2331        di = NULL;
2332    }
2333    return C_OK;
2334
2335werr:
2336    if (di) dictReleaseIterator(di);
2337    return C_ERR;
2338}

2340/* Write a sequence of commands able to fully rebuild the dataset into
2341 * "filename". Used both by REWRITEAOF and BGREWRITEAOF.
2342 *
2343 * In order to minimize the number of commands needed in the rewritten
2344 * log Redis uses variadic commands when possible, such as RPUSH, SADD
2345 * and ZADD. However at max AOF_REWRITE_ITEMS_PER_CMD items per time
2346 * are inserted using a single command. */
2347int rewriteAppendOnlyFile(char *filename) {
2348    rio aof;
2349    FILE *fp = NULL;
2350    char tmpfile[256];
2351
2352    /* Note that we have to use a different temp name here compared to the
2353     * one used by rewriteAppendOnlyFileBackground() function. */
2354    snprintf(tmpfile,256,"temp-rewriteaof-%d.aof", (int) getpid());
2355    fp = fopen(tmpfile,"w");
2356    if (!fp) {
2357        serverLog(LL_WARNING, "Opening the temp file for AOF rewrite in rewriteAppendOnlyFile(): %s", strerror(errno));
2358        return C_ERR;
2359    }
2360
2361    rioInitWithFile(&aof,fp);
2362
2363    if (server.aof_rewrite_incremental_fsync) {
2364        rioSetAutoSync(&aof,REDIS_AUTOSYNC_BYTES);
2365        rioSetReclaimCache(&aof,1);
2366    }
2367
2368    startSaving(RDBFLAGS_AOF_PREAMBLE);
2369
2370    if (server.aof_use_rdb_preamble) {
2371        int error;
2372        if (rdbSaveRio(SLAVE_REQ_NONE,&aof,&error,RDBFLAGS_AOF_PREAMBLE,NULL) == C_ERR) {
2373            errno = error;
2374            goto werr;
2375        }
2376    } else {
2377        if (rewriteAppendOnlyFileRio(&aof) == C_ERR) goto werr;
2378    }
2379
2380    /* Make sure data will not remain on the OS's output buffers */
2381    if (fflush(fp)) goto werr;

2383    if (reclaimFilePageCache(fileno(fp), 0, 0) == -1) {
2384        /* A minor error. Just log to know what happens */
2385        serverLog(LL_NOTICE,"Unable to reclaim page cache: %s", strerror(errno));
2386    }
2387    if (fclose(fp)) { fp = NULL; goto werr; }
2388    fp = NULL;
2389
2390    /* Use RENAME to make sure the DB file is changed atomically only
2391     * if the generate DB file is ok. */
2392    if (rename(tmpfile,filename) == -1) {
2393        serverLog(LL_WARNING,"Error moving temp append only file on the final destination: %s", strerror(errno));
2394        unlink(tmpfile);
2395        stopSaving(0);
2396        return C_ERR;
2397    }
2398    stopSaving(1);
2399
2400    return C_OK;
2401
2402werr:
2403    serverLog(LL_WARNING,"Write error writing append only file on disk: %s", strerror(errno));
2404    if (fp) fclose(fp);

2406    stopSaving(0);
2407    return C_ERR;

2413/* This is how rewriting of the append only file in background works:
2414 *
2415 * 1) The user calls BGREWRITEAOF
2416 * 2) Redis calls this function, that forks():
2417 *    2a) the child rewrite the append only file in a temp file.
2418 *    2b) the parent open a new INCR AOF file to continue writing.
2419 * 3) When the child finished '2a' exists.
2420 * 4) The parent will trap the exit code, if it's OK, it will:
2421 *    4a) get a new BASE file name and mark the previous (if we have) as the HISTORY type
2422 *    4b) rename(2) the temp file in new BASE file name
2423 *    4c) mark the rewritten INCR AOFs as history type
2424 *    4d) persist AOF manifest file
2425 *    4e) Delete the history files use bio
2426 */
2427int rewriteAppendOnlyFileBackground(void) {
2428    pid_t childpid;
2429
2430    if (hasActiveChildProcess()) return C_ERR;
2431
2432    if (dirCreateIfMissing(server.aof_dirname) == -1) {
2433        serverLog(LL_WARNING, "Can't open or create append-only dir %s: %s",
2434            server.aof_dirname, strerror(errno));
2435        server.aof_lastbgrewrite_status = C_ERR;
2436        return C_ERR;
2437    }
2438
2439    /* We set aof_selected_db to -1 in order to force the next call to the
2440     * feedAppendOnlyFile() to issue a SELECT command. */
2441    server.aof_selected_db = -1;
2442    flushAppendOnlyFile(1);
2443    if (openNewIncrAofForAppend() != C_OK) {
2444        server.aof_lastbgrewrite_status = C_ERR;
2445        return C_ERR;
2446    }
2447
2448    if (server.aof_state == AOF_WAIT_REWRITE) {
2449        /* Wait for all bio jobs related to AOF to drain. This prevents a race
2450         * between updates to `fsynced_reploff_pending` of the worker thread, belonging
2451         * to the previous AOF, and the new one. This concern is specific for a full
2452         * sync scenario where we don't wanna risk the ACKed replication offset
2453         * jumping backwards or forward when switching to a different master. */
2454        bioDrainWorker(BIO_AOF_FSYNC);
2455
2456        /* Set the initial repl_offset, which will be applied to fsynced_reploff
2457         * when AOFRW finishes (after possibly being updated by a bio thread) */
2458        atomicSet(server.fsynced_reploff_pending, server.master_repl_offset);
2459        server.fsynced_reploff = 0;
2460    }
2461
2462    server.stat_aof_rewrites++;
2463
2464    if ((childpid = redisFork(CHILD_TYPE_AOF)) == 0) {
2465        char tmpfile[256];
2466
2467        /* Child */
2468        redisSetProcTitle("redis-aof-rewrite");
2469        redisSetCpuAffinity(server.aof_rewrite_cpulist);
2470        snprintf(tmpfile,256,"temp-rewriteaof-bg-%d.aof", (int) getpid());
2471        if (rewriteAppendOnlyFile(tmpfile) == C_OK) {
2472            serverLog(LL_NOTICE,
2473                "Successfully created the temporary AOF base file %s", tmpfile);
2474            sendChildCowInfo(CHILD_INFO_TYPE_AOF_COW_SIZE, "AOF rewrite");
2475            exitFromChild(0);
2476        } else {
2477            exitFromChild(1);
2478        }
2479    } else {
2480        /* Parent */
2481        if (childpid == -1) {
2482            server.aof_lastbgrewrite_status = C_ERR;
2483            serverLog(LL_WARNING,
2484                "Can't rewrite append only file in background: fork: %s",
2485                strerror(errno));
2486            return C_ERR;
2487        }
2488        serverLog(LL_NOTICE,
2489            "Background append only file rewriting started by pid %ld",(long) childpid);
2490        server.aof_rewrite_scheduled = 0;
2491        server.aof_rewrite_time_start = time(NULL);
2492        return C_OK;
2493    }
2494    return C_OK; /* unreached */
2495}

2582/* A background append only file rewriting (BGREWRITEAOF) terminated its work.
2583 * Handle this. */
2584void backgroundRewriteDoneHandler(int exitcode, int bysignal) {
2585    if (!bysignal && exitcode == 0) {
2586        char tmpfile[256];
2587        long long now = ustime();
2588        sds new_base_filepath = NULL;
2589        sds new_incr_filepath = NULL;
2590        aofManifest *temp_am;
2591        mstime_t latency;
2592
2593        serverLog(LL_NOTICE,
2594            "Background AOF rewrite terminated with success");
2595
2596        snprintf(tmpfile, 256, "temp-rewriteaof-bg-%d.aof",
2597            (int)server.child_pid);
2598
2599        serverAssert(server.aof_manifest != NULL);
2600
2601        /* Dup a temporary aof_manifest for subsequent modifications. */
2602        temp_am = aofManifestDup(server.aof_manifest);
2603
2604        /* Get a new BASE file name and mark the previous (if we have)
2605         * as the HISTORY type. */
2606        sds new_base_filename = getNewBaseFileNameAndMarkPreAsHistory(temp_am);
2607        serverAssert(new_base_filename != NULL);
2608        new_base_filepath = makePath(server.aof_dirname, new_base_filename);
2609
2610        /* Rename the temporary aof file to 'new_base_filename'. */
2611        latencyStartMonitor(latency);
2612        if (rename(tmpfile, new_base_filepath) == -1) {
2613            serverLog(LL_WARNING,
2614                "Error trying to rename the temporary AOF base file %s into %s: %s",
2615                tmpfile,
2616                new_base_filepath,
2617                strerror(errno));
2618            aofManifestFree(temp_am);
2619            sdsfree(new_base_filepath);
2620            server.aof_lastbgrewrite_status = C_ERR;
2621            server.stat_aofrw_consecutive_failures++;
2622            goto cleanup;
2623        }
2624        latencyEndMonitor(latency);
2625        latencyAddSampleIfNeeded("aof-rename", latency);
2626        serverLog(LL_NOTICE,
2627            "Successfully renamed the temporary AOF base file %s into %s", tmpfile, new_base_filename);
2628
2629        /* Rename the temporary incr aof file to 'new_incr_filename'. */
2630        if (server.aof_state == AOF_WAIT_REWRITE) {
2631            /* Get temporary incr aof name. */
2632            sds temp_incr_aof_name = getTempIncrAofName();
2633            sds temp_incr_filepath = makePath(server.aof_dirname, temp_incr_aof_name);
2634            /* Get next new incr aof name. */
2635            sds new_incr_filename = getNewIncrAofName(temp_am);
2636            new_incr_filepath = makePath(server.aof_dirname, new_incr_filename);
2637            latencyStartMonitor(latency);
2638            if (rename(temp_incr_filepath, new_incr_filepath) == -1) {
2639                serverLog(LL_WARNING,
2640                    "Error trying to rename the temporary AOF incr file %s into %s: %s",
2641                    temp_incr_filepath,
2642                    new_incr_filepath,
2643                    strerror(errno));
2644                bg_unlink(new_base_filepath);
2645                sdsfree(new_base_filepath);
2646                aofManifestFree(temp_am);
2647                sdsfree(temp_incr_filepath);
2648                sdsfree(new_incr_filepath);
2649                sdsfree(temp_incr_aof_name);
2650                server.aof_lastbgrewrite_status = C_ERR;
2651                server.stat_aofrw_consecutive_failures++;
2652                goto cleanup;
2653            }
2654            latencyEndMonitor(latency);
2655            latencyAddSampleIfNeeded("aof-rename", latency);
2656            serverLog(LL_NOTICE,
2657                "Successfully renamed the temporary AOF incr file %s into %s", temp_incr_aof_name, new_incr_filename);
2658            sdsfree(temp_incr_filepath);
2659            sdsfree(temp_incr_aof_name);
2660        }
2661
2662        /* Change the AOF file type in 'incr_aof_list' from AOF_FILE_TYPE_INCR
2663         * to AOF_FILE_TYPE_HIST, and move them to the 'history_aof_list'. */
2664        markRewrittenIncrAofAsHistory(temp_am);
2665
2666        /* Persist our modifications. */
2667        if (persistAofManifest(temp_am) == C_ERR) {
2668            bg_unlink(new_base_filepath);
2669            aofManifestFree(temp_am);
2670            sdsfree(new_base_filepath);
2671            if (new_incr_filepath) {
2672                bg_unlink(new_incr_filepath);
2673                sdsfree(new_incr_filepath);
2674            }
2675            server.aof_lastbgrewrite_status = C_ERR;
2676            server.stat_aofrw_consecutive_failures++;
2677            goto cleanup;
2678        }
2679        sdsfree(new_base_filepath);
2680        if (new_incr_filepath) sdsfree(new_incr_filepath);
2681
2682        /* We can safely let `server.aof_manifest` point to 'temp_am' and free the previous one. */
2683        aofManifestFreeAndUpdate(temp_am);
2684
2685        if (server.aof_state != AOF_OFF) {
2686            /* AOF enabled. */
2687            server.aof_current_size = getAppendOnlyFileSize(new_base_filename, NULL) + server.aof_last_incr_size;
2688            server.aof_rewrite_base_size = server.aof_current_size;
2689        }
2690
2691        /* We don't care about the return value of `aofDelHistoryFiles`, because the history
2692         * deletion failure will not cause any problems. */
2693        aofDelHistoryFiles();
2694
2695        server.aof_lastbgrewrite_status = C_OK;
2696        server.stat_aofrw_consecutive_failures = 0;
2697
2698        serverLog(LL_NOTICE, "Background AOF rewrite finished successfully");
2699        /* Change state from WAIT_REWRITE to ON if needed */
2700        if (server.aof_state == AOF_WAIT_REWRITE) {
2701            server.aof_state = AOF_ON;
2702
2703            /* Update the fsynced replication offset that just now become valid.
2704             * This could either be the one we took in startAppendOnly, or a
2705             * newer one set by the bio thread. */
2706            long long fsynced_reploff_pending;
2707            atomicGet(server.fsynced_reploff_pending, fsynced_reploff_pending);
2708            server.fsynced_reploff = fsynced_reploff_pending;
2709        }
2710
2711        serverLog(LL_VERBOSE,
2712            "Background AOF rewrite signal handler took %lldus", ustime()-now);
2713    } else if (!bysignal && exitcode != 0) {
2714        server.aof_lastbgrewrite_status = C_ERR;
2715        server.stat_aofrw_consecutive_failures++;
2716
2717        serverLog(LL_WARNING,
2718            "Background AOF rewrite terminated with error");
2719    } else {
2720        /* SIGUSR1 is whitelisted, so we have a way to kill a child without
2721         * triggering an error condition. */
2722        if (bysignal != SIGUSR1) {
2723            server.aof_lastbgrewrite_status = C_ERR;
2724            server.stat_aofrw_consecutive_failures++;
2725        }
2726
2727        serverLog(LL_WARNING,
2728            "Background AOF rewrite terminated by signal %d", bysignal);
2729    }
2730
2731cleanup:
2732    aofRemoveTempFile(server.child_pid);
2733    /* Clear AOF buffer and delete temp incr aof for next rewrite. */
2734    if (server.aof_state == AOF_WAIT_REWRITE) {
2735        sdsfree(server.aof_buf);
2736        server.aof_buf = sdsempty();
2737        aofDelTempIncrAofFile();
2738    }
2739    server.aof_rewrite_time_last = time(NULL)-server.aof_rewrite_time_start;
2740    server.aof_rewrite_time_start = -1;
2741    /* Schedule a new rewrite if we are waiting for it to switch the AOF ON. */
2742    if (server.aof_state == AOF_WAIT_REWRITE)
2743        server.aof_rewrite_scheduled = 1;
2744}
