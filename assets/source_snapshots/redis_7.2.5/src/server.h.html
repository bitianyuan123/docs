259/* Key-spec flags *
260 * -------------- */
261/* The following refer what the command actually does with the value or metadata
262 * of the key, and not necessarily the user data or how it affects it.
263 * Each key-spec may must have exactly one of these. Any operation that's not
264 * distinctly deletion, overwrite or read-only would be marked as RW. */
265#define CMD_KEY_RO (1ULL<<0)     /* Read-Only - Reads the value of the key, but
266                                  * doesn't necessarily returns it. */
267#define CMD_KEY_RW (1ULL<<1)     /* Read-Write - Modifies the data stored in the
268                                  * value of the key or its metadata. */
269#define CMD_KEY_OW (1ULL<<2)     /* Overwrite - Overwrites the data stored in
270                                  * the value of the key. */
271#define CMD_KEY_RM (1ULL<<3)     /* Deletes the key. */
272/* The following refer to user data inside the value of the key, not the metadata
273 * like LRU, type, cardinality. It refers to the logical operation on the user's
274 * data (actual input strings / TTL), being used / returned / copied / changed,
275 * It doesn't refer to modification or returning of metadata (like type, count,
276 * presence of data). Any write that's not INSERT or DELETE, would be an UPDATE.
277 * Each key-spec may have one of the writes with or without access, or none: */
278#define CMD_KEY_ACCESS (1ULL<<4) /* Returns, copies or uses the user data from
279                                  * the value of the key. */
280#define CMD_KEY_UPDATE (1ULL<<5) /* Updates data to the value, new value may
281                                  * depend on the old value. */
282#define CMD_KEY_INSERT (1ULL<<6) /* Adds data to the value with no chance of
283                                  * modification or deletion of existing data. */
284#define CMD_KEY_DELETE (1ULL<<7) /* Explicitly deletes some content
285                                  * from the value of the key. */

287#define CMD_KEY_NOT_KEY (1ULL<<8)     /* A 'fake' key that should be routed

289                                       * excluded from other key checks. */

935/* Replication buffer blocks is the list of replBufBlock.
936 *
937 * +--------------+       +--------------+       +--------------+
938 * | refcount = 1 |  ...  | refcount = 0 |  ...  | refcount = 2 |

940 *      |                                            /       \
941 *      |                                           /         \

943 *  Repl Backlog                               Replica_A    Replica_B
944 *
945 * Each replica or replication backlog increments only the refcount of the
946 * 'ref_repl_buf_node' which it points to. So when replica walks to the next
947 * node, it should first increase the next node's refcount, and when we trim
948 * the replication buffer nodes, we remove node always from the head node which
949 * refcount is 0. If the refcount of the head node is not 0, we must stop
950 * trimming and never iterate the next node. */

2073/* Key specs definitions.
2074 *
2075 * Brief: This is a scheme that tries to describe the location
2076 * of key arguments better than the old [first,last,step] scheme
2077 * which is limited and doesn't fit many commands.
2078 *
2079 * There are two steps:
2080 * 1. begin_search (BS): in which index should we start searching for keys?
2081 * 2. find_keys (FK): relative to the output of BS, how can we will which args are keys?
2082 *
2083 * There are two types of BS:
2084 * 1. index: key args start at a constant index
2085 * 2. keyword: key args start just after a specific keyword
2086 *
2087 * There are two kinds of FK:
2088 * 1. range: keys end at a specific index (or relative to the last argument)
2089 * 2. keynum: there's an arg that contains the number of key args somewhere before the keys themselves
2090 */

2094    KSPEC_BS_INVALID = 0, /* Must be 0 */

2096    KSPEC_BS_INDEX,
2097    KSPEC_BS_KEYWORD

2115        struct {

2120            /* The keyword that indicates the beginning of key args */

2148            int firstkey;

2150            int keystep;

2219/* Redis command structure.
2220 *
2221 * Note that the command table is in commands.c and it is auto-generated.
2222 *
2223 * This is the meaning of the flags:
2224 *
2225 * CMD_WRITE:       Write command (may modify the key space).
2226 *
2227 * CMD_READONLY:    Commands just reading from keys without changing the content.
2228 *                  Note that commands that don't read from the keyspace such as
2229 *                  TIME, SELECT, INFO, administrative commands, and connection
2230 *                  or transaction related commands (multi, exec, discard, ...)
2231 *                  are not flagged as read-only commands, since they affect the
2232 *                  server or the connection in other ways.
2233 *
2234 * CMD_DENYOOM:     May increase memory usage once called. Don't allow if out
2235 *                  of memory.
2236 *
2237 * CMD_ADMIN:       Administrative command, like SAVE or SHUTDOWN.
2238 *
2239 * CMD_PUBSUB:      Pub/Sub related command.
2240 *
2241 * CMD_NOSCRIPT:    Command not allowed in scripts.
2242 *
2243 * CMD_BLOCKING:    The command has the potential to block the client.
2244 *
2245 * CMD_LOADING:     Allow the command while loading the database.
2246 *
2247 * CMD_NO_ASYNC_LOADING: Deny during async loading (when a replica uses diskless
2248 *                       sync swapdb, and allows access to the old dataset)
2249 *
2250 * CMD_STALE:       Allow the command while a slave has stale data but is not
2251 *                  allowed to serve this data. Normally no command is accepted
2252 *                  in this condition but just a few.
2253 *
2254 * CMD_SKIP_MONITOR:  Do not automatically propagate the command on MONITOR.
2255 *
2256 * CMD_SKIP_SLOWLOG:  Do not automatically propagate the command to the slowlog.
2257 *
2258 * CMD_ASKING:      Perform an implicit ASKING for this command, so the
2259 *                  command will be accepted in cluster mode if the slot is marked
2260 *                  as 'importing'.
2261 *
2262 * CMD_FAST:        Fast command: O(1) or O(log(N)) command that should never
2263 *                  delay its execution as long as the kernel scheduler is giving
2264 *                  us time. Note that commands that may trigger a DEL as a side
2265 *                  effect (like SET) are not fast commands.
2266 *
2267 * CMD_NO_AUTH:     Command doesn't require authentication
2268 *
2269 * CMD_MAY_REPLICATE:   Command may produce replication traffic, but should be
2270 *                      allowed under circumstances where write commands are disallowed.
2271 *                      Examples include PUBLISH, which replicates pubsub messages,and
2272 *                      EVAL, which may execute write commands, which are replicated,
2273 *                      or may just execute read commands. A command can not be marked
2274 *                      both CMD_WRITE and CMD_MAY_REPLICATE
2275 *
2276 * CMD_SENTINEL:    This command is present in sentinel mode.
2277 *
2278 * CMD_ONLY_SENTINEL: This command is present only when in sentinel mode.
2279 *                    And should be removed from redis.
2280 *
2281 * CMD_NO_MANDATORY_KEYS: This key arguments for this command are optional.
2282 *
2283 * CMD_NO_MULTI: The command is not allowed inside a transaction
2284 *
2285 * CMD_ALLOW_BUSY: The command can run while another command is running for
2286 *                 a long time (timedout script, module command that yields)
2287 *
2288 * CMD_TOUCHES_ARBITRARY_KEYS: The command may touch (and cause lazy-expire)
2289 *                             arbitrary key (i.e not provided in argv)
2290 *
2291 * The following additional flags are only used in order to put commands
2292 * in a specific ACL category. Commands can have multiple ACL categories.
2293 * See redis.conf for the exact meaning of each.
2294 *
2295 * @keyspace, @read, @write, @set, @sortedset, @list, @hash, @string, @bitmap,
2296 * @hyperloglog, @stream, @admin, @fast, @slow, @pubsub, @blocking, @dangerous,
2297 * @connection, @transaction, @scripting, @geo.
2298 *
2299 * Note that:
2300 *
2301 * 1) The read-only flag implies the @read ACL category.
2302 * 2) The write flag implies the @write ACL category.
2303 * 3) The fast flag implies the @fast ACL category.
2304 * 4) The admin flag implies the @admin and @dangerous ACL category.
2305 * 5) The pub-sub flag implies the @pubsub ACL category.
2306 * 6) The lack of fast flag implies the @slow ACL category.
