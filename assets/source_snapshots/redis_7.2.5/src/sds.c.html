45static inline int sdsHdrSize(char type) {
46    switch(type&SDS_TYPE_MASK) {
47        case SDS_TYPE_5:
48            return sizeof(struct sdshdr5);
49        case SDS_TYPE_8:
50            return sizeof(struct sdshdr8);
51        case SDS_TYPE_16:
52            return sizeof(struct sdshdr16);
53        case SDS_TYPE_32:
54            return sizeof(struct sdshdr32);
55        case SDS_TYPE_64:
56            return sizeof(struct sdshdr64);
57    }
58    return 0;
59}

61static inline char sdsReqType(size_t string_size) {
62    if (string_size < 1<<5)
63        return SDS_TYPE_5;
64    if (string_size < 1<<8)
65        return SDS_TYPE_8;
66    if (string_size < 1<<16)
67        return SDS_TYPE_16;
68#if (LONG_MAX == LLONG_MAX)
69    if (string_size < 1ll<<32)
70        return SDS_TYPE_32;
71    return SDS_TYPE_64;
72#else
73    return SDS_TYPE_32;
74#endif
75}

77static inline size_t sdsTypeMaxSize(char type) {
78    if (type == SDS_TYPE_5)
79        return (1<<5) - 1;
80    if (type == SDS_TYPE_8)
81        return (1<<8) - 1;
82    if (type == SDS_TYPE_16)
83        return (1<<16) - 1;
84#if (LONG_MAX == LLONG_MAX)
85    if (type == SDS_TYPE_32)
86        return (1ll<<32) - 1;
87#endif
88    return -1; /* this is equivalent to the max SDS_TYPE_64 or SDS_TYPE_32 */
89}

91/* Create a new sds string with the content specified by the 'init' pointer
92 * and 'initlen'.
93 * If NULL is used for 'init' the string is initialized with zero bytes.
94 * If SDS_NOINIT is used, the buffer is left uninitialized;
95 *
96 * The string is always null-terminated (all the sds strings are, always) so
97 * even if you create an sds string with:
98 *
99 * mystring = sdsnewlen("abc",3);
100 *
101 * You can print the string with printf() as there is an implicit \0 at the
102 * end of the string. However the string is binary safe and can contain
103 * \0 characters in the middle, as the length is stored in the sds header. */
104sds _sdsnewlen(const void *init, size_t initlen, int trymalloc) {
105    void *sh;
106    sds s;
107    char type = sdsReqType(initlen);
108    /* Empty strings are usually created in order to append. Use type 8
109     * since type 5 is not good at this. */
110    if (type == SDS_TYPE_5 && initlen == 0) type = SDS_TYPE_8;
111    int hdrlen = sdsHdrSize(type);
112    unsigned char *fp; /* flags pointer. */
113    size_t usable;
114
115    assert(initlen + hdrlen + 1 > initlen); /* Catch size_t overflow */
116    sh = trymalloc?
117        s_trymalloc_usable(hdrlen+initlen+1, &usable) :
118        s_malloc_usable(hdrlen+initlen+1, &usable);
119    if (sh == NULL) return NULL;
120    if (init==SDS_NOINIT)
121        init = NULL;
122    else if (!init)
123        memset(sh, 0, hdrlen+initlen+1);
124    s = (char*)sh+hdrlen;
125    fp = ((unsigned char*)s)-1;
126    usable = usable-hdrlen-1;
127    if (usable > sdsTypeMaxSize(type))
128        usable = sdsTypeMaxSize(type);
129    switch(type) {
130        case SDS_TYPE_5: {
131            *fp = type | (initlen << SDS_TYPE_BITS);
132            break;
133        }
134        case SDS_TYPE_8: {
135            SDS_HDR_VAR(8,s);
136            sh->len = initlen;
137            sh->alloc = usable;
138            *fp = type;
139            break;
140        }
141        case SDS_TYPE_16: {
142            SDS_HDR_VAR(16,s);
143            sh->len = initlen;
144            sh->alloc = usable;
145            *fp = type;
146            break;
147        }
148        case SDS_TYPE_32: {
149            SDS_HDR_VAR(32,s);
150            sh->len = initlen;
151            sh->alloc = usable;
152            *fp = type;
153            break;
154        }
155        case SDS_TYPE_64: {
156            SDS_HDR_VAR(64,s);
157            sh->len = initlen;
158            sh->alloc = usable;
159            *fp = type;
160            break;
161        }
162    }
163    if (initlen && init)
164        memcpy(s, init, initlen);
165    s[initlen] = '\0';
166    return s;
167}

200/* Set the sds string length to the length as obtained with strlen(), so
201 * considering as content only up to the first null term character.
202 *
203 * This function is useful when the sds string is hacked manually in some
204 * way, like in the following example:
205 *
206 * s = sdsnew("foobar");
207 * s[2] = '\0';
208 * sdsupdatelen(s);
209 * printf("%d\n", sdslen(s));
210 *
211 * The output will be "2", but if we comment out the call to sdsupdatelen()
212 * the output will be "6" as the string was modified but the logical length
213 * remains 6 bytes. */
214void sdsupdatelen(sds s) {
215    size_t reallen = strlen(s);
216    sdssetlen(s, reallen);

228/* Enlarge the free space at the end of the sds string so that the caller
229 * is sure that after calling this function can overwrite up to addlen
230 * bytes after the end of the string, plus one more byte for nul term.
231 * If there's already sufficient free space, this function returns without any
232 * action, if there isn't sufficient free space, it'll allocate what's missing,
233 * and possibly more:
234 * When greedy is 1, enlarge more than needed, to avoid need for future reallocs
235 * on incremental growth.
236 * When greedy is 0, enlarge just enough so that there's free space for 'addlen'.
237 *
238 * Note: this does not change the *length* of the sds string as returned
239 * by sdslen(), but only the free buffer space we have. */
240sds _sdsMakeRoomFor(sds s, size_t addlen, int greedy) {
241    void *sh, *newsh;
242    size_t avail = sdsavail(s);
243    size_t len, newlen, reqlen;
244    char type, oldtype = s[-1] & SDS_TYPE_MASK;
245    int hdrlen;
246    size_t usable;
247
248    /* Return ASAP if there is enough space left. */
249    if (avail >= addlen) return s;
250
251    len = sdslen(s);
252    sh = (char*)s-sdsHdrSize(oldtype);
253    reqlen = newlen = (len+addlen);
254    assert(newlen > len);   /* Catch size_t overflow */
255    if (greedy == 1) {
256        if (newlen < SDS_MAX_PREALLOC)
257            newlen *= 2;
258        else
259            newlen += SDS_MAX_PREALLOC;
260    }
261
262    type = sdsReqType(newlen);
263
264    /* Don't use type 5: the user is appending to the string and type 5 is
265     * not able to remember empty space, so sdsMakeRoomFor() must be called
266     * at every appending operation. */
267    if (type == SDS_TYPE_5) type = SDS_TYPE_8;
268
269    hdrlen = sdsHdrSize(type);
270    assert(hdrlen + newlen + 1 > reqlen);  /* Catch size_t overflow */
271    if (oldtype==type) {
272        newsh = s_realloc_usable(sh, hdrlen+newlen+1, &usable);
273        if (newsh == NULL) return NULL;
274        s = (char*)newsh+hdrlen;
275    } else {
276        /* Since the header size changes, need to move the string forward,
277         * and can't use realloc */
278        newsh = s_malloc_usable(hdrlen+newlen+1, &usable);
279        if (newsh == NULL) return NULL;
280        memcpy((char*)newsh+hdrlen, s, len+1);
281        s_free(sh);
282        s = (char*)newsh+hdrlen;
283        s[-1] = type;
284        sdssetlen(s, len);
285    }
286    usable = usable-hdrlen-1;
287    if (usable > sdsTypeMaxSize(type))
288        usable = sdsTypeMaxSize(type);
289    sdssetalloc(s, usable);
290    return s;
291}

314/* Resize the allocation, this can make the allocation bigger or smaller,
315 * if the size is smaller than currently used len, the data will be truncated.
316 *
317 * The when the would_regrow argument is set to 1, it prevents the use of
318 * SDS_TYPE_5, which is desired when the sds is likely to be changed again.
319 *
320 * The sdsAlloc size will be set to the requested size regardless of the actual
321 * allocation size, this is done in order to avoid repeated calls to this
322 * function when the caller detects that it has excess space. */
323sds sdsResize(sds s, size_t size, int would_regrow) {
324    void *sh, *newsh;
325    char type, oldtype = s[-1] & SDS_TYPE_MASK;
326    int hdrlen, oldhdrlen = sdsHdrSize(oldtype);
327    size_t len = sdslen(s);
328    sh = (char*)s-oldhdrlen;
329
330    /* Return ASAP if the size is already good. */
331    if (sdsalloc(s) == size) return s;
332
333    /* Truncate len if needed. */
334    if (size < len) len = size;
335
336    /* Check what would be the minimum SDS header that is just good enough to
337     * fit this string. */
338    type = sdsReqType(size);
339    if (would_regrow) {
340        /* Don't use type 5, it is not good for strings that are expected to grow back. */
341        if (type == SDS_TYPE_5) type = SDS_TYPE_8;
342    }
343    hdrlen = sdsHdrSize(type);
344
345    /* If the type is the same, or can hold the size in it with low overhead
346     * (larger than SDS_TYPE_8), we just realloc(), letting the allocator
347     * to do the copy only if really needed. Otherwise if the change is
348     * huge, we manually reallocate the string to use the different header
349     * type. */
350    int use_realloc = (oldtype==type || (type < oldtype && type > SDS_TYPE_8));
351    size_t newlen = use_realloc ? oldhdrlen+size+1 : hdrlen+size+1;
352
353    if (use_realloc) {
354        int alloc_already_optimal = 0;
355        #if defined(USE_JEMALLOC)
356            /* je_nallocx returns the expected allocation size for the newlen.
357             * We aim to avoid calling realloc() when using Jemalloc if there is no
358             * change in the allocation size, as it incurs a cost even if the
359             * allocation size stays the same. */
360            alloc_already_optimal = (je_nallocx(newlen, 0) == zmalloc_size(sh));
361        #endif
362        if (!alloc_already_optimal) {
363            newsh = s_realloc(sh, newlen);
364            if (newsh == NULL) return NULL;
365            s = (char*)newsh+oldhdrlen;
366        }
367    } else {
368        newsh = s_malloc(newlen);
369        if (newsh == NULL) return NULL;
370        memcpy((char*)newsh+hdrlen, s, len);
371        s_free(sh);
372        s = (char*)newsh+hdrlen;
373        s[-1] = type;
374    }
375    s[len] = 0;
376    sdssetlen(s, len);
377    sdssetalloc(s, size);
378    return s;
379}

399/* Increment the sds length and decrements the left free space at the
400 * end of the string according to 'incr'. Also set the null term
401 * in the new end of the string.
402 *
403 * This function is used in order to fix the string length after the
404 * user calls sdsMakeRoomFor(), writes something after the end of
405 * the current string, and finally needs to set the new length.
406 *
407 * Note: it is possible to use a negative increment in order to
408 * right-trim the string.
409 *
410 * Usage example:
411 *
412 * Using sdsIncrLen() and sdsMakeRoomFor() it is possible to mount the
413 * following schema, to cat bytes coming from the kernel to the end of an
414 * sds string without copying into an intermediate buffer:
415 *
416 * oldlen = sdslen(s);
417 * s = sdsMakeRoomFor(s, BUFFER_SIZE);
418 * nread = read(fd, s+oldlen, BUFFER_SIZE);
419 * ... check for nread <= 0 and handle it ...
420 * sdsIncrLen(s, nread);
421 */
422void sdsIncrLen(sds s, ssize_t incr) {
423    unsigned char flags = s[-1];
424    size_t len;
425    switch(flags&SDS_TYPE_MASK) {
426        case SDS_TYPE_5: {
427            unsigned char *fp = ((unsigned char*)s)-1;
428            unsigned char oldlen = SDS_TYPE_5_LEN(flags);
429            assert((incr > 0 && oldlen+incr < 32) || (incr < 0 && oldlen >= (unsigned int)(-incr)));
430            *fp = SDS_TYPE_5 | ((oldlen+incr) << SDS_TYPE_BITS);
431            len = oldlen+incr;
432            break;
433        }
434        case SDS_TYPE_8: {
435            SDS_HDR_VAR(8,s);
436            assert((incr >= 0 && sh->alloc-sh->len >= incr) || (incr < 0 && sh->len >= (unsigned int)(-incr)));
437            len = (sh->len += incr);
438            break;
439        }
440        case SDS_TYPE_16: {
441            SDS_HDR_VAR(16,s);
442            assert((incr >= 0 && sh->alloc-sh->len >= incr) || (incr < 0 && sh->len >= (unsigned int)(-incr)));
443            len = (sh->len += incr);
444            break;
445        }
446        case SDS_TYPE_32: {
447            SDS_HDR_VAR(32,s);
448            assert((incr >= 0 && sh->alloc-sh->len >= (unsigned int)incr) || (incr < 0 && sh->len >= (unsigned int)(-incr)));
449            len = (sh->len += incr);
450            break;
451        }
452        case SDS_TYPE_64: {
453            SDS_HDR_VAR(64,s);
454            assert((incr >= 0 && sh->alloc-sh->len >= (uint64_t)incr) || (incr < 0 && sh->len >= (uint64_t)(-incr)));
455            len = (sh->len += incr);
456            break;
457        }
458        default: len = 0; /* Just to avoid compilation warnings. */
459    }
460    s[len] = '\0';
461}

463/* Grow the sds to have the specified length. Bytes that were not part of
464 * the original length of the sds will be set to zero.
465 *
466 * if the specified length is smaller than the current length, no operation
467 * is performed. */
468sds sdsgrowzero(sds s, size_t len) {
469    size_t curlen = sdslen(s);
470
471    if (len <= curlen) return s;
472    s = sdsMakeRoomFor(s,len-curlen);
473    if (s == NULL) return NULL;
474
475    /* Make sure added region doesn't contain garbage */
476    memset(s+curlen,0,(len-curlen+1)); /* also set trailing \0 byte */
477    sdssetlen(s, len);
478    return s;
479}

481/* Append the specified binary-safe string pointed by 't' of 'len' bytes to the
482 * end of the specified sds string 's'.
483 *
484 * After the call, the passed sds string is no longer valid and all the
485 * references must be substituted with the new pointer returned by the call. */
486sds sdscatlen(sds s, const void *t, size_t len) {
487    size_t curlen = sdslen(s);
488
489    s = sdsMakeRoomFor(s,len);
490    if (s == NULL) return NULL;
491    memcpy(s+curlen, t, len);
492    sdssetlen(s, curlen+len);
493    s[curlen+len] = '\0';
494    return s;
495}

513/* Destructively modify the sds string 's' to hold the specified binary
514 * safe string pointed by 't' of length 'len' bytes. */
515sds sdscpylen(sds s, const char *t, size_t len) {
516    if (sdsalloc(s) < len) {
517        s = sdsMakeRoomFor(s,len-sdslen(s));
518        if (s == NULL) return NULL;
519    }
520    memcpy(s, t, len);
521    s[len] = '\0';
522    sdssetlen(s, len);
523    return s;
524}

543/* Like sdscatprintf() but gets va_list instead of being variadic. */
544sds sdscatvprintf(sds s, const char *fmt, va_list ap) {
545    va_list cpy;
546    char staticbuf[1024], *buf = staticbuf, *t;
547    size_t buflen = strlen(fmt)*2;
548    int bufstrlen;
549
550    /* We try to start using a static buffer for speed.
551     * If not possible we revert to heap allocation. */
552    if (buflen > sizeof(staticbuf)) {
553        buf = s_malloc(buflen);
554        if (buf == NULL) return NULL;
555    } else {
556        buflen = sizeof(staticbuf);
557    }
558
559    /* Alloc enough space for buffer and \0 after failing to
560     * fit the string in the current buffer size. */
561    while(1) {
562        va_copy(cpy,ap);
563        bufstrlen = vsnprintf(buf, buflen, fmt, cpy);
564        va_end(cpy);
565        if (bufstrlen < 0) {
566            if (buf != staticbuf) s_free(buf);
567            return NULL;
568        }
569        if (((size_t)bufstrlen) >= buflen) {
570            if (buf != staticbuf) s_free(buf);
571            buflen = ((size_t)bufstrlen) + 1;
572            buf = s_malloc(buflen);
573            if (buf == NULL) return NULL;
574            continue;
575        }
576        break;
577    }
578
579    /* Finally concat the obtained string to the SDS string and return it. */
580    t = sdscatlen(s, buf, bufstrlen);
581    if (buf != staticbuf) s_free(buf);
582    return t;
583}

585/* Append to the sds string 's' a string obtained using printf-alike format
586 * specifier.
587 *
588 * After the call, the modified sds string is no longer valid and all the
589 * references must be substituted with the new pointer returned by the call.
590 *
591 * Example:
592 *
593 * s = sdsnew("Sum is: ");
594 * s = sdscatprintf(s,"%d+%d = %d",a,b,a+b).
595 *
596 * Often you need to create a string from scratch with the printf-alike
597 * format. When this is the need, just use sdsempty() as the target string:
598 *
599 * s = sdscatprintf(sdsempty(), "... your format ...", args);
600 */
601sds sdscatprintf(sds s, const char *fmt, ...) {
602    va_list ap;
603    char *t;
604    va_start(ap, fmt);
605    t = sdscatvprintf(s,fmt,ap);
606    va_end(ap);
607    return t;
608}

610/* This function is similar to sdscatprintf, but much faster as it does
611 * not rely on sprintf() family functions implemented by the libc that
612 * are often very slow. Moreover directly handling the sds string as
613 * new data is concatenated provides a performance improvement.
614 *
615 * However this function only handles an incompatible subset of printf-alike
616 * format specifiers:
617 *
618 * %s - C String
619 * %S - SDS string
620 * %i - signed int
621 * %I - 64 bit signed integer (long long, int64_t)
622 * %u - unsigned int
623 * %U - 64 bit unsigned integer (unsigned long long, uint64_t)
624 * %% - Verbatim "%" character.
625 */
626sds sdscatfmt(sds s, char const *fmt, ...) {
627    size_t initlen = sdslen(s);
628    const char *f = fmt;
629    long i;
630    va_list ap;
631
632    /* To avoid continuous reallocations, let's start with a buffer that
633     * can hold at least two times the format string itself. It's not the
634     * best heuristic but seems to work in practice. */
635    s = sdsMakeRoomFor(s, strlen(fmt)*2);
636    va_start(ap,fmt);
637    f = fmt;    /* Next format specifier byte to process. */
638    i = initlen; /* Position of the next byte to write to dest str. */
639    while(*f) {
640        char next, *str;
641        size_t l;
642        long long num;
643        unsigned long long unum;
644
645        /* Make sure there is always space for at least 1 char. */
646        if (sdsavail(s)==0) {
647            s = sdsMakeRoomFor(s,1);
648        }
649
650        switch(*f) {
651        case '%':
652            next = *(f+1);
653            if (next == '\0') break;
654            f++;
655            switch(next) {
656            case 's':
657            case 'S':
658                str = va_arg(ap,char*);
659                l = (next == 's') ? strlen(str) : sdslen(str);
660                if (sdsavail(s) < l) {
661                    s = sdsMakeRoomFor(s,l);
662                }
663                memcpy(s+i,str,l);
664                sdsinclen(s,l);
665                i += l;
666                break;
667            case 'i':
668            case 'I':
669                if (next == 'i')
670                    num = va_arg(ap,int);
671                else
672                    num = va_arg(ap,long long);
673                {
674                    char buf[LONG_STR_SIZE];
675                    l = ll2string(buf,sizeof(buf),num);
676                    if (sdsavail(s) < l) {
677                        s = sdsMakeRoomFor(s,l);
678                    }
679                    memcpy(s+i,buf,l);
680                    sdsinclen(s,l);
681                    i += l;
682                }
683                break;
684            case 'u':
685            case 'U':
686                if (next == 'u')
687                    unum = va_arg(ap,unsigned int);
688                else
689                    unum = va_arg(ap,unsigned long long);
690                {
691                    char buf[LONG_STR_SIZE];
692                    l = ull2string(buf,sizeof(buf),unum);
693                    if (sdsavail(s) < l) {
694                        s = sdsMakeRoomFor(s,l);
695                    }
696                    memcpy(s+i,buf,l);
697                    sdsinclen(s,l);
698                    i += l;
699                }
700                break;
701            default: /* Handle %% and generally %<unknown>. */
702                s[i++] = next;
703                sdsinclen(s,1);
704                break;
705            }
706            break;
707        default:
708            s[i++] = *f;
709            sdsinclen(s,1);
710            break;
711        }
712        f++;
713    }
714    va_end(ap);
715
716    /* Add null-term */
717    s[i] = '\0';
718    return s;
719}

721/* Remove the part of the string from left and from right composed just of
722 * contiguous characters found in 'cset', that is a null terminated C string.
723 *
724 * After the call, the modified sds string is no longer valid and all the
725 * references must be substituted with the new pointer returned by the call.
726 *
727 * Example:
728 *
729 * s = sdsnew("AA...AA.a.aa.aHelloWorld     :::");
730 * s = sdstrim(s,"Aa. :");
731 * printf("%s\n", s);
732 *
733 * Output will be just "HelloWorld".
734 */
735sds sdstrim(sds s, const char *cset) {
736    char *end, *sp, *ep;
737    size_t len;
738
739    sp = s;
740    ep = end = s+sdslen(s)-1;
741    while(sp <= end && strchr(cset, *sp)) sp++;
742    while(ep > sp && strchr(cset, *ep)) ep--;
743    len = (ep-sp)+1;
744    if (s != sp) memmove(s, sp, len);
745    s[len] = '\0';
746    sdssetlen(s,len);
747    return s;
748}

750/* Changes the input string to be a subset of the original.
751 * It does not release the free space in the string, so a call to
752 * sdsRemoveFreeSpace may be wise after. */
753void sdssubstr(sds s, size_t start, size_t len) {
754    /* Clamp out of range input */
755    size_t oldlen = sdslen(s);
756    if (start >= oldlen) start = len = 0;
757    if (len > oldlen-start) len = oldlen-start;
758
759    /* Move the data */
760    if (len) memmove(s, s+start, len);
761    s[len] = 0;
762    sdssetlen(s,len);
763}

765/* Turn the string into a smaller (or equal) string containing only the
766 * substring specified by the 'start' and 'end' indexes.
767 *
768 * start and end can be negative, where -1 means the last character of the
769 * string, -2 the penultimate character, and so forth.
770 *
771 * The interval is inclusive, so the start and end characters will be part
772 * of the resulting string.
773 *
774 * The string is modified in-place.
775 *
776 * NOTE: this function can be misleading and can have unexpected behaviour,
777 * specifically when you want the length of the new string to be 0.
778 * Having start==end will result in a string with one character.
779 * please consider using sdssubstr instead.
780 *
781 * Example:
782 *
783 * s = sdsnew("Hello World");
784 * sdsrange(s,1,-1); => "ello World"
785 */
786void sdsrange(sds s, ssize_t start, ssize_t end) {
787    size_t newlen, len = sdslen(s);
788    if (len == 0) return;
789    if (start < 0)
790        start = len + start;
791    if (end < 0)
792        end = len + end;
793    newlen = (start > end) ? 0 : (end-start)+1;
794    sdssubstr(s, start, newlen);
795}

811/* Compare two sds strings s1 and s2 with memcmp().
812 *
813 * Return value:
814 *
815 *     positive if s1 > s2.
816 *     negative if s1 < s2.
817 *     0 if s1 and s2 are exactly the same binary string.
818 *
819 * If two strings share exactly the same prefix, but one of the two has
820 * additional characters, the longer string is considered to be greater than
821 * the smaller one. */
822int sdscmp(const sds s1, const sds s2) {
823    size_t l1, l2, minlen;
824    int cmp;
825
826    l1 = sdslen(s1);
827    l2 = sdslen(s2);
828    minlen = (l1 < l2) ? l1 : l2;
829    cmp = memcmp(s1,s2,minlen);
830    if (cmp == 0) return l1>l2? 1: (l1<l2? -1: 0);
831    return cmp;
832}

834/* Split 's' with separator in 'sep'. An array
835 * of sds strings is returned. *count will be set
836 * by reference to the number of tokens returned.
837 *
838 * On out of memory, zero length string, zero length
839 * separator, NULL is returned.
840 *
841 * Note that 'sep' is able to split a string using
842 * a multi-character separator. For example
843 * sdssplit("foo_-_bar","_-_"); will return two
844 * elements "foo" and "bar".
845 *
846 * This version of the function is binary-safe but
847 * requires length arguments. sdssplit() is just the
848 * same function but for zero-terminated strings.
849 */
850sds *sdssplitlen(const char *s, ssize_t len, const char *sep, int seplen, int *count) {
851    int elements = 0, slots = 5;
852    long start = 0, j;
853    sds *tokens;
854
855    if (seplen < 1 || len <= 0) {
856        *count = 0;
857        return NULL;
858    }
859    tokens = s_malloc(sizeof(sds)*slots);
860    if (tokens == NULL) return NULL;
861
862    for (j = 0; j < (len-(seplen-1)); j++) {
863        /* make sure there is room for the next element and the final one */
864        if (slots < elements+2) {
865            sds *newtokens;
866
867            slots *= 2;
868            newtokens = s_realloc(tokens,sizeof(sds)*slots);
869            if (newtokens == NULL) goto cleanup;
870            tokens = newtokens;
871        }
872        /* search the separator */
873        if ((seplen == 1 && *(s+j) == sep[0]) || (memcmp(s+j,sep,seplen) == 0)) {
874            tokens[elements] = sdsnewlen(s+start,j-start);
875            if (tokens[elements] == NULL) goto cleanup;
876            elements++;
877            start = j+seplen;
878            j = j+seplen-1; /* skip the separator */
879        }
880    }
881    /* Add the final element. We are sure there is room in the tokens array. */
882    tokens[elements] = sdsnewlen(s+start,len-start);
883    if (tokens[elements] == NULL) goto cleanup;
884    elements++;
885    *count = elements;
886    return tokens;
887
888cleanup:
889    {
890        int i;
891        for (i = 0; i < elements; i++) sdsfree(tokens[i]);
892        s_free(tokens);
893        *count = 0;
894        return NULL;
895    }
896}

898/* Free the result returned by sdssplitlen(), or do nothing if 'tokens' is NULL. */
899void sdsfreesplitres(sds *tokens, int count) {
900    if (!tokens) return;
901    while(count--)
902        sdsfree(tokens[count]);
903    s_free(tokens);
904}

906/* Append to the sds string "s" an escaped string representation where
907 * all the non-printable characters (tested with isprint()) are turned into
908 * escapes in the form "\n\r\a...." or "\x<hex-number>".
909 *
910 * After the call, the modified sds string is no longer valid and all the
911 * references must be substituted with the new pointer returned by the call. */
912sds sdscatrepr(sds s, const char *p, size_t len) {
913    s = sdsMakeRoomFor(s, len + 2);
914    s = sdscatlen(s,"\"",1);
915    while(len--) {
916        switch(*p) {
917        case '\\':
918        case '"':
919            s = sdscatprintf(s,"\\%c",*p);
920            break;
921        case '\n': s = sdscatlen(s,"\\n",2); break;
922        case '\r': s = sdscatlen(s,"\\r",2); break;
923        case '\t': s = sdscatlen(s,"\\t",2); break;
924        case '\a': s = sdscatlen(s,"\\a",2); break;
925        case '\b': s = sdscatlen(s,"\\b",2); break;
926        default:
927            if (isprint(*p))
928                s = sdscatlen(s, p, 1);
929            else
930                s = sdscatprintf(s,"\\x%02x",(unsigned char)*p);
931            break;
932        }
933        p++;
934    }
935    return sdscatlen(s,"\"",1);
936}

938/* Returns one if the string contains characters to be escaped
939 * by sdscatrepr(), zero otherwise.
940 *
941 * Typically, this should be used to help protect aggregated strings in a way
942 * that is compatible with sdssplitargs(). For this reason, also spaces will be
943 * treated as needing an escape.
944 */
945int sdsneedsrepr(const sds s) {
946    size_t len = sdslen(s);
947    const char *p = s;
948
949    while (len--) {
950        if (*p == '\\' || *p == '"' || *p == '\n' || *p == '\r' ||
951            *p == '\t' || *p == '\a' || *p == '\b' || !isprint(*p) || isspace(*p)) return 1;
952        p++;
953    }
954
955    return 0;
956}

958/* Helper function for sdssplitargs() that returns non zero if 'c'
959 * is a valid hex digit. */
960int is_hex_digit(char c) {
961    return (c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') ||
962           (c >= 'A' && c <= 'F');
963}

965/* Helper function for sdssplitargs() that converts a hex digit into an
966 * integer from 0 to 15 */
967int hex_digit_to_int(char c) {
968    switch(c) {
969    case '0': return 0;
970    case '1': return 1;
971    case '2': return 2;
972    case '3': return 3;
973    case '4': return 4;
974    case '5': return 5;
975    case '6': return 6;
976    case '7': return 7;
977    case '8': return 8;
978    case '9': return 9;
979    case 'a': case 'A': return 10;
980    case 'b': case 'B': return 11;
981    case 'c': case 'C': return 12;
982    case 'd': case 'D': return 13;
983    case 'e': case 'E': return 14;
984    case 'f': case 'F': return 15;
985    default: return 0;
986    }

989/* Split a line into arguments, where every argument can be in the
990 * following programming-language REPL-alike form:
991 *
992 * foo bar "newline are supported\n" and "\xff\x00otherstuff"
993 *
994 * The number of arguments is stored into *argc, and an array
995 * of sds is returned.
996 *
997 * The caller should free the resulting array of sds strings with
998 * sdsfreesplitres().
999 *
1000 * Note that sdscatrepr() is able to convert back a string into
1001 * a quoted string in the same format sdssplitargs() is able to parse.
1002 *
1003 * The function returns the allocated tokens on success, even when the
1004 * input string is empty, or NULL if the input contains unbalanced
1005 * quotes or closed quotes followed by non space characters
1006 * as in: "foo"bar or "foo'
1007 */
1008sds *sdssplitargs(const char *line, int *argc) {
1009    const char *p = line;
1010    char *current = NULL;
1011    char **vector = NULL;
1012
1013    *argc = 0;
1014    while(1) {
1015        /* skip blanks */
1016        while(*p && isspace(*p)) p++;
1017        if (*p) {
1018            /* get a token */
1019            int inq=0;  /* set to 1 if we are in "quotes" */
1020            int insq=0; /* set to 1 if we are in 'single quotes' */
1021            int done=0;
1022
1023            if (current == NULL) current = sdsempty();
1024            while(!done) {
1025                if (inq) {
1026                    if (*p == '\\' && *(p+1) == 'x' &&
1027                                             is_hex_digit(*(p+2)) &&
1028                                             is_hex_digit(*(p+3)))
1029                    {
1030                        unsigned char byte;
1031
1032                        byte = (hex_digit_to_int(*(p+2))*16)+
1033                                hex_digit_to_int(*(p+3));
1034                        current = sdscatlen(current,(char*)&byte,1);
1035                        p += 3;
1036                    } else if (*p == '\\' && *(p+1)) {
1037                        char c;
1038
1039                        p++;
1040                        switch(*p) {
1041                        case 'n': c = '\n'; break;
1042                        case 'r': c = '\r'; break;
1043                        case 't': c = '\t'; break;
1044                        case 'b': c = '\b'; break;
1045                        case 'a': c = '\a'; break;
1046                        default: c = *p; break;
1047                        }
1048                        current = sdscatlen(current,&c,1);
1049                    } else if (*p == '"') {
1050                        /* closing quote must be followed by a space or
1051                         * nothing at all. */
1052                        if (*(p+1) && !isspace(*(p+1))) goto err;
1053                        done=1;
1054                    } else if (!*p) {
1055                        /* unterminated quotes */
1056                        goto err;
1057                    } else {
1058                        current = sdscatlen(current,p,1);
1059                    }
1060                } else if (insq) {
1061                    if (*p == '\\' && *(p+1) == '\'') {
1062                        p++;
1063                        current = sdscatlen(current,"'",1);
1064                    } else if (*p == '\'') {
1065                        /* closing quote must be followed by a space or
1066                         * nothing at all. */
1067                        if (*(p+1) && !isspace(*(p+1))) goto err;
1068                        done=1;
1069                    } else if (!*p) {
1070                        /* unterminated quotes */
1071                        goto err;
1072                    } else {
1073                        current = sdscatlen(current,p,1);
1074                    }
1075                } else {
1076                    switch(*p) {
1077                    case ' ':
1078                    case '\n':
1079                    case '\r':
1080                    case '\t':
1081                    case '\0':
1082                        done=1;
1083                        break;
1084                    case '"':
1085                        inq=1;
1086                        break;
1087                    case '\'':
1088                        insq=1;
1089                        break;
1090                    default:
1091                        current = sdscatlen(current,p,1);
1092                        break;
1093                    }
1094                }
1095                if (*p) p++;
1096            }
1097            /* add the token to the vector */
1098            vector = s_realloc(vector,((*argc)+1)*sizeof(char*));
1099            vector[*argc] = current;
1100            (*argc)++;
1101            current = NULL;
1102        } else {
1103            /* Even on empty input string return something not NULL. */
1104            if (vector == NULL) vector = s_malloc(sizeof(void*));
1105            return vector;
1106        }
1107    }
1108
1109err:
1110    while((*argc)--)
1111        sdsfree(vector[*argc]);
1112    s_free(vector);
1113    if (current) sdsfree(current);
1114    *argc = 0;
1115    return NULL;
1116}

1118/* Modify the string substituting all the occurrences of the set of
1119 * characters specified in the 'from' string to the corresponding character
1120 * in the 'to' array.
1121 *
1122 * For instance: sdsmapchars(mystring, "ho", "01", 2)
1123 * will have the effect of turning the string "hello" into "0ell1".
1124 *
1125 * The function returns the sds string pointer, that is always the same
1126 * as the input pointer since no resize is needed. */
1127sds sdsmapchars(sds s, const char *from, const char *to, size_t setlen) {
1128    size_t j, i, l = sdslen(s);
1129
1130    for (j = 0; j < l; j++) {
1131        for (i = 0; i < setlen; i++) {
1132            if (s[j] == from[i]) {
1133                s[j] = to[i];
1134                break;
1135            }
1136        }
1137    }
1138    return s;
1139}

1175/* Perform expansion of a template string and return the result as a newly
1176 * allocated sds.
1177 *
1178 * Template variables are specified using curly brackets, e.g. {variable}.
1179 * An opening bracket can be quoted by repeating it twice.
1180 */
1181sds sdstemplate(const char *template, sdstemplate_callback_t cb_func, void *cb_arg)
1182{
1183    sds res = sdsempty();
1184    const char *p = template;
1185
1186    while (*p) {
1187        /* Find next variable, copy everything until there */
1188        const char *sv = strchr(p, '{');
1189        if (!sv) {
1190            /* Not found: copy till rest of template and stop */
1191            res = sdscat(res, p);
1192            break;
1193        } else if (sv > p) {
1194            /* Found: copy anything up to the beginning of the variable */
1195            res = sdscatlen(res, p, sv - p);
1196        }
1197
1198        /* Skip into variable name, handle premature end or quoting */
1199        sv++;
1200        if (!*sv) goto error;       /* Premature end of template */
1201        if (*sv == '{') {
1202            /* Quoted '{' */
1203            p = sv + 1;
1204            res = sdscat(res, "{");
1205            continue;
1206        }
1207
1208        /* Find end of variable name, handle premature end of template */
1209        const char *ev = strchr(sv, '}');
1210        if (!ev) goto error;
1211
1212        /* Pass variable name to callback and obtain value. If callback failed,
1213         * abort. */
1214        sds varname = sdsnewlen(sv, ev - sv);
1215        sds value = cb_func(varname, cb_arg);
1216        sdsfree(varname);
1217        if (!value) goto error;
1218
1219        /* Append value to result and continue */
1220        res = sdscat(res, value);
1221        sdsfree(value);
1222        p = ev + 1;
1223    }
1224
1225    return res;
1226
1227error:
1228    sdsfree(res);
1229    return NULL;
1230}
