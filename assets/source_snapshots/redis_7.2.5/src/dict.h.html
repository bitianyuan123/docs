51typedef struct dictType {
52    uint64_t (*hashFunction)(const void *key);
53    void *(*keyDup)(dict *d, const void *key);
54    void *(*valDup)(dict *d, const void *obj);
55    int (*keyCompare)(dict *d, const void *key1, const void *key2);
56    void (*keyDestructor)(dict *d, void *key);
57    void (*valDestructor)(dict *d, void *obj);
58    int (*expandAllowed)(size_t moreMem, double usedRatio);
59    /* Flags */
60    /* The 'no_value' flag, if set, indicates that values are not used, i.e. the
61     * dict is a set. When this flag is set, it's not possible to access the
62     * value of a dictEntry and it's also impossible to use dictSetKey(). Entry
63     * metadata can also not be used. */
64    unsigned int no_value:1;
65    /* If no_value = 1 and all keys are odd (LSB=1), setting keys_are_odd = 1
66     * enables one more optimization: to store a key without an allocated
67     * dictEntry. */
68    unsigned int keys_are_odd:1;
69    /* TODO: Add a 'keys_are_even' flag and use a similar optimization if that
70     * flag is set. */
71
72    /* Allow each dict and dictEntry to carry extra caller-defined metadata. The
73     * extra memory is initialized to 0 when allocated. */
74    size_t (*dictEntryMetadataBytes)(dict *d);
75    size_t (*dictMetadataBytes)(void);
76    /* Optional callback called after an entry has been reallocated (due to
77     * active defrag). Only called if the entry has metadata. */
78    void (*afterReplaceEntry)(dict *d, dictEntry *entry);
79} dictType;

81#define DICTHT_SIZE(exp) ((exp) == -1 ? 0 : (unsigned long)1<<(exp))
82#define DICTHT_SIZE_MASK(exp) ((exp) == -1 ? 0 : (DICTHT_SIZE(exp))-1)
83
84struct dict {
85    dictType *type;
86
87    dictEntry **ht_table[2];
88    unsigned long ht_used[2];
89
90    long rehashidx; /* rehashing not in progress if rehashidx == -1 */
91
92    /* Keep small vars at end for optimal (minimal) struct padding */
93    int16_t pauserehash; /* If >0 rehashing is paused (<0 indicates coding error) */
94    signed char ht_size_exp[2]; /* exponent of size. (size = 1<<exp) */
95
96    void *metadata[];           /* An arbitrary number of bytes (starting at a
97                                 * pointer-aligned address) of size as defined
98                                 * by dictType's dictEntryBytes. */
99};

101/* If safe is set to 1 this is a safe iterator, that means, you can call
102 * dictAdd, dictFind, and other functions against the dictionary even while
103 * iterating. Otherwise it is a non safe iterator, and only dictNext()
104 * should be called while iterating. */
105typedef struct dictIterator {
106    dict *d;
107    long index;
108    int table, safe;
109    dictEntry *entry, *nextEntry;
110    /* unsafe iterator fingerprint for misuse detection. */
111    unsigned long long fingerprint;
112} dictIterator;

127#define dictFreeVal(d, entry) do {                     \
128    if ((d)->type->valDestructor)                      \
129        (d)->type->valDestructor((d), dictGetVal(entry)); \
130   } while(0)
131
132#define dictFreeKey(d, entry) \
133    if ((d)->type->keyDestructor) \
134        (d)->type->keyDestructor((d), dictGetKey(entry))

136#define dictCompareKeys(d, key1, key2) \
137    (((d)->type->keyCompare) ? \
