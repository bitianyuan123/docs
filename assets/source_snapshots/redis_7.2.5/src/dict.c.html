50/* Using dictEnableResize() / dictDisableResize() we make possible to disable
51 * resizing and rehashing of the hash table as needed. This is very important
52 * for Redis, as we use copy-on-write and don't want to move too much memory
53 * around when there is a child performing saving operations.
54 *
55 * Note that even when dict_can_resize is set to DICT_RESIZE_AVOID, not all
56 * resizes are prevented: a hash table is still allowed to grow if the ratio
57 * between the number of elements and the buckets > dict_force_resize_ratio. */

119/* The 3 least significant bits in a pointer to a dictEntry determines what the
120 * pointer actually points to. If the least bit is set, it's a key. Otherwise,
121 * the bit pattern of the least 3 significant bits mark the kind of entry. */
122
123#define ENTRY_PTR_MASK     7 /* 111 */
124#define ENTRY_PTR_NORMAL   0 /* 000 */
125#define ENTRY_PTR_NO_VALUE 2 /* 010 */
126
127/* Returns 1 if the entry pointer is a pointer to a key, rather than to an
128 * allocated entry. Returns 0 otherwise. */
129static inline int entryIsKey(const dictEntry *de) {
130    return (uintptr_t)(void *)de & 1;
131}
132
133/* Returns 1 if the pointer is actually a pointer to a dictEntry struct. Returns
134 * 0 otherwise. */
135static inline int entryIsNormal(const dictEntry *de) {
136    return ((uintptr_t)(void *)de & ENTRY_PTR_MASK) == ENTRY_PTR_NORMAL;
137}
138
139/* Returns 1 if the entry is a special entry with key and next, but without
140 * value. Returns 0 otherwise. */
141static inline int entryIsNoValue(const dictEntry *de) {
142    return ((uintptr_t)(void *)de & ENTRY_PTR_MASK) == ENTRY_PTR_NO_VALUE;
143}

145/* Creates an entry without a value field. */
146static inline dictEntry *createEntryNoValue(void *key, dictEntry *next) {
147    dictEntryNoValue *entry = zmalloc(sizeof(*entry));
148    entry->key = key;
149    entry->next = next;
150    return (dictEntry *)(void *)((uintptr_t)(void *)entry | ENTRY_PTR_NO_VALUE);
151}

153static inline dictEntry *encodeMaskedPtr(const void *ptr, unsigned int bits) {
154    assert(((uintptr_t)ptr & ENTRY_PTR_MASK) == 0);
155    return (dictEntry *)(void *)((uintptr_t)ptr | bits);
156}

158static inline void *decodeMaskedPtr(const dictEntry *de) {
159    assert(!entryIsKey(de));
160    return (void *)((uintptr_t)(void *)de & ~ENTRY_PTR_MASK);
161}

208/* Resize the table to the minimal size that contains all the elements,
209 * but with the invariant of a USED/BUCKETS ratio near to <= 1 */
210int dictResize(dict *d)
211{
212    unsigned long minimal;
213
214    if (dict_can_resize != DICT_RESIZE_ENABLE || dictIsRehashing(d)) return DICT_ERR;
215    minimal = d->ht_used[0];
216    if (minimal < DICT_HT_INITIAL_SIZE)
217        minimal = DICT_HT_INITIAL_SIZE;
218    return dictExpand(d, minimal);
219}

221/* Expand or create the hash table,
222 * when malloc_failed is non-NULL, it'll avoid panic if malloc fails (in which case it'll be set to 1).
223 * Returns DICT_OK if expand was performed, and DICT_ERR if skipped. */
224int _dictExpand(dict *d, unsigned long size, int* malloc_failed)
225{
226    if (malloc_failed) *malloc_failed = 0;
227
228    /* the size is invalid if it is smaller than the number of
229     * elements already inside the hash table */
230    if (dictIsRehashing(d) || d->ht_used[0] > size)
231        return DICT_ERR;
232
233    /* the new hash table */
234    dictEntry **new_ht_table;
235    unsigned long new_ht_used;
236    signed char new_ht_size_exp = _dictNextExp(size);
237
238    /* Detect overflows */
239    size_t newsize = 1ul<<new_ht_size_exp;
240    if (newsize < size || newsize * sizeof(dictEntry*) < newsize)
241        return DICT_ERR;
242
243    /* Rehashing to the same table size is not useful. */
244    if (new_ht_size_exp == d->ht_size_exp[0]) return DICT_ERR;
245
246    /* Allocate the new hash table and initialize all pointers to NULL */
247    if (malloc_failed) {
248        new_ht_table = ztrycalloc(newsize*sizeof(dictEntry*));
249        *malloc_failed = new_ht_table == NULL;
250        if (*malloc_failed)
251            return DICT_ERR;
252    } else
253        new_ht_table = zcalloc(newsize*sizeof(dictEntry*));
254
255    new_ht_used = 0;
256
257    /* Is this the first initialization? If so it's not really a rehashing
258     * we just set the first hash table so that it can accept keys. */
259    if (d->ht_table[0] == NULL) {
260        d->ht_size_exp[0] = new_ht_size_exp;
261        d->ht_used[0] = new_ht_used;
262        d->ht_table[0] = new_ht_table;
263        return DICT_OK;
264    }
265
266    /* Prepare a second hash table for incremental rehashing */
267    d->ht_size_exp[1] = new_ht_size_exp;
268    d->ht_used[1] = new_ht_used;
269    d->ht_table[1] = new_ht_table;
270    d->rehashidx = 0;
271    return DICT_OK;
272}

286/* Performs N steps of incremental rehashing. Returns 1 if there are still
287 * keys to move from the old to the new hash table, otherwise 0 is returned.
288 *
289 * Note that a rehashing step consists in moving a bucket (that may have more
290 * than one key as we use chaining) from the old to the new hash table, however
291 * since part of the hash table may be composed of empty spaces, it is not
292 * guaranteed that this function will rehash even a single bucket, since it
293 * will visit at max N*10 empty buckets in total, otherwise the amount of
294 * work it does would be unbound and the function may block for a long time. */
295int dictRehash(dict *d, int n) {
296    int empty_visits = n*10; /* Max number of empty buckets to visit. */
297    unsigned long s0 = DICTHT_SIZE(d->ht_size_exp[0]);
298    unsigned long s1 = DICTHT_SIZE(d->ht_size_exp[1]);
299    if (dict_can_resize == DICT_RESIZE_FORBID || !dictIsRehashing(d)) return 0;
300    if (dict_can_resize == DICT_RESIZE_AVOID && 
301        ((s1 > s0 && s1 / s0 < dict_force_resize_ratio) ||
302         (s1 < s0 && s0 / s1 < dict_force_resize_ratio)))
303    {
304        return 0;
305    }
306
307    while(n-- && d->ht_used[0] != 0) {
308        dictEntry *de, *nextde;
309
310        /* Note that rehashidx can't overflow as we are sure there are more
311         * elements because ht[0].used != 0 */
312        assert(DICTHT_SIZE(d->ht_size_exp[0]) > (unsigned long)d->rehashidx);
313        while(d->ht_table[0][d->rehashidx] == NULL) {
314            d->rehashidx++;
315            if (--empty_visits == 0) return 1;
316        }
317        de = d->ht_table[0][d->rehashidx];
318        /* Move all the keys in this bucket from the old to the new hash HT */
319        while(de) {
320            uint64_t h;
321
322            nextde = dictGetNext(de);
323            void *key = dictGetKey(de);
324            /* Get the index in the new hash table */
325            if (d->ht_size_exp[1] > d->ht_size_exp[0]) {
326                h = dictHashKey(d, key) & DICTHT_SIZE_MASK(d->ht_size_exp[1]);
327            } else {
328                /* We're shrinking the table. The tables sizes are powers of
329                 * two, so we simply mask the bucket index in the larger table
330                 * to get the bucket index in the smaller table. */
331                h = d->rehashidx & DICTHT_SIZE_MASK(d->ht_size_exp[1]);
332            }
333            if (d->type->no_value) {
334                if (d->type->keys_are_odd && !d->ht_table[1][h]) {
335                    /* Destination bucket is empty and we can store the key
336                     * directly without an allocated entry. Free the old entry
337                     * if it's an allocated entry.
338                     *
339                     * TODO: Add a flag 'keys_are_even' and if set, we can use
340                     * this optimization for these dicts too. We can set the LSB
341                     * bit when stored as a dict entry and clear it again when
342                     * we need the key back. */
343                    assert(entryIsKey(key));
344                    if (!entryIsKey(de)) zfree(decodeMaskedPtr(de));
345                    de = key;
346                } else if (entryIsKey(de)) {
347                    /* We don't have an allocated entry but we need one. */
348                    de = createEntryNoValue(key, d->ht_table[1][h]);
349                } else {
350                    /* Just move the existing entry to the destination table and
351                     * update the 'next' field. */
352                    assert(entryIsNoValue(de));
353                    dictSetNext(de, d->ht_table[1][h]);
354                }
355            } else {
356                dictSetNext(de, d->ht_table[1][h]);
357            }
358            d->ht_table[1][h] = de;
359            d->ht_used[0]--;
360            d->ht_used[1]++;
361            de = nextde;
362        }
363        d->ht_table[0][d->rehashidx] = NULL;
364        d->rehashidx++;
365    }
366
367    /* Check if we already rehashed the whole table... */
368    if (d->ht_used[0] == 0) {
369        zfree(d->ht_table[0]);
370        /* Copy the new ht onto the old one */
371        d->ht_table[0] = d->ht_table[1];
372        d->ht_used[0] = d->ht_used[1];
373        d->ht_size_exp[0] = d->ht_size_exp[1];
374        _dictReset(d, 1);
375        d->rehashidx = -1;
376        return 0;
377    }
378
379    /* More to rehash... */
380    return 1;
381}

390/* Rehash in ms+"delta" milliseconds. The value of "delta" is larger 
391 * than 0, and is smaller than 1 in most cases. The exact upper bound 
392 * depends on the running time of dictRehash(d,100).*/
393int dictRehashMilliseconds(dict *d, int ms) {
394    if (d->pauserehash > 0) return 0;
395
396    long long start = timeInMilliseconds();
397    int rehashes = 0;
398
399    while(dictRehash(d,100)) {
400        rehashes += 100;
401        if (timeInMilliseconds()-start > ms) break;
402    }
403    return rehashes;
404}

408 * middle of a rehashing we can't mess with the two hash tables otherwise

412 * dictionary so that the hash table automatically migrates from H1 to H2

433/* Low level add or find:
434 * This function adds the entry but instead of setting a value returns the
435 * dictEntry structure to the user, that will make sure to fill the value
436 * field as they wish.
437 *
438 * This function is also directly exposed to the user API to be called
439 * mainly in order to store non-pointers inside the hash value, example:
440 *
441 * entry = dictAddRaw(dict,mykey,NULL);
442 * if (entry != NULL) dictSetSignedIntegerVal(entry,1000);
443 *
444 * Return values:
445 *
446 * If key already exists NULL is returned, and "*existing" is populated
447 * with the existing entry if existing is not NULL.
448 *
449 * If key was added, the hash entry is returned to be manipulated by the caller.
450 */
451dictEntry *dictAddRaw(dict *d, void *key, dictEntry **existing)
452{
453    /* Get the position for the new key or NULL if the key already exists. */
454    void *position = dictFindPositionForInsert(d, key, existing);
455    if (!position) return NULL;
456
457    /* Dup the key if necessary. */
458    if (d->type->keyDup) key = d->type->keyDup(d, key);
459
460    return dictInsertAtPosition(d, key, position);
461}

463/* Adds a key in the dict's hashtable at the position returned by a preceding
464 * call to dictFindPositionForInsert. This is a low level function which allows
465 * splitting dictAddRaw in two parts. Normally, dictAddRaw or dictAdd should be
466 * used instead. */
467dictEntry *dictInsertAtPosition(dict *d, void *key, void *position) {
468    dictEntry **bucket = position; /* It's a bucket, but the API hides that. */
469    dictEntry *entry;
470    /* If rehashing is ongoing, we insert in table 1, otherwise in table 0.
471     * Assert that the provided bucket is the right table. */
472    int htidx = dictIsRehashing(d) ? 1 : 0;
473    assert(bucket >= &d->ht_table[htidx][0] &&
474           bucket <= &d->ht_table[htidx][DICTHT_SIZE_MASK(d->ht_size_exp[htidx])]);
475    size_t metasize = dictEntryMetadataSize(d);
476    if (d->type->no_value) {
477        assert(!metasize); /* Entry metadata + no value not supported. */
478        if (d->type->keys_are_odd && !*bucket) {
479            /* We can store the key directly in the destination bucket without the
480             * allocated entry.
481             *
482             * TODO: Add a flag 'keys_are_even' and if set, we can use this
483             * optimization for these dicts too. We can set the LSB bit when
484             * stored as a dict entry and clear it again when we need the key
485             * back. */
486            entry = key;
487            assert(entryIsKey(entry));
488        } else {
489            /* Allocate an entry without value. */
490            entry = createEntryNoValue(key, *bucket);
491        }
492    } else {
493        /* Allocate the memory and store the new entry.
494         * Insert the element in top, with the assumption that in a database
495         * system it is more likely that recently added entries are accessed
496         * more frequently. */
497        entry = zmalloc(sizeof(*entry) + metasize);
498        assert(entryIsNormal(entry)); /* Check alignment of allocation */
499        if (metasize > 0) {
500            memset(dictEntryMetadata(entry), 0, metasize);
501        }
502        entry->key = key;
503        entry->next = *bucket;
504    }
505    *bucket = entry;
506    d->ht_used[htidx]++;
507
508    return entry;
509}

511/* Add or Overwrite:
512 * Add an element, discarding the old value if the key already exists.
513 * Return 1 if the key was added from scratch, 0 if there was already an
514 * element with such key and dictReplace() just performed a value update
515 * operation. */
516int dictReplace(dict *d, void *key, void *val)
517{
518    dictEntry *entry, *existing;
519
520    /* Try to add the element. If the key
521     * does not exists dictAdd will succeed. */
522    entry = dictAddRaw(d,key,&existing);
523    if (entry) {
524        dictSetVal(d, entry, val);
525        return 1;
526    }
527
528    /* Set the new value and free the old one. Note that it is important
529     * to do that in this order, as the value may just be exactly the same
530     * as the previous one. In this context, think to reference counting,
531     * you want to increment (set), and then decrement (free), and not the
532     * reverse. */
533    void *oldval = dictGetVal(existing);
534    dictSetVal(d, existing, val);
535    if (d->type->valDestructor)
536        d->type->valDestructor(d, oldval);
537    return 0;
538}

540/* Add or Find:
541 * dictAddOrFind() is simply a version of dictAddRaw() that always
542 * returns the hash entry of the specified key, even if the key already

544 * existing key is returned.)

553/* Search and remove an element. This is a helper function for
554 * dictDelete() and dictUnlink(), please check the top comment
555 * of those functions. */
556static dictEntry *dictGenericDelete(dict *d, const void *key, int nofree) {
557    uint64_t h, idx;
558    dictEntry *he, *prevHe;
559    int table;
560
561    /* dict is empty */
562    if (dictSize(d) == 0) return NULL;
563
564    if (dictIsRehashing(d)) _dictRehashStep(d);
565    h = dictHashKey(d, key);
566
567    for (table = 0; table <= 1; table++) {
568        idx = h & DICTHT_SIZE_MASK(d->ht_size_exp[table]);
569        he = d->ht_table[table][idx];
570        prevHe = NULL;
571        while(he) {
572            void *he_key = dictGetKey(he);
573            if (key == he_key || dictCompareKeys(d, key, he_key)) {
574                /* Unlink the element from the list */
575                if (prevHe)
576                    dictSetNext(prevHe, dictGetNext(he));
577                else
578                    d->ht_table[table][idx] = dictGetNext(he);
579                if (!nofree) {
580                    dictFreeUnlinkedEntry(d, he);
581                }
582                d->ht_used[table]--;
583                return he;
584            }
585            prevHe = he;
586            he = dictGetNext(he);
587        }
588        if (!dictIsRehashing(d)) break;
589    }
590    return NULL; /* not found */
591}

599/* Remove an element from the table, but without actually releasing
600 * the key, value and dictionary entry. The dictionary entry is returned
601 * if the element was found (and unlinked from the table), and the user
602 * should later call `dictFreeUnlinkedEntry()` with it in order to release it.
603 * Otherwise if the key is not found, NULL is returned.
604 *
605 * This function is useful when we want to remove something from the hash
606 * table but want to use its value before actually deleting the entry.
607 * Without this function the pattern would require two lookups:
608 *
609 *  entry = dictFind(...);
610 *  // Do something with entry
611 *  dictDelete(dictionary,entry);
612 *
613 * Thanks to this function it is possible to avoid this, and use
614 * instead:
615 *

633/* Destroy an entire dictionary */
634int _dictClear(dict *d, int htidx, void(callback)(dict*)) {
635    unsigned long i;
636
637    /* Free all the elements */
638    for (i = 0; i < DICTHT_SIZE(d->ht_size_exp[htidx]) && d->ht_used[htidx] > 0; i++) {
639        dictEntry *he, *nextHe;
640
641        if (callback && (i & 65535) == 0) callback(d);
642
643        if ((he = d->ht_table[htidx][i]) == NULL) continue;
644        while(he) {
645            nextHe = dictGetNext(he);
646            dictFreeKey(d, he);
647            dictFreeVal(d, he);
648            if (!entryIsKey(he)) zfree(decodeMaskedPtr(he));
649            d->ht_used[htidx]--;
650            he = nextHe;
651        }
652    }
653    /* Free the table and the allocated cache structure */
654    zfree(d->ht_table[htidx]);
655    /* Re-initialize the table */
656    _dictReset(d, htidx);
657    return DICT_OK; /* never fails */
658}

668dictEntry *dictFind(dict *d, const void *key)
669{
670    dictEntry *he;
671    uint64_t h, idx, table;
672
673    if (dictSize(d) == 0) return NULL; /* dict is empty */
674    if (dictIsRehashing(d)) _dictRehashStep(d);
675    h = dictHashKey(d, key);
676    for (table = 0; table <= 1; table++) {
677        idx = h & DICTHT_SIZE_MASK(d->ht_size_exp[table]);
678        he = d->ht_table[table][idx];
679        while(he) {
680            void *he_key = dictGetKey(he);
681            if (key == he_key || dictCompareKeys(d, key, he_key))
682                return he;
683            he = dictGetNext(he);
684        }
685        if (!dictIsRehashing(d)) return NULL;
686    }
687    return NULL;
688}

691    dictEntry *he;
692
693    he = dictFind(d,key);
694    return he ? dictGetVal(he) : NULL;
695}

697/* Find an element from the table, also get the plink of the entry. The entry
698 * is returned if the element is found, and the user should later call
699 * `dictTwoPhaseUnlinkFree` with it in order to unlink and release it. Otherwise if
700 * the key is not found, NULL is returned. These two functions should be used in pair.
701 * `dictTwoPhaseUnlinkFind` pauses rehash and `dictTwoPhaseUnlinkFree` resumes rehash.
702 *
703 * We can use like this:
704 *
705 * dictEntry *de = dictTwoPhaseUnlinkFind(db->dict,key->ptr,&plink, &table);
706 * // Do something, but we can't modify the dict
707 * dictTwoPhaseUnlinkFree(db->dict,de,plink,table); // We don't need to lookup again
708 *
709 * If we want to find an entry before delete this entry, this an optimization to avoid
710 * dictFind followed by dictDelete. i.e. the first API is a find, and it gives some info
711 * to the second one to avoid repeating the lookup
712 */
713dictEntry *dictTwoPhaseUnlinkFind(dict *d, const void *key, dictEntry ***plink, int *table_index) {
714    uint64_t h, idx, table;
715
716    if (dictSize(d) == 0) return NULL; /* dict is empty */
717    if (dictIsRehashing(d)) _dictRehashStep(d);
718    h = dictHashKey(d, key);
719
720    for (table = 0; table <= 1; table++) {
721        idx = h & DICTHT_SIZE_MASK(d->ht_size_exp[table]);
722        dictEntry **ref = &d->ht_table[table][idx];
723        while (ref && *ref) {
724            void *de_key = dictGetKey(*ref);
725            if (key == de_key || dictCompareKeys(d, key, de_key)) {
726                *table_index = table;
727                *plink = ref;
728                dictPauseRehashing(d);
729                return *ref;
730            }
731            ref = dictGetNextRef(*ref);
732        }
733        if (!dictIsRehashing(d)) return NULL;
734    }
735    return NULL;
736}

738void dictTwoPhaseUnlinkFree(dict *d, dictEntry *he, dictEntry **plink, int table_index) {
739    if (he == NULL) return;
740    d->ht_used[table_index]--;
741    *plink = dictGetNext(he);
742    dictFreeKey(d, he);
743    dictFreeVal(d, he);
744    if (!entryIsKey(he)) zfree(decodeMaskedPtr(he));
745    dictResumeRehashing(d);
746}

866/* A fingerprint is a 64 bit number that represents the state of the dictionary
867 * at a given time, it's just a few dict properties xored together.
868 * When an unsafe iterator is initialized, we get the dict fingerprint, and check
869 * the fingerprint again when the iterator is released.
870 * If the two fingerprints are different it means that the user of the iterator
871 * performed forbidden operations against the dictionary while iterating. */
872unsigned long long dictFingerprint(dict *d) {
873    unsigned long long integers[6], hash = 0;
874    int j;
875
876    integers[0] = (long) d->ht_table[0];
877    integers[1] = d->ht_size_exp[0];
878    integers[2] = d->ht_used[0];
879    integers[3] = (long) d->ht_table[1];
880    integers[4] = d->ht_size_exp[1];
881    integers[5] = d->ht_used[1];
882
883    /* We hash N integers by summing every successive integer with the integer
884     * hashing of the previous sum. Basically:
885     *
886     * Result = hash(hash(hash(int1)+int2)+int3) ...
887     *
888     * This way the same set of integers in a different order will (likely) hash
889     * to a different number. */
890    for (j = 0; j < 6; j++) {
891        hash += integers[j];
892        /* For the hashing step we use Tomas Wang's 64 bit integer hash. */
893        hash = (~hash) + (hash << 21); // hash = (hash << 21) - hash - 1;
894        hash = hash ^ (hash >> 24);
895        hash = (hash + (hash << 3)) + (hash << 8); // hash * 265
896        hash = hash ^ (hash >> 14);
897        hash = (hash + (hash << 2)) + (hash << 4); // hash * 21
898        hash = hash ^ (hash >> 28);
899        hash = hash + (hash << 31);
900    }
901    return hash;
902}

944dictEntry *dictNext(dictIterator *iter)
945{
946    while (1) {
947        if (iter->entry == NULL) {
948            if (iter->index == -1 && iter->table == 0) {
949                if (iter->safe)
950                    dictPauseRehashing(iter->d);
951                else
952                    iter->fingerprint = dictFingerprint(iter->d);
953            }
954            iter->index++;
955            if (iter->index >= (long) DICTHT_SIZE(iter->d->ht_size_exp[iter->table])) {
956                if (dictIsRehashing(iter->d) && iter->table == 0) {
957                    iter->table++;
958                    iter->index = 0;
959                } else {
960                    break;
961                }
962            }
963            iter->entry = iter->d->ht_table[iter->table][iter->index];
964        } else {
965            iter->entry = iter->nextEntry;
966        }
967        if (iter->entry) {
968            /* We need to save the 'next' here, the iterator user
969             * may delete the entry we are returning. */
970            iter->nextEntry = dictGetNext(iter->entry);
971            return iter->entry;
972        }
973    }
974    return NULL;
975}

983/* Return a random entry from the hash table. Useful to
984 * implement randomized algorithms */
985dictEntry *dictGetRandomKey(dict *d)
986{
987    dictEntry *he, *orighe;
988    unsigned long h;
989    int listlen, listele;
990
991    if (dictSize(d) == 0) return NULL;
992    if (dictIsRehashing(d)) _dictRehashStep(d);
993    if (dictIsRehashing(d)) {
994        unsigned long s0 = DICTHT_SIZE(d->ht_size_exp[0]);
995        do {
996            /* We are sure there are no elements in indexes from 0
997             * to rehashidx-1 */
998            h = d->rehashidx + (randomULong() % (dictSlots(d) - d->rehashidx));
999            he = (h >= s0) ? d->ht_table[1][h - s0] : d->ht_table[0][h];
1000        } while(he == NULL);
1001    } else {
1002        unsigned long m = DICTHT_SIZE_MASK(d->ht_size_exp[0]);
1003        do {
1004            h = randomULong() & m;
1005            he = d->ht_table[0][h];
1006        } while(he == NULL);
1007    }
1008
1009    /* Now we found a non empty bucket, but it is a linked
1010     * list and we need to get a random element from the list.
1011     * The only sane way to do so is counting the elements and
1012     * select a random index. */
1013    listlen = 0;
1014    orighe = he;
1015    while(he) {
1016        he = dictGetNext(he);
1017        listlen++;
1018    }
1019    listele = random() % listlen;
1020    he = orighe;
1021    while(listele--) he = dictGetNext(he);
1022    return he;
1023}

1025/* This function samples the dictionary to return a few keys from random
1026 * locations.
1027 *
1028 * It does not guarantee to return all the keys specified in 'count', nor
1029 * it does guarantee to return non-duplicated elements, however it will make
1030 * some effort to do both things.
1031 *
1032 * Returned pointers to hash table entries are stored into 'des' that
1033 * points to an array of dictEntry pointers. The array must have room for
1034 * at least 'count' elements, that is the argument we pass to the function
1035 * to tell how many random elements we need.
1036 *
1037 * The function returns the number of items stored into 'des', that may
1038 * be less than 'count' if the hash table has less than 'count' elements
1039 * inside, or if not enough elements were found in a reasonable amount of
1040 * steps.
1041 *
1042 * Note that this function is not suitable when you need a good distribution
1043 * of the returned items, but only when you need to "sample" a given number
1044 * of continuous elements to run some kind of algorithm or to produce
1045 * statistics. However the function is much faster than dictGetRandomKey()
1046 * at producing N elements. */
1047unsigned int dictGetSomeKeys(dict *d, dictEntry **des, unsigned int count) {
1048    unsigned long j; /* internal hash table id, 0 or 1. */
1049    unsigned long tables; /* 1 or 2 tables? */
1050    unsigned long stored = 0, maxsizemask;
1051    unsigned long maxsteps;
1052
1053    if (dictSize(d) < count) count = dictSize(d);
1054    maxsteps = count*10;
1055
1056    /* Try to do a rehashing work proportional to 'count'. */
1057    for (j = 0; j < count; j++) {
1058        if (dictIsRehashing(d))
1059            _dictRehashStep(d);
1060        else
1061            break;
1062    }
1063
1064    tables = dictIsRehashing(d) ? 2 : 1;
1065    maxsizemask = DICTHT_SIZE_MASK(d->ht_size_exp[0]);
1066    if (tables > 1 && maxsizemask < DICTHT_SIZE_MASK(d->ht_size_exp[1]))
1067        maxsizemask = DICTHT_SIZE_MASK(d->ht_size_exp[1]);
1068
1069    /* Pick a random point inside the larger table. */
1070    unsigned long i = randomULong() & maxsizemask;
1071    unsigned long emptylen = 0; /* Continuous empty entries so far. */
1072    while(stored < count && maxsteps--) {
1073        for (j = 0; j < tables; j++) {
1074            /* Invariant of the dict.c rehashing: up to the indexes already
1075             * visited in ht[0] during the rehashing, there are no populated
1076             * buckets, so we can skip ht[0] for indexes between 0 and idx-1. */
1077            if (tables == 2 && j == 0 && i < (unsigned long) d->rehashidx) {
1078                /* Moreover, if we are currently out of range in the second
1079                 * table, there will be no elements in both tables up to
1080                 * the current rehashing index, so we jump if possible.
1081                 * (this happens when going from big to small table). */
1082                if (i >= DICTHT_SIZE(d->ht_size_exp[1]))
1083                    i = d->rehashidx;
1084                else
1085                    continue;
1086            }
1087            if (i >= DICTHT_SIZE(d->ht_size_exp[j])) continue; /* Out of range for this table. */
1088            dictEntry *he = d->ht_table[j][i];
1089
1090            /* Count contiguous empty buckets, and jump to other
1091             * locations if they reach 'count' (with a minimum of 5). */
1092            if (he == NULL) {
1093                emptylen++;
1094                if (emptylen >= 5 && emptylen > count) {
1095                    i = randomULong() & maxsizemask;
1096                    emptylen = 0;
1097                }
1098            } else {
1099                emptylen = 0;
1100                while (he) {
1101                    /* Collect all the elements of the buckets found non empty while iterating.
1102                     * To avoid the issue of being unable to sample the end of a long chain,
1103                     * we utilize the Reservoir Sampling algorithm to optimize the sampling process.
1104                     * This means that even when the maximum number of samples has been reached,
1105                     * we continue sampling until we reach the end of the chain.
1106                     * See https://en.wikipedia.org/wiki/Reservoir_sampling. */
1107                    if (stored < count) {
1108                        des[stored] = he;
1109                    } else {
1110                        unsigned long r = randomULong() % (stored + 1);
1111                        if (r < count) des[r] = he;
1112                    }
1113
1114                    he = dictGetNext(he);
1115                    stored++;
1116                }
1117                if (stored >= count) goto end;
1118            }
1119        }
1120        i = (i+1) & maxsizemask;
1121    }
1122
1123end:
1124    return stored > count ? count : stored;
1125}

1128/* Reallocate the dictEntry, key and value allocations in a bucket using the
1129 * provided allocation functions in order to defrag them. */
1130static void dictDefragBucket(dict *d, dictEntry **bucketref, dictDefragFunctions *defragfns) {
1131    dictDefragAllocFunction *defragalloc = defragfns->defragAlloc;
1132    dictDefragAllocFunction *defragkey = defragfns->defragKey;
1133    dictDefragAllocFunction *defragval = defragfns->defragVal;
1134    while (bucketref && *bucketref) {
1135        dictEntry *de = *bucketref, *newde = NULL;
1136        void *newkey = defragkey ? defragkey(dictGetKey(de)) : NULL;
1137        void *newval = defragval ? defragval(dictGetVal(de)) : NULL;
1138        if (entryIsKey(de)) {
1139            if (newkey) *bucketref = newkey;
1140            assert(entryIsKey(*bucketref));
1141        } else if (entryIsNoValue(de)) {
1142            dictEntryNoValue *entry = decodeEntryNoValue(de), *newentry;
1143            if ((newentry = defragalloc(entry))) {
1144                newde = encodeMaskedPtr(newentry, ENTRY_PTR_NO_VALUE);
1145                entry = newentry;
1146            }
1147            if (newkey) entry->key = newkey;
1148        } else {
1149            assert(entryIsNormal(de));
1150            newde = defragalloc(de);
1151            if (newde) de = newde;
1152            if (newkey) de->key = newkey;
1153            if (newval) de->v.val = newval;
1154        }
1155        if (newde) {
1156            *bucketref = newde;
1157            if (d->type->afterReplaceEntry)
1158                d->type->afterReplaceEntry(d, newde);
1159        }
1160        bucketref = dictGetNextRef(*bucketref);
1161    }
1162}

1164/* This is like dictGetRandomKey() from the POV of the API, but will do more
1165 * work to ensure a better distribution of the returned element.
1166 *
1167 * This function improves the distribution because the dictGetRandomKey()
1168 * problem is that it selects a random bucket, then it selects a random
1169 * element from the chain in the bucket. However elements being in different
1170 * chain lengths will have different probabilities of being reported. With
1171 * this function instead what we do is to consider a "linear" range of the table
1172 * that may be constituted of N buckets with chains of different lengths
1173 * appearing one after the other. Then we report a random element in the range.
1174 * In this way we smooth away the problem of different chain lengths. */
1175#define GETFAIR_NUM_ENTRIES 15
1176dictEntry *dictGetFairRandomKey(dict *d) {
1177    dictEntry *entries[GETFAIR_NUM_ENTRIES];
1178    unsigned int count = dictGetSomeKeys(d,entries,GETFAIR_NUM_ENTRIES);
1179    /* Note that dictGetSomeKeys() may return zero elements in an unlucky
1180     * run() even if there are actually elements inside the hash table. So
1181     * when we get zero, we call the true dictGetRandomKey() that will always
1182     * yield the element if the hash table has at least one. */
1183    if (count == 0) return dictGetRandomKey(d);
1184    unsigned int idx = rand() % count;
1185    return entries[idx];
1186}

1188/* Function to reverse bits. Algorithm from:
1189 * http://graphics.stanford.edu/~seander/bithacks.html#ReverseParallel */
1190static unsigned long rev(unsigned long v) {
1191    unsigned long s = CHAR_BIT * sizeof(v); // bit size; must be power of 2
1192    unsigned long mask = ~0UL;
1193    while ((s >>= 1) > 0) {
1194        mask ^= (mask << s);
1195        v = ((v >> s) & mask) | ((v << s) & ~mask);
1196    }
1197    return v;
1198}

1200/* dictScan() is used to iterate over the elements of a dictionary.
1201 *
1202 * Iterating works the following way:
1203 *
1204 * 1) Initially you call the function using a cursor (v) value of 0.
1205 * 2) The function performs one step of the iteration, and returns the
1206 *    new cursor value you must use in the next call.
1207 * 3) When the returned cursor is 0, the iteration is complete.
1208 *
1209 * The function guarantees all elements present in the
1210 * dictionary get returned between the start and end of the iteration.
1211 * However it is possible some elements get returned multiple times.
1212 *
1213 * For every element returned, the callback argument 'fn' is
1214 * called with 'privdata' as first argument and the dictionary entry
1215 * 'de' as second argument.
1216 *
1217 * HOW IT WORKS.
1218 *
1219 * The iteration algorithm was designed by Pieter Noordhuis.
1220 * The main idea is to increment a cursor starting from the higher order
1221 * bits. That is, instead of incrementing the cursor normally, the bits
1222 * of the cursor are reversed, then the cursor is incremented, and finally
1223 * the bits are reversed again.
1224 *
1225 * This strategy is needed because the hash table may be resized between
1226 * iteration calls.
1227 *
1228 * dict.c hash tables are always power of two in size, and they
1229 * use chaining, so the position of an element in a given table is given
1230 * by computing the bitwise AND between Hash(key) and SIZE-1
1231 * (where SIZE-1 is always the mask that is equivalent to taking the rest
1232 *  of the division between the Hash of the key and SIZE).
1233 *
1234 * For example if the current hash table size is 16, the mask is
1235 * (in binary) 1111. The position of a key in the hash table will always be
1236 * the last four bits of the hash output, and so forth.
1237 *
1238 * WHAT HAPPENS IF THE TABLE CHANGES IN SIZE?
1239 *
1240 * If the hash table grows, elements can go anywhere in one multiple of
1241 * the old bucket: for example let's say we already iterated with
1242 * a 4 bit cursor 1100 (the mask is 1111 because hash table size = 16).
1243 *
1244 * If the hash table will be resized to 64 elements, then the new mask will
1245 * be 111111. The new buckets you obtain by substituting in ??1100
1246 * with either 0 or 1 can be targeted only by keys we already visited
1247 * when scanning the bucket 1100 in the smaller hash table.
1248 *
1249 * By iterating the higher bits first, because of the inverted counter, the
1250 * cursor does not need to restart if the table size gets bigger. It will
1251 * continue iterating using cursors without '1100' at the end, and also
1252 * without any other combination of the final 4 bits already explored.
1253 *
1254 * Similarly when the table size shrinks over time, for example going from
1255 * 16 to 8, if a combination of the lower three bits (the mask for size 8
1256 * is 111) were already completely explored, it would not be visited again
1257 * because we are sure we tried, for example, both 0111 and 1111 (all the
1258 * variations of the higher bit) so we don't need to test it again.
1259 *
1260 * WAIT... YOU HAVE *TWO* TABLES DURING REHASHING!
1261 *
1262 * Yes, this is true, but we always iterate the smaller table first, then
1263 * we test all the expansions of the current cursor into the larger
1264 * table. For example if the current cursor is 101 and we also have a
1265 * larger table of size 16, we also test (0)101 and (1)101 inside the larger
1266 * table. This reduces the problem back to having only one table, where
1267 * the larger one, if it exists, is just an expansion of the smaller one.
1268 *
1269 * LIMITATIONS
1270 *
1271 * This iterator is completely stateless, and this is a huge advantage,
1272 * including no additional memory used.
1273 *
1274 * The disadvantages resulting from this design are:
1275 *
1276 * 1) It is possible we return elements more than once. However this is usually
1277 *    easy to deal with in the application level.
1278 * 2) The iterator must return multiple elements per call, as it needs to always
1279 *    return all the keys chained in a given bucket, and all the expansions, so
1280 *    we are sure we don't miss keys moving during rehashing.
1281 * 3) The reverse cursor is somewhat hard to understand at first, but this
1282 *    comment is supposed to help.
1283 */
1284unsigned long dictScan(dict *d,
1285                       unsigned long v,
1286                       dictScanFunction *fn,
1287                       void *privdata)
1288{
1289    return dictScanDefrag(d, v, fn, NULL, privdata);
1290}

1292/* Like dictScan, but additionally reallocates the memory used by the dict
1293 * entries using the provided allocation function. This feature was added for
1294 * the active defrag feature.
1295 *
1296 * The 'defragfns' callbacks are called with a pointer to memory that callback
1297 * can reallocate. The callbacks should return a new memory address or NULL,
1298 * where NULL means that no reallocation happened and the old memory is still
1299 * valid. */
1300unsigned long dictScanDefrag(dict *d,
1301                             unsigned long v,
1302                             dictScanFunction *fn,
1303                             dictDefragFunctions *defragfns,
1304                             void *privdata)
1305{
1306    int htidx0, htidx1;
1307    const dictEntry *de, *next;
1308    unsigned long m0, m1;
1309
1310    if (dictSize(d) == 0) return 0;
1311
1312    /* This is needed in case the scan callback tries to do dictFind or alike. */
1313    dictPauseRehashing(d);
1314
1315    if (!dictIsRehashing(d)) {
1316        htidx0 = 0;
1317        m0 = DICTHT_SIZE_MASK(d->ht_size_exp[htidx0]);
1318
1319        /* Emit entries at cursor */
1320        if (defragfns) {
1321            dictDefragBucket(d, &d->ht_table[htidx0][v & m0], defragfns);
1322        }
1323        de = d->ht_table[htidx0][v & m0];
1324        while (de) {
1325            next = dictGetNext(de);
1326            fn(privdata, de);
1327            de = next;
1328        }
1329
1330        /* Set unmasked bits so incrementing the reversed cursor
1331         * operates on the masked bits */
1332        v |= ~m0;
1333
1334        /* Increment the reverse cursor */
1335        v = rev(v);
1336        v++;
1337        v = rev(v);
1338
1339    } else {
1340        htidx0 = 0;
1341        htidx1 = 1;
1342
1343        /* Make sure t0 is the smaller and t1 is the bigger table */
1344        if (DICTHT_SIZE(d->ht_size_exp[htidx0]) > DICTHT_SIZE(d->ht_size_exp[htidx1])) {
1345            htidx0 = 1;
1346            htidx1 = 0;
1347        }
1348
1349        m0 = DICTHT_SIZE_MASK(d->ht_size_exp[htidx0]);
1350        m1 = DICTHT_SIZE_MASK(d->ht_size_exp[htidx1]);
1351
1352        /* Emit entries at cursor */
1353        if (defragfns) {
1354            dictDefragBucket(d, &d->ht_table[htidx0][v & m0], defragfns);
1355        }
1356        de = d->ht_table[htidx0][v & m0];
1357        while (de) {
1358            next = dictGetNext(de);
1359            fn(privdata, de);
1360            de = next;
1361        }
1362
1363        /* Iterate over indices in larger table that are the expansion
1364         * of the index pointed to by the cursor in the smaller table */
1365        do {
1366            /* Emit entries at cursor */
1367            if (defragfns) {
1368                dictDefragBucket(d, &d->ht_table[htidx1][v & m1], defragfns);
1369            }
1370            de = d->ht_table[htidx1][v & m1];
1371            while (de) {
1372                next = dictGetNext(de);
1373                fn(privdata, de);
1374                de = next;
1375            }
1376
1377            /* Increment the reverse cursor not covered by the smaller mask.*/
1378            v |= ~m1;
1379            v = rev(v);
1380            v++;
1381            v = rev(v);
1382
1383            /* Continue while bits covered by mask difference is non-zero */
1384        } while (v & (m0 ^ m1));
1385    }
1386
1387    dictResumeRehashing(d);
1388
1389    return v;
1390}

1394/* Because we may need to allocate huge memory chunk at once when dict
1395 * expands, we will check this allocation is allowed or not if the dict
1396 * type has expandAllowed member function. */
1397static int dictTypeExpandAllowed(dict *d) {
1398    if (d->type->expandAllowed == NULL) return 1;
1399    return d->type->expandAllowed(
1400                    DICTHT_SIZE(_dictNextExp(d->ht_used[0] + 1)) * sizeof(dictEntry*),
1401                    (double)d->ht_used[0] / DICTHT_SIZE(d->ht_size_exp[0]));
1402}

1404/* Expand the hash table if needed */
1405static int _dictExpandIfNeeded(dict *d)
1406{
1407    /* Incremental rehashing already in progress. Return. */
1408    if (dictIsRehashing(d)) return DICT_OK;
1409
1410    /* If the hash table is empty expand it to the initial size. */
1411    if (DICTHT_SIZE(d->ht_size_exp[0]) == 0) return dictExpand(d, DICT_HT_INITIAL_SIZE);
1412
1413    /* If we reached the 1:1 ratio, and we are allowed to resize the hash
1414     * table (global setting) or we should avoid it but the ratio between
1415     * elements/buckets is over the "safe" threshold, we resize doubling
1416     * the number of buckets. */
1417    if ((dict_can_resize == DICT_RESIZE_ENABLE &&
1418         d->ht_used[0] >= DICTHT_SIZE(d->ht_size_exp[0])) ||
1419        (dict_can_resize != DICT_RESIZE_FORBID &&
1420         d->ht_used[0] / DICTHT_SIZE(d->ht_size_exp[0]) > dict_force_resize_ratio))
1421    {
1422        if (!dictTypeExpandAllowed(d))
1423            return DICT_OK;
1424        return dictExpand(d, d->ht_used[0] + 1);
1425    }
1426    return DICT_OK;
1427}

1429/* Our hash table capability is a power of two */
1430static signed char _dictNextExp(unsigned long size)
1431{
1432    if (size <= DICT_HT_INITIAL_SIZE) return DICT_HT_INITIAL_EXP;
1433    if (size >= LONG_MAX) return (8*sizeof(long)-1);
1434
1435    return 8*sizeof(long) - __builtin_clzl(size-1);
1436}

1438/* Finds and returns the position within the dict where the provided key should
1439 * be inserted using dictInsertAtPosition if the key does not already exist in
1440 * the dict. If the key exists in the dict, NULL is returned and the optional
1441 * 'existing' entry pointer is populated, if provided. */
1442void *dictFindPositionForInsert(dict *d, const void *key, dictEntry **existing) {
1443    unsigned long idx, table;
1444    dictEntry *he;
1445    uint64_t hash = dictHashKey(d, key);
1446    if (existing) *existing = NULL;
1447    if (dictIsRehashing(d)) _dictRehashStep(d);
1448
1449    /* Expand the hash table if needed */
1450    if (_dictExpandIfNeeded(d) == DICT_ERR)
1451        return NULL;
1452    for (table = 0; table <= 1; table++) {
1453        idx = hash & DICTHT_SIZE_MASK(d->ht_size_exp[table]);
1454        /* Search if this slot does not already contain the given key */
1455        he = d->ht_table[table][idx];
1456        while(he) {
1457            void *he_key = dictGetKey(he);
1458            if (key == he_key || dictCompareKeys(d, key, he_key)) {
1459                if (existing) *existing = he;
1460                return NULL;
1461            }
1462            he = dictGetNext(he);
1463        }
1464        if (!dictIsRehashing(d)) break;
1465    }
1466
1467    /* If we are in the process of rehashing the hash table, the bucket is
1468     * always returned in the context of the second (new) hash table. */
1469    dictEntry **bucket = &d->ht_table[dictIsRehashing(d) ? 1 : 0][idx];
1470    return bucket;
1471}

1488/* Finds the dictEntry using pointer and pre-calculated hash.
1489 * oldkey is a dead pointer and should not be accessed.
1490 * the hash value should be provided using dictGetHash.
1491 * no string / key comparison is performed.
1492 * return value is a pointer to the dictEntry if found, or NULL if not found. */
1493dictEntry *dictFindEntryByPtrAndHash(dict *d, const void *oldptr, uint64_t hash) {
1494    dictEntry *he;
1495    unsigned long idx, table;
1496
1497    if (dictSize(d) == 0) return NULL; /* dict is empty */
1498    for (table = 0; table <= 1; table++) {
1499        idx = hash & DICTHT_SIZE_MASK(d->ht_size_exp[table]);
1500        he = d->ht_table[table][idx];
1501        while(he) {
1502            if (oldptr == dictGetKey(he))
1503                return he;
1504            he = dictGetNext(he);
1505        }
1506        if (!dictIsRehashing(d)) return NULL;
1507    }
1508    return NULL;
1509}

1513#define DICT_STATS_VECTLEN 50
1514size_t _dictGetStatsHt(char *buf, size_t bufsize, dict *d, int htidx, int full) {
1515    unsigned long i, slots = 0, chainlen, maxchainlen = 0;
1516    unsigned long totchainlen = 0;
1517    unsigned long clvector[DICT_STATS_VECTLEN];
1518    size_t l = 0;
1519
1520    if (d->ht_used[htidx] == 0) {
1521        return snprintf(buf,bufsize,
1522            "Hash table %d stats (%s):\n"
1523            "No stats available for empty dictionaries\n",
1524            htidx, (htidx == 0) ? "main hash table" : "rehashing target");
1525    }
1526
1527    if (!full) {
1528        l += snprintf(buf+l,bufsize-l,
1529            "Hash table %d stats (%s):\n"
1530            " table size: %lu\n"
1531            " number of elements: %lu\n",
1532            htidx, (htidx == 0) ? "main hash table" : "rehashing target",
1533            DICTHT_SIZE(d->ht_size_exp[htidx]), d->ht_used[htidx]);
1534
1535        /* Make sure there is a NULL term at the end. */
1536        buf[bufsize-1] = '\0';
1537        /* Unlike snprintf(), return the number of characters actually written. */
1538        return strlen(buf);
1539    }
1540
1541    /* Compute stats. */
1542    for (i = 0; i < DICT_STATS_VECTLEN; i++) clvector[i] = 0;
1543    for (i = 0; i < DICTHT_SIZE(d->ht_size_exp[htidx]); i++) {
1544        dictEntry *he;
1545
1546        if (d->ht_table[htidx][i] == NULL) {
1547            clvector[0]++;
1548            continue;
1549        }
1550        slots++;
1551        /* For each hash entry on this slot... */
1552        chainlen = 0;
1553        he = d->ht_table[htidx][i];
1554        while(he) {
1555            chainlen++;
1556            he = dictGetNext(he);
1557        }
1558        clvector[(chainlen < DICT_STATS_VECTLEN) ? chainlen : (DICT_STATS_VECTLEN-1)]++;
1559        if (chainlen > maxchainlen) maxchainlen = chainlen;
1560        totchainlen += chainlen;
1561    }
1562
1563    /* Generate human readable stats. */
1564    l += snprintf(buf+l,bufsize-l,
1565        "Hash table %d stats (%s):\n"
1566        " table size: %lu\n"
1567        " number of elements: %lu\n"
1568        " different slots: %lu\n"
1569        " max chain length: %lu\n"
1570        " avg chain length (counted): %.02f\n"
1571        " avg chain length (computed): %.02f\n"
1572        " Chain length distribution:\n",
1573        htidx, (htidx == 0) ? "main hash table" : "rehashing target",
1574        DICTHT_SIZE(d->ht_size_exp[htidx]), d->ht_used[htidx], slots, maxchainlen,
1575        (float)totchainlen/slots, (float)d->ht_used[htidx]/slots);
1576
1577    for (i = 0; i < DICT_STATS_VECTLEN-1; i++) {
1578        if (clvector[i] == 0) continue;
1579        if (l >= bufsize) break;
1580        l += snprintf(buf+l,bufsize-l,
1581            "   %ld: %ld (%.02f%%)\n",
1582            i, clvector[i], ((float)clvector[i]/DICTHT_SIZE(d->ht_size_exp[htidx]))*100);
1583    }
1584
1585    /* Make sure there is a NULL term at the end. */
1586    buf[bufsize-1] = '\0';
1587    /* Unlike snprintf(), return the number of characters actually written. */
1588    return strlen(buf);
1589}
