67void rdbReportError(int corruption_error, int linenum, char *reason, ...) {
68    va_list ap;
69    char msg[1024];
70    int len;
71
72    len = snprintf(msg,sizeof(msg),
73        "Internal error in RDB reading offset %llu, function at rdb.c:%d -> ",
74        (unsigned long long)server.loading_loaded_bytes, linenum);
75    va_start(ap,reason);
76    vsnprintf(msg+len,sizeof(msg)-len,reason,ap);
77    va_end(ap);
78
79    if (isRestoreContext()) {
80        /* If we're in the context of a RESTORE command, just propagate the error. */
81        /* log in VERBOSE, and return (don't exit). */
82        serverLog(LL_VERBOSE, "%s", msg);
83        return;
84    } else if (rdbCheckMode) {
85        /* If we're inside the rdb checker, let it handle the error. */
86        rdbCheckError("%s",msg);
87    } else if (rdbFileBeingLoaded) {
88        /* If we're loading an rdb file form disk, run rdb check (and exit) */
89        serverLog(LL_WARNING, "%s", msg);
90        char *argv[2] = {"",rdbFileBeingLoaded};
91        if (anetIsFifo(argv[1])) {
92            /* Cannot check RDB FIFO because we cannot reopen the FIFO and check already streamed data. */
93            rdbCheckError("Cannot check RDB that is a FIFO: %s", argv[1]);
94            return;
95        }
96        redis_check_rdb_main(2,argv,NULL);
97    } else if (corruption_error) {
98        /* In diskless loading, in case of corrupt file, log and exit. */
99        serverLog(LL_WARNING, "%s. Failure loading rdb format", msg);
100    } else {
101        /* In diskless loading, in case of a short read (not a corrupt
102         * file), log and proceed (don't exit). */
103        serverLog(LL_WARNING, "%s. Failure loading rdb format from socket, assuming connection error, resuming operation.", msg);
104        return;
105    }
106    serverLog(LL_WARNING, "Terminating server after rdb file reading failure.");
107    exit(1);
108}

146/* This function loads a time from the RDB file. It gets the version of the
147 * RDB because, unfortunately, before Redis 5 (RDB version 9), the function
148 * failed to convert data to/from little endian, so RDB files with keys having
149 * expires could not be shared between big endian and little endian systems
150 * (because the expire time will be totally wrong). The fix for this is just
151 * to call memrev64ifbe(), however if we fix this for all the RDB versions,
152 * this call will introduce an incompatibility for big endian systems:
153 * after upgrading to Redis version 5 they will no longer be able to load their
154 * own old RDB files. Because of that, we instead fix the function only for new
155 * RDB versions, and load older RDB versions as we used to do in the past,
156 * allowing big endian systems to load their own old RDB files.
157 *
158 * On I/O error the function returns LLONG_MAX, however if this is also a
159 * valid stored value, the caller should use rioGetReadError() to check for
160 * errors after calling this function. */
161long long rdbLoadMillisecondTime(rio *rdb, int rdbver) {
162    int64_t t64;
163    if (rioRead(rdb,&t64,8) == 0) return LLONG_MAX;
164    if (rdbver >= 9) /* Check the top comment of this function. */
165        memrev64ifbe(&t64); /* Convert in big endian if the system is BE. */
166    return (long long)t64;
167}

169/* Saves an encoded length. The first two bits in the first byte are used to
170 * hold the encoding type. See the RDB_* definitions for more information
171 * on the types of encoding. */
172int rdbSaveLen(rio *rdb, uint64_t len) {
173    unsigned char buf[2];
174    size_t nwritten;
175
176    if (len < (1<<6)) {
177        /* Save a 6 bit len */
178        buf[0] = (len&0xFF)|(RDB_6BITLEN<<6);
179        if (rdbWriteRaw(rdb,buf,1) == -1) return -1;
180        nwritten = 1;
181    } else if (len < (1<<14)) {
182        /* Save a 14 bit len */
183        buf[0] = ((len>>8)&0xFF)|(RDB_14BITLEN<<6);
184        buf[1] = len&0xFF;
185        if (rdbWriteRaw(rdb,buf,2) == -1) return -1;
186        nwritten = 2;
187    } else if (len <= UINT32_MAX) {
188        /* Save a 32 bit len */
189        buf[0] = RDB_32BITLEN;
190        if (rdbWriteRaw(rdb,buf,1) == -1) return -1;
191        uint32_t len32 = htonl(len);
192        if (rdbWriteRaw(rdb,&len32,4) == -1) return -1;
193        nwritten = 1+4;
194    } else {
195        /* Save a 64 bit len */
196        buf[0] = RDB_64BITLEN;
197        if (rdbWriteRaw(rdb,buf,1) == -1) return -1;
198        len = htonu64(len);
199        if (rdbWriteRaw(rdb,&len,8) == -1) return -1;
200        nwritten = 1+8;
201    }
202    return nwritten;
203}

206/* Load an encoded length. If the loaded length is a normal length as stored
207 * with rdbSaveLen(), the read length is set to '*lenptr'. If instead the
208 * loaded length describes a special encoding that follows, then '*isencoded'
209 * is set to 1 and the encoding format is stored at '*lenptr'.
210 *
211 * See the RDB_ENC_* definitions in rdb.h for more information on special
212 * encodings.
213 *
214 * The function returns -1 on error, 0 on success. */
215int rdbLoadLenByRef(rio *rdb, int *isencoded, uint64_t *lenptr) {
216    unsigned char buf[2];
217    int type;
218
219    if (isencoded) *isencoded = 0;
220    if (rioRead(rdb,buf,1) == 0) return -1;
221    type = (buf[0]&0xC0)>>6;
222    if (type == RDB_ENCVAL) {
223        /* Read a 6 bit encoding type. */
224        if (isencoded) *isencoded = 1;
225        *lenptr = buf[0]&0x3F;
226    } else if (type == RDB_6BITLEN) {
227        /* Read a 6 bit len. */
228        *lenptr = buf[0]&0x3F;
229    } else if (type == RDB_14BITLEN) {
230        /* Read a 14 bit len. */
231        if (rioRead(rdb,buf+1,1) == 0) return -1;
232        *lenptr = ((buf[0]&0x3F)<<8)|buf[1];
233    } else if (buf[0] == RDB_32BITLEN) {
234        /* Read a 32 bit len. */
235        uint32_t len;
236        if (rioRead(rdb,&len,4) == 0) return -1;
237        *lenptr = ntohl(len);
238    } else if (buf[0] == RDB_64BITLEN) {
239        /* Read a 64 bit len. */
240        uint64_t len;
241        if (rioRead(rdb,&len,8) == 0) return -1;
242        *lenptr = ntohu64(len);
243    } else {
244        rdbReportCorruptRDB(
245            "Unknown length encoding %d in rdbLoadLen()",type);
246        return -1; /* Never reached. */
247    }
248    return 0;
249}

262/* Encodes the "value" argument as integer when it fits in the supported ranges
263 * for encoded types. If the function successfully encodes the integer, the
264 * representation is stored in the buffer pointer to by "enc" and the string
265 * length is returned. Otherwise 0 is returned. */
266int rdbEncodeInteger(long long value, unsigned char *enc) {
267    if (value >= -(1<<7) && value <= (1<<7)-1) {
268        enc[0] = (RDB_ENCVAL<<6)|RDB_ENC_INT8;
269        enc[1] = value&0xFF;
270        return 2;
271    } else if (value >= -(1<<15) && value <= (1<<15)-1) {
272        enc[0] = (RDB_ENCVAL<<6)|RDB_ENC_INT16;
273        enc[1] = value&0xFF;
274        enc[2] = (value>>8)&0xFF;
275        return 3;
276    } else if (value >= -((long long)1<<31) && value <= ((long long)1<<31)-1) {
277        enc[0] = (RDB_ENCVAL<<6)|RDB_ENC_INT32;
278        enc[1] = value&0xFF;
279        enc[2] = (value>>8)&0xFF;
280        enc[3] = (value>>16)&0xFF;
281        enc[4] = (value>>24)&0xFF;
282        return 5;
283    } else {
284        return 0;
285    }
286}

288/* Loads an integer-encoded object with the specified encoding type "enctype".
289 * The returned value changes according to the flags, see
290 * rdbGenericLoadStringObject() for more info. */
291void *rdbLoadIntegerObject(rio *rdb, int enctype, int flags, size_t *lenptr) {
292    int plain = flags & RDB_LOAD_PLAIN;
293    int sds = flags & RDB_LOAD_SDS;
294    int encode = flags & RDB_LOAD_ENC;
295    unsigned char enc[4];
296    long long val;
297
298    if (enctype == RDB_ENC_INT8) {
299        if (rioRead(rdb,enc,1) == 0) return NULL;
300        val = (signed char)enc[0];
301    } else if (enctype == RDB_ENC_INT16) {
302        uint16_t v;
303        if (rioRead(rdb,enc,2) == 0) return NULL;
304        v = ((uint32_t)enc[0])|
305            ((uint32_t)enc[1]<<8);
306        val = (int16_t)v;
307    } else if (enctype == RDB_ENC_INT32) {
308        uint32_t v;
309        if (rioRead(rdb,enc,4) == 0) return NULL;
310        v = ((uint32_t)enc[0])|
311            ((uint32_t)enc[1]<<8)|
312            ((uint32_t)enc[2]<<16)|
313            ((uint32_t)enc[3]<<24);
314        val = (int32_t)v;
315    } else {
316        rdbReportCorruptRDB("Unknown RDB integer encoding type %d",enctype);
317        return NULL; /* Never reached. */
318    }
319    if (plain || sds) {
320        char buf[LONG_STR_SIZE], *p;
321        int len = ll2string(buf,sizeof(buf),val);
322        if (lenptr) *lenptr = len;
323        p = plain ? zmalloc(len) : sdsnewlen(SDS_NOINIT,len);
324        memcpy(p,buf,len);
325        return p;
326    } else if (encode) {
327        return createStringObjectFromLongLongForValue(val);
328    } else {
329        return createStringObjectFromLongLongWithSds(val);
330    }
331}

345ssize_t rdbSaveLzfBlob(rio *rdb, void *data, size_t compress_len,
346                       size_t original_len) {
347    unsigned char byte;
348    ssize_t n, nwritten = 0;
349
350    /* Data compressed! Let's save it on disk */
351    byte = (RDB_ENCVAL<<6)|RDB_ENC_LZF;
352    if ((n = rdbWriteRaw(rdb,&byte,1)) == -1) goto writeerr;
353    nwritten += n;
354
355    if ((n = rdbSaveLen(rdb,compress_len)) == -1) goto writeerr;
356    nwritten += n;
357
358    if ((n = rdbSaveLen(rdb,original_len)) == -1) goto writeerr;
359    nwritten += n;
360
361    if ((n = rdbWriteRaw(rdb,data,compress_len)) == -1) goto writeerr;
362    nwritten += n;
363
364    return nwritten;
365
366writeerr:
367    return -1;
368}

370ssize_t rdbSaveLzfStringObject(rio *rdb, unsigned char *s, size_t len) {
371    size_t comprlen, outlen;
372    void *out;
373
374    /* We require at least four bytes compression for this to be worth it */
375    if (len <= 4) return 0;
376    outlen = len-4;
377    if ((out = zmalloc(outlen+1)) == NULL) return 0;
378    comprlen = lzf_compress(s, len, out, outlen);
379    if (comprlen == 0) {
380        zfree(out);
381        return 0;
382    }
383    ssize_t nwritten = rdbSaveLzfBlob(rdb, out, comprlen, len);
384    zfree(out);
385    return nwritten;
386}

388/* Load an LZF compressed string in RDB format. The returned value
389 * changes according to 'flags'. For more info check the
390 * rdbGenericLoadStringObject() function. */
391void *rdbLoadLzfStringObject(rio *rdb, int flags, size_t *lenptr) {
392    int plain = flags & RDB_LOAD_PLAIN;
393    int sds = flags & RDB_LOAD_SDS;
394    uint64_t len, clen;
395    unsigned char *c = NULL;
396    char *val = NULL;
397
398    if ((clen = rdbLoadLen(rdb,NULL)) == RDB_LENERR) return NULL;
399    if ((len = rdbLoadLen(rdb,NULL)) == RDB_LENERR) return NULL;
400    if ((c = ztrymalloc(clen)) == NULL) {
401        serverLog(isRestoreContext()? LL_VERBOSE: LL_WARNING, "rdbLoadLzfStringObject failed allocating %llu bytes", (unsigned long long)clen);
402        goto err;
403    }
404
405    /* Allocate our target according to the uncompressed size. */
406    if (plain) {
407        val = ztrymalloc(len);
408    } else {
409        val = sdstrynewlen(SDS_NOINIT,len);
410    }
411    if (!val) {
412        serverLog(isRestoreContext()? LL_VERBOSE: LL_WARNING, "rdbLoadLzfStringObject failed allocating %llu bytes", (unsigned long long)len);
413        goto err;
414    }
415
416    if (lenptr) *lenptr = len;
417
418    /* Load the compressed representation and uncompress it to target. */
419    if (rioRead(rdb,c,clen) == 0) goto err;
420    if (lzf_decompress(c,clen,val,len) != len) {
421        rdbReportCorruptRDB("Invalid LZF compressed string");
422        goto err;
423    }
424    zfree(c);
425
426    if (plain || sds) {
427        return val;
428    } else {
429        return createObject(OBJ_STRING,val);
430    }
431err:
432    zfree(c);
433    if (plain)
434        zfree(val);
435    else
436        sdsfree(val);
437    return NULL;
438}

440/* Save a string object as [len][data] on disk. If the object is a string
441 * representation of an integer value we try to save it in a special form */
442ssize_t rdbSaveRawString(rio *rdb, unsigned char *s, size_t len) {
443    int enclen;
444    ssize_t n, nwritten = 0;
445
446    /* Try integer encoding */
447    if (len <= 11) {
448        unsigned char buf[5];
449        if ((enclen = rdbTryIntegerEncoding((char*)s,len,buf)) > 0) {
450            if (rdbWriteRaw(rdb,buf,enclen) == -1) return -1;
451            return enclen;
452        }
453    }
454
455    /* Try LZF compression - under 20 bytes it's unable to compress even
456     * aaaaaaaaaaaaaaaaaa so skip it */
457    if (server.rdb_compression && len > 20) {
458        n = rdbSaveLzfStringObject(rdb,s,len);
459        if (n == -1) return -1;
460        if (n > 0) return n;
461        /* Return value of 0 means data can't be compressed, save the old way */
462    }
463
464    /* Store verbatim */
465    if ((n = rdbSaveLen(rdb,len)) == -1) return -1;
466    nwritten += n;
467    if (len > 0) {
468        if (rdbWriteRaw(rdb,s,len) == -1) return -1;
469        nwritten += len;
470    }
471    return nwritten;
472}

474/* Save a long long value as either an encoded string or a string. */
475ssize_t rdbSaveLongLongAsStringObject(rio *rdb, long long value) {
476    unsigned char buf[32];
477    ssize_t n, nwritten = 0;
478    int enclen = rdbEncodeInteger(value,buf);
479    if (enclen > 0) {
480        return rdbWriteRaw(rdb,buf,enclen);
481    } else {
482        /* Encode as string */
483        enclen = ll2string((char*)buf,32,value);
484        serverAssert(enclen < 32);
485        if ((n = rdbSaveLen(rdb,enclen)) == -1) return -1;
486        nwritten += n;
487        if ((n = rdbWriteRaw(rdb,buf,enclen)) == -1) return -1;
488        nwritten += n;
489    }
490    return nwritten;
491}

505/* Load a string object from an RDB file according to flags:
506 *
507 * RDB_LOAD_NONE (no flags): load an RDB object, unencoded.
508 * RDB_LOAD_ENC: If the returned type is a Redis object, try to
509 *               encode it in a special way to be more memory
510 *               efficient. When this flag is passed the function
511 *               no longer guarantees that obj->ptr is an SDS string.
512 * RDB_LOAD_PLAIN: Return a plain string allocated with zmalloc()
513 *                 instead of a Redis object with an sds in it.
514 * RDB_LOAD_SDS: Return an SDS string instead of a Redis object.
515 *
516 * On I/O error NULL is returned.
517 */
518void *rdbGenericLoadStringObject(rio *rdb, int flags, size_t *lenptr) {
519    int plain = flags & RDB_LOAD_PLAIN;
520    int sds = flags & RDB_LOAD_SDS;
521    int isencoded;
522    unsigned long long len;
523
524    len = rdbLoadLen(rdb,&isencoded);
525    if (len == RDB_LENERR) return NULL;
526
527    if (isencoded) {
528        switch(len) {
529        case RDB_ENC_INT8:
530        case RDB_ENC_INT16:
531        case RDB_ENC_INT32:
532            return rdbLoadIntegerObject(rdb,len,flags,lenptr);
533        case RDB_ENC_LZF:
534            return rdbLoadLzfStringObject(rdb,flags,lenptr);
535        default:
536            rdbReportCorruptRDB("Unknown RDB string encoding type %llu",len);
537            return NULL;
538        }
539    }
540
541    if (plain || sds) {
542        void *buf = plain ? ztrymalloc(len) : sdstrynewlen(SDS_NOINIT,len);
543        if (!buf) {
544            serverLog(isRestoreContext()? LL_VERBOSE: LL_WARNING, "rdbGenericLoadStringObject failed allocating %llu bytes", len);
545            return NULL;
546        }
547        if (lenptr) *lenptr = len;
548        if (len && rioRead(rdb,buf,len) == 0) {
549            if (plain)
550                zfree(buf);
551            else
552                sdsfree(buf);
553            return NULL;
554        }
555        return buf;
556    } else {
557        robj *o = tryCreateStringObject(SDS_NOINIT,len);
558        if (!o) {
559            serverLog(isRestoreContext()? LL_VERBOSE: LL_WARNING, "rdbGenericLoadStringObject failed allocating %llu bytes", len);
560            return NULL;
561        }
562        if (len && rioRead(rdb,o->ptr,len) == 0) {
563            decrRefCount(o);
564            return NULL;
565        }
566        return o;
567    }
568}

578/* Save a double value. Doubles are saved as strings prefixed by an unsigned
579 * 8 bit integer specifying the length of the representation.
580 * This 8 bit integer has special values in order to specify the following
581 * conditions:
582 * 253: not a number
583 * 254: + inf
584 * 255: - inf
585 */
586int rdbSaveDoubleValue(rio *rdb, double val) {
587    unsigned char buf[128];
588    int len;
589
590    if (isnan(val)) {
591        buf[0] = 253;
592        len = 1;
593    } else if (!isfinite(val)) {
594        len = 1;
595        buf[0] = (val < 0) ? 255 : 254;
596    } else {
597        long long lvalue;
598        /* Integer printing function is much faster, check if we can safely use it. */
599        if (double2ll(val, &lvalue))
600            ll2string((char*)buf+1,sizeof(buf)-1,lvalue);
601        else {
602            const int dlen = fpconv_dtoa(val, (char*)buf+1);
603            buf[dlen+1] = '\0';
604        }
605        buf[0] = strlen((char*)buf+1);
606        len = buf[0]+1;
607    }
608    return rdbWriteRaw(rdb,buf,len);
609}

611/* For information about double serialization check rdbSaveDoubleValue() */
612int rdbLoadDoubleValue(rio *rdb, double *val) {
613    char buf[256];
614    unsigned char len;
615
616    if (rioRead(rdb,&len,1) == 0) return -1;
617    switch(len) {
618    case 255: *val = R_NegInf; return 0;
619    case 254: *val = R_PosInf; return 0;
620    case 253: *val = R_Nan; return 0;
621    default:
622        if (rioRead(rdb,buf,len) == 0) return -1;
623        buf[len] = '\0';
624        if (sscanf(buf, "%lg", val)!=1) return -1;
625        return 0;
626    }
627}

712/* This helper function serializes a consumer group Pending Entries List (PEL)
713 * into the RDB file. The 'nacks' argument tells the function if also persist
714 * the information about the not acknowledged message, or if to persist
715 * just the IDs: this is useful because for the global consumer group PEL
716 * we serialized the NACKs as well, but when serializing the local consumer
717 * PELs we just add the ID, that will be resolved inside the global PEL to
718 * put a reference to the same structure. */
719ssize_t rdbSaveStreamPEL(rio *rdb, rax *pel, int nacks) {
720    ssize_t n, nwritten = 0;
721
722    /* Number of entries in the PEL. */
723    if ((n = rdbSaveLen(rdb,raxSize(pel))) == -1) return -1;
724    nwritten += n;
725
726    /* Save each entry. */
727    raxIterator ri;
728    raxStart(&ri,pel);
729    raxSeek(&ri,"^",NULL,0);
730    while(raxNext(&ri)) {
731        /* We store IDs in raw form as 128 big big endian numbers, like
732         * they are inside the radix tree key. */
733        if ((n = rdbWriteRaw(rdb,ri.key,sizeof(streamID))) == -1) {
734            raxStop(&ri);
735            return -1;
736        }
737        nwritten += n;
738
739        if (nacks) {
740            streamNACK *nack = ri.data;
741            if ((n = rdbSaveMillisecondTime(rdb,nack->delivery_time)) == -1) {
742                raxStop(&ri);
743                return -1;
744            }
745            nwritten += n;
746            if ((n = rdbSaveLen(rdb,nack->delivery_count)) == -1) {
747                raxStop(&ri);
748                return -1;
749            }
750            nwritten += n;
751            /* We don't save the consumer name: we'll save the pending IDs
752             * for each consumer in the consumer PEL, and resolve the consumer
753             * at loading time. */
754        }
755    }
756    raxStop(&ri);
757    return nwritten;
758}

760/* Serialize the consumers of a stream consumer group into the RDB. Helper
761 * function for the stream data type serialization. What we do here is to
762 * persist the consumer metadata, and it's PEL, for each consumer. */
763size_t rdbSaveStreamConsumers(rio *rdb, streamCG *cg) {
764    ssize_t n, nwritten = 0;
765
766    /* Number of consumers in this consumer group. */
767    if ((n = rdbSaveLen(rdb,raxSize(cg->consumers))) == -1) return -1;
768    nwritten += n;
769
770    /* Save each consumer. */
771    raxIterator ri;
772    raxStart(&ri,cg->consumers);
773    raxSeek(&ri,"^",NULL,0);
774    while(raxNext(&ri)) {
775        streamConsumer *consumer = ri.data;
776
777        /* Consumer name. */
778        if ((n = rdbSaveRawString(rdb,ri.key,ri.key_len)) == -1) {
779            raxStop(&ri);
780            return -1;
781        }
782        nwritten += n;
783
784        /* Seen time. */
785        if ((n = rdbSaveMillisecondTime(rdb,consumer->seen_time)) == -1) {
786            raxStop(&ri);
787            return -1;
788        }
789        nwritten += n;
790
791        /* Active time. */
792        if ((n = rdbSaveMillisecondTime(rdb,consumer->active_time)) == -1) {
793            raxStop(&ri);
794            return -1;
795        }
796        nwritten += n;
797
798        /* Consumer PEL, without the ACKs (see last parameter of the function
799         * passed with value of 0), at loading time we'll lookup the ID
800         * in the consumer group global PEL and will put a reference in the
801         * consumer local PEL. */
802        if ((n = rdbSaveStreamPEL(rdb,consumer->pel,0)) == -1) {
803            raxStop(&ri);
804            return -1;
805        }
806        nwritten += n;
807    }
808    raxStop(&ri);
809    return nwritten;
810}

812/* Save a Redis object.
813 * Returns -1 on error, number of bytes written on success. */
814ssize_t rdbSaveObject(rio *rdb, robj *o, robj *key, int dbid) {
815    ssize_t n = 0, nwritten = 0;
816
817    if (o->type == OBJ_STRING) {
818        /* Save a string value */
819        if ((n = rdbSaveStringObject(rdb,o)) == -1) return -1;
820        nwritten += n;
821    } else if (o->type == OBJ_LIST) {
822        /* Save a list value */
823        if (o->encoding == OBJ_ENCODING_QUICKLIST) {
824            quicklist *ql = o->ptr;
825            quicklistNode *node = ql->head;
826
827            if ((n = rdbSaveLen(rdb,ql->len)) == -1) return -1;
828            nwritten += n;
829
830            while(node) {
831                if ((n = rdbSaveLen(rdb,node->container)) == -1) return -1;
832                nwritten += n;
833
834                if (quicklistNodeIsCompressed(node)) {
835                    void *data;
836                    size_t compress_len = quicklistGetLzf(node, &data);
837                    if ((n = rdbSaveLzfBlob(rdb,data,compress_len,node->sz)) == -1) return -1;
838                    nwritten += n;
839                } else {
840                    if ((n = rdbSaveRawString(rdb,node->entry,node->sz)) == -1) return -1;
841                    nwritten += n;
842                }
843                node = node->next;
844            }
845        } else if (o->encoding == OBJ_ENCODING_LISTPACK) {
846            unsigned char *lp = o->ptr;
847
848            /* Save list listpack as a fake quicklist that only has a single node. */
849            if ((n = rdbSaveLen(rdb,1)) == -1) return -1;
850            nwritten += n;
851            if ((n = rdbSaveLen(rdb,QUICKLIST_NODE_CONTAINER_PACKED)) == -1) return -1;
852            nwritten += n;
853            if ((n = rdbSaveRawString(rdb,lp,lpBytes(lp))) == -1) return -1;
854            nwritten += n;
855        } else {
856            serverPanic("Unknown list encoding");
857        }
858    } else if (o->type == OBJ_SET) {
859        /* Save a set value */
860        if (o->encoding == OBJ_ENCODING_HT) {
861            dict *set = o->ptr;
862            dictIterator *di = dictGetIterator(set);
863            dictEntry *de;
864
865            if ((n = rdbSaveLen(rdb,dictSize(set))) == -1) {
866                dictReleaseIterator(di);
867                return -1;
868            }
869            nwritten += n;
870
871            while((de = dictNext(di)) != NULL) {
872                sds ele = dictGetKey(de);
873                if ((n = rdbSaveRawString(rdb,(unsigned char*)ele,sdslen(ele)))
874                    == -1)
875                {
876                    dictReleaseIterator(di);
877                    return -1;
878                }
879                nwritten += n;
880            }
881            dictReleaseIterator(di);
882        } else if (o->encoding == OBJ_ENCODING_INTSET) {
883            size_t l = intsetBlobLen((intset*)o->ptr);
884
885            if ((n = rdbSaveRawString(rdb,o->ptr,l)) == -1) return -1;
886            nwritten += n;
887        } else if (o->encoding == OBJ_ENCODING_LISTPACK) {
888            size_t l = lpBytes((unsigned char *)o->ptr);
889            if ((n = rdbSaveRawString(rdb, o->ptr, l)) == -1) return -1;
890            nwritten += n;
891        } else {
892            serverPanic("Unknown set encoding");
893        }
894    } else if (o->type == OBJ_ZSET) {
895        /* Save a sorted set value */
896        if (o->encoding == OBJ_ENCODING_LISTPACK) {
897            size_t l = lpBytes((unsigned char*)o->ptr);
898
899            if ((n = rdbSaveRawString(rdb,o->ptr,l)) == -1) return -1;
900            nwritten += n;
901        } else if (o->encoding == OBJ_ENCODING_SKIPLIST) {
902            zset *zs = o->ptr;
903            zskiplist *zsl = zs->zsl;
904
905            if ((n = rdbSaveLen(rdb,zsl->length)) == -1) return -1;
906            nwritten += n;
907
908            /* We save the skiplist elements from the greatest to the smallest
909             * (that's trivial since the elements are already ordered in the
910             * skiplist): this improves the load process, since the next loaded
911             * element will always be the smaller, so adding to the skiplist
912             * will always immediately stop at the head, making the insertion
913             * O(1) instead of O(log(N)). */
914            zskiplistNode *zn = zsl->tail;
915            while (zn != NULL) {
916                if ((n = rdbSaveRawString(rdb,
917                    (unsigned char*)zn->ele,sdslen(zn->ele))) == -1)
918                {
919                    return -1;
920                }
921                nwritten += n;
922                if ((n = rdbSaveBinaryDoubleValue(rdb,zn->score)) == -1)
923                    return -1;
924                nwritten += n;
925                zn = zn->backward;
926            }
927        } else {
928            serverPanic("Unknown sorted set encoding");
929        }
930    } else if (o->type == OBJ_HASH) {
931        /* Save a hash value */
932        if (o->encoding == OBJ_ENCODING_LISTPACK) {
933            size_t l = lpBytes((unsigned char*)o->ptr);
934
935            if ((n = rdbSaveRawString(rdb,o->ptr,l)) == -1) return -1;
936            nwritten += n;
937        } else if (o->encoding == OBJ_ENCODING_HT) {
938            dictIterator *di = dictGetIterator(o->ptr);
939            dictEntry *de;
940
941            if ((n = rdbSaveLen(rdb,dictSize((dict*)o->ptr))) == -1) {
942                dictReleaseIterator(di);
943                return -1;
944            }
945            nwritten += n;
946
947            while((de = dictNext(di)) != NULL) {
948                sds field = dictGetKey(de);
949                sds value = dictGetVal(de);
950
951                if ((n = rdbSaveRawString(rdb,(unsigned char*)field,
952                        sdslen(field))) == -1)
953                {
954                    dictReleaseIterator(di);
955                    return -1;
956                }
957                nwritten += n;
958                if ((n = rdbSaveRawString(rdb,(unsigned char*)value,
959                        sdslen(value))) == -1)
960                {
961                    dictReleaseIterator(di);
962                    return -1;
963                }
964                nwritten += n;
965            }
966            dictReleaseIterator(di);
967        } else {
968            serverPanic("Unknown hash encoding");
969        }
970    } else if (o->type == OBJ_STREAM) {
971        /* Store how many listpacks we have inside the radix tree. */
972        stream *s = o->ptr;
973        rax *rax = s->rax;
974        if ((n = rdbSaveLen(rdb,raxSize(rax))) == -1) return -1;
975        nwritten += n;
976
977        /* Serialize all the listpacks inside the radix tree as they are,
978         * when loading back, we'll use the first entry of each listpack
979         * to insert it back into the radix tree. */
980        raxIterator ri;
981        raxStart(&ri,rax);
982        raxSeek(&ri,"^",NULL,0);
983        while (raxNext(&ri)) {
984            unsigned char *lp = ri.data;
985            size_t lp_bytes = lpBytes(lp);
986            if ((n = rdbSaveRawString(rdb,ri.key,ri.key_len)) == -1) {
987                raxStop(&ri);
988                return -1;
989            }
990            nwritten += n;
991            if ((n = rdbSaveRawString(rdb,lp,lp_bytes)) == -1) {
992                raxStop(&ri);
993                return -1;
994            }
995            nwritten += n;
996        }
997        raxStop(&ri);
998
999        /* Save the number of elements inside the stream. We cannot obtain
1000         * this easily later, since our macro nodes should be checked for
1001         * number of items: not a great CPU / space tradeoff. */
1002        if ((n = rdbSaveLen(rdb,s->length)) == -1) return -1;
1003        nwritten += n;
1004        /* Save the last entry ID. */
1005        if ((n = rdbSaveLen(rdb,s->last_id.ms)) == -1) return -1;
1006        nwritten += n;
1007        if ((n = rdbSaveLen(rdb,s->last_id.seq)) == -1) return -1;
1008        nwritten += n;
1009        /* Save the first entry ID. */
1010        if ((n = rdbSaveLen(rdb,s->first_id.ms)) == -1) return -1;
1011        nwritten += n;
1012        if ((n = rdbSaveLen(rdb,s->first_id.seq)) == -1) return -1;
1013        nwritten += n;
1014        /* Save the maximal tombstone ID. */
1015        if ((n = rdbSaveLen(rdb,s->max_deleted_entry_id.ms)) == -1) return -1;
1016        nwritten += n;
1017        if ((n = rdbSaveLen(rdb,s->max_deleted_entry_id.seq)) == -1) return -1;
1018        nwritten += n;
1019        /* Save the offset. */
1020        if ((n = rdbSaveLen(rdb,s->entries_added)) == -1) return -1;
1021        nwritten += n;
1022
1023        /* The consumer groups and their clients are part of the stream
1024         * type, so serialize every consumer group. */
1025
1026        /* Save the number of groups. */
1027        size_t num_cgroups = s->cgroups ? raxSize(s->cgroups) : 0;
1028        if ((n = rdbSaveLen(rdb,num_cgroups)) == -1) return -1;
1029        nwritten += n;
1030
1031        if (num_cgroups) {
1032            /* Serialize each consumer group. */
1033            raxStart(&ri,s->cgroups);
1034            raxSeek(&ri,"^",NULL,0);
1035            while(raxNext(&ri)) {
1036                streamCG *cg = ri.data;
1037
1038                /* Save the group name. */
1039                if ((n = rdbSaveRawString(rdb,ri.key,ri.key_len)) == -1) {
1040                    raxStop(&ri);
1041                    return -1;
1042                }
1043                nwritten += n;
1044
1045                /* Last ID. */
1046                if ((n = rdbSaveLen(rdb,cg->last_id.ms)) == -1) {
1047                    raxStop(&ri);
1048                    return -1;
1049                }
1050                nwritten += n;
1051                if ((n = rdbSaveLen(rdb,cg->last_id.seq)) == -1) {
1052                    raxStop(&ri);
1053                    return -1;
1054                }
1055                nwritten += n;
1056                
1057                /* Save the group's logical reads counter. */
1058                if ((n = rdbSaveLen(rdb,cg->entries_read)) == -1) {
1059                    raxStop(&ri);
1060                    return -1;
1061                }
1062                nwritten += n;
1063
1064                /* Save the global PEL. */
1065                if ((n = rdbSaveStreamPEL(rdb,cg->pel,1)) == -1) {
1066                    raxStop(&ri);
1067                    return -1;
1068                }
1069                nwritten += n;
1070
1071                /* Save the consumers of this group. */
1072                if ((n = rdbSaveStreamConsumers(rdb,cg)) == -1) {
1073                    raxStop(&ri);
1074                    return -1;
1075                }
1076                nwritten += n;
1077            }
1078            raxStop(&ri);
1079        }
1080    } else if (o->type == OBJ_MODULE) {
1081        /* Save a module-specific value. */
1082        RedisModuleIO io;
1083        moduleValue *mv = o->ptr;
1084        moduleType *mt = mv->type;
1085
1086        /* Write the "module" identifier as prefix, so that we'll be able
1087         * to call the right module during loading. */
1088        int retval = rdbSaveLen(rdb,mt->id);
1089        if (retval == -1) return -1;
1090        moduleInitIOContext(io,mt,rdb,key,dbid);
1091        io.bytes += retval;
1092
1093        /* Then write the module-specific representation + EOF marker. */
1094        mt->rdb_save(&io,mv->value);
1095        retval = rdbSaveLen(rdb,RDB_MODULE_OPCODE_EOF);
1096        if (retval == -1)
1097            io.error = 1;
1098        else
1099            io.bytes += retval;
1100
1101        if (io.ctx) {
1102            moduleFreeContext(io.ctx);
1103            zfree(io.ctx);
1104        }
1105        return io.error ? -1 : (ssize_t)io.bytes;
1106    } else {
1107        serverPanic("Unknown object type");
1108    }
1109    return nwritten;
1110}

1122/* Save a key-value pair, with expire time, type, key, value.
1123 * On error -1 is returned.
1124 * On success if the key was actually saved 1 is returned. */
1125int rdbSaveKeyValuePair(rio *rdb, robj *key, robj *val, long long expiretime, int dbid) {
1126    int savelru = server.maxmemory_policy & MAXMEMORY_FLAG_LRU;
1127    int savelfu = server.maxmemory_policy & MAXMEMORY_FLAG_LFU;
1128
1129    /* Save the expire time */
1130    if (expiretime != -1) {
1131        if (rdbSaveType(rdb,RDB_OPCODE_EXPIRETIME_MS) == -1) return -1;
1132        if (rdbSaveMillisecondTime(rdb,expiretime) == -1) return -1;
1133    }
1134
1135    /* Save the LRU info. */
1136    if (savelru) {
1137        uint64_t idletime = estimateObjectIdleTime(val);
1138        idletime /= 1000; /* Using seconds is enough and requires less space.*/
1139        if (rdbSaveType(rdb,RDB_OPCODE_IDLE) == -1) return -1;
1140        if (rdbSaveLen(rdb,idletime) == -1) return -1;
1141    }
1142
1143    /* Save the LFU info. */
1144    if (savelfu) {
1145        uint8_t buf[1];
1146        buf[0] = LFUDecrAndReturn(val);
1147        /* We can encode this in exactly two bytes: the opcode and an 8
1148         * bit counter, since the frequency is logarithmic with a 0-255 range.
1149         * Note that we do not store the halving time because to reset it
1150         * a single time when loading does not affect the frequency much. */
1151        if (rdbSaveType(rdb,RDB_OPCODE_FREQ) == -1) return -1;
1152        if (rdbWriteRaw(rdb,buf,1) == -1) return -1;
1153    }
1154
1155    /* Save type, key, value */
1156    if (rdbSaveObjectType(rdb,val) == -1) return -1;
1157    if (rdbSaveStringObject(rdb,key) == -1) return -1;
1158    if (rdbSaveObject(rdb,val,key,dbid) == -1) return -1;
1159
1160    /* Delay return if required (for testing) */
1161    if (server.rdb_key_save_delay)
1162        debugDelay(server.rdb_key_save_delay);

1216ssize_t rdbSaveSingleModuleAux(rio *rdb, int when, moduleType *mt) {
1217    /* Save a module-specific aux value. */
1218    RedisModuleIO io;
1219    int retval = 0;
1220    moduleInitIOContext(io,mt,rdb,NULL,-1);
1221
1222    /* We save the AUX field header in a temporary buffer so we can support aux_save2 API.
1223     * If aux_save2 is used the buffer will be flushed at the first time the module will perform
1224     * a write operation to the RDB and will be ignored is case there was no writes. */
1225    rio aux_save_headers_rio;
1226    rioInitWithBuffer(&aux_save_headers_rio, sdsempty());
1227
1228    if (rdbSaveType(&aux_save_headers_rio, RDB_OPCODE_MODULE_AUX) == -1) goto error;
1229
1230    /* Write the "module" identifier as prefix, so that we'll be able
1231     * to call the right module during loading. */
1232    if (rdbSaveLen(&aux_save_headers_rio,mt->id) == -1) goto error;
1233
1234    /* write the 'when' so that we can provide it on loading. add a UINT opcode
1235     * for backwards compatibility, everything after the MT needs to be prefixed
1236     * by an opcode. */
1237    if (rdbSaveLen(&aux_save_headers_rio,RDB_MODULE_OPCODE_UINT) == -1) goto error;
1238    if (rdbSaveLen(&aux_save_headers_rio,when) == -1) goto error;
1239
1240    /* Then write the module-specific representation + EOF marker. */
1241    if (mt->aux_save2) {
1242        io.pre_flush_buffer = aux_save_headers_rio.io.buffer.ptr;
1243        mt->aux_save2(&io,when);
1244        if (io.pre_flush_buffer) {
1245            /* aux_save did not save any data to the RDB.
1246             * We will avoid saving any data related to this aux type
1247             * to allow loading this RDB if the module is not present. */
1248            sdsfree(io.pre_flush_buffer);
1249            io.pre_flush_buffer = NULL;
1250            return 0;
1251        }
1252    } else {
1253        /* Write headers now, aux_save does not do lazy saving of the headers. */
1254        retval = rdbWriteRaw(rdb, aux_save_headers_rio.io.buffer.ptr, sdslen(aux_save_headers_rio.io.buffer.ptr));
1255        if (retval == -1) goto error;
1256        io.bytes += retval;
1257        sdsfree(aux_save_headers_rio.io.buffer.ptr);
1258        mt->aux_save(&io,when);
1259    }
1260    retval = rdbSaveLen(rdb,RDB_MODULE_OPCODE_EOF);
1261    serverAssert(!io.pre_flush_buffer);
1262    if (retval == -1)
1263        io.error = 1;
1264    else
1265        io.bytes += retval;
1266
1267    if (io.ctx) {
1268        moduleFreeContext(io.ctx);
1269        zfree(io.ctx);
1270    }
1271    if (io.error)
1272        return -1;
1273    return io.bytes;
1274error:
1275    sdsfree(aux_save_headers_rio.io.buffer.ptr);
1276    return -1;
1277}

1300ssize_t rdbSaveDb(rio *rdb, int dbid, int rdbflags, long *key_counter) {
1301    dictIterator *di;
1302    dictEntry *de;
1303    ssize_t written = 0;
1304    ssize_t res;
1305    static long long info_updated_time = 0;
1306    char *pname = (rdbflags & RDBFLAGS_AOF_PREAMBLE) ? "AOF rewrite" :  "RDB";
1307
1308    redisDb *db = server.db + dbid;
1309    dict *d = db->dict;
1310    if (dictSize(d) == 0) return 0;
1311    di = dictGetSafeIterator(d);
1312
1313    /* Write the SELECT DB opcode */
1314    if ((res = rdbSaveType(rdb,RDB_OPCODE_SELECTDB)) < 0) goto werr;
1315    written += res;
1316    if ((res = rdbSaveLen(rdb, dbid)) < 0) goto werr;
1317    written += res;
1318
1319    /* Write the RESIZE DB opcode. */
1320    uint64_t db_size, expires_size;
1321    db_size = dictSize(db->dict);
1322    expires_size = dictSize(db->expires);
1323    if ((res = rdbSaveType(rdb,RDB_OPCODE_RESIZEDB)) < 0) goto werr;
1324    written += res;
1325    if ((res = rdbSaveLen(rdb,db_size)) < 0) goto werr;
1326    written += res;
1327    if ((res = rdbSaveLen(rdb,expires_size)) < 0) goto werr;
1328    written += res;
1329
1330    /* Iterate this DB writing every entry */
1331    while((de = dictNext(di)) != NULL) {
1332        sds keystr = dictGetKey(de);
1333        robj key, *o = dictGetVal(de);
1334        long long expire;
1335        size_t rdb_bytes_before_key = rdb->processed_bytes;
1336
1337        initStaticStringObject(key,keystr);
1338        expire = getExpire(db,&key);
1339        if ((res = rdbSaveKeyValuePair(rdb, &key, o, expire, dbid)) < 0) goto werr;
1340        written += res;
1341
1342        /* In fork child process, we can try to release memory back to the
1343         * OS and possibly avoid or decrease COW. We give the dismiss
1344         * mechanism a hint about an estimated size of the object we stored. */
1345        size_t dump_size = rdb->processed_bytes - rdb_bytes_before_key;
1346        if (server.in_fork_child) dismissObject(o, dump_size);
1347
1348        /* Update child info every 1 second (approximately).
1349         * in order to avoid calling mstime() on each iteration, we will
1350         * check the diff every 1024 keys */
1351        if (((*key_counter)++ & 1023) == 0) {
1352            long long now = mstime();
1353            if (now - info_updated_time >= 1000) {
1354                sendChildInfo(CHILD_INFO_TYPE_CURRENT_INFO, *key_counter, pname);
1355                info_updated_time = now;
1356            }
1357        }
1358    }
1359
1360    dictReleaseIterator(di);
1361    return written;
1362
1363werr:
1364    dictReleaseIterator(di);
1365    return -1;
1366}

1368/* Produces a dump of the database in RDB format sending it to the specified
1369 * Redis I/O channel. On success C_OK is returned, otherwise C_ERR
1370 * is returned and part of the output, or all the output, can be
1371 * missing because of I/O errors.
1372 *
1373 * When the function returns C_ERR and if 'error' is not NULL, the
1374 * integer pointed by 'error' is set to the value of errno just after the I/O
1375 * error. */
1376int rdbSaveRio(int req, rio *rdb, int *error, int rdbflags, rdbSaveInfo *rsi) {
1377    char magic[10];
1378    uint64_t cksum;
1379    long key_counter = 0;
1380    int j;
1381
1382    if (server.rdb_checksum)
1383        rdb->update_cksum = rioGenericUpdateChecksum;
1384    snprintf(magic,sizeof(magic),"REDIS%04d",RDB_VERSION);
1385    if (rdbWriteRaw(rdb,magic,9) == -1) goto werr;
1386    if (rdbSaveInfoAuxFields(rdb,rdbflags,rsi) == -1) goto werr;
1387    if (!(req & SLAVE_REQ_RDB_EXCLUDE_DATA) && rdbSaveModulesAux(rdb, REDISMODULE_AUX_BEFORE_RDB) == -1) goto werr;
1388
1389    /* save functions */
1390    if (!(req & SLAVE_REQ_RDB_EXCLUDE_FUNCTIONS) && rdbSaveFunctions(rdb) == -1) goto werr;
1391
1392    /* save all databases, skip this if we're in functions-only mode */
1393    if (!(req & SLAVE_REQ_RDB_EXCLUDE_DATA)) {
1394        for (j = 0; j < server.dbnum; j++) {
1395            if (rdbSaveDb(rdb, j, rdbflags, &key_counter) == -1) goto werr;
1396        }
1397    }
1398
1399    if (!(req & SLAVE_REQ_RDB_EXCLUDE_DATA) && rdbSaveModulesAux(rdb, REDISMODULE_AUX_AFTER_RDB) == -1) goto werr;
1400
1401    /* EOF opcode */
1402    if (rdbSaveType(rdb,RDB_OPCODE_EOF) == -1) goto werr;
1403
1404    /* CRC64 checksum. It will be zero if checksum computation is disabled, the
1405     * loading code skips the check in this case. */
1406    cksum = rdb->cksum;
1407    memrev64ifbe(&cksum);
1408    if (rioWrite(rdb,&cksum,8) == 0) goto werr;
1409    return C_OK;
1410
1411werr:
1412    if (error) *error = errno;
1413    return C_ERR;
1414}

1416/* This is just a wrapper to rdbSaveRio() that additionally adds a prefix
1417 * and a suffix to the generated RDB dump. The prefix is:
1418 *
1419 * $EOF:<40 bytes unguessable hex string>\r\n
1420 *
1421 * While the suffix is the 40 bytes hex string we announced in the prefix.
1422 * This way processes receiving the payload can understand when it ends
1423 * without doing any processing of the content. */
1424int rdbSaveRioWithEOFMark(int req, rio *rdb, int *error, rdbSaveInfo *rsi) {
1425    char eofmark[RDB_EOF_MARK_SIZE];
1426
1427    startSaving(RDBFLAGS_REPLICATION);
1428    getRandomHexChars(eofmark,RDB_EOF_MARK_SIZE);
1429    if (error) *error = 0;
1430    if (rioWrite(rdb,"$EOF:",5) == 0) goto werr;
1431    if (rioWrite(rdb,eofmark,RDB_EOF_MARK_SIZE) == 0) goto werr;
1432    if (rioWrite(rdb,"\r\n",2) == 0) goto werr;
1433    if (rdbSaveRio(req,rdb,error,RDBFLAGS_NONE,rsi) == C_ERR) goto werr;
1434    if (rioWrite(rdb,eofmark,RDB_EOF_MARK_SIZE) == 0) goto werr;
1435    stopSaving(1);
1436    return C_OK;
1437
1438werr: /* Write error. */
1439    /* Set 'error' only if not already set by rdbSaveRio() call. */
1440    if (error && *error == 0) *error = errno;
1441    stopSaving(0);
1442    return C_ERR;
1443}

1445static int rdbSaveInternal(int req, const char *filename, rdbSaveInfo *rsi, int rdbflags) {
1446    char cwd[MAXPATHLEN]; /* Current working dir path for error messages. */
1447    rio rdb;
1448    int error = 0;
1449    int saved_errno;
1450    char *err_op;    /* For a detailed log */
1451
1452    FILE *fp = fopen(filename,"w");
1453    if (!fp) {
1454        saved_errno = errno;
1455        char *str_err = strerror(errno);
1456        char *cwdp = getcwd(cwd,MAXPATHLEN);
1457        serverLog(LL_WARNING,
1458            "Failed opening the temp RDB file %s (in server root dir %s) "
1459            "for saving: %s",
1460            filename,
1461            cwdp ? cwdp : "unknown",
1462            str_err);
1463        errno = saved_errno;
1464        return C_ERR;
1465    }
1466
1467    rioInitWithFile(&rdb,fp);
1468
1469    if (server.rdb_save_incremental_fsync) {
1470        rioSetAutoSync(&rdb,REDIS_AUTOSYNC_BYTES);
1471        if (!(rdbflags & RDBFLAGS_KEEP_CACHE)) rioSetReclaimCache(&rdb,1);
1472    }
1473
1474    if (rdbSaveRio(req,&rdb,&error,rdbflags,rsi) == C_ERR) {
1475        errno = error;
1476        err_op = "rdbSaveRio";
1477        goto werr;
1478    }
1479
1480    /* Make sure data will not remain on the OS's output buffers */
1481    if (fflush(fp)) { err_op = "fflush"; goto werr; }
1482    if (fsync(fileno(fp))) { err_op = "fsync"; goto werr; }
1483    if (!(rdbflags & RDBFLAGS_KEEP_CACHE) && reclaimFilePageCache(fileno(fp), 0, 0) == -1) {
1484        serverLog(LL_NOTICE,"Unable to reclaim cache after saving RDB: %s", strerror(errno));
1485    }
1486    if (fclose(fp)) { fp = NULL; err_op = "fclose"; goto werr; }
1487
1488    return C_OK;
1489
1490werr:
1491    saved_errno = errno;
1492    serverLog(LL_WARNING,"Write error while saving DB to the disk(%s): %s", err_op, strerror(errno));
1493    if (fp) fclose(fp);
1494    unlink(filename);
1495    errno = saved_errno;
1496    return C_ERR;
1497}

1621 * mode and we find a module value of type 2 that can be parsed without
1622 * the need of the actual module. The value is parsed for errors, finally

1638                rdbReportCorruptRDB(

1646                    "Error reading float from module %s value", modulename);

1649            double val;
1650            if (rdbLoadBinaryDoubleValue(rdb,&val) == -1) {
1651                rdbReportCorruptRDB(
1652                    "Error reading double from module %s value", modulename);

1654        }
1655    }

1660 * Check that the ziplist doesn't have duplicate hash field names.
1661 * The ziplist element pointed by 'p' will be converted and stored into listpack. */
1662static int _ziplistPairsEntryConvertAndValidate(unsigned char *p, unsigned int head_count, void *userdata) {
1663    unsigned char *str;
1664    unsigned int slen;
1665    long long vll;
1666

1668        long count;
1669        dict *fields;
1670        unsigned char **lp;

1702 * listpack and storing it at 'lp'.
1703 * The function is safe to call on non-validated ziplists, it returns 0
1704 * when encounter an integrity validation issue. */
1705int ziplistPairsConvertAndValidateIntegrity(unsigned char *zl, size_t size, unsigned char **lp) {
1706    /* Keep track of the field names to locate duplicate ones */
1707    struct {
1708        long count;
1709        dict *fields; /* Initialisation at the first callback. */
1710        unsigned char **lp;
1711    } data = {0, NULL, lp};
1712
1713    int ret = ziplistValidateIntegrity(zl, size, 1, _ziplistPairsEntryConvertAndValidate, &data);
1714
1715    /* make sure we have an even number of records. */
1716    if (data.count & 1)
1717        ret = 0;
1718
1719    if (data.fields) dictRelease(data.fields);
1720    return ret;
1721}

1723/* callback for ziplistValidateIntegrity.
1724 * The ziplist element pointed by 'p' will be converted and stored into listpack. */
1725static int _ziplistEntryConvertAndValidate(unsigned char *p, unsigned int head_count, void *userdata) {
1726    UNUSED(head_count);
1727    unsigned char *str;
1728    unsigned int slen;
1729    long long vll;
1730    unsigned char **lp = (unsigned char**)userdata;
1731
1732    if (!ziplistGet(p, &str, &slen, &vll)) return 0;
1733
1734    if (str)
1735        *lp = lpAppend(*lp, (unsigned char*)str, slen);
1736    else
1737        *lp = lpAppendInteger(*lp, vll);

1762/* callback for to check the listpack doesn't have duplicate records */
1763static int _lpEntryValidation(unsigned char *p, unsigned int head_count, void *userdata) {
1764    struct {
1765        int pairs;
1766        long count;
1767        dict *fields;
1768    } *data = userdata;
1769
1770    if (data->fields == NULL) {
1771        data->fields = dictCreate(&hashDictType);
1772        dictExpand(data->fields, data->pairs ? head_count/2 : head_count);
1773    }
1774
1775    /* If we're checking pairs, then even records are field names. Otherwise
1776     * we're checking all elements. Add to dict and check that's not a dup */
1777    if (!data->pairs || ((data->count) & 1) == 0) {
1778        unsigned char *str;
1779        int64_t slen;
1780        unsigned char buf[LP_INTBUF_SIZE];
1781
1782        str = lpGet(p, &slen, buf);
1783        sds field = sdsnewlen(str, slen);
1784        if (dictAdd(data->fields, field, NULL) != DICT_OK) {
1785            /* Duplicate, return an error */
1786            sdsfree(field);
1787            return 0;
1788        }
1789    }
1790
1791    (data->count)++;
1792    return 1;
1793}

1795/* Validate the integrity of the listpack structure.
1796 * when `deep` is 0, only the integrity of the header is validated.
1797 * when `deep` is 1, we scan all the entries one by one.
1798 * when `pairs` is 0, all elements need to be unique (it's a set)
1799 * when `pairs` is 1, odd elements need to be unique (it's a key-value map) */
1800int lpValidateIntegrityAndDups(unsigned char *lp, size_t size, int deep, int pairs) {
1801    if (!deep)
1802        return lpValidateIntegrity(lp, size, 0, NULL, NULL);
1803
1804    /* Keep track of the field names to locate duplicate ones */
1805    struct {
1806        int pairs;
1807        long count;
1808        dict *fields; /* Initialisation at the first callback. */
1809    } data = {pairs, 0, NULL};
1810
1811    int ret = lpValidateIntegrity(lp, size, 1, _lpEntryValidation, &data);
1812
1813    /* make sure we have an even number of records. */
1814    if (pairs && data.count & 1)
1815        ret = 0;
1816
1817    if (data.fields) dictRelease(data.fields);
1818    return ret;
1819}

2283                        if (vlen > maxlen) maxlen = vlen;
2284
2285                        /* search for duplicate records */
2286                        sds field = sdstrynewlen(fstr, flen);
2287                        if (!field || dictAdd(dupSearchDict, field, NULL) != DICT_OK ||
2288                            !lpSafeToAdd(lp, (size_t)flen + vlen)) {
2289                            rdbReportCorruptRDB("Hash zipmap with dup elements, or big length (%u)", flen);
2290                            dictRelease(dupSearchDict);
2291                            sdsfree(field);
2292                            zfree(encoded);
2293                            o->ptr = NULL;
2294                            decrRefCount(o);
2295                            return NULL;
2296                        }
2297
2298                        lp = lpAppend(lp, fstr, flen);
2299                        lp = lpAppend(lp, vstr, vlen);
2300                    }
2301
2302                    dictRelease(dupSearchDict);
2303                    zfree(o->ptr);
2304                    o->ptr = lp;
2305                    o->type = OBJ_HASH;
2306                    o->encoding = OBJ_ENCODING_LISTPACK;
2307
2308                    if (hashTypeLength(o) > server.hash_max_listpack_entries ||
2309                        maxlen > server.hash_max_listpack_value)
2310                    {
2311                        hashTypeConvert(o, OBJ_ENCODING_HT);
2312                    }
2313                }
2314                break;
2315            case RDB_TYPE_LIST_ZIPLIST: 
2316                {
2317                    quicklist *ql = quicklistNew(server.list_max_listpack_size,
2318                                                 server.list_compress_depth);
2319
2320                    if (!ziplistValidateIntegrity(encoded, encoded_len, 1,
2321                            _listZiplistEntryConvertAndValidate, ql))
2322                    {
2323                        rdbReportCorruptRDB("List ziplist integrity check failed.");
2324                        zfree(encoded);
2325                        o->ptr = NULL;
2326                        decrRefCount(o);
2327                        quicklistRelease(ql);
2328                        return NULL;
2329                    }
2330
2331                    if (ql->len == 0) {
2332                        zfree(encoded);
2333                        o->ptr = NULL;
2334                        decrRefCount(o);
2335                        quicklistRelease(ql);
2336                        goto emptykey;
2337                    }
2338
2339                    zfree(encoded);
2340                    o->type = OBJ_LIST;
2341                    o->ptr = ql;
2342                    o->encoding = OBJ_ENCODING_QUICKLIST;
2343                    break;
2344                }
2345            case RDB_TYPE_SET_INTSET:
2346                if (deep_integrity_validation) server.stat_dump_payload_sanitizations++;
2347                if (!intsetValidateIntegrity(encoded, encoded_len, deep_integrity_validation)) {
2348                    rdbReportCorruptRDB("Intset integrity check failed.");
2349                    zfree(encoded);
2350                    o->ptr = NULL;
2351                    decrRefCount(o);
2352                    return NULL;
2353                }
2354                o->type = OBJ_SET;
2355                o->encoding = OBJ_ENCODING_INTSET;
2356                if (intsetLen(o->ptr) > server.set_max_intset_entries)
2357                    setTypeConvert(o,OBJ_ENCODING_HT);
2358                break;
2359            case RDB_TYPE_SET_LISTPACK:
2360                if (deep_integrity_validation) server.stat_dump_payload_sanitizations++;
2361                if (!lpValidateIntegrityAndDups(encoded, encoded_len, deep_integrity_validation, 0)) {
2362                    rdbReportCorruptRDB("Set listpack integrity check failed.");
2363                    zfree(encoded);
2364                    o->ptr = NULL;
2365                    decrRefCount(o);
2366                    return NULL;
2367                }
2368                o->type = OBJ_SET;
2369                o->encoding = OBJ_ENCODING_LISTPACK;
2370
2371                if (setTypeSize(o) == 0) {
2372                    zfree(encoded);
2373                    o->ptr = NULL;
2374                    decrRefCount(o);
2375                    goto emptykey;
2376                }
2377                if (setTypeSize(o) > server.set_max_listpack_entries)
2378                    setTypeConvert(o, OBJ_ENCODING_HT);
2379                break;
2380            case RDB_TYPE_ZSET_ZIPLIST:
2381                {
2382                    unsigned char *lp = lpNew(encoded_len);
2383                    if (!ziplistPairsConvertAndValidateIntegrity(encoded, encoded_len, &lp)) {
2384                        rdbReportCorruptRDB("Zset ziplist integrity check failed.");
2385                        zfree(lp);
2386                        zfree(encoded);
2387                        o->ptr = NULL;
2388                        decrRefCount(o);
2389                        return NULL;
2390                    }
2391
2392                    zfree(o->ptr);
2393                    o->type = OBJ_ZSET;
2394                    o->ptr = lp;
2395                    o->encoding = OBJ_ENCODING_LISTPACK;
2396                    if (zsetLength(o) == 0) {
2397                        decrRefCount(o);
2398                        goto emptykey;
2399                    }
2400
2401                    if (zsetLength(o) > server.zset_max_listpack_entries)
2402                        zsetConvert(o,OBJ_ENCODING_SKIPLIST);
2403                    else
2404                        o->ptr = lpShrinkToFit(o->ptr);
2405                    break;
2406                }
2407            case RDB_TYPE_ZSET_LISTPACK:
2408                if (deep_integrity_validation) server.stat_dump_payload_sanitizations++;
2409                if (!lpValidateIntegrityAndDups(encoded, encoded_len, deep_integrity_validation, 1)) {
2410                    rdbReportCorruptRDB("Zset listpack integrity check failed.");
2411                    zfree(encoded);
2412                    o->ptr = NULL;
2413                    decrRefCount(o);
2414                    return NULL;
2415                }
2416                o->type = OBJ_ZSET;
2417                o->encoding = OBJ_ENCODING_LISTPACK;
2418                if (zsetLength(o) == 0) {
2419                    decrRefCount(o);
2420                    goto emptykey;
2421                }
2422
2423                if (zsetLength(o) > server.zset_max_listpack_entries)
2424                    zsetConvert(o,OBJ_ENCODING_SKIPLIST);
2425                break;
2426            case RDB_TYPE_HASH_ZIPLIST:
2427                {
2428                    unsigned char *lp = lpNew(encoded_len);
2429                    if (!ziplistPairsConvertAndValidateIntegrity(encoded, encoded_len, &lp)) {
2430                        rdbReportCorruptRDB("Hash ziplist integrity check failed.");
2431                        zfree(lp);
2432                        zfree(encoded);
2433                        o->ptr = NULL;
2434                        decrRefCount(o);
2435                        return NULL;
2436                    }
2437
2438                    zfree(o->ptr);
2439                    o->ptr = lp;
2440                    o->type = OBJ_HASH;
2441                    o->encoding = OBJ_ENCODING_LISTPACK;
2442                    if (hashTypeLength(o) == 0) {
2443                        decrRefCount(o);
2444                        goto emptykey;
2445                    }
2446
2447                    if (hashTypeLength(o) > server.hash_max_listpack_entries)
2448                        hashTypeConvert(o, OBJ_ENCODING_HT);
2449                    else
2450                        o->ptr = lpShrinkToFit(o->ptr);
2451                    break;
2452                }
2453            case RDB_TYPE_HASH_LISTPACK:
2454                if (deep_integrity_validation) server.stat_dump_payload_sanitizations++;
2455                if (!lpValidateIntegrityAndDups(encoded, encoded_len, deep_integrity_validation, 1)) {
2456                    rdbReportCorruptRDB("Hash listpack integrity check failed.");
2457                    zfree(encoded);
2458                    o->ptr = NULL;
2459                    decrRefCount(o);
2460                    return NULL;
2461                }
2462                o->type = OBJ_HASH;
2463                o->encoding = OBJ_ENCODING_LISTPACK;
2464                if (hashTypeLength(o) == 0) {
2465                    decrRefCount(o);
2466                    goto emptykey;
2467                }
2468
2469                if (hashTypeLength(o) > server.hash_max_listpack_entries)
2470                    hashTypeConvert(o, OBJ_ENCODING_HT);
2471                break;
2472            default:
2473                /* totally unreachable */
2474                rdbReportCorruptRDB("Unknown RDB encoding type %d",rdbtype);
2475                break;
2476        }
2477    } else if (rdbtype == RDB_TYPE_STREAM_LISTPACKS ||
2478               rdbtype == RDB_TYPE_STREAM_LISTPACKS_2 ||
2479               rdbtype == RDB_TYPE_STREAM_LISTPACKS_3)
2480    {
2481        o = createStreamObject();
2482        stream *s = o->ptr;
2483        uint64_t listpacks = rdbLoadLen(rdb,NULL);
2484        if (listpacks == RDB_LENERR) {
2485            rdbReportReadError("Stream listpacks len loading failed.");
2486            decrRefCount(o);
2487            return NULL;
2488        }
2489
2490        while(listpacks--) {
2491            /* Get the master ID, the one we'll use as key of the radix tree
2492             * node: the entries inside the listpack itself are delta-encoded
2493             * relatively to this ID. */
2494            sds nodekey = rdbGenericLoadStringObject(rdb,RDB_LOAD_SDS,NULL);
2495            if (nodekey == NULL) {
2496                rdbReportReadError("Stream master ID loading failed: invalid encoding or I/O error.");
2497                decrRefCount(o);
2498                return NULL;
2499            }
2500            if (sdslen(nodekey) != sizeof(streamID)) {
2501                rdbReportCorruptRDB("Stream node key entry is not the "
2502                                        "size of a stream ID");
2503                sdsfree(nodekey);
2504                decrRefCount(o);
2505                return NULL;
2506            }
2507
2508            /* Load the listpack. */
2509            size_t lp_size;
2510            unsigned char *lp =
2511                rdbGenericLoadStringObject(rdb,RDB_LOAD_PLAIN,&lp_size);
2512            if (lp == NULL) {
2513                rdbReportReadError("Stream listpacks loading failed.");
2514                sdsfree(nodekey);
2515                decrRefCount(o);
2516                return NULL;
2517            }
2518            if (deep_integrity_validation) server.stat_dump_payload_sanitizations++;
2519            if (!streamValidateListpackIntegrity(lp, lp_size, deep_integrity_validation)) {
2520                rdbReportCorruptRDB("Stream listpack integrity check failed.");
2521                sdsfree(nodekey);
2522                decrRefCount(o);
2523                zfree(lp);
2524                return NULL;
2525            }
2526
2527            unsigned char *first = lpFirst(lp);
2528            if (first == NULL) {
2529                /* Serialized listpacks should never be empty, since on
2530                 * deletion we should remove the radix tree key if the
2531                 * resulting listpack is empty. */
2532                rdbReportCorruptRDB("Empty listpack inside stream");
2533                sdsfree(nodekey);
2534                decrRefCount(o);
2535                zfree(lp);
2536                return NULL;
2537            }
2538
2539            /* Insert the key in the radix tree. */
2540            int retval = raxTryInsert(s->rax,
2541                (unsigned char*)nodekey,sizeof(streamID),lp,NULL);
2542            sdsfree(nodekey);
2543            if (!retval) {
2544                rdbReportCorruptRDB("Listpack re-added with existing key");
2545                decrRefCount(o);
2546                zfree(lp);
2547                return NULL;
2548            }
2549        }
2550        /* Load total number of items inside the stream. */
2551        s->length = rdbLoadLen(rdb,NULL);
2552
2553        /* Load the last entry ID. */
2554        s->last_id.ms = rdbLoadLen(rdb,NULL);
2555        s->last_id.seq = rdbLoadLen(rdb,NULL);
2556        
2557        if (rdbtype >= RDB_TYPE_STREAM_LISTPACKS_2) {
2558            /* Load the first entry ID. */
2559            s->first_id.ms = rdbLoadLen(rdb,NULL);
2560            s->first_id.seq = rdbLoadLen(rdb,NULL);
2561
2562            /* Load the maximal deleted entry ID. */
2563            s->max_deleted_entry_id.ms = rdbLoadLen(rdb,NULL);
2564            s->max_deleted_entry_id.seq = rdbLoadLen(rdb,NULL);
2565
2566            /* Load the offset. */
2567            s->entries_added = rdbLoadLen(rdb,NULL);
2568        } else {
2569            /* During migration the offset can be initialized to the stream's
2570             * length. At this point, we also don't care about tombstones
2571             * because CG offsets will be later initialized as well. */
2572            s->max_deleted_entry_id.ms = 0;
2573            s->max_deleted_entry_id.seq = 0;
2574            s->entries_added = s->length;
2575            
2576            /* Since the rax is already loaded, we can find the first entry's
2577             * ID. */ 
2578            streamGetEdgeID(s,1,1,&s->first_id);
2579        }
2580
2581        if (rioGetReadError(rdb)) {
2582            rdbReportReadError("Stream object metadata loading failed.");
2583            decrRefCount(o);
2584            return NULL;
2585        }
2586
2587        if (s->length && !raxSize(s->rax)) {
2588            rdbReportCorruptRDB("Stream length inconsistent with rax entries");
2589            decrRefCount(o);
2590            return NULL;
2591        }
2592
2593        /* Consumer groups loading */
2594        uint64_t cgroups_count = rdbLoadLen(rdb,NULL);
2595        if (cgroups_count == RDB_LENERR) {
2596            rdbReportReadError("Stream cgroup count loading failed.");
2597            decrRefCount(o);
2598            return NULL;
2599        }
2600        while(cgroups_count--) {
2601            /* Get the consumer group name and ID. We can then create the
2602             * consumer group ASAP and populate its structure as
2603             * we read more data. */
2604            streamID cg_id;
2605            sds cgname = rdbGenericLoadStringObject(rdb,RDB_LOAD_SDS,NULL);
2606            if (cgname == NULL) {
2607                rdbReportReadError(
2608                    "Error reading the consumer group name from Stream");
2609                decrRefCount(o);
2610                return NULL;
2611            }
2612
2613            cg_id.ms = rdbLoadLen(rdb,NULL);
2614            cg_id.seq = rdbLoadLen(rdb,NULL);
2615            if (rioGetReadError(rdb)) {
2616                rdbReportReadError("Stream cgroup ID loading failed.");
2617                sdsfree(cgname);
2618                decrRefCount(o);
2619                return NULL;
2620            }
2621            
2622            /* Load group offset. */
2623            uint64_t cg_offset;
2624            if (rdbtype >= RDB_TYPE_STREAM_LISTPACKS_2) {
2625                cg_offset = rdbLoadLen(rdb,NULL);
2626                if (rioGetReadError(rdb)) {
2627                    rdbReportReadError("Stream cgroup offset loading failed.");
2628                    sdsfree(cgname);
2629                    decrRefCount(o);
2630                    return NULL;
2631                }
2632            } else {
2633                cg_offset = streamEstimateDistanceFromFirstEverEntry(s,&cg_id);
2634            }
2635
2636            streamCG *cgroup = streamCreateCG(s,cgname,sdslen(cgname),&cg_id,cg_offset);
2637            if (cgroup == NULL) {
2638                rdbReportCorruptRDB("Duplicated consumer group name %s",
2639                                         cgname);
2640                decrRefCount(o);
2641                sdsfree(cgname);
2642                return NULL;
2643            }
2644            sdsfree(cgname);
2645
2646            /* Load the global PEL for this consumer group, however we'll
2647             * not yet populate the NACK structures with the message
2648             * owner, since consumers for this group and their messages will
2649             * be read as a next step. So for now leave them not resolved
2650             * and later populate it. */
2651            uint64_t pel_size = rdbLoadLen(rdb,NULL);
2652            if (pel_size == RDB_LENERR) {
2653                rdbReportReadError("Stream PEL size loading failed.");
2654                decrRefCount(o);
2655                return NULL;
2656            }
2657            while(pel_size--) {
2658                unsigned char rawid[sizeof(streamID)];
2659                if (rioRead(rdb,rawid,sizeof(rawid)) == 0) {
2660                    rdbReportReadError("Stream PEL ID loading failed.");
2661                    decrRefCount(o);
2662                    return NULL;
2663                }
2664                streamNACK *nack = streamCreateNACK(NULL);
2665                nack->delivery_time = rdbLoadMillisecondTime(rdb,RDB_VERSION);
2666                nack->delivery_count = rdbLoadLen(rdb,NULL);
2667                if (rioGetReadError(rdb)) {
2668                    rdbReportReadError("Stream PEL NACK loading failed.");
2669                    decrRefCount(o);
2670                    streamFreeNACK(nack);
2671                    return NULL;
2672                }
2673                if (!raxTryInsert(cgroup->pel,rawid,sizeof(rawid),nack,NULL)) {
2674                    rdbReportCorruptRDB("Duplicated global PEL entry "
2675                                            "loading stream consumer group");
2676                    decrRefCount(o);
2677                    streamFreeNACK(nack);
2678                    return NULL;
2679                }
2680            }
2681
2682            /* Now that we loaded our global PEL, we need to load the
2683             * consumers and their local PELs. */
2684            uint64_t consumers_num = rdbLoadLen(rdb,NULL);
2685            if (consumers_num == RDB_LENERR) {
2686                rdbReportReadError("Stream consumers num loading failed.");
2687                decrRefCount(o);
2688                return NULL;
2689            }
2690            while(consumers_num--) {
2691                sds cname = rdbGenericLoadStringObject(rdb,RDB_LOAD_SDS,NULL);
2692                if (cname == NULL) {
2693                    rdbReportReadError(
2694                        "Error reading the consumer name from Stream group.");
2695                    decrRefCount(o);
2696                    return NULL;
2697                }
2698                streamConsumer *consumer = streamCreateConsumer(cgroup,cname,NULL,0,
2699                                                        SCC_NO_NOTIFY|SCC_NO_DIRTIFY);
2700                sdsfree(cname);
2701                if (!consumer) {
2702                    rdbReportCorruptRDB("Duplicate stream consumer detected.");
2703                    decrRefCount(o);
2704                    return NULL;
2705                }
2706
2707                consumer->seen_time = rdbLoadMillisecondTime(rdb,RDB_VERSION);
2708                if (rioGetReadError(rdb)) {
2709                    rdbReportReadError("Stream short read reading seen time.");
2710                    decrRefCount(o);
2711                    return NULL;
2712                }
2713
2714                if (rdbtype >= RDB_TYPE_STREAM_LISTPACKS_3) {
2715                    consumer->active_time = rdbLoadMillisecondTime(rdb,RDB_VERSION);
2716                    if (rioGetReadError(rdb)) {
2717                        rdbReportReadError("Stream short read reading active time.");
2718                        decrRefCount(o);
2719                        return NULL;
2720                    }
2721                } else {
2722                    /* That's the best estimate we got */
2723                    consumer->active_time = consumer->seen_time;
2724                }
2725
2726                /* Load the PEL about entries owned by this specific
2727                 * consumer. */
2728                pel_size = rdbLoadLen(rdb,NULL);
2729                if (pel_size == RDB_LENERR) {
2730                    rdbReportReadError(
2731                        "Stream consumer PEL num loading failed.");
2732                    decrRefCount(o);
2733                    return NULL;
2734                }
2735                while(pel_size--) {
2736                    unsigned char rawid[sizeof(streamID)];
2737                    if (rioRead(rdb,rawid,sizeof(rawid)) == 0) {
2738                        rdbReportReadError(
2739                            "Stream short read reading PEL streamID.");
2740                        decrRefCount(o);
2741                        return NULL;
2742                    }
2743                    streamNACK *nack = raxFind(cgroup->pel,rawid,sizeof(rawid));
2744                    if (nack == raxNotFound) {
2745                        rdbReportCorruptRDB("Consumer entry not found in "
2746                                                "group global PEL");
2747                        decrRefCount(o);
2748                        return NULL;
2749                    }
2750
2751                    /* Set the NACK consumer, that was left to NULL when
2752                     * loading the global PEL. Then set the same shared
2753                     * NACK structure also in the consumer-specific PEL. */
2754                    nack->consumer = consumer;
2755                    if (!raxTryInsert(consumer->pel,rawid,sizeof(rawid),nack,NULL)) {
2756                        rdbReportCorruptRDB("Duplicated consumer PEL entry "
2757                                                " loading a stream consumer "
2758                                                "group");
2759                        decrRefCount(o);
2760                        streamFreeNACK(nack);
2761                        return NULL;
2762                    }
2763                }
2764            }
2765
2766            /* Verify that each PEL eventually got a consumer assigned to it. */
2767            if (deep_integrity_validation) {
2768                raxIterator ri_cg_pel;
2769                raxStart(&ri_cg_pel,cgroup->pel);
2770                raxSeek(&ri_cg_pel,"^",NULL,0);
2771                while(raxNext(&ri_cg_pel)) {
2772                    streamNACK *nack = ri_cg_pel.data;
2773                    if (!nack->consumer) {
2774                        raxStop(&ri_cg_pel);
2775                        rdbReportCorruptRDB("Stream CG PEL entry without consumer");
2776                        decrRefCount(o);
2777                        return NULL;
2778                    }
2779                }
2780                raxStop(&ri_cg_pel);
2781            }
2782        }
2783    } else if (rdbtype == RDB_TYPE_MODULE_PRE_GA) {
2784            rdbReportCorruptRDB("Pre-release module format not supported");
2785            return NULL;
2786    } else if (rdbtype == RDB_TYPE_MODULE_2) {
2787        uint64_t moduleid = rdbLoadLen(rdb,NULL);
2788        if (rioGetReadError(rdb)) {
2789            rdbReportReadError("Short read module id");
2790            return NULL;
2791        }
2792        moduleType *mt = moduleTypeLookupModuleByID(moduleid);
2793
2794        if (rdbCheckMode) {
2795            char name[10];
2796            moduleTypeNameByID(name,moduleid);
2797            return rdbLoadCheckModuleValue(rdb,name);
2798        }
2799
2800        if (mt == NULL) {
2801            char name[10];
2802            moduleTypeNameByID(name,moduleid);
2803            rdbReportCorruptRDB("The RDB file contains module data I can't load: no matching module type '%s'", name);
2804            return NULL;
2805        }
2806        RedisModuleIO io;
2807        robj keyobj;
2808        initStaticStringObject(keyobj,key);
2809        moduleInitIOContext(io,mt,rdb,&keyobj,dbid);
2810        /* Call the rdb_load method of the module providing the 10 bit
2811         * encoding version in the lower 10 bits of the module ID. */
2812        void *ptr = mt->rdb_load(&io,moduleid&1023);
2813        if (io.ctx) {
2814            moduleFreeContext(io.ctx);
2815            zfree(io.ctx);
2816        }
2817
2818        /* Module v2 serialization has an EOF mark at the end. */
2819        uint64_t eof = rdbLoadLen(rdb,NULL);
2820        if (eof == RDB_LENERR) {
2821            if (ptr) {
2822                o = createModuleObject(mt,ptr); /* creating just in order to easily destroy */
2823                decrRefCount(o);
2824            }
2825            return NULL;
2826        }
2827        if (eof != RDB_MODULE_OPCODE_EOF) {
2828            rdbReportCorruptRDB("The RDB file contains module data for the module '%s' that is not terminated by "
2829                                "the proper module value EOF marker", moduleTypeModuleName(mt));
2830            if (ptr) {
2831                o = createModuleObject(mt,ptr); /* creating just in order to easily destroy */
2832                decrRefCount(o);
2833            }
2834            return NULL;
2835        }
2836
2837        if (ptr == NULL) {
2838            rdbReportCorruptRDB("The RDB file contains module data for the module type '%s', that the responsible "
2839                                "module is not able to load. Check for modules log above for additional clues.",
2840                                moduleTypeModuleName(mt));
2841            return NULL;
2842        }
2843        o = createModuleObject(mt,ptr);
2844    } else {
2845        rdbReportReadError("Unknown RDB encoding type %d",rdbtype);
2846        return NULL;
2847    }
2848    if (error) *error = 0;
2849    return o;
2850
2851emptykey:
2852    if (error) *error = RDB_LOAD_ERR_EMPTY_KEY;
2853    return NULL;
2854}

3024    uint64_t dbid = 0;

3034    buf[9] = '\0';
3035    if (memcmp(buf,"REDIS",5) != 0) {
3036        serverLog(LL_WARNING,"Wrong signature trying to load DB from file");
3037        return C_ERR;
3038    }

3040    if (rdbver < 1 || rdbver > RDB_VERSION) {
3041        serverLog(LL_WARNING,"Can't handle RDB format version %d",rdbver);

3046    long long lru_idle = -1, lfu_freq = -1, expiretime = -1, now = mstime();

3059             * to load. Note that after loading an expire we need to
3060             * load the actual type, and continue. */
3061            expiretime = rdbLoadTime(rdb);
3062            expiretime *= 1000;
3063            if (rioGetReadError(rdb)) goto eoferr;
3064            continue; /* Read next opcode. */
3065        } else if (type == RDB_OPCODE_EXPIRETIME_MS) {
3066            /* EXPIRETIME_MS: milliseconds precision expire times introduced
3067             * with RDB v3. Like EXPIRETIME but no with more precision. */

3069            if (rioGetReadError(rdb)) goto eoferr;
3070            continue; /* Read next opcode. */
3071        } else if (type == RDB_OPCODE_FREQ) {
3072            /* FREQ: LFU frequency. */
3073            uint8_t byte;
3074            if (rioRead(rdb,&byte,1) == 0) goto eoferr;
3075            lfu_freq = byte;
3076            continue; /* Read next opcode. */
3077        } else if (type == RDB_OPCODE_IDLE) {
3078            /* IDLE: LRU idle time. */
3079            uint64_t qword;
3080            if ((qword = rdbLoadLen(rdb,NULL)) == RDB_LENERR) goto eoferr;
3081            lru_idle = qword;
3082            continue; /* Read next opcode. */
3083        } else if (type == RDB_OPCODE_EOF) {
3084            /* EOF: End of file, exit the main loop. */
3085            break;

3095            }

3097            continue; /* Read next opcode. */

3100             * selected data base, in order to avoid useless rehashing. */

3131            } else if (!strcasecmp(auxkey->ptr,"repl-id")) {

3138            } else if (!strcasecmp(auxkey->ptr,"lua")) {
3139                /* Won't load the script back in memory anymore. */
3140            } else if (!strcasecmp(auxkey->ptr,"redis-ver")) {
3141                serverLog(LL_NOTICE,"Loading RDB produced by version %s",
3142                    (char*)auxval->ptr);

3145                if (age < 0) age = 0;
3146                serverLog(LL_NOTICE,"RDB age %ld seconds",
3147                    (unsigned long) age);
3148            } else if (!strcasecmp(auxkey->ptr,"used-mem")) {
3149                long long usedmem = strtoll(auxval->ptr,NULL,10);
3150                serverLog(LL_NOTICE,"RDB memory usage when created %.2f Mb",
3151                    (double) usedmem / (1024*1024));
3152                server.loading_rdb_used_mem = usedmem;
3153            } else if (!strcasecmp(auxkey->ptr,"aof-preamble")) {
3154                long long haspreamble = strtoll(auxval->ptr,NULL,10);
3155                if (haspreamble) serverLog(LL_NOTICE,"RDB has an AOF tail");
3156            } else if (!strcasecmp(auxkey->ptr, "aof-base")) {
3157                long long isbase = strtoll(auxval->ptr, NULL, 10);
3158                if (isbase) serverLog(LL_NOTICE, "RDB is base AOF");
3159            } else if (!strcasecmp(auxkey->ptr,"redis-bits")) {
3160                /* Just ignored. */
3161            } else {

3163                 * contract. */
3164                serverLog(LL_DEBUG,"Unrecognized RDB AUX field: '%s'",

3169            decrRefCount(auxval);
3170            continue; /* Read type again. */
3171        } else if (type == RDB_OPCODE_MODULE_AUX) {

3173             * Such data can be potentially be stored both before and after the
3174             * RDB keys-values section. */

3176            int when_opcode = rdbLoadLen(rdb,NULL);
3177            int when = rdbLoadLen(rdb,NULL);
3178            if (rioGetReadError(rdb)) goto eoferr;
3179            if (when_opcode != RDB_MODULE_OPCODE_UINT) {
3180                rdbReportReadError("bad when_opcode");
3181                goto eoferr;
3182            }

3184            char name[10];

3186

3188                /* Unknown module. */
3189                serverLog(LL_WARNING,"The RDB file contains AUX module data I can't load: no matching module '%s'", name);
3190                exit(1);
3191            } else if (!rdbCheckMode && mt != NULL) {
3192                if (!mt->aux_load) {
3193                    /* Module doesn't support AUX. */
3194                    serverLog(LL_WARNING,"The RDB file contains module AUX data, but the module '%s' doesn't seem to support it.", name);
3195                    exit(1);
3196                }
3197
3198                RedisModuleIO io;
3199                moduleInitIOContext(io,mt,rdb,NULL,-1);
3200                /* Call the rdb_load method of the module providing the 10 bit
3201                 * encoding version in the lower 10 bits of the module ID. */

3203                if (io.ctx) {
3204                    moduleFreeContext(io.ctx);
3205                    zfree(io.ctx);
3206                }

3208                    moduleTypeNameByID(name,moduleid);
3209                    serverLog(LL_WARNING,"The RDB file contains module AUX data for the module type '%s', that the responsible module is not able to load. Check for modules log above for additional clues.", name);

3211                }

3213                if (eof != RDB_MODULE_OPCODE_EOF) {
3214                    serverLog(LL_WARNING,"The RDB file contains module AUX data for the module '%s' that is not terminated by the proper module value EOF marker", name);

3216                }
3217                continue;
3218            } else {
3219                /* RDB check mode. */
3220                robj *aux = rdbLoadCheckModuleValue(rdb,name);
3221                decrRefCount(aux);
3222                continue; /* Read next opcode. */
3223            }
3224        } else if (type == RDB_OPCODE_FUNCTION_PRE_GA) {
3225            rdbReportCorruptRDB("Pre-release function format not supported.");
3226            exit(1);
3227        } else if (type == RDB_OPCODE_FUNCTION2) {
3228            sds err = NULL;
3229            if (rdbFunctionLoad(rdb, rdbver, rdb_loading_ctx->functions_lib_ctx, rdbflags, &err) != C_OK) {
3230                serverLog(LL_WARNING,"Failed loading library, %s", err);
3231                sdsfree(err);

3292                    /* This flag is useful for DEBUG RELOAD special modes.
3293                     * When it's set we allow new keys to replace the current
3294                     * keys with the same name. */
3295                    dbSyncDelete(db,&keyobj);
3296                    dbAddRDBLoad(db,key,val);
3297                } else {

3322         * opcodes before the key, so that we start from scratch again. */
3323        expiretime = -1;
3324        lfu_freq = -1;

3326    }

3337                serverLog(LL_WARNING,"Wrong RDB checksum expected: (%llx) but "
3338                    "got (%llx). Aborting now.",
3339                        (unsigned long long)expected,

3342                return C_ERR;
3343            }
3344        }
3345    }
3346
3347    if (empty_keys_skipped) {

3349            "Done loading RDB, keys loaded: %lld, keys expired: %lld, empty keys skipped: %lld.",

3353            "Done loading RDB, keys loaded: %lld, keys expired: %lld.",
3354                server.rdb_last_load_keys_loaded, server.rdb_last_load_keys_expired);

3359     * this will in turn either abort Redis in most cases, or if we are loading
3360     * the RDB file from a socket during initial SYNC (diskless replica mode),
3361     * we'll report the error to the caller, so that we can retry. */
3362eoferr:

3364        "Short read or OOM loading DB. Unrecoverable error, aborting now.");

3367}

3406    }

3502/* Spawn an RDB child that writes the RDB to the sockets of the slaves
3503 * that are currently in SLAVE_STATE_WAIT_BGSAVE_START state. */
3504int rdbSaveToSlavesSockets(int req, rdbSaveInfo *rsi) {
3505    listNode *ln;
3506    listIter li;
3507    pid_t childpid;
3508    int pipefds[2], rdb_pipe_write, safe_to_exit_pipe;
3509
3510    if (hasActiveChildProcess()) return C_ERR;
3511
3512    /* Even if the previous fork child exited, don't start a new one until we
3513     * drained the pipe. */
3514    if (server.rdb_pipe_conns) return C_ERR;
3515
3516    /* Before to fork, create a pipe that is used to transfer the rdb bytes to
3517     * the parent, we can't let it write directly to the sockets, since in case
3518     * of TLS we must let the parent handle a continuous TLS state when the
3519     * child terminates and parent takes over. */
3520    if (anetPipe(pipefds, O_NONBLOCK, 0) == -1) return C_ERR;
3521    server.rdb_pipe_read = pipefds[0]; /* read end */
3522    rdb_pipe_write = pipefds[1]; /* write end */
3523
3524    /* create another pipe that is used by the parent to signal to the child
3525     * that it can exit. */
3526    if (anetPipe(pipefds, 0, 0) == -1) {
3527        close(rdb_pipe_write);
3528        close(server.rdb_pipe_read);
3529        return C_ERR;
3530    }
3531    safe_to_exit_pipe = pipefds[0]; /* read end */
3532    server.rdb_child_exit_pipe = pipefds[1]; /* write end */
3533
3534    /* Collect the connections of the replicas we want to transfer
3535     * the RDB to, which are i WAIT_BGSAVE_START state. */
3536    server.rdb_pipe_conns = zmalloc(sizeof(connection *)*listLength(server.slaves));
3537    server.rdb_pipe_numconns = 0;
3538    server.rdb_pipe_numconns_writing = 0;
3539    listRewind(server.slaves,&li);
3540    while((ln = listNext(&li))) {
3541        client *slave = ln->value;
3542        if (slave->replstate == SLAVE_STATE_WAIT_BGSAVE_START) {
3543            /* Check slave has the exact requirements */
3544            if (slave->slave_req != req)
3545                continue;
3546            server.rdb_pipe_conns[server.rdb_pipe_numconns++] = slave->conn;
3547            replicationSetupSlaveForFullResync(slave,getPsyncInitialOffset());
3548        }
3549    }
3550
3551    /* Create the child process. */
3552    if ((childpid = redisFork(CHILD_TYPE_RDB)) == 0) {
3553        /* Child */
3554        int retval, dummy;
3555        rio rdb;
3556
3557        rioInitWithFd(&rdb,rdb_pipe_write);
3558
3559        /* Close the reading part, so that if the parent crashes, the child will
3560         * get a write error and exit. */
3561        close(server.rdb_pipe_read);
3562
3563        redisSetProcTitle("redis-rdb-to-slaves");
3564        redisSetCpuAffinity(server.bgsave_cpulist);
3565
3566        retval = rdbSaveRioWithEOFMark(req,&rdb,NULL,rsi);
3567        if (retval == C_OK && rioFlush(&rdb) == 0)
3568            retval = C_ERR;
3569
3570        if (retval == C_OK) {
3571            sendChildCowInfo(CHILD_INFO_TYPE_RDB_COW_SIZE, "RDB");
3572        }
3573
3574        rioFreeFd(&rdb);
3575        /* wake up the reader, tell it we're done. */
3576        close(rdb_pipe_write);
3577        close(server.rdb_child_exit_pipe); /* close write end so that we can detect the close on the parent. */
3578        /* hold exit until the parent tells us it's safe. we're not expecting
3579         * to read anything, just get the error when the pipe is closed. */
3580        dummy = read(safe_to_exit_pipe, pipefds, 1);
3581        UNUSED(dummy);
3582        exitFromChild((retval == C_OK) ? 0 : 1);
3583    } else {
3584        /* Parent */
3585        if (childpid == -1) {
3586            serverLog(LL_WARNING,"Can't save in background: fork: %s",
3587                strerror(errno));
3588
3589            /* Undo the state change. The caller will perform cleanup on
3590             * all the slaves in BGSAVE_START state, but an early call to
3591             * replicationSetupSlaveForFullResync() turned it into BGSAVE_END */
3592            listRewind(server.slaves,&li);
3593            while((ln = listNext(&li))) {
3594                client *slave = ln->value;
3595                if (slave->replstate == SLAVE_STATE_WAIT_BGSAVE_END) {
3596                    slave->replstate = SLAVE_STATE_WAIT_BGSAVE_START;
3597                }
3598            }
3599            close(rdb_pipe_write);
3600            close(server.rdb_pipe_read);
3601            zfree(server.rdb_pipe_conns);
3602            server.rdb_pipe_conns = NULL;
3603            server.rdb_pipe_numconns = 0;
3604            server.rdb_pipe_numconns_writing = 0;
3605        } else {
3606            serverLog(LL_NOTICE,"Background RDB transfer started by pid %ld",
3607                (long) childpid);
3608            server.rdb_save_time_start = time(NULL);
3609            server.rdb_child_type = RDB_CHILD_TYPE_SOCKET;
3610            close(rdb_pipe_write); /* close write in parent so that it can detect the close on the child. */
3611            if (aeCreateFileEvent(server.el, server.rdb_pipe_read, AE_READABLE, rdbPipeReadHandler,NULL) == AE_ERR) {
3612                serverPanic("Unrecoverable error creating server.rdb_pipe_read file event.");
3613            }
3614        }
3615        close(safe_to_exit_pipe);
3616        return (childpid == -1) ? C_ERR : C_OK;
3617    }
3618    return C_OK; /* Unreached. */
3619}

3675/* Populate the rdbSaveInfo structure used to persist the replication
3676 * information inside the RDB file. Currently the structure explicitly
3677 * contains just the currently selected DB from the master stream, however
3678 * if the rdbSave*() family functions receive a NULL rsi structure also
3679 * the Replication ID/offset is not saved. The function populates 'rsi'
3680 * that is normally stack-allocated in the caller, returns the populated
3681 * pointer if the instance has a valid master client, otherwise NULL
3682 * is returned, and the RDB saving will not persist any replication related
3683 * information. */
3684rdbSaveInfo *rdbPopulateSaveInfo(rdbSaveInfo *rsi) {
3685    rdbSaveInfo rsi_init = RDB_SAVE_INFO_INIT;
3686    *rsi = rsi_init;
3687
3688    /* If the instance is a master, we can populate the replication info
3689     * only when repl_backlog is not NULL. If the repl_backlog is NULL,
3690     * it means that the instance isn't in any replication chains. In this
3691     * scenario the replication info is useless, because when a slave
3692     * connects to us, the NULL repl_backlog will trigger a full
3693     * synchronization, at the same time we will use a new replid and clear
3694     * replid2. */
3695    if (!server.masterhost && server.repl_backlog) {
3696        /* Note that when server.slaveseldb is -1, it means that this master
3697         * didn't apply any write commands after a full synchronization.
3698         * So we can let repl_stream_db be 0, this allows a restarted slave
3699         * to reload replication ID/offset, it's safe because the next write
3700         * command must generate a SELECT statement. */
3701        rsi->repl_stream_db = server.slaveseldb == -1 ? 0 : server.slaveseldb;
3702        return rsi;
3703    }
3704
3705    /* If the instance is a slave we need a connected master
3706     * in order to fetch the currently selected DB. */
3707    if (server.master) {
3708        rsi->repl_stream_db = server.master->db->id;
3709        return rsi;
3710    }
3711
3712    /* If we have a cached master we can use it in order to populate the
3713     * replication selected DB info inside the RDB file: the slave can
3714     * increment the master_repl_offset only from data arriving from the
3715     * master, so if we are disconnected the offset in the cached master
3716     * is valid. */
3717    if (server.cached_master) {
3718        rsi->repl_stream_db = server.cached_master->db->id;
3719        return rsi;
3720    }
3721    return NULL;
