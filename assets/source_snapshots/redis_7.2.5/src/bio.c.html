1/* Background I/O service for Redis.
2 *
3 * This file implements operations that we need to perform in the background.
4 * Currently there is only a single operation, that is a background close(2)
5 * system call. This is needed as when the process is the last owner of a
6 * reference to a file closing it means unlinking it, and the deletion of the
7 * file is slow, blocking the server.
8 *
9 * In the future we'll either continue implementing new things we need or
10 * we'll switch to libeio. However there are probably long term uses for this
11 * file as we may want to put here Redis specific background tasks (for instance
12 * it is not impossible that we'll need a non blocking FLUSHDB/FLUSHALL
13 * implementation).
14 *
15 * DESIGN
16 * ------
17 *
18 * The design is simple: We have a structure representing a job to perform,
19 * and several worker threads and job queues. Every job type is assigned to
20 * a specific worker thread, and a single worker may handle several different
21 * job types.
22 * Every thread waits for new jobs in its queue, and processes every job
23 * sequentially.
24 *
25 * Jobs handled by the same worker are guaranteed to be processed from the
26 * least-recently-inserted to the most-recently-inserted (older jobs processed
27 * first).
28 *
29 * Currently there is no way for the creator of the job to be notified about
30 * the completion of the operation, this will only be added when/if needed.

87/* This structure represents a background Job. It is only used locally to this
88 * file as the API does not expose the internals at all. */
89typedef union bio_job {
90    struct {
91        int type; /* Job-type tag. This needs to appear as the first element in all union members. */
92    } header;
93
94    /* Job specific arguments.*/
95    struct {
96        int type;
97        int fd; /* Fd for file based background jobs */
98        long long offset; /* A job-specific offset, if applicable */
99        unsigned need_fsync:1; /* A flag to indicate that a fsync is required before
100                                * the file is closed. */
101        unsigned need_reclaim_cache:1; /* A flag to indicate that reclaim cache is required before
102                                * the file is closed. */
103    } fd_args;
104
105    struct {
106        int type;
107        lazy_free_fn *free_fn; /* Function that will free the provided arguments */
108        void *free_args[]; /* List of arguments to be passed to the free function */
109    } free_args;
110} bio_job;

114/* Make sure we have enough stack to perform all the things we do in the
115 * main thread. */
116#define REDIS_THREAD_STACK_SIZE (1024*1024*4)
117
118/* Initialize the background system, spawning the thread. */
119void bioInit(void) {
120    pthread_attr_t attr;
121    pthread_t thread;
122    size_t stacksize;
123    unsigned long j;
124
125    /* Initialization of state vars and objects */
126    for (j = 0; j < BIO_WORKER_NUM; j++) {
127        pthread_mutex_init(&bio_mutex[j],NULL);
128        pthread_cond_init(&bio_newjob_cond[j],NULL);
129        bio_jobs[j] = listCreate();
130    }
131
132    /* Set the stack size as by default it may be small in some system */
133    pthread_attr_init(&attr);
134    pthread_attr_getstacksize(&attr,&stacksize);
135    if (!stacksize) stacksize = 1; /* The world is full of Solaris Fixes */
136    while (stacksize < REDIS_THREAD_STACK_SIZE) stacksize *= 2;
137    pthread_attr_setstacksize(&attr, stacksize);
138
139    /* Ready to spawn our threads. We use the single argument the thread
140     * function accepts in order to pass the job ID the thread is
141     * responsible for. */
142    for (j = 0; j < BIO_WORKER_NUM; j++) {
143        void *arg = (void*)(unsigned long) j;
144        if (pthread_create(&thread,&attr,bioProcessBackgroundJobs,arg) != 0) {
145            serverLog(LL_WARNING, "Fatal: Can't initialize Background Jobs. Error message: %s", strerror(errno));
146            exit(1);
147        }
148        bio_threads[j] = thread;
149    }
150}

152void bioSubmitJob(int type, bio_job *job) {
153    job->header.type = type;
154    unsigned long worker = bio_job_to_worker[type];
155    pthread_mutex_lock(&bio_mutex[worker]);
156    listAddNodeTail(bio_jobs[worker],job);
157    bio_jobs_counter[type]++;
158    pthread_cond_signal(&bio_newjob_cond[worker]);
159    pthread_mutex_unlock(&bio_mutex[worker]);
160}

162void bioCreateLazyFreeJob(lazy_free_fn free_fn, int arg_count, ...) {
163    va_list valist;
164    /* Allocate memory for the job structure and all required
165     * arguments */
166    bio_job *job = zmalloc(sizeof(*job) + sizeof(void *) * (arg_count));
167    job->free_args.free_fn = free_fn;
168
169    va_start(valist, arg_count);
170    for (int i = 0; i < arg_count; i++) {
171        job->free_args.free_args[i] = va_arg(valist, void *);
172    }
173    va_end(valist);
174    bioSubmitJob(BIO_LAZY_FREE, job);
175}

205void *bioProcessBackgroundJobs(void *arg) {
206    bio_job *job;
207    unsigned long worker = (unsigned long) arg;
208    sigset_t sigset;
209
210    /* Check that the worker is within the right interval. */
211    serverAssert(worker < BIO_WORKER_NUM);
212
213    redis_set_thread_title(bio_worker_title[worker]);
214
215    redisSetCpuAffinity(server.bio_cpulist);
216
217    makeThreadKillable();
218
219    pthread_mutex_lock(&bio_mutex[worker]);
220    /* Block SIGALRM so we are sure that only the main thread will
221     * receive the watchdog signal. */
222    sigemptyset(&sigset);
223    sigaddset(&sigset, SIGALRM);
224    if (pthread_sigmask(SIG_BLOCK, &sigset, NULL))
225        serverLog(LL_WARNING,
226            "Warning: can't mask SIGALRM in bio.c thread: %s", strerror(errno));
227
228    while(1) {
229        listNode *ln;
230
231        /* The loop always starts with the lock hold. */
232        if (listLength(bio_jobs[worker]) == 0) {
233            pthread_cond_wait(&bio_newjob_cond[worker], &bio_mutex[worker]);
234            continue;
235        }
236        /* Get the job from the queue. */
237        ln = listFirst(bio_jobs[worker]);
238        job = ln->value;
239        /* It is now possible to unlock the background system as we know have
240         * a stand alone job structure to process.*/
241        pthread_mutex_unlock(&bio_mutex[worker]);
242
243        /* Process the job accordingly to its type. */
244        int job_type = job->header.type;
245
246        if (job_type == BIO_CLOSE_FILE) {
247            if (job->fd_args.need_fsync &&
248                redis_fsync(job->fd_args.fd) == -1 &&
249                errno != EBADF && errno != EINVAL)
250            {
251                serverLog(LL_WARNING, "Fail to fsync the AOF file: %s",strerror(errno));
252            }
253            if (job->fd_args.need_reclaim_cache) {
254                if (reclaimFilePageCache(job->fd_args.fd, 0, 0) == -1) {
255                    serverLog(LL_NOTICE,"Unable to reclaim page cache: %s", strerror(errno));
256                }
257            }
258            close(job->fd_args.fd);
259        } else if (job_type == BIO_AOF_FSYNC || job_type == BIO_CLOSE_AOF) {
260            /* The fd may be closed by main thread and reused for another
261             * socket, pipe, or file. We just ignore these errno because
262             * aof fsync did not really fail. */
263            if (redis_fsync(job->fd_args.fd) == -1 &&
264                errno != EBADF && errno != EINVAL)
265            {
266                int last_status;
267                atomicGet(server.aof_bio_fsync_status,last_status);
268                atomicSet(server.aof_bio_fsync_status,C_ERR);
269                atomicSet(server.aof_bio_fsync_errno,errno);
270                if (last_status == C_OK) {
271                    serverLog(LL_WARNING,
272                        "Fail to fsync the AOF file: %s",strerror(errno));
273                }
274            } else {
275                atomicSet(server.aof_bio_fsync_status,C_OK);
276                atomicSet(server.fsynced_reploff_pending, job->fd_args.offset);
277            }
278
279            if (job->fd_args.need_reclaim_cache) {
280                if (reclaimFilePageCache(job->fd_args.fd, 0, 0) == -1) {
281                    serverLog(LL_NOTICE,"Unable to reclaim page cache: %s", strerror(errno));
282                }
283            }
284            if (job_type == BIO_CLOSE_AOF)
285                close(job->fd_args.fd);
286        } else if (job_type == BIO_LAZY_FREE) {
287            job->free_args.free_fn(job->free_args.free_args);
288        } else {
289            serverPanic("Wrong job type in bioProcessBackgroundJobs().");
290        }
291        zfree(job);
292
293        /* Lock again before reiterating the loop, if there are no longer
294         * jobs to process we'll block again in pthread_cond_wait(). */
295        pthread_mutex_lock(&bio_mutex[worker]);
296        listDelNode(bio_jobs[worker], ln);
297        bio_jobs_counter[job_type]--;
298        pthread_cond_signal(&bio_newjob_cond[worker]);
299    }
300}

302/* Return the number of pending jobs of the specified type. */
303unsigned long bioPendingJobsOfType(int type) {
304    unsigned int worker = bio_job_to_worker[type];
305
306    pthread_mutex_lock(&bio_mutex[worker]);
307    unsigned long val = bio_jobs_counter[type];
308    pthread_mutex_unlock(&bio_mutex[worker]);
309
310    return val;
311}

313/* Wait for the job queue of the worker for jobs of specified type to become empty. */
314void bioDrainWorker(int job_type) {
315    unsigned long worker = bio_job_to_worker[job_type];
316
317    pthread_mutex_lock(&bio_mutex[worker]);
318    while (listLength(bio_jobs[worker]) > 0) {
319        pthread_cond_wait(&bio_newjob_cond[worker], &bio_mutex[worker]);
320    }
321    pthread_mutex_unlock(&bio_mutex[worker]);
322}

324/* Kill the running bio threads in an unclean way. This function should be
325 * used only when it's critical to stop the threads for some reason.
326 * Currently Redis does this only on crash (for instance on SIGSEGV) in order
327 * to perform a fast memory check without other threads messing with memory. */
328void bioKillThreads(void) {
329    int err;
330    unsigned long j;
331
332    for (j = 0; j < BIO_WORKER_NUM; j++) {
333        if (bio_threads[j] == pthread_self()) continue;
334        if (bio_threads[j] && pthread_cancel(bio_threads[j]) == 0) {
335            if ((err = pthread_join(bio_threads[j],NULL)) != 0) {
336                serverLog(LL_WARNING,
337                    "Bio worker thread #%lu can not be joined: %s",
338                        j, strerror(err));
339            } else {
340                serverLog(LL_WARNING,
341                    "Bio worker thread #%lu terminated",j);
342            }
343        }
344    }
