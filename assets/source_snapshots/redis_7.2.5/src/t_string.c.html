40static int checkStringLength(client *c, long long size, long long append) {
41    if (mustObeyClient(c))
42        return C_OK;
43    /* 'uint64_t' cast is there just to prevent undefined behavior on overflow */
44    long long total = (uint64_t)size + append;
45    /* Test configured max-bulk-len represending a limit of the biggest string object,
46     * and also test for overflow. */
47    if (total > server.proto_max_bulk_len || total < size || total < append) {
48        addReplyError(c,"string exceeds maximum allowed size (proto-max-bulk-len)");
49        return C_ERR;
50    }
51    return C_OK;
52}

54/* The setGenericCommand() function implements the SET operation with different
55 * options and variants. This function is called in order to implement the
56 * following commands: SET, SETEX, PSETEX, SETNX, GETSET.
57 *
58 * 'flags' changes the behavior of the command (NX, XX or GET, see below).
59 *
60 * 'expire' represents an expire to set in form of a Redis object as passed
61 * by the user. It is interpreted according to the specified 'unit'.
62 *
63 * 'ok_reply' and 'abort_reply' is what the function will reply to the client
64 * if the operation is performed, or when it is not because of NX or
65 * XX flags.
66 *
67 * If ok_reply is NULL "+OK" is used.
68 * If abort_reply is NULL, "$-1" is used. */
69
70#define OBJ_NO_FLAGS 0
71#define OBJ_SET_NX (1<<0)          /* Set if key not exists. */
72#define OBJ_SET_XX (1<<1)          /* Set if key exists. */
73#define OBJ_EX (1<<2)              /* Set if time in seconds is given */
74#define OBJ_PX (1<<3)              /* Set if time in ms in given */
75#define OBJ_KEEPTTL (1<<4)         /* Set and keep the ttl */
76#define OBJ_SET_GET (1<<5)         /* Set if want to get key before set */
77#define OBJ_EXAT (1<<6)            /* Set if timestamp in second is given */
78#define OBJ_PXAT (1<<7)            /* Set if timestamp in ms is given */
79#define OBJ_PERSIST (1<<8)         /* Set if we need to remove the ttl */
80

84void setGenericCommand(client *c, int flags, robj *key, robj *val, robj *expire, int unit, robj *ok_reply, robj *abort_reply) {
85    long long milliseconds = 0; /* initialized to avoid any harmness warning */
86    int found = 0;
87    int setkey_flags = 0;
88
89    if (expire && getExpireMillisecondsOrReply(c, expire, flags, unit, &milliseconds) != C_OK) {
90        return;
91    }
92
93    if (flags & OBJ_SET_GET) {
94        if (getGenericCommand(c) == C_ERR) return;
95    }
96
97    found = (lookupKeyWrite(c->db,key) != NULL);
98
99    if ((flags & OBJ_SET_NX && found) ||
100        (flags & OBJ_SET_XX && !found))
101    {
102        if (!(flags & OBJ_SET_GET)) {
103            addReply(c, abort_reply ? abort_reply : shared.null[c->resp]);
104        }
105        return;
106    }
107
108    /* When expire is not NULL, we avoid deleting the TTL so it can be updated later instead of being deleted and then created again. */
109    setkey_flags |= ((flags & OBJ_KEEPTTL) || expire) ? SETKEY_KEEPTTL : 0;
110    setkey_flags |= found ? SETKEY_ALREADY_EXIST : SETKEY_DOESNT_EXIST;
111
112    setKey(c,c->db,key,val,setkey_flags);
113    server.dirty++;
114    notifyKeyspaceEvent(NOTIFY_STRING,"set",key,c->db->id);
115
116    if (expire) {
117        setExpire(c,c->db,key,milliseconds);
118        /* Propagate as SET Key Value PXAT millisecond-timestamp if there is
119         * EX/PX/EXAT flag. */
120        if (!(flags & OBJ_PXAT)) {
121            robj *milliseconds_obj = createStringObjectFromLongLong(milliseconds);
122            rewriteClientCommandVector(c, 5, shared.set, key, val, shared.pxat, milliseconds_obj);
123            decrRefCount(milliseconds_obj);
124        }
125        notifyKeyspaceEvent(NOTIFY_GENERIC,"expire",key,c->db->id);
126    }
127
128    if (!(flags & OBJ_SET_GET)) {
129        addReply(c, ok_reply ? ok_reply : shared.ok);
130    }
131
132    /* Propagate without the GET argument (Isn't needed if we had expire since in that case we completely re-written the command argv) */
133    if ((flags & OBJ_SET_GET) && !expire) {
134        int argc = 0;
135        int j;
136        robj **argv = zmalloc((c->argc-1)*sizeof(robj*));
137        for (j=0; j < c->argc; j++) {
138            char *a = c->argv[j]->ptr;
139            /* Skip GET which may be repeated multiple times. */
140            if (j >= 3 &&
141                (a[0] == 'g' || a[0] == 'G') &&
142                (a[1] == 'e' || a[1] == 'E') &&
143                (a[2] == 't' || a[2] == 'T') && a[3] == '\0')
144                continue;
145            argv[argc++] = c->argv[j];
146            incrRefCount(c->argv[j]);
147        }
148        replaceClientCommandVector(c, argc, argv);
149    }
150}

152/*
153 * Extract the `expire` argument of a given GET/SET command as an absolute timestamp in milliseconds.
154 *
155 * "client" is the client that sent the `expire` argument.
156 * "expire" is the `expire` argument to be extracted.
157 * "flags" represents the behavior of the command (e.g. PX or EX).
158 * "unit" is the original unit of the given `expire` argument (e.g. UNIT_SECONDS).
159 * "milliseconds" is output argument.
160 *
161 * If return C_OK, "milliseconds" output argument will be set to the resulting absolute timestamp.
162 * If return C_ERR, an error reply has been added to the given client.
163 */
164static int getExpireMillisecondsOrReply(client *c, robj *expire, int flags, int unit, long long *milliseconds) {
165    int ret = getLongLongFromObjectOrReply(c, expire, milliseconds, NULL);
166    if (ret != C_OK) {
167        return ret;
168    }
169
170    if (*milliseconds <= 0 || (unit == UNIT_SECONDS && *milliseconds > LLONG_MAX / 1000)) {
171        /* Negative value provided or multiplication is gonna overflow. */
172        addReplyErrorExpireTime(c);
173        return C_ERR;
174    }
175
176    if (unit == UNIT_SECONDS) *milliseconds *= 1000;
177
178    if ((flags & OBJ_PX) || (flags & OBJ_EX)) {
179        *milliseconds += commandTimeSnapshot();
180    }
181
182    if (*milliseconds <= 0) {
183        /* Overflow detected. */
184        addReplyErrorExpireTime(c);
185        return C_ERR;
186    }
187
188    return C_OK;
189}

193/*
194 * The parseExtendedStringArgumentsOrReply() function performs the common validation for extended
195 * string arguments used in SET and GET command.
196 *
197 * Get specific commands - PERSIST/DEL
198 * Set specific commands - XX/NX/GET
199 * Common commands - EX/EXAT/PX/PXAT/KEEPTTL
200 *
201 * Function takes pointers to client, flags, unit, pointer to pointer of expire obj if needed
202 * to be determined and command_type which can be COMMAND_GET or COMMAND_SET.
203 *
204 * If there are any syntax violations C_ERR is returned else C_OK is returned.
205 *
206 * Input flags are updated upon parsing the arguments. Unit and expire are updated if there are any
207 * EX/EXAT/PX/PXAT arguments. Unit is updated to millisecond if PX/PXAT is set.
208 */
209int parseExtendedStringArgumentsOrReply(client *c, int *flags, int *unit, robj **expire, int command_type) {
210
211    int j = command_type == COMMAND_GET ? 2 : 3;
212    for (; j < c->argc; j++) {
213        char *opt = c->argv[j]->ptr;
214        robj *next = (j == c->argc-1) ? NULL : c->argv[j+1];
215
216        if ((opt[0] == 'n' || opt[0] == 'N') &&
217            (opt[1] == 'x' || opt[1] == 'X') && opt[2] == '\0' &&
218            !(*flags & OBJ_SET_XX) && (command_type == COMMAND_SET))
219        {
220            *flags |= OBJ_SET_NX;
221        } else if ((opt[0] == 'x' || opt[0] == 'X') &&
222                   (opt[1] == 'x' || opt[1] == 'X') && opt[2] == '\0' &&
223                   !(*flags & OBJ_SET_NX) && (command_type == COMMAND_SET))
224        {
225            *flags |= OBJ_SET_XX;
226        } else if ((opt[0] == 'g' || opt[0] == 'G') &&
227                   (opt[1] == 'e' || opt[1] == 'E') &&
228                   (opt[2] == 't' || opt[2] == 'T') && opt[3] == '\0' &&
229                   (command_type == COMMAND_SET))
230        {
231            *flags |= OBJ_SET_GET;
232        } else if (!strcasecmp(opt, "KEEPTTL") && !(*flags & OBJ_PERSIST) &&
233            !(*flags & OBJ_EX) && !(*flags & OBJ_EXAT) &&
234            !(*flags & OBJ_PX) && !(*flags & OBJ_PXAT) && (command_type == COMMAND_SET))
235        {
236            *flags |= OBJ_KEEPTTL;
237        } else if (!strcasecmp(opt,"PERSIST") && (command_type == COMMAND_GET) &&
238               !(*flags & OBJ_EX) && !(*flags & OBJ_EXAT) &&
239               !(*flags & OBJ_PX) && !(*flags & OBJ_PXAT) &&
240               !(*flags & OBJ_KEEPTTL))
241        {
242            *flags |= OBJ_PERSIST;
243        } else if ((opt[0] == 'e' || opt[0] == 'E') &&
244                   (opt[1] == 'x' || opt[1] == 'X') && opt[2] == '\0' &&
245                   !(*flags & OBJ_KEEPTTL) && !(*flags & OBJ_PERSIST) &&
246                   !(*flags & OBJ_EXAT) && !(*flags & OBJ_PX) &&
247                   !(*flags & OBJ_PXAT) && next)
248        {
249            *flags |= OBJ_EX;
250            *expire = next;
251            j++;
252        } else if ((opt[0] == 'p' || opt[0] == 'P') &&
253                   (opt[1] == 'x' || opt[1] == 'X') && opt[2] == '\0' &&
254                   !(*flags & OBJ_KEEPTTL) && !(*flags & OBJ_PERSIST) &&
255                   !(*flags & OBJ_EX) && !(*flags & OBJ_EXAT) &&
256                   !(*flags & OBJ_PXAT) && next)
257        {
258            *flags |= OBJ_PX;
259            *unit = UNIT_MILLISECONDS;
260            *expire = next;
261            j++;
262        } else if ((opt[0] == 'e' || opt[0] == 'E') &&
263                   (opt[1] == 'x' || opt[1] == 'X') &&
264                   (opt[2] == 'a' || opt[2] == 'A') &&
265                   (opt[3] == 't' || opt[3] == 'T') && opt[4] == '\0' &&
266                   !(*flags & OBJ_KEEPTTL) && !(*flags & OBJ_PERSIST) &&
267                   !(*flags & OBJ_EX) && !(*flags & OBJ_PX) &&
268                   !(*flags & OBJ_PXAT) && next)
269        {
270            *flags |= OBJ_EXAT;
271            *expire = next;
272            j++;
273        } else if ((opt[0] == 'p' || opt[0] == 'P') &&
274                   (opt[1] == 'x' || opt[1] == 'X') &&
275                   (opt[2] == 'a' || opt[2] == 'A') &&
276                   (opt[3] == 't' || opt[3] == 'T') && opt[4] == '\0' &&
277                   !(*flags & OBJ_KEEPTTL) && !(*flags & OBJ_PERSIST) &&
278                   !(*flags & OBJ_EX) && !(*flags & OBJ_EXAT) &&
279                   !(*flags & OBJ_PX) && next)
280        {
281            *flags |= OBJ_PXAT;
282            *unit = UNIT_MILLISECONDS;
283            *expire = next;
284            j++;
285        } else {
286            addReplyErrorObject(c,shared.syntaxerr);
287            return C_ERR;
288        }
289    }
290    return C_OK;
291}

341/*
342 * GETEX <key> [PERSIST][EX seconds][PX milliseconds][EXAT seconds-timestamp][PXAT milliseconds-timestamp]
343 *
344 * The getexCommand() function implements extended options and variants of the GET command. Unlike GET
345 * command this command is not read-only.
346 *
347 * The default behavior when no options are specified is same as GET and does not alter any TTL.
348 *
349 * Only one of the below options can be used at a given time.
350 *
351 * 1. PERSIST removes any TTL associated with the key.
352 * 2. EX Set expiry TTL in seconds.
353 * 3. PX Set expiry TTL in milliseconds.
354 * 4. EXAT Same like EX instead of specifying the number of seconds representing the TTL
355 *      (time to live), it takes an absolute Unix timestamp
356 * 5. PXAT Same like PX instead of specifying the number of milliseconds representing the TTL
357 *      (time to live), it takes an absolute Unix timestamp
358 *
359 * Command would either return the bulk string, error or nil.
360 */
361void getexCommand(client *c) {
362    robj *expire = NULL;
363    int unit = UNIT_SECONDS;
364    int flags = OBJ_NO_FLAGS;
365
366    if (parseExtendedStringArgumentsOrReply(c,&flags,&unit,&expire,COMMAND_GET) != C_OK) {
367        return;
368    }
369
370    robj *o;
371
372    if ((o = lookupKeyReadOrReply(c,c->argv[1],shared.null[c->resp])) == NULL)
373        return;
374
375    if (checkType(c,o,OBJ_STRING)) {
376        return;
377    }
378
379    /* Validate the expiration time value first */
380    long long milliseconds = 0;
381    if (expire && getExpireMillisecondsOrReply(c, expire, flags, unit, &milliseconds) != C_OK) {
382        return;
383    }
384
385    /* We need to do this before we expire the key or delete it */
386    addReplyBulk(c,o);
387
388    /* This command is never propagated as is. It is either propagated as PEXPIRE[AT],DEL,UNLINK or PERSIST.
389     * This why it doesn't need special handling in feedAppendOnlyFile to convert relative expire time to absolute one. */
390    if (((flags & OBJ_PXAT) || (flags & OBJ_EXAT)) && checkAlreadyExpired(milliseconds)) {
391        /* When PXAT/EXAT absolute timestamp is specified, there can be a chance that timestamp
392         * has already elapsed so delete the key in that case. */
393        int deleted = dbGenericDelete(c->db, c->argv[1], server.lazyfree_lazy_expire, DB_FLAG_KEY_EXPIRED);
394        serverAssert(deleted);
395        robj *aux = server.lazyfree_lazy_expire ? shared.unlink : shared.del;
396        rewriteClientCommandVector(c,2,aux,c->argv[1]);
397        signalModifiedKey(c, c->db, c->argv[1]);
398        notifyKeyspaceEvent(NOTIFY_GENERIC, "del", c->argv[1], c->db->id);
399        server.dirty++;
400    } else if (expire) {
401        setExpire(c,c->db,c->argv[1],milliseconds);
402        /* Propagate as PXEXPIREAT millisecond-timestamp if there is
403         * EX/PX/EXAT/PXAT flag and the key has not expired. */
404        robj *milliseconds_obj = createStringObjectFromLongLong(milliseconds);
405        rewriteClientCommandVector(c,3,shared.pexpireat,c->argv[1],milliseconds_obj);
406        decrRefCount(milliseconds_obj);
407        signalModifiedKey(c, c->db, c->argv[1]);
408        notifyKeyspaceEvent(NOTIFY_GENERIC,"expire",c->argv[1],c->db->id);
409        server.dirty++;
410    } else if (flags & OBJ_PERSIST) {
411        if (removeExpire(c->db, c->argv[1])) {
412            signalModifiedKey(c, c->db, c->argv[1]);
413            rewriteClientCommandVector(c, 2, shared.persist, c->argv[1]);
414            notifyKeyspaceEvent(NOTIFY_GENERIC,"persist",c->argv[1],c->db->id);
415            server.dirty++;
416        }
417    }
418}

442void setrangeCommand(client *c) {
443    robj *o;
444    long offset;
445    sds value = c->argv[3]->ptr;
446
447    if (getLongFromObjectOrReply(c,c->argv[2],&offset,NULL) != C_OK)
448        return;
449
450    if (offset < 0) {
451        addReplyError(c,"offset is out of range");
452        return;
453    }
454
455    o = lookupKeyWrite(c->db,c->argv[1]);
456    if (o == NULL) {
457        /* Return 0 when setting nothing on a non-existing string */
458        if (sdslen(value) == 0) {
459            addReply(c,shared.czero);
460            return;
461        }
462
463        /* Return when the resulting string exceeds allowed size */
464        if (checkStringLength(c,offset,sdslen(value)) != C_OK)
465            return;
466
467        o = createObject(OBJ_STRING,sdsnewlen(NULL, offset+sdslen(value)));
468        dbAdd(c->db,c->argv[1],o);
469    } else {
470        size_t olen;
471
472        /* Key exists, check type */
473        if (checkType(c,o,OBJ_STRING))
474            return;
475
476        /* Return existing string length when setting nothing */
477        olen = stringObjectLen(o);
478        if (sdslen(value) == 0) {
479            addReplyLongLong(c,olen);
480            return;
481        }
482
483        /* Return when the resulting string exceeds allowed size */
484        if (checkStringLength(c,offset,sdslen(value)) != C_OK)
485            return;
486
487        /* Create a copy when the object is shared or encoded. */
488        o = dbUnshareStringValue(c->db,c->argv[1],o);
489    }
490
491    if (sdslen(value) > 0) {
492        o->ptr = sdsgrowzero(o->ptr,offset+sdslen(value));
493        memcpy((char*)o->ptr+offset,value,sdslen(value));
494        signalModifiedKey(c,c->db,c->argv[1]);
495        notifyKeyspaceEvent(NOTIFY_STRING,
496            "setrange",c->argv[1],c->db->id);
497        server.dirty++;
498    }
499    addReplyLongLong(c,sdslen(o->ptr));
500}

502void getrangeCommand(client *c) {
503    robj *o;
504    long long start, end;
505    char *str, llbuf[32];
506    size_t strlen;
507
508    if (getLongLongFromObjectOrReply(c,c->argv[2],&start,NULL) != C_OK)
509        return;
510    if (getLongLongFromObjectOrReply(c,c->argv[3],&end,NULL) != C_OK)
511        return;
512    if ((o = lookupKeyReadOrReply(c,c->argv[1],shared.emptybulk)) == NULL ||
513        checkType(c,o,OBJ_STRING)) return;
514
515    if (o->encoding == OBJ_ENCODING_INT) {
516        str = llbuf;
517        strlen = ll2string(llbuf,sizeof(llbuf),(long)o->ptr);
518    } else {
519        str = o->ptr;
520        strlen = sdslen(str);
521    }
522
523    /* Convert negative indexes */
524    if (start < 0 && end < 0 && start > end) {
525        addReply(c,shared.emptybulk);
526        return;
527    }
528    if (start < 0) start = strlen+start;
529    if (end < 0) end = strlen+end;
530    if (start < 0) start = 0;
531    if (end < 0) end = 0;
532    if ((unsigned long long)end >= strlen) end = strlen-1;
533
534    /* Precondition: end >= 0 && end < strlen, so the only condition where
535     * nothing can be returned is: start > end. */
536    if (start > end || strlen == 0) {
537        addReply(c,shared.emptybulk);
538    } else {
539        addReplyBulkCBuffer(c,(char*)str+start,end-start+1);
540    }
541}

543void mgetCommand(client *c) {
544    int j;
545
546    addReplyArrayLen(c,c->argc-1);
547    for (j = 1; j < c->argc; j++) {
548        robj *o = lookupKeyRead(c->db,c->argv[j]);
549        if (o == NULL) {
550            addReplyNull(c);
551        } else {
552            if (o->type != OBJ_STRING) {
553                addReplyNull(c);
554            } else {
555                addReplyBulk(c,o);
556            }
557        }
558    }
559}

561void msetGenericCommand(client *c, int nx) {
562    int j;
563
564    if ((c->argc % 2) == 0) {
565        addReplyErrorArity(c);
566        return;
567    }
568
569    /* Handle the NX flag. The MSETNX semantic is to return zero and don't
570     * set anything if at least one key already exists. */
571    if (nx) {
572        for (j = 1; j < c->argc; j += 2) {
573            if (lookupKeyWrite(c->db,c->argv[j]) != NULL) {
574                addReply(c, shared.czero);
575                return;
576            }
577        }
578    }
579
580    int setkey_flags = nx ? SETKEY_DOESNT_EXIST : 0;
581    for (j = 1; j < c->argc; j += 2) {
582        c->argv[j+1] = tryObjectEncoding(c->argv[j+1]);
583        setKey(c, c->db, c->argv[j], c->argv[j + 1], setkey_flags);
584        notifyKeyspaceEvent(NOTIFY_STRING,"set",c->argv[j],c->db->id);
585        /* In MSETNX, It could be that we're overriding the same key, we can't be sure it doesn't exist. */
586        if (nx)
587            setkey_flags = SETKEY_ADD_OR_UPDATE;
588    }
589    server.dirty += (c->argc-1)/2;
590    addReply(c, nx ? shared.cone : shared.ok);
591}

601void incrDecrCommand(client *c, long long incr) {
602    long long value, oldvalue;
603    robj *o, *new;
604
605    o = lookupKeyWrite(c->db,c->argv[1]);
606    if (checkType(c,o,OBJ_STRING)) return;
607    if (getLongLongFromObjectOrReply(c,o,&value,NULL) != C_OK) return;
608
609    oldvalue = value;
610    if ((incr < 0 && oldvalue < 0 && incr < (LLONG_MIN-oldvalue)) ||
611        (incr > 0 && oldvalue > 0 && incr > (LLONG_MAX-oldvalue))) {
612        addReplyError(c,"increment or decrement would overflow");
613        return;
614    }
615    value += incr;
616
617    if (o && o->refcount == 1 && o->encoding == OBJ_ENCODING_INT &&
618        (value < 0 || value >= OBJ_SHARED_INTEGERS) &&
619        value >= LONG_MIN && value <= LONG_MAX)
620    {
621        new = o;
622        o->ptr = (void*)((long)value);
623    } else {
624        new = createStringObjectFromLongLongForValue(value);
625        if (o) {
626            dbReplaceValue(c->db,c->argv[1],new);
627        } else {
628            dbAdd(c->db,c->argv[1],new);
629        }
630    }
631    signalModifiedKey(c,c->db,c->argv[1]);
632    notifyKeyspaceEvent(NOTIFY_STRING,"incrby",c->argv[1],c->db->id);
633    server.dirty++;
634    addReplyLongLong(c, value);
635}

665    long double incr, value;

667
668    o = lookupKeyWrite(c->db,c->argv[1]);
669    if (checkType(c,o,OBJ_STRING)) return;
670    if (getLongDoubleFromObjectOrReply(c,o,&value,NULL) != C_OK ||

672        return;

677        return;
678    }

681        dbReplaceValue(c->db,c->argv[1],new);

691     * will not create differences in replicas or after an AOF restart. */
692    rewriteClientCommandArgument(c,0,shared.set);
693    rewriteClientCommandArgument(c,2,new);

697void appendCommand(client *c) {
698    size_t totlen;
699    robj *o, *append;
700
701    o = lookupKeyWrite(c->db,c->argv[1]);
702    if (o == NULL) {
703        /* Create the key */
704        c->argv[2] = tryObjectEncoding(c->argv[2]);
705        dbAdd(c->db,c->argv[1],c->argv[2]);
706        incrRefCount(c->argv[2]);
707        totlen = stringObjectLen(c->argv[2]);
708    } else {
709        /* Key exists, check type */
710        if (checkType(c,o,OBJ_STRING))
711            return;
712
713        /* "append" is an argument, so always an sds */
714        append = c->argv[2];
715        if (checkStringLength(c,stringObjectLen(o),sdslen(append->ptr)) != C_OK)
716            return;
717
718        /* Append the value */
719        o = dbUnshareStringValue(c->db,c->argv[1],o);
720        o->ptr = sdscatlen(o->ptr,append->ptr,sdslen(append->ptr));
721        totlen = sdslen(o->ptr);
722    }
723    signalModifiedKey(c,c->db,c->argv[1]);
724    notifyKeyspaceEvent(NOTIFY_STRING,"append",c->argv[1],c->db->id);
725    server.dirty++;
726    addReplyLongLong(c,totlen);
727}

736/* LCS key1 key2 [LEN] [IDX] [MINMATCHLEN <len>] [WITHMATCHLEN] */
737void lcsCommand(client *c) {
738    uint32_t i, j;
739    long long minmatchlen = 0;
740    sds a = NULL, b = NULL;
741    int getlen = 0, getidx = 0, withmatchlen = 0;
742    robj *obja = NULL, *objb = NULL;
743
744    obja = lookupKeyRead(c->db,c->argv[1]);
745    objb = lookupKeyRead(c->db,c->argv[2]);
746    if ((obja && obja->type != OBJ_STRING) ||
747        (objb && objb->type != OBJ_STRING))
748    {
749        addReplyError(c,
750            "The specified keys must contain string values");
751        /* Don't cleanup the objects, we need to do that
752         * only after calling getDecodedObject(). */
753        obja = NULL;
754        objb = NULL;
755        goto cleanup;
756    }
757    obja = obja ? getDecodedObject(obja) : createStringObject("",0);
758    objb = objb ? getDecodedObject(objb) : createStringObject("",0);
759    a = obja->ptr;
760    b = objb->ptr;
761
762    for (j = 3; j < (uint32_t)c->argc; j++) {
763        char *opt = c->argv[j]->ptr;
764        int moreargs = (c->argc-1) - j;
765
766        if (!strcasecmp(opt,"IDX")) {
767            getidx = 1;
768        } else if (!strcasecmp(opt,"LEN")) {
769            getlen = 1;
770        } else if (!strcasecmp(opt,"WITHMATCHLEN")) {
771            withmatchlen = 1;
772        } else if (!strcasecmp(opt,"MINMATCHLEN") && moreargs) {
773            if (getLongLongFromObjectOrReply(c,c->argv[j+1],&minmatchlen,NULL)
774                != C_OK) goto cleanup;
775            if (minmatchlen < 0) minmatchlen = 0;
776            j++;
777        } else {
778            addReplyErrorObject(c,shared.syntaxerr);
779            goto cleanup;
780        }
781    }
782
783    /* Complain if the user passed ambiguous parameters. */
784    if (getlen && getidx) {
785        addReplyError(c,
786            "If you want both the length and indexes, please just use IDX.");
787        goto cleanup;
788    }
789
790    /* Detect string truncation or later overflows. */
791    if (sdslen(a) >= UINT32_MAX-1 || sdslen(b) >= UINT32_MAX-1) {
792        addReplyError(c, "String too long for LCS");
793        goto cleanup;
794    }
795
796    /* Compute the LCS using the vanilla dynamic programming technique of
797     * building a table of LCS(x,y) substrings. */
798    uint32_t alen = sdslen(a);
799    uint32_t blen = sdslen(b);
800
801    /* Setup an uint32_t array to store at LCS[i,j] the length of the
802     * LCS A0..i-1, B0..j-1. Note that we have a linear array here, so
803     * we index it as LCS[j+(blen+1)*i] */
804    #define LCS(A,B) lcs[(B)+((A)*(blen+1))]
805
806    /* Try to allocate the LCS table, and abort on overflow or insufficient memory. */
807    unsigned long long lcssize = (unsigned long long)(alen+1)*(blen+1); /* Can't overflow due to the size limits above. */
808    unsigned long long lcsalloc = lcssize * sizeof(uint32_t);
809    uint32_t *lcs = NULL;
810    if (lcsalloc < SIZE_MAX && lcsalloc / lcssize == sizeof(uint32_t)) {
811        if (lcsalloc > (size_t)server.proto_max_bulk_len) {
812            addReplyError(c, "Insufficient memory, transient memory for LCS exceeds proto-max-bulk-len");
813            goto cleanup;
814        }
815        lcs = ztrymalloc(lcsalloc);
816    }
817    if (!lcs) {
818        addReplyError(c, "Insufficient memory, failed allocating transient memory for LCS");
819        goto cleanup;
820    }
821
822    /* Start building the LCS table. */
823    for (uint32_t i = 0; i <= alen; i++) {
824        for (uint32_t j = 0; j <= blen; j++) {
825            if (i == 0 || j == 0) {
826                /* If one substring has length of zero, the
827                 * LCS length is zero. */
828                LCS(i,j) = 0;
829            } else if (a[i-1] == b[j-1]) {
830                /* The len LCS (and the LCS itself) of two
831                 * sequences with the same final character, is the
832                 * LCS of the two sequences without the last char
833                 * plus that last char. */
834                LCS(i,j) = LCS(i-1,j-1)+1;
835            } else {
836                /* If the last character is different, take the longest
837                 * between the LCS of the first string and the second
838                 * minus the last char, and the reverse. */
839                uint32_t lcs1 = LCS(i-1,j);
840                uint32_t lcs2 = LCS(i,j-1);
841                LCS(i,j) = lcs1 > lcs2 ? lcs1 : lcs2;
842            }
843        }
844    }
845
846    /* Store the actual LCS string in "result" if needed. We create
847     * it backward, but the length is already known, we store it into idx. */
848    uint32_t idx = LCS(alen,blen);
849    sds result = NULL;        /* Resulting LCS string. */
850    void *arraylenptr = NULL; /* Deferred length of the array for IDX. */
851    uint32_t arange_start = alen, /* alen signals that values are not set. */
852             arange_end = 0,
853             brange_start = 0,
854             brange_end = 0;
855
856    /* Do we need to compute the actual LCS string? Allocate it in that case. */
857    int computelcs = getidx || !getlen;
858    if (computelcs) result = sdsnewlen(SDS_NOINIT,idx);
859
860    /* Start with a deferred array if we have to emit the ranges. */
861    uint32_t arraylen = 0;  /* Number of ranges emitted in the array. */
862    if (getidx) {
863        addReplyMapLen(c,2);
864        addReplyBulkCString(c,"matches");
865        arraylenptr = addReplyDeferredLen(c);
866    }
867
868    i = alen, j = blen;
869    while (computelcs && i > 0 && j > 0) {
870        int emit_range = 0;
871        if (a[i-1] == b[j-1]) {
872            /* If there is a match, store the character and reduce
873             * the indexes to look for a new match. */
874            result[idx-1] = a[i-1];
875
876            /* Track the current range. */
877            if (arange_start == alen) {
878                arange_start = i-1;
879                arange_end = i-1;
880                brange_start = j-1;
881                brange_end = j-1;
882            } else {
883                /* Let's see if we can extend the range backward since
884                 * it is contiguous. */
885                if (arange_start == i && brange_start == j) {
886                    arange_start--;
887                    brange_start--;
888                } else {
889                    emit_range = 1;
890                }
891            }
892            /* Emit the range if we matched with the first byte of
893             * one of the two strings. We'll exit the loop ASAP. */
894            if (arange_start == 0 || brange_start == 0) emit_range = 1;
895            idx--; i--; j--;
896        } else {
897            /* Otherwise reduce i and j depending on the largest
898             * LCS between, to understand what direction we need to go. */
899            uint32_t lcs1 = LCS(i-1,j);
900            uint32_t lcs2 = LCS(i,j-1);
901            if (lcs1 > lcs2)
902                i--;
903            else
904                j--;
905            if (arange_start != alen) emit_range = 1;
906        }
907
908        /* Emit the current range if needed. */
909        uint32_t match_len = arange_end - arange_start + 1;
910        if (emit_range) {
911            if (minmatchlen == 0 || match_len >= minmatchlen) {
912                if (arraylenptr) {
913                    addReplyArrayLen(c,2+withmatchlen);
914                    addReplyArrayLen(c,2);
915                    addReplyLongLong(c,arange_start);
916                    addReplyLongLong(c,arange_end);
917                    addReplyArrayLen(c,2);
918                    addReplyLongLong(c,brange_start);
919                    addReplyLongLong(c,brange_end);
920                    if (withmatchlen) addReplyLongLong(c,match_len);
921                    arraylen++;
922                }
923            }
924            arange_start = alen; /* Restart at the next match. */
925        }
926    }
927
928    /* Signal modified key, increment dirty, ... */
929
930    /* Reply depending on the given options. */
931    if (arraylenptr) {
932        addReplyBulkCString(c,"len");
933        addReplyLongLong(c,LCS(alen,blen));
934        setDeferredArrayLen(c,arraylenptr,arraylen);
935    } else if (getlen) {
936        addReplyLongLong(c,LCS(alen,blen));
937    } else {
938        addReplyBulkSds(c,result);
939        result = NULL;
940    }
941
942    /* Cleanup. */
943    sdsfree(result);
944    zfree(lcs);
945
946cleanup:
947    if (obja) decrRefCount(obja);
948    if (objb) decrRefCount(objb);
949    return;
950}
