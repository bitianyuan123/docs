67aeEventLoop *aeCreateEventLoop(int setsize) {
68    aeEventLoop *eventLoop;
69    int i;
70
71    monotonicInit();    /* just in case the calling app didn't initialize */
72
73    if ((eventLoop = zmalloc(sizeof(*eventLoop))) == NULL) goto err;
74    eventLoop->events = zmalloc(sizeof(aeFileEvent)*setsize);
75    eventLoop->fired = zmalloc(sizeof(aeFiredEvent)*setsize);
76    if (eventLoop->events == NULL || eventLoop->fired == NULL) goto err;
77    eventLoop->setsize = setsize;
78    eventLoop->timeEventHead = NULL;
79    eventLoop->timeEventNextId = 0;
80    eventLoop->stop = 0;
81    eventLoop->maxfd = -1;
82    eventLoop->beforesleep = NULL;
83    eventLoop->aftersleep = NULL;
84    eventLoop->flags = 0;
85    if (aeApiCreate(eventLoop) == -1) goto err;
86    /* Events with mask == AE_NONE are not set. So let's initialize the
87     * vector with it. */
88    for (i = 0; i < setsize; i++)
89        eventLoop->events[i].mask = AE_NONE;
90    return eventLoop;
91
92err:
93    if (eventLoop) {
94        zfree(eventLoop->events);
95        zfree(eventLoop->fired);
96        zfree(eventLoop);
97    }
98    return NULL;
99}

118/* Resize the maximum set size of the event loop.
119 * If the requested set size is smaller than the current set size, but
120 * there is already a file descriptor in use that is >= the requested
121 * set size minus one, AE_ERR is returned and the operation is not
122 * performed at all.
123 *
124 * Otherwise AE_OK is returned and the operation is successful. */
125int aeResizeSetSize(aeEventLoop *eventLoop, int setsize) {
126    int i;
127
128    if (setsize == eventLoop->setsize) return AE_OK;
129    if (eventLoop->maxfd >= setsize) return AE_ERR;
130    if (aeApiResize(eventLoop,setsize) == -1) return AE_ERR;
131
132    eventLoop->events = zrealloc(eventLoop->events,sizeof(aeFileEvent)*setsize);
133    eventLoop->fired = zrealloc(eventLoop->fired,sizeof(aeFiredEvent)*setsize);
134    eventLoop->setsize = setsize;
135
136    /* Make sure that if we created new slots, they are initialized with
137     * an AE_NONE mask. */
138    for (i = eventLoop->maxfd+1; i < setsize; i++)
139        eventLoop->events[i].mask = AE_NONE;
140    return AE_OK;
141}

143void aeDeleteEventLoop(aeEventLoop *eventLoop) {
144    aeApiFree(eventLoop);
145    zfree(eventLoop->events);
146    zfree(eventLoop->fired);
147
148    /* Free the time events list. */
149    aeTimeEvent *next_te, *te = eventLoop->timeEventHead;
150    while (te) {
151        next_te = te->next;
152        zfree(te);
153        te = next_te;
154    }
155    zfree(eventLoop);
156}

162int aeCreateFileEvent(aeEventLoop *eventLoop, int fd, int mask,
163        aeFileProc *proc, void *clientData)
164{
165    if (fd >= eventLoop->setsize) {
166        errno = ERANGE;
167        return AE_ERR;
168    }
169    aeFileEvent *fe = &eventLoop->events[fd];
170
171    if (aeApiAddEvent(eventLoop, fd, mask) == -1)
172        return AE_ERR;
173    fe->mask |= mask;
174    if (mask & AE_READABLE) fe->rfileProc = proc;
175    if (mask & AE_WRITABLE) fe->wfileProc = proc;
176    fe->clientData = clientData;
177    if (fd > eventLoop->maxfd)
178        eventLoop->maxfd = fd;
179    return AE_OK;
180}

182void aeDeleteFileEvent(aeEventLoop *eventLoop, int fd, int mask)
183{
184    if (fd >= eventLoop->setsize) return;
185    aeFileEvent *fe = &eventLoop->events[fd];
186    if (fe->mask == AE_NONE) return;
187
188    /* We want to always remove AE_BARRIER if set when AE_WRITABLE
189     * is removed. */
190    if (mask & AE_WRITABLE) mask |= AE_BARRIER;
191
192    aeApiDelEvent(eventLoop, fd, mask);
193    fe->mask = fe->mask & (~mask);
194    if (fd == eventLoop->maxfd && fe->mask == AE_NONE) {
195        /* Update the max fd */
196        int j;
197
198        for (j = eventLoop->maxfd-1; j >= 0; j--)
199            if (eventLoop->events[j].mask != AE_NONE) break;
200        eventLoop->maxfd = j;
201    }
202}

219long long aeCreateTimeEvent(aeEventLoop *eventLoop, long long milliseconds,
220        aeTimeProc *proc, void *clientData,
221        aeEventFinalizerProc *finalizerProc)
222{
223    long long id = eventLoop->timeEventNextId++;
224    aeTimeEvent *te;
225
226    te = zmalloc(sizeof(*te));
227    if (te == NULL) return AE_ERR;
228    te->id = id;
229    te->when = getMonotonicUs() + milliseconds * 1000;
230    te->timeProc = proc;
231    te->finalizerProc = finalizerProc;
232    te->clientData = clientData;
233    te->prev = NULL;
234    te->next = eventLoop->timeEventHead;
235    te->refcount = 0;
236    if (te->next)
237        te->next->prev = te;
238    eventLoop->timeEventHead = te;
239    return id;
240}

242int aeDeleteTimeEvent(aeEventLoop *eventLoop, long long id)
243{
244    aeTimeEvent *te = eventLoop->timeEventHead;
245    while(te) {
246        if (te->id == id) {
247            te->id = AE_DELETED_EVENT_ID;
248            return AE_OK;
249        }
250        te = te->next;
251    }
252    return AE_ERR; /* NO event with the specified ID found */
253}

255/* How many microseconds until the first timer should fire.
256 * If there are no timers, -1 is returned.
257 *
258 * Note that's O(N) since time events are unsorted.
259 * Possible optimizations (not needed by Redis so far, but...):
260 * 1) Insert the event in order, so that the nearest is just the head.
261 *    Much better but still insertion or deletion of timers is O(N).
262 * 2) Use a skiplist to have this operation as O(1) and insertion as O(log(N)).
263 */
264static int64_t usUntilEarliestTimer(aeEventLoop *eventLoop) {
265    aeTimeEvent *te = eventLoop->timeEventHead;
266    if (te == NULL) return -1;
267
268    aeTimeEvent *earliest = NULL;
269    while (te) {
270        if ((!earliest || te->when < earliest->when) && te->id != AE_DELETED_EVENT_ID)
271            earliest = te;
272        te = te->next;
273    }
274
275    monotime now = getMonotonicUs();
276    return (now >= earliest->when) ? 0 : earliest->when - now;
277}

279/* Process time events */
280static int processTimeEvents(aeEventLoop *eventLoop) {
281    int processed = 0;
282    aeTimeEvent *te;
283    long long maxId;
284
285    te = eventLoop->timeEventHead;
286    maxId = eventLoop->timeEventNextId-1;
287    monotime now = getMonotonicUs();
288    while(te) {
289        long long id;
290
291        /* Remove events scheduled for deletion. */
292        if (te->id == AE_DELETED_EVENT_ID) {
293            aeTimeEvent *next = te->next;
294            /* If a reference exists for this timer event,
295             * don't free it. This is currently incremented
296             * for recursive timerProc calls */
297            if (te->refcount) {
298                te = next;
299                continue;
300            }
301            if (te->prev)
302                te->prev->next = te->next;
303            else
304                eventLoop->timeEventHead = te->next;
305            if (te->next)
306                te->next->prev = te->prev;
307            if (te->finalizerProc) {
308                te->finalizerProc(eventLoop, te->clientData);
309                now = getMonotonicUs();
310            }
311            zfree(te);
312            te = next;
313            continue;
314        }
315
316        /* Make sure we don't process time events created by time events in
317         * this iteration. Note that this check is currently useless: we always
318         * add new timers on the head, however if we change the implementation
319         * detail, this check may be useful again: we keep it here for future
320         * defense. */
321        if (te->id > maxId) {
322            te = te->next;
323            continue;
324        }
325
326        if (te->when <= now) {
327            int retval;
328
329            id = te->id;
330            te->refcount++;
331            retval = te->timeProc(eventLoop, id, te->clientData);
332            te->refcount--;
333            processed++;
334            now = getMonotonicUs();
335            if (retval != AE_NOMORE) {
336                te->when = now + (monotime)retval * 1000;
337            } else {
338                te->id = AE_DELETED_EVENT_ID;
339            }
340        }
341        te = te->next;
342    }
343    return processed;
344}

346/* Process every pending time event, then every pending file event
347 * (that may be registered by time event callbacks just processed).
348 * Without special flags the function sleeps until some file event
349 * fires, or when the next time event occurs (if any).
350 *
351 * If flags is 0, the function does nothing and returns.
352 * if flags has AE_ALL_EVENTS set, all the kind of events are processed.
353 * if flags has AE_FILE_EVENTS set, file events are processed.
354 * if flags has AE_TIME_EVENTS set, time events are processed.
355 * if flags has AE_DONT_WAIT set, the function returns ASAP once all
356 * the events that can be handled without a wait are processed.
357 * if flags has AE_CALL_AFTER_SLEEP set, the aftersleep callback is called.
358 * if flags has AE_CALL_BEFORE_SLEEP set, the beforesleep callback is called.
359 *
360 * The function returns the number of events processed. */
361int aeProcessEvents(aeEventLoop *eventLoop, int flags)
362{
363    int processed = 0, numevents;
364
365    /* Nothing to do? return ASAP */
366    if (!(flags & AE_TIME_EVENTS) && !(flags & AE_FILE_EVENTS)) return 0;
367
368    /* Note that we want to call aeApiPoll() even if there are no
369     * file events to process as long as we want to process time
370     * events, in order to sleep until the next time event is ready
371     * to fire. */
372    if (eventLoop->maxfd != -1 ||
373        ((flags & AE_TIME_EVENTS) && !(flags & AE_DONT_WAIT))) {
374        int j;
375        struct timeval tv, *tvp = NULL; /* NULL means infinite wait. */
376        int64_t usUntilTimer;
377
378        if (eventLoop->beforesleep != NULL && (flags & AE_CALL_BEFORE_SLEEP))
379            eventLoop->beforesleep(eventLoop);
380
381        /* The eventLoop->flags may be changed inside beforesleep.
382         * So we should check it after beforesleep be called. At the same time,
383         * the parameter flags always should have the highest priority.
384         * That is to say, once the parameter flag is set to AE_DONT_WAIT,
385         * no matter what value eventLoop->flags is set to, we should ignore it. */
386        if ((flags & AE_DONT_WAIT) || (eventLoop->flags & AE_DONT_WAIT)) {
387            tv.tv_sec = tv.tv_usec = 0;
388            tvp = &tv;
389        } else if (flags & AE_TIME_EVENTS) {
390            usUntilTimer = usUntilEarliestTimer(eventLoop);
391            if (usUntilTimer >= 0) {
392                tv.tv_sec = usUntilTimer / 1000000;
393                tv.tv_usec = usUntilTimer % 1000000;
394                tvp = &tv;
395            }
396        }
397        /* Call the multiplexing API, will return only on timeout or when
398         * some event fires. */
399        numevents = aeApiPoll(eventLoop, tvp);
400
401        /* Don't process file events if not requested. */
402        if (!(flags & AE_FILE_EVENTS)) {
403            numevents = 0;
404        }
405
406        /* After sleep callback. */
407        if (eventLoop->aftersleep != NULL && flags & AE_CALL_AFTER_SLEEP)
408            eventLoop->aftersleep(eventLoop);
409
410        for (j = 0; j < numevents; j++) {
411            int fd = eventLoop->fired[j].fd;
412            aeFileEvent *fe = &eventLoop->events[fd];
413            int mask = eventLoop->fired[j].mask;
414            int fired = 0; /* Number of events fired for current fd. */
415
416            /* Normally we execute the readable event first, and the writable
417             * event later. This is useful as sometimes we may be able
418             * to serve the reply of a query immediately after processing the
419             * query.
420             *
421             * However if AE_BARRIER is set in the mask, our application is
422             * asking us to do the reverse: never fire the writable event
423             * after the readable. In such a case, we invert the calls.
424             * This is useful when, for instance, we want to do things
425             * in the beforeSleep() hook, like fsyncing a file to disk,
426             * before replying to a client. */
427            int invert = fe->mask & AE_BARRIER;
428
429            /* Note the "fe->mask & mask & ..." code: maybe an already
430             * processed event removed an element that fired and we still
431             * didn't processed, so we check if the event is still valid.
432             *
433             * Fire the readable event if the call sequence is not
434             * inverted. */
435            if (!invert && fe->mask & mask & AE_READABLE) {
436                fe->rfileProc(eventLoop,fd,fe->clientData,mask);
437                fired++;
438                fe = &eventLoop->events[fd]; /* Refresh in case of resize. */
439            }
440
441            /* Fire the writable event. */
442            if (fe->mask & mask & AE_WRITABLE) {
443                if (!fired || fe->wfileProc != fe->rfileProc) {
444                    fe->wfileProc(eventLoop,fd,fe->clientData,mask);
445                    fired++;
446                }
447            }
448
449            /* If we have to invert the call, fire the readable event now
450             * after the writable one. */
451            if (invert) {
452                fe = &eventLoop->events[fd]; /* Refresh in case of resize. */
453                if ((fe->mask & mask & AE_READABLE) &&
454                    (!fired || fe->wfileProc != fe->rfileProc))
455                {
456                    fe->rfileProc(eventLoop,fd,fe->clientData,mask);
457                    fired++;
458                }
459            }
460
461            processed++;
462        }
463    }
464    /* Check time events */
465    if (flags & AE_TIME_EVENTS)
466        processed += processTimeEvents(eventLoop);
467
468    return processed; /* return the number of processed file/time events */
469}

471/* Wait for milliseconds until the given file descriptor becomes
472 * writable/readable/exception */
473int aeWait(int fd, int mask, long long milliseconds) {
474    struct pollfd pfd;
475    int retmask = 0, retval;
476
477    memset(&pfd, 0, sizeof(pfd));
478    pfd.fd = fd;
479    if (mask & AE_READABLE) pfd.events |= POLLIN;
480    if (mask & AE_WRITABLE) pfd.events |= POLLOUT;
481
482    if ((retval = poll(&pfd, 1, milliseconds))== 1) {
483        if (pfd.revents & POLLIN) retmask |= AE_READABLE;
484        if (pfd.revents & POLLOUT) retmask |= AE_WRITABLE;
485        if (pfd.revents & POLLERR) retmask |= AE_WRITABLE;
486        if (pfd.revents & POLLHUP) retmask |= AE_WRITABLE;
487        return retmask;
488    } else {
489        return retval;
490    }
491}

493void aeMain(aeEventLoop *eventLoop) {
494    eventLoop->stop = 0;
495    while (!eventLoop->stop) {
496        aeProcessEvents(eventLoop, AE_ALL_EVENTS|
497                                   AE_CALL_BEFORE_SLEEP|
498                                   AE_CALL_AFTER_SLEEP);
499    }
500}
