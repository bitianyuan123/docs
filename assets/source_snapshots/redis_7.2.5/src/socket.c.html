33/* The connections module provides a lean abstraction of network connections
34 * to avoid direct socket and async event management across the Redis code base.
35 *
36 * It does NOT provide advanced connection features commonly found in similar
37 * libraries such as complete in/out buffer management, throttling, etc. These
38 * functions remain in networking.c.
39 *
40 * The primary goal is to allow transparent handling of TCP and TLS based
41 * connections. To do so, connections have the following properties:
42 *
43 * 1. A connection may live before its corresponding socket exists.  This
44 *    allows various context and configuration setting to be handled before
45 *    establishing the actual connection.
46 * 2. The caller may register/unregister logical read/write handlers to be
47 *    called when the connection has data to read from/can accept writes.
48 *    These logical handlers may or may not correspond to actual AE events,
49 *    depending on the implementation (for TCP they are; for TLS they aren't).
50 */
51
52static ConnectionType CT_Socket;
53
54/* When a connection is created we must know its type already, but the
55 * underlying socket may or may not exist:
56 *
57 * - For accepted connections, it exists as we do not model the listen/accept
58 *   part; So caller calls connCreateSocket() followed by connAccept().
59 * - For outgoing connections, the socket is created by the connection module
60 *   itself; So caller calls connCreateSocket() followed by connConnect(),
61 *   which registers a connect callback that fires on connected/error state
62 *   (and after any transport level handshake was done).
63 *
64 * NOTE: An earlier version relied on connections being part of other structs
65 * and not independently allocated. This could lead to further optimizations
66 * like using container_of(), etc.  However it was discontinued in favor of
67 * this approach for these reasons:
68 *
69 * 1. In some cases conns are created/handled outside the context of the
70 * containing struct, in which case it gets a bit awkward to copy them.
71 * 2. Future implementations may wish to allocate arbitrary data for the
72 * connection.
73 * 3. The container_of() approach is anyway risky because connections may
74 * be embedded in different structs, not just client.
75 */
76
77static connection *connCreateSocket(void) {
78    connection *conn = zcalloc(sizeof(connection));
79    conn->type = &CT_Socket;
80    conn->fd = -1;
81    conn->iovcnt = IOV_MAX;
82
83    return conn;
84}

86/* Create a new socket-type connection that is already associated with
87 * an accepted connection.
88 *
89 * The socket is not ready for I/O until connAccept() was called and
90 * invoked the connection-level accept handler.
91 *
92 * Callers should use connGetState() and verify the created connection
93 * is not in an error state (which is not possible for a socket connection,
94 * but could but possible with other protocols).
95 */
96static connection *connCreateAcceptedSocket(int fd, void *priv) {
97    UNUSED(priv);
98    connection *conn = connCreateSocket();
99    conn->fd = fd;
100    conn->state = CONN_STATE_ACCEPTING;
101    return conn;
102}

104static int connSocketConnect(connection *conn, const char *addr, int port, const char *src_addr,
105        ConnectionCallbackFunc connect_handler) {
106    int fd = anetTcpNonBlockBestEffortBindConnect(NULL,addr,port,src_addr);
107    if (fd == -1) {
108        conn->state = CONN_STATE_ERROR;
109        conn->last_errno = errno;
110        return C_ERR;
111    }
112
113    conn->fd = fd;
114    conn->state = CONN_STATE_CONNECTING;
115
116    conn->conn_handler = connect_handler;
117    aeCreateFileEvent(server.el, conn->fd, AE_WRITABLE,
118            conn->type->ae_handler, conn);
119
120    return C_OK;
121}

135/* Close the connection and free resources. */
136static void connSocketClose(connection *conn) {
137    if (conn->fd != -1) {
138        aeDeleteFileEvent(server.el,conn->fd, AE_READABLE | AE_WRITABLE);
139        close(conn->fd);
140        conn->fd = -1;
141    }
142
143    /* If called from within a handler, schedule the close but
144     * keep the connection until the handler returns.
145     */
146    if (connHasRefs(conn)) {
147        conn->flags |= CONN_FLAG_CLOSE_SCHEDULED;
148        return;
149    }
150
151    zfree(conn);
152}

154static int connSocketWrite(connection *conn, const void *data, size_t data_len) {
155    int ret = write(conn->fd, data, data_len);
156    if (ret < 0 && errno != EAGAIN) {
157        conn->last_errno = errno;
158
159        /* Don't overwrite the state of a connection that is not already
160         * connected, not to mess with handler callbacks.
161         */
162        if (errno != EINTR && conn->state == CONN_STATE_CONNECTED)
163            conn->state = CONN_STATE_ERROR;
164    }
165
166    return ret;
167}

184static int connSocketRead(connection *conn, void *buf, size_t buf_len) {
185    int ret = read(conn->fd, buf, buf_len);
186    if (!ret) {
187        conn->state = CONN_STATE_CLOSED;
188    } else if (ret < 0 && errno != EAGAIN) {
189        conn->last_errno = errno;
190
191        /* Don't overwrite the state of a connection that is not already
192         * connected, not to mess with handler callbacks.
193         */
194        if (errno != EINTR && conn->state == CONN_STATE_CONNECTED)
195            conn->state = CONN_STATE_ERROR;
196    }
197
198    return ret;
199}

201static int connSocketAccept(connection *conn, ConnectionCallbackFunc accept_handler) {
202    int ret = C_OK;
203
204    if (conn->state != CONN_STATE_ACCEPTING) return C_ERR;
205    conn->state = CONN_STATE_CONNECTED;
206
207    connIncrRefs(conn);
208    if (!callHandler(conn, accept_handler)) ret = C_ERR;
209    connDecrRefs(conn);
210
211    return ret;
212}

214/* Register a write handler, to be called when the connection is writable.
215 * If NULL, the existing handler is removed.
216 *
217 * The barrier flag indicates a write barrier is requested, resulting with
218 * CONN_FLAG_WRITE_BARRIER set. This will ensure that the write handler is
219 * always called before and not after the read handler in a single event
220 * loop.
221 */
222static int connSocketSetWriteHandler(connection *conn, ConnectionCallbackFunc func, int barrier) {
223    if (func == conn->write_handler) return C_OK;
224
225    conn->write_handler = func;
226    if (barrier)
227        conn->flags |= CONN_FLAG_WRITE_BARRIER;
228    else
229        conn->flags &= ~CONN_FLAG_WRITE_BARRIER;
230    if (!conn->write_handler)
231        aeDeleteFileEvent(server.el,conn->fd,AE_WRITABLE);
232    else
233        if (aeCreateFileEvent(server.el,conn->fd,AE_WRITABLE,
234                    conn->type->ae_handler,conn) == AE_ERR) return C_ERR;
235    return C_OK;
236}

238/* Register a read handler, to be called when the connection is readable.
239 * If NULL, the existing handler is removed.
240 */
241static int connSocketSetReadHandler(connection *conn, ConnectionCallbackFunc func) {
242    if (func == conn->read_handler) return C_OK;
243
244    conn->read_handler = func;
245    if (!conn->read_handler)
246        aeDeleteFileEvent(server.el,conn->fd,AE_READABLE);
247    else
248        if (aeCreateFileEvent(server.el,conn->fd,
249                    AE_READABLE,conn->type->ae_handler,conn) == AE_ERR) return C_ERR;
250    return C_OK;
251}

257static void connSocketEventHandler(struct aeEventLoop *el, int fd, void *clientData, int mask)
258{
259    UNUSED(el);
260    UNUSED(fd);
261    connection *conn = clientData;
262
263    if (conn->state == CONN_STATE_CONNECTING &&
264            (mask & AE_WRITABLE) && conn->conn_handler) {
265
266        int conn_error = anetGetError(conn->fd);
267        if (conn_error) {
268            conn->last_errno = conn_error;
269            conn->state = CONN_STATE_ERROR;
270        } else {
271            conn->state = CONN_STATE_CONNECTED;
272        }
273
274        if (!conn->write_handler) aeDeleteFileEvent(server.el,conn->fd,AE_WRITABLE);
275
276        if (!callHandler(conn, conn->conn_handler)) return;
277        conn->conn_handler = NULL;
278    }
279
280    /* Normally we execute the readable event first, and the writable
281     * event later. This is useful as sometimes we may be able
282     * to serve the reply of a query immediately after processing the
283     * query.
284     *
285     * However if WRITE_BARRIER is set in the mask, our application is
286     * asking us to do the reverse: never fire the writable event
287     * after the readable. In such a case, we invert the calls.
288     * This is useful when, for instance, we want to do things
289     * in the beforeSleep() hook, like fsync'ing a file to disk,
290     * before replying to a client. */
291    int invert = conn->flags & CONN_FLAG_WRITE_BARRIER;
292
293    int call_write = (mask & AE_WRITABLE) && conn->write_handler;
294    int call_read = (mask & AE_READABLE) && conn->read_handler;
295
296    /* Handle normal I/O flows */
297    if (!invert && call_read) {
298        if (!callHandler(conn, conn->read_handler)) return;
299    }
300    /* Fire the writable event. */
301    if (call_write) {
302        if (!callHandler(conn, conn->write_handler)) return;
303    }
304    /* If we have to invert the call, fire the readable event now
305     * after the writable one. */
306    if (invert && call_read) {
307        if (!callHandler(conn, conn->read_handler)) return;
308    }
309}

311static void connSocketAcceptHandler(aeEventLoop *el, int fd, void *privdata, int mask) {
312    int cport, cfd, max = MAX_ACCEPTS_PER_CALL;
313    char cip[NET_IP_STR_LEN];
314    UNUSED(el);
315    UNUSED(mask);
316    UNUSED(privdata);
317
318    while(max--) {
319        cfd = anetTcpAccept(server.neterr, fd, cip, sizeof(cip), &cport);
320        if (cfd == ANET_ERR) {
321            if (errno != EWOULDBLOCK)
322                serverLog(LL_WARNING,
323                    "Accepting client connection: %s", server.neterr);
324            return;
325        }
326        serverLog(LL_VERBOSE,"Accepted %s:%d", cip, cport);
327        acceptCommonHandler(connCreateAcceptedSocket(cfd, NULL),0,cip);
328    }
329}

331static int connSocketAddr(connection *conn, char *ip, size_t ip_len, int *port, int remote) {
332    if (anetFdToString(conn->fd, ip, ip_len, port, remote) == 0)
333        return C_OK;
334
335    conn->last_errno = errno;
336    return C_ERR;
337}

339static int connSocketIsLocal(connection *conn) {
340    char cip[NET_IP_STR_LEN + 1] = { 0 };
341
342    if (connSocketAddr(conn, cip, sizeof(cip) - 1, NULL, 1) == C_ERR)
343        return -1;
344
345    return !strncmp(cip, "127.", 4) || !strcmp(cip, "::1");
346}

352static int connSocketBlockingConnect(connection *conn, const char *addr, int port, long long timeout) {
353    int fd = anetTcpNonBlockConnect(NULL,addr,port);
354    if (fd == -1) {
355        conn->state = CONN_STATE_ERROR;
356        conn->last_errno = errno;
357        return C_ERR;
358    }
359
360    if ((aeWait(fd, AE_WRITABLE, timeout) & AE_WRITABLE) == 0) {
361        conn->state = CONN_STATE_ERROR;
362        conn->last_errno = ETIMEDOUT;
363    }
364
365    conn->fd = fd;
366    conn->state = CONN_STATE_CONNECTED;
367    return C_OK;
368}
