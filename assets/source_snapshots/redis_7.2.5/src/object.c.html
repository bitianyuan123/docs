45    o->type = type;

53void initObjectLRUOrLFU(robj *o) {
54    if (o->refcount == OBJ_SHARED_REFCOUNT)
55        return;
56    /* Set the LRU to the current lruclock (minutes resolution), or
57     * alternatively the LFU counter. */
58    if (server.maxmemory_policy & MAXMEMORY_FLAG_LFU) {
59        o->lru = (LFUGetTimeInMinutes() << 8) | LFU_INIT_VAL;
60    } else {
61        o->lru = LRU_CLOCK();
62    }
63    return;
64}

66/* Set a special refcount in the object to make it "shared":
67 * incrRefCount and decrRefCount() will test for this special refcount
68 * and will not touch the object. This way it is free to access shared
69 * objects such as small integers from different threads without any
70 * mutex.
71 *
72 * A common pattern to create shared objects:
73 *
74 * robj *myobject = makeObjectShared(createObject(...));
75 *
76 */
77robj *makeObjectShared(robj *o) {
78    serverAssert(o->refcount == 1);
79    o->refcount = OBJ_SHARED_REFCOUNT;
80    return o;
81}

89/* Create a string object with encoding OBJ_ENCODING_EMBSTR, that is
90 * an object where the sds string is actually an unmodifiable string
91 * allocated in the same chunk as the object itself. */
92robj *createEmbeddedStringObject(const char *ptr, size_t len) {
93    robj *o = zmalloc(sizeof(robj)+sizeof(struct sdshdr8)+len+1);
94    struct sdshdr8 *sh = (void*)(o+1);
95
96    o->type = OBJ_STRING;
97    o->encoding = OBJ_ENCODING_EMBSTR;
98    o->ptr = sh+1;
99    o->refcount = 1;
100    o->lru = 0;
101
102    sh->len = len;
103    sh->alloc = len;
104    sh->flags = SDS_TYPE_8;
105    if (ptr == SDS_NOINIT)
106        sh->buf[len] = '\0';
107    else if (ptr) {
108        memcpy(sh->buf,ptr,len);
109        sh->buf[len] = '\0';
110    } else {
111        memset(sh->buf,0,len+1);
112    }
113    return o;
114}

116/* Create a string object with EMBSTR encoding if it is smaller than
117 * OBJ_ENCODING_EMBSTR_SIZE_LIMIT, otherwise the RAW encoding is
118 * used.
119 *
120 * The current limit of 44 is chosen so that the biggest string object
121 * we allocate as EMBSTR will still fit into the 64 byte arena of jemalloc. */
122#define OBJ_ENCODING_EMBSTR_SIZE_LIMIT 44
123robj *createStringObject(const char *ptr, size_t len) {
124    if (len <= OBJ_ENCODING_EMBSTR_SIZE_LIMIT)
125        return createEmbeddedStringObject(ptr,len);
126    else
127        return createRawStringObject(ptr,len);
128}

145/* Create a string object from a long long value according to the specified flag. */
146#define LL2STROBJ_AUTO 0       /* automatically create the optimal string object */
147#define LL2STROBJ_NO_SHARED 1  /* disallow shared objects */
148#define LL2STROBJ_NO_INT_ENC 2 /* disallow integer encoded objects. */
149robj *createStringObjectFromLongLongWithOptions(long long value, int flag) {
150    robj *o;
151
152    if (value >= 0 && value < OBJ_SHARED_INTEGERS && flag == LL2STROBJ_AUTO) {
153        o = shared.integers[value];
154    } else {
155        if ((value >= LONG_MIN && value <= LONG_MAX) && flag != LL2STROBJ_NO_INT_ENC) {
156            o = createObject(OBJ_STRING, NULL);
157            o->encoding = OBJ_ENCODING_INT;
158            o->ptr = (void*)((long)value);
159        } else {
160            char buf[LONG_STR_SIZE];
161            int len = ll2string(buf, sizeof(buf), value);
162            o = createStringObject(buf, len);
163        }
164    }
165    return o;
166}

174/* The function avoids returning a shared integer when LFU/LRU info
175 * are needed, that is, when the object is used as a value in the key
176 * space(for instance when the INCR command is used), and Redis is
177 * configured to evict based on LFU/LRU, so we want LFU/LRU values
178 * specific for each key. */
179robj *createStringObjectFromLongLongForValue(long long value) {
180    if (server.maxmemory == 0 || !(server.maxmemory_policy & MAXMEMORY_FLAG_NO_SHARED_INTEGERS)) {
181        /* If the maxmemory policy permits, we can still return shared integers */
182        return createStringObjectFromLongLongWithOptions(value, LL2STROBJ_AUTO);
183    } else {
184        return createStringObjectFromLongLongWithOptions(value, LL2STROBJ_NO_SHARED);
185    }
186}

206/* Duplicate a string object, with the guarantee that the returned object
207 * has the same encoding as the original one.
208 *
209 * This function also guarantees that duplicating a small integer object
210 * (or a string object that contains a representation of a small integer)
211 * will always result in a fresh object that is unshared (refcount == 1).
212 *
213 * The resulting object always has refcount set to 1. */
214robj *dupStringObject(const robj *o) {
215    robj *d;
216
217    serverAssert(o->type == OBJ_STRING);
218
219    switch(o->encoding) {
220    case OBJ_ENCODING_RAW:
221        return createRawStringObject(o->ptr,sdslen(o->ptr));
222    case OBJ_ENCODING_EMBSTR:
223        return createEmbeddedStringObject(o->ptr,sdslen(o->ptr));
224    case OBJ_ENCODING_INT:
225        d = createObject(OBJ_STRING, NULL);
226        d->encoding = OBJ_ENCODING_INT;
227        d->ptr = o->ptr;
228        return d;
229    default:
230        serverPanic("Wrong encoding.");
231        break;
232    }
233}

380void incrRefCount(robj *o) {
381    if (o->refcount < OBJ_FIRST_SPECIAL_REFCOUNT) {
382        o->refcount++;
383    } else {
384        if (o->refcount == OBJ_SHARED_REFCOUNT) {
385            /* Nothing to do: this refcount is immutable. */
386        } else if (o->refcount == OBJ_STATIC_REFCOUNT) {
387            serverPanic("You tried to retain an object allocated in the stack");
388        }
389    }
390}

392void decrRefCount(robj *o) {
393    if (o->refcount == 1) {
394        switch(o->type) {
395        case OBJ_STRING: freeStringObject(o); break;
396        case OBJ_LIST: freeListObject(o); break;
397        case OBJ_SET: freeSetObject(o); break;
398        case OBJ_ZSET: freeZsetObject(o); break;
399        case OBJ_HASH: freeHashObject(o); break;
400        case OBJ_MODULE: freeModuleObject(o); break;
401        case OBJ_STREAM: freeStreamObject(o); break;
402        default: serverPanic("Unknown object type"); break;
403        }
404        zfree(o);
405    } else {
406        if (o->refcount <= 0) serverPanic("decrRefCount against refcount <= 0");
407        if (o->refcount != OBJ_SHARED_REFCOUNT) o->refcount--;
408    }
409}

423/* See dismissObject() */
424void dismissListObject(robj *o, size_t size_hint) {
425    if (o->encoding == OBJ_ENCODING_QUICKLIST) {
426        quicklist *ql = o->ptr;
427        serverAssert(ql->len != 0);
428        /* We iterate all nodes only when average node size is bigger than a
429         * page size, and there's a high chance we'll actually dismiss something. */
430        if (size_hint / ql->len >= server.page_size) {
431            quicklistNode *node = ql->head;
432            while (node) {
433                if (quicklistNodeIsCompressed(node)) {
434                    dismissMemory(node->entry, ((quicklistLZF*)node->entry)->sz);
435                } else {
436                    dismissMemory(node->entry, node->sz);
437                }
438                node = node->next;
439            }
440        }
441    } else if (o->encoding == OBJ_ENCODING_LISTPACK) {
442        dismissMemory(o->ptr, lpBytes((unsigned char*)o->ptr));
443    } else {
444        serverPanic("Unknown list encoding type");
445    }
446}

448/* See dismissObject() */
449void dismissSetObject(robj *o, size_t size_hint) {
450    if (o->encoding == OBJ_ENCODING_HT) {
451        dict *set = o->ptr;
452        serverAssert(dictSize(set) != 0);
453        /* We iterate all nodes only when average member size is bigger than a
454         * page size, and there's a high chance we'll actually dismiss something. */
455        if (size_hint / dictSize(set) >= server.page_size) {
456            dictEntry *de;
457            dictIterator *di = dictGetIterator(set);
458            while ((de = dictNext(di)) != NULL) {
459                dismissSds(dictGetKey(de));
460            }
461            dictReleaseIterator(di);
462        }
463
464        /* Dismiss hash table memory. */
465        dismissMemory(set->ht_table[0], DICTHT_SIZE(set->ht_size_exp[0])*sizeof(dictEntry*));
466        dismissMemory(set->ht_table[1], DICTHT_SIZE(set->ht_size_exp[1])*sizeof(dictEntry*));
467    } else if (o->encoding == OBJ_ENCODING_INTSET) {
468        dismissMemory(o->ptr, intsetBlobLen((intset*)o->ptr));
469    } else if (o->encoding == OBJ_ENCODING_LISTPACK) {
470        dismissMemory(o->ptr, lpBytes((unsigned char *)o->ptr));
471    } else {
472        serverPanic("Unknown set encoding type");
473    }
474}

476/* See dismissObject() */
477void dismissZsetObject(robj *o, size_t size_hint) {
478    if (o->encoding == OBJ_ENCODING_SKIPLIST) {
479        zset *zs = o->ptr;
480        zskiplist *zsl = zs->zsl;
481        serverAssert(zsl->length != 0);
482        /* We iterate all nodes only when average member size is bigger than a
483         * page size, and there's a high chance we'll actually dismiss something. */
484        if (size_hint / zsl->length >= server.page_size) {
485            zskiplistNode *zn = zsl->tail;
486            while (zn != NULL) {
487                dismissSds(zn->ele);
488                zn = zn->backward;
489            }
490        }
491
492        /* Dismiss hash table memory. */
493        dict *d = zs->dict;
494        dismissMemory(d->ht_table[0], DICTHT_SIZE(d->ht_size_exp[0])*sizeof(dictEntry*));
495        dismissMemory(d->ht_table[1], DICTHT_SIZE(d->ht_size_exp[1])*sizeof(dictEntry*));
496    } else if (o->encoding == OBJ_ENCODING_LISTPACK) {
497        dismissMemory(o->ptr, lpBytes((unsigned char*)o->ptr));
498    } else {
499        serverPanic("Unknown zset encoding type");
500    }
501}

503/* See dismissObject() */
504void dismissHashObject(robj *o, size_t size_hint) {
505    if (o->encoding == OBJ_ENCODING_HT) {
506        dict *d = o->ptr;
507        serverAssert(dictSize(d) != 0);
508        /* We iterate all fields only when average field/value size is bigger than
509         * a page size, and there's a high chance we'll actually dismiss something. */
510        if (size_hint / dictSize(d) >= server.page_size) {
511            dictEntry *de;
512            dictIterator *di = dictGetIterator(d);
513            while ((de = dictNext(di)) != NULL) {
514                /* Only dismiss values memory since the field size
515                 * usually is small. */
516                dismissSds(dictGetVal(de));
517            }
518            dictReleaseIterator(di);
519        }
520
521        /* Dismiss hash table memory. */
522        dismissMemory(d->ht_table[0], DICTHT_SIZE(d->ht_size_exp[0])*sizeof(dictEntry*));
523        dismissMemory(d->ht_table[1], DICTHT_SIZE(d->ht_size_exp[1])*sizeof(dictEntry*));
524    } else if (o->encoding == OBJ_ENCODING_LISTPACK) {
525        dismissMemory(o->ptr, lpBytes((unsigned char*)o->ptr));
526    } else {
527        serverPanic("Unknown hash encoding type");
528    }
529}

531/* See dismissObject() */
532void dismissStreamObject(robj *o, size_t size_hint) {
533    stream *s = o->ptr;
534    rax *rax = s->rax;
535    if (raxSize(rax) == 0) return;
536
537    /* Iterate only on stream entries, although size_hint may include serialized
538     * consumer groups info, but usually, stream entries take up most of
539     * the space. */
540    if (size_hint / raxSize(rax) >= server.page_size) {
541        raxIterator ri;
542        raxStart(&ri,rax);
543        raxSeek(&ri,"^",NULL,0);
544        while (raxNext(&ri)) {
545            dismissMemory(ri.data, lpBytes(ri.data));
546        }
547        raxStop(&ri);
548    }
549}

551/* When creating a snapshot in a fork child process, the main process and child
552 * process share the same physical memory pages, and if / when the parent
553 * modifies any keys due to write traffic, it'll cause CoW which consume
554 * physical memory. In the child process, after serializing the key and value,
555 * the data is definitely not accessed again, so to avoid unnecessary CoW, we
556 * try to release their memory back to OS. see dismissMemory().
557 *
558 * Because of the cost of iterating all node/field/member/entry of complex data
559 * types, we iterate and dismiss them only when approximate average we estimate
560 * the size of an individual allocation is more than a page size of OS.
561 * 'size_hint' is the size of serialized value. This method is not accurate, but
562 * it can reduce unnecessary iteration for complex data types that are probably
563 * not going to release any memory. */
564void dismissObject(robj *o, size_t size_hint) {
565    /* madvise(MADV_DONTNEED) may not work if Transparent Huge Pages is enabled. */
566    if (server.thp_enabled) return;
567
568    /* Currently we use zmadvise_dontneed only when we use jemalloc with Linux.
569     * so we avoid these pointless loops when they're not going to do anything. */
570#if defined(USE_JEMALLOC) && defined(__linux__)
571    if (o->refcount != 1) return;
572    switch(o->type) {
573        case OBJ_STRING: dismissStringObject(o); break;
574        case OBJ_LIST: dismissListObject(o, size_hint); break;
575        case OBJ_SET: dismissSetObject(o, size_hint); break;
576        case OBJ_ZSET: dismissZsetObject(o, size_hint); break;
577        case OBJ_HASH: dismissHashObject(o, size_hint); break;
578        case OBJ_STREAM: dismissStreamObject(o, size_hint); break;
579        default: break;
580    }
581#else
582    UNUSED(o); UNUSED(size_hint);
583#endif
584}

616/* Optimize the SDS string inside the string object to require little space,
617 * in case there is more than 10% of free space at the end of the SDS. */
618void trimStringObjectIfNeeded(robj *o, int trim_small_values) {
619    if (o->encoding != OBJ_ENCODING_RAW) return;
620    /* A string may have free space in the following cases:
621     * 1. When an arg len is greater than PROTO_MBULK_BIG_ARG the query buffer may be used directly as the SDS string.
622     * 2. When utilizing the argument caching mechanism in Lua. 
623     * 3. When calling from RM_TrimStringAllocation (trim_small_values is true). */
624    size_t len = sdslen(o->ptr);
625    if (len >= PROTO_MBULK_BIG_ARG ||
626        trim_small_values||
627        (server.executing_client && server.executing_client->flags & CLIENT_SCRIPT && len < LUA_CMD_OBJCACHE_MAX_LEN)) {
628        if (sdsavail(o->ptr) > len/10) {
629            o->ptr = sdsRemoveFreeSpace(o->ptr, 0);
630        }
631    }
632}

634/* Try to encode a string object in order to save space */
635robj *tryObjectEncodingEx(robj *o, int try_trim) {
636    long value;
637    sds s = o->ptr;
638    size_t len;
639
640    /* Make sure this is a string object, the only type we encode
641     * in this function. Other types use encoded memory efficient
642     * representations but are handled by the commands implementing
643     * the type. */
644    serverAssertWithInfo(NULL,o,o->type == OBJ_STRING);
645
646    /* We try some specialized encoding only for objects that are
647     * RAW or EMBSTR encoded, in other words objects that are still
648     * in represented by an actually array of chars. */
649    if (!sdsEncodedObject(o)) return o;
650
651    /* It's not safe to encode shared objects: shared objects can be shared
652     * everywhere in the "object space" of Redis and may end in places where
653     * they are not handled. We handle them only as values in the keyspace. */
654     if (o->refcount > 1) return o;
655
656    /* Check if we can represent this string as a long integer.
657     * Note that we are sure that a string larger than 20 chars is not
658     * representable as a 32 nor 64 bit integer. */
659    len = sdslen(s);
660    if (len <= 20 && string2l(s,len,&value)) {
661        /* This object is encodable as a long. Try to use a shared object.
662         * Note that we avoid using shared integers when maxmemory is used
663         * because every object needs to have a private LRU field for the LRU
664         * algorithm to work well. */
665        if ((server.maxmemory == 0 ||
666            !(server.maxmemory_policy & MAXMEMORY_FLAG_NO_SHARED_INTEGERS)) &&
667            value >= 0 &&
668            value < OBJ_SHARED_INTEGERS)
669        {
670            decrRefCount(o);
671            return shared.integers[value];
672        } else {
673            if (o->encoding == OBJ_ENCODING_RAW) {
674                sdsfree(o->ptr);
675                o->encoding = OBJ_ENCODING_INT;
676                o->ptr = (void*) value;
677                return o;
678            } else if (o->encoding == OBJ_ENCODING_EMBSTR) {
679                decrRefCount(o);
680                return createStringObjectFromLongLongForValue(value);
681            }
682        }
683    }
684
685    /* If the string is small and is still RAW encoded,
686     * try the EMBSTR encoding which is more efficient.
687     * In this representation the object and the SDS string are allocated
688     * in the same chunk of memory to save space and cache misses. */
689    if (len <= OBJ_ENCODING_EMBSTR_SIZE_LIMIT) {
690        robj *emb;
691
692        if (o->encoding == OBJ_ENCODING_EMBSTR) return o;
693        emb = createEmbeddedStringObject(s,sdslen(s));
694        decrRefCount(o);
695        return emb;
696    }
697
698    /* We can't encode the object...
699     * Do the last try, and at least optimize the SDS string inside */
700    if (try_trim)
701        trimStringObjectIfNeeded(o, 0);
702
703    /* Return the original object. */
704    return o;
705}

707robj *tryObjectEncoding(robj *o) {
708    return tryObjectEncodingEx(o, 1);
709}

731/* Compare two string objects via strcmp() or strcoll() depending on flags.
732 * Note that the objects may be integer-encoded. In such a case we
733 * use ll2string() to get a string representation of the numbers on the stack
734 * and compare the strings, it's much faster than calling getDecodedObject().
735 *
736 * Important note: when REDIS_COMPARE_BINARY is used a binary-safe comparison
737 * is used. */
738
739#define REDIS_COMPARE_BINARY (1<<0)
740#define REDIS_COMPARE_COLL (1<<1)
741
742int compareStringObjectsWithFlags(const robj *a, const robj *b, int flags) {
743    serverAssertWithInfo(NULL,a,a->type == OBJ_STRING && b->type == OBJ_STRING);
744    char bufa[128], bufb[128], *astr, *bstr;
745    size_t alen, blen, minlen;
746
747    if (a == b) return 0;
748    if (sdsEncodedObject(a)) {
749        astr = a->ptr;
750        alen = sdslen(astr);
751    } else {
752        alen = ll2string(bufa,sizeof(bufa),(long) a->ptr);
753        astr = bufa;
754    }
755    if (sdsEncodedObject(b)) {
756        bstr = b->ptr;
757        blen = sdslen(bstr);
758    } else {
759        blen = ll2string(bufb,sizeof(bufb),(long) b->ptr);
760        bstr = bufb;
761    }
762    if (flags & REDIS_COMPARE_COLL) {
763        return strcoll(astr,bstr);
764    } else {
765        int cmp;
766
767        minlen = (alen < blen) ? alen : blen;
768        cmp = memcmp(astr,bstr,minlen);
769        if (cmp == 0) return alen-blen;
770        return cmp;
771    }
772}

774/* Wrapper for compareStringObjectsWithFlags() using binary comparison. */
775int compareStringObjects(const robj *a, const robj *b) {
776    return compareStringObjectsWithFlags(a,b,REDIS_COMPARE_BINARY);
777}

779/* Wrapper for compareStringObjectsWithFlags() using collation. */
780int collateStringObjects(const robj *a, const robj *b) {
781    return compareStringObjectsWithFlags(a,b,REDIS_COMPARE_COLL);

785 * point of view of a string comparison, otherwise 0 is returned. Note that
786 * this function is faster then checking for (compareStringObject(a,b) == 0)
787 * because it can perform some more optimization. */
788int equalStringObjects(robj *a, robj *b) {
789    if (a->encoding == OBJ_ENCODING_INT &&

793        return a->ptr == b->ptr;
794    } else {
795        return compareStringObjects(a,b) == 0;
796    }

974/* This is a helper function with the goal of estimating the memory
975 * size of a radix tree that is used to store Stream IDs.
976 *
977 * Note: to guess the size of the radix tree is not trivial, so we
978 * approximate it considering 16 bytes of data overhead for each
979 * key (the ID), and then adding the number of bare nodes, plus some
980 * overhead due by the data and child pointers. This secret recipe
981 * was obtained by checking the average radix tree created by real
982 * workloads, and then adjusting the constants to get numbers that
983 * more or less match the real memory usage.
984 *
985 * Actually the number of nodes and keys may be different depending
986 * on the insertion speed and thus the ability of the radix tree
987 * to compress prefixes. */
988size_t streamRadixTreeMemoryUsage(rax *rax) {
989    size_t size = sizeof(*rax);
990    size = rax->numele * sizeof(streamID);
991    size += rax->numnodes * sizeof(raxNode);
992    /* Add a fixed overhead due to the aux data pointer, children, ... */
993    size += rax->numnodes * sizeof(long)*30;
994    return size;
995}

997/* Returns the size in bytes consumed by the key's value in RAM.
998 * Note that the returned value is just an approximation, especially in the
999 * case of aggregated data types where only "sample_size" elements
1000 * are checked and averaged to estimate the total size. */
1001#define OBJ_COMPUTE_SIZE_DEF_SAMPLES 5 /* Default sample size. */
1002size_t objectComputeSize(robj *key, robj *o, size_t sample_size, int dbid) {
1003    sds ele, ele2;
1004    dict *d;
1005    dictIterator *di;
1006    struct dictEntry *de;
1007    size_t asize = 0, elesize = 0, samples = 0;
1008
1009    if (o->type == OBJ_STRING) {
1010        if(o->encoding == OBJ_ENCODING_INT) {
1011            asize = sizeof(*o);
1012        } else if(o->encoding == OBJ_ENCODING_RAW) {
1013            asize = sdsZmallocSize(o->ptr)+sizeof(*o);
1014        } else if(o->encoding == OBJ_ENCODING_EMBSTR) {
1015            asize = zmalloc_size((void *)o);
1016        } else {
1017            serverPanic("Unknown string encoding");
1018        }
1019    } else if (o->type == OBJ_LIST) {
1020        if (o->encoding == OBJ_ENCODING_QUICKLIST) {
1021            quicklist *ql = o->ptr;
1022            quicklistNode *node = ql->head;
1023            asize = sizeof(*o)+sizeof(quicklist);
1024            do {
1025                elesize += sizeof(quicklistNode)+zmalloc_size(node->entry);
1026                samples++;
1027            } while ((node = node->next) && samples < sample_size);
1028            asize += (double)elesize/samples*ql->len;
1029        } else if (o->encoding == OBJ_ENCODING_LISTPACK) {
1030            asize = sizeof(*o)+zmalloc_size(o->ptr);
1031        } else {
1032            serverPanic("Unknown list encoding");
1033        }
1034    } else if (o->type == OBJ_SET) {
1035        if (o->encoding == OBJ_ENCODING_HT) {
1036            d = o->ptr;
1037            di = dictGetIterator(d);
1038            asize = sizeof(*o)+sizeof(dict)+(sizeof(struct dictEntry*)*dictSlots(d));
1039            while((de = dictNext(di)) != NULL && samples < sample_size) {
1040                ele = dictGetKey(de);
1041                elesize += dictEntryMemUsage() + sdsZmallocSize(ele);
1042                samples++;
1043            }
1044            dictReleaseIterator(di);
1045            if (samples) asize += (double)elesize/samples*dictSize(d);
1046        } else if (o->encoding == OBJ_ENCODING_INTSET) {
1047            asize = sizeof(*o)+zmalloc_size(o->ptr);
1048        } else if (o->encoding == OBJ_ENCODING_LISTPACK) {
1049            asize = sizeof(*o)+zmalloc_size(o->ptr);
1050        } else {
1051            serverPanic("Unknown set encoding");
1052        }
1053    } else if (o->type == OBJ_ZSET) {
1054        if (o->encoding == OBJ_ENCODING_LISTPACK) {
1055            asize = sizeof(*o)+zmalloc_size(o->ptr);
1056        } else if (o->encoding == OBJ_ENCODING_SKIPLIST) {
1057            d = ((zset*)o->ptr)->dict;
1058            zskiplist *zsl = ((zset*)o->ptr)->zsl;
1059            zskiplistNode *znode = zsl->header->level[0].forward;
1060            asize = sizeof(*o)+sizeof(zset)+sizeof(zskiplist)+sizeof(dict)+
1061                    (sizeof(struct dictEntry*)*dictSlots(d))+
1062                    zmalloc_size(zsl->header);
1063            while(znode != NULL && samples < sample_size) {
1064                elesize += sdsZmallocSize(znode->ele);
1065                elesize += dictEntryMemUsage()+zmalloc_size(znode);
1066                samples++;
1067                znode = znode->level[0].forward;
1068            }
1069            if (samples) asize += (double)elesize/samples*dictSize(d);
1070        } else {
1071            serverPanic("Unknown sorted set encoding");
1072        }
1073    } else if (o->type == OBJ_HASH) {
1074        if (o->encoding == OBJ_ENCODING_LISTPACK) {
1075            asize = sizeof(*o)+zmalloc_size(o->ptr);
1076        } else if (o->encoding == OBJ_ENCODING_HT) {
1077            d = o->ptr;
1078            di = dictGetIterator(d);
1079            asize = sizeof(*o)+sizeof(dict)+(sizeof(struct dictEntry*)*dictSlots(d));
1080            while((de = dictNext(di)) != NULL && samples < sample_size) {
1081                ele = dictGetKey(de);
1082                ele2 = dictGetVal(de);
1083                elesize += sdsZmallocSize(ele) + sdsZmallocSize(ele2);
1084                elesize += dictEntryMemUsage();
1085                samples++;
1086            }
1087            dictReleaseIterator(di);
1088            if (samples) asize += (double)elesize/samples*dictSize(d);
1089        } else {
1090            serverPanic("Unknown hash encoding");
1091        }
1092    } else if (o->type == OBJ_STREAM) {
1093        stream *s = o->ptr;
1094        asize = sizeof(*o)+sizeof(*s);
1095        asize += streamRadixTreeMemoryUsage(s->rax);
1096
1097        /* Now we have to add the listpacks. The last listpack is often non
1098         * complete, so we estimate the size of the first N listpacks, and
1099         * use the average to compute the size of the first N-1 listpacks, and
1100         * finally add the real size of the last node. */
1101        raxIterator ri;
1102        raxStart(&ri,s->rax);
1103        raxSeek(&ri,"^",NULL,0);
1104        size_t lpsize = 0, samples = 0;
1105        while(samples < sample_size && raxNext(&ri)) {
1106            unsigned char *lp = ri.data;
1107            /* Use the allocated size, since we overprovision the node initially. */
1108            lpsize += zmalloc_size(lp);
1109            samples++;
1110        }
1111        if (s->rax->numele <= samples) {
1112            asize += lpsize;
1113        } else {
1114            if (samples) lpsize /= samples; /* Compute the average. */
1115            asize += lpsize * (s->rax->numele-1);
1116            /* No need to check if seek succeeded, we enter this branch only
1117             * if there are a few elements in the radix tree. */
1118            raxSeek(&ri,"$",NULL,0);
1119            raxNext(&ri);
1120            /* Use the allocated size, since we overprovision the node initially. */
1121            asize += zmalloc_size(ri.data);
1122        }
1123        raxStop(&ri);
1124
1125        /* Consumer groups also have a non trivial memory overhead if there
1126         * are many consumers and many groups, let's count at least the
1127         * overhead of the pending entries in the groups and consumers
1128         * PELs. */
1129        if (s->cgroups) {
1130            raxStart(&ri,s->cgroups);
1131            raxSeek(&ri,"^",NULL,0);
1132            while(raxNext(&ri)) {
1133                streamCG *cg = ri.data;
1134                asize += sizeof(*cg);
1135                asize += streamRadixTreeMemoryUsage(cg->pel);
1136                asize += sizeof(streamNACK)*raxSize(cg->pel);
1137
1138                /* For each consumer we also need to add the basic data
1139                 * structures and the PEL memory usage. */
1140                raxIterator cri;
1141                raxStart(&cri,cg->consumers);
1142                raxSeek(&cri,"^",NULL,0);
1143                while(raxNext(&cri)) {
1144                    streamConsumer *consumer = cri.data;
1145                    asize += sizeof(*consumer);
1146                    asize += sdslen(consumer->name);
1147                    asize += streamRadixTreeMemoryUsage(consumer->pel);
1148                    /* Don't count NACKs again, they are shared with the
1149                     * consumer group PEL. */
1150                }
1151                raxStop(&cri);
1152            }
1153            raxStop(&ri);
1154        }
1155    } else if (o->type == OBJ_MODULE) {
1156        asize = moduleGetMemUsage(key, o, sample_size, dbid);
1157    } else {
1158        serverPanic("Unknown object type");
1159    }
1160    return asize;
1161}

1169/* Return a struct redisMemOverhead filled with memory overhead
1170 * information used for the MEMORY OVERHEAD and INFO command. The returned
1171 * structure pointer should be freed calling freeMemoryOverheadData(). */
1172struct redisMemOverhead *getMemoryOverheadData(void) {
1173    int j;
1174    size_t mem_total = 0;
1175    size_t mem = 0;
1176    size_t zmalloc_used = zmalloc_used_memory();
1177    struct redisMemOverhead *mh = zcalloc(sizeof(*mh));
1178
1179    mh->total_allocated = zmalloc_used;
1180    mh->startup_allocated = server.initial_memory_usage;
1181    mh->peak_allocated = server.stat_peak_memory;
1182    mh->total_frag =
1183        (float)server.cron_malloc_stats.process_rss / server.cron_malloc_stats.zmalloc_used;
1184    mh->total_frag_bytes =
1185        server.cron_malloc_stats.process_rss - server.cron_malloc_stats.zmalloc_used;
1186    mh->allocator_frag =
1187        (float)server.cron_malloc_stats.allocator_active / server.cron_malloc_stats.allocator_allocated;
1188    mh->allocator_frag_bytes =
1189        server.cron_malloc_stats.allocator_active - server.cron_malloc_stats.allocator_allocated;
1190    mh->allocator_rss =
1191        (float)server.cron_malloc_stats.allocator_resident / server.cron_malloc_stats.allocator_active;
1192    mh->allocator_rss_bytes =
1193        server.cron_malloc_stats.allocator_resident - server.cron_malloc_stats.allocator_active;
1194    mh->rss_extra =
1195        (float)server.cron_malloc_stats.process_rss / server.cron_malloc_stats.allocator_resident;
1196    mh->rss_extra_bytes =
1197        server.cron_malloc_stats.process_rss - server.cron_malloc_stats.allocator_resident;
1198
1199    mem_total += server.initial_memory_usage;
1200
1201    /* Replication backlog and replicas share one global replication buffer,
1202     * only if replication buffer memory is more than the repl backlog setting,
1203     * we consider the excess as replicas' memory. Otherwise, replication buffer
1204     * memory is the consumption of repl backlog. */
1205    if (listLength(server.slaves) &&
1206        (long long)server.repl_buffer_mem > server.repl_backlog_size)
1207    {
1208        mh->clients_slaves = server.repl_buffer_mem - server.repl_backlog_size;
1209        mh->repl_backlog = server.repl_backlog_size;
1210    } else {
1211        mh->clients_slaves = 0;
1212        mh->repl_backlog = server.repl_buffer_mem;
1213    }
1214    if (server.repl_backlog) {
1215        /* The approximate memory of rax tree for indexed blocks. */
1216        mh->repl_backlog +=
1217            server.repl_backlog->blocks_index->numnodes * sizeof(raxNode) +
1218            raxSize(server.repl_backlog->blocks_index) * sizeof(void*);
1219    }
1220    mem_total += mh->repl_backlog;
1221    mem_total += mh->clients_slaves;
1222
1223    /* Computing the memory used by the clients would be O(N) if done
1224     * here online. We use our values computed incrementally by
1225     * updateClientMemoryUsage(). */
1226    mh->clients_normal = server.stat_clients_type_memory[CLIENT_TYPE_MASTER]+
1227                         server.stat_clients_type_memory[CLIENT_TYPE_PUBSUB]+
1228                         server.stat_clients_type_memory[CLIENT_TYPE_NORMAL];
1229    mem_total += mh->clients_normal;
1230
1231    mh->cluster_links = server.stat_cluster_links_memory;
1232    mem_total += mh->cluster_links;
1233
1234    mem = 0;
1235    if (server.aof_state != AOF_OFF) {
1236        mem += sdsZmallocSize(server.aof_buf);
1237    }
1238    mh->aof_buffer = mem;
1239    mem_total+=mem;
1240
1241    mem = evalScriptsMemory();
1242    mh->lua_caches = mem;
1243    mem_total+=mem;
1244    mh->functions_caches = functionsMemoryOverhead();
1245    mem_total+=mh->functions_caches;
1246
1247    for (j = 0; j < server.dbnum; j++) {
1248        redisDb *db = server.db+j;
1249        long long keyscount = dictSize(db->dict);
1250        if (keyscount==0) continue;
1251
1252        mh->total_keys += keyscount;
1253        mh->db = zrealloc(mh->db,sizeof(mh->db[0])*(mh->num_dbs+1));
1254        mh->db[mh->num_dbs].dbid = j;
1255
1256        mem = dictMemUsage(db->dict) +
1257              dictSize(db->dict) * sizeof(robj);
1258        mh->db[mh->num_dbs].overhead_ht_main = mem;
1259        mem_total+=mem;
1260
1261        mem = dictMemUsage(db->expires);
1262        mh->db[mh->num_dbs].overhead_ht_expires = mem;
1263        mem_total+=mem;
1264
1265        /* Account for the slot to keys map in cluster mode */
1266        mem = dictSize(db->dict) * dictEntryMetadataSize(db->dict) +
1267              dictMetadataSize(db->dict);
1268        mh->db[mh->num_dbs].overhead_ht_slot_to_keys = mem;
1269        mem_total+=mem;
1270
1271        mh->num_dbs++;
1272    }
1273
1274    mh->overhead_total = mem_total;
1275    mh->dataset = zmalloc_used - mem_total;
1276    mh->peak_perc = (float)zmalloc_used*100/mh->peak_allocated;
1277
1278    /* Metrics computed after subtracting the startup memory from
1279     * the total memory. */
1280    size_t net_usage = 1;
1281    if (zmalloc_used > mh->startup_allocated)
1282        net_usage = zmalloc_used - mh->startup_allocated;
1283    mh->dataset_perc = (float)mh->dataset*100/net_usage;
1284    mh->bytes_per_key = mh->total_keys ? (net_usage / mh->total_keys) : 0;
1285
1286    return mh;
1287}

1395        }

1411/* Set the object LRU/LFU depending on server.maxmemory_policy.
1412 * The lfu_freq arg is only relevant if policy is MAXMEMORY_FLAG_LFU.
1413 * The lru_idle and lru_clock args are only relevant if policy
1414 * is MAXMEMORY_FLAG_LRU.
1415 * Either or both of them may be <0, in that case, nothing is set. */
1416int objectSetLRUOrLFU(robj *val, long long lfu_freq, long long lru_idle,
1417                       long long lru_clock, int lru_multiplier) {
1418    if (server.maxmemory_policy & MAXMEMORY_FLAG_LFU) {
1419        if (lfu_freq >= 0) {
1420            serverAssert(lfu_freq <= 255);
1421            val->lru = (LFUGetTimeInMinutes()<<8) | lfu_freq;
1422            return 1;
1423        }
1424    } else if (lru_idle >= 0) {
1425        /* Provided LRU idle time is in seconds. Scale
1426         * according to the LRU clock resolution this Redis
1427         * instance was compiled with (normally 1000 ms, so the
1428         * below statement will expand to lru_idle*1000/1000. */
1429        lru_idle = lru_idle*lru_multiplier/LRU_CLOCK_RESOLUTION;
1430        long lru_abs = lru_clock - lru_idle; /* Absolute access time. */
1431        /* If the LRU field underflows (since lru_clock is a wrapping clock),
1432         * we need to make it positive again. This be handled by the unwrapping
1433         * code in estimateObjectIdleTime. I.e. imagine a day when lru_clock
1434         * wrap arounds (happens once in some 6 months), and becomes a low
1435         * value, like 10, an lru_idle of 1000 should be near LRU_CLOCK_MAX. */
1436        if (lru_abs < 0)
1437            lru_abs += LRU_CLOCK_MAX;
1438        val->lru = lru_abs;
1439        return 1;
1440    }
1441    return 0;
1442}
