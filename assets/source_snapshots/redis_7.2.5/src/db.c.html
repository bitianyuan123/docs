52/* Update LFU when an object is accessed.
53 * Firstly, decrement the counter if the decrement time is reached.
54 * Then logarithmically increment the counter, and update the access time. */
55void updateLFU(robj *val) {
56    unsigned long counter = LFUDecrAndReturn(val);
57    counter = LFULogIncr(counter);
58    val->lru = (LFUGetTimeInMinutes()<<8) | counter;
59}

61/* Lookup a key for read or write operations, or return NULL if the key is not
62 * found in the specified DB. This function implements the functionality of
63 * lookupKeyRead(), lookupKeyWrite() and their ...WithFlags() variants.
64 *
65 * Side-effects of calling this function:
66 *
67 * 1. A key gets expired if it reached it's TTL.
68 * 2. The key's last access time is updated.
69 * 3. The global keys hits/misses stats are updated (reported in INFO).
70 * 4. If keyspace notifications are enabled, a "keymiss" notification is fired.
71 *
72 * Flags change the behavior of this command:
73 *
74 *  LOOKUP_NONE (or zero): No special flags are passed.
75 *  LOOKUP_NOTOUCH: Don't alter the last access time of the key.
76 *  LOOKUP_NONOTIFY: Don't trigger keyspace event on key miss.
77 *  LOOKUP_NOSTATS: Don't increment key hits/misses counters.
78 *  LOOKUP_WRITE: Prepare the key for writing (delete expired keys even on
79 *                replicas, use separate keyspace stats and events (TODO)).
80 *  LOOKUP_NOEXPIRE: Perform expiration check, but avoid deleting the key,
81 *                   so that we don't have to propagate the deletion.
82 *
83 * Note: this function also returns NULL if the key is logically expired but
84 * still existing, in case this is a replica and the LOOKUP_WRITE is not set.
85 * Even if the key expiry is master-driven, we can correctly report a key is
86 * expired on replicas even if the master is lagging expiring our key via DELs
87 * in the replication link. */
88robj *lookupKey(redisDb *db, robj *key, int flags) {
89    dictEntry *de = dictFind(db->dict,key->ptr);
90    robj *val = NULL;
91    if (de) {
92        val = dictGetVal(de);
93        /* Forcing deletion of expired keys on a replica makes the replica
94         * inconsistent with the master. We forbid it on readonly replicas, but
95         * we have to allow it on writable replicas to make write commands
96         * behave consistently.
97         *
98         * It's possible that the WRITE flag is set even during a readonly
99         * command, since the command may trigger events that cause modules to
100         * perform additional writes. */
101        int is_ro_replica = server.masterhost && server.repl_slave_ro;
102        int expire_flags = 0;
103        if (flags & LOOKUP_WRITE && !is_ro_replica)
104            expire_flags |= EXPIRE_FORCE_DELETE_EXPIRED;
105        if (flags & LOOKUP_NOEXPIRE)
106            expire_flags |= EXPIRE_AVOID_DELETE_EXPIRED;
107        if (expireIfNeeded(db, key, expire_flags)) {
108            /* The key is no longer valid. */
109            val = NULL;
110        }
111    }
112
113    if (val) {
114        /* Update the access time for the ageing algorithm.
115         * Don't do it if we have a saving child, as this will trigger
116         * a copy on write madness. */
117        if (server.current_client && server.current_client->flags & CLIENT_NO_TOUCH &&
118            server.current_client->cmd->proc != touchCommand)
119            flags |= LOOKUP_NOTOUCH;
120        if (!hasActiveChildProcess() && !(flags & LOOKUP_NOTOUCH)){
121            if (server.maxmemory_policy & MAXMEMORY_FLAG_LFU) {
122                updateLFU(val);
123            } else {
124                val->lru = LRU_CLOCK();
125            }
126        }
127
128        if (!(flags & (LOOKUP_NOSTATS | LOOKUP_WRITE)))
129            server.stat_keyspace_hits++;
130        /* TODO: Use separate hits stats for WRITE */
131    } else {
132        if (!(flags & (LOOKUP_NONOTIFY | LOOKUP_WRITE)))
133            notifyKeyspaceEvent(NOTIFY_KEY_MISS, "keymiss", key, db->id);
134        if (!(flags & (LOOKUP_NOSTATS | LOOKUP_WRITE)))
135            server.stat_keyspace_misses++;
136        /* TODO: Use separate misses stats and notify event for WRITE */
137    }
138
139    return val;
140}

188/* Add the key to the DB. It's up to the caller to increment the reference
189 * counter of the value if needed.
190 *
191 * If the update_if_existing argument is false, the the program is aborted
192 * if the key already exists, otherwise, it can fall back to dbOverwite. */
193static void dbAddInternal(redisDb *db, robj *key, robj *val, int update_if_existing) {
194    dictEntry *existing;
195    dictEntry *de = dictAddRaw(db->dict, key->ptr, &existing);
196    if (update_if_existing && existing) {
197        dbSetValue(db, key, val, 1, existing);
198        return;
199    }
200    serverAssertWithInfo(NULL, key, de != NULL);
201    dictSetKey(db->dict, de, sdsdup(key->ptr));
202    initObjectLRUOrLFU(val);
203    dictSetVal(db->dict, de, val);
204    signalKeyAsReady(db, key, val->type);
205    if (server.cluster_enabled) slotToKeyAddEntry(de, db);
206    notifyKeyspaceEvent(NOTIFY_NEW,"new",key,db->id);
207}

213/* This is a special version of dbAdd() that is used only when loading
214 * keys from the RDB file: the key is passed as an SDS string that is
215 * retained by the function (and not freed by the caller).
216 *
217 * Moreover this function will not abort if the key is already busy, to
218 * give more control to the caller, nor will signal the key as ready
219 * since it is not useful in this context.
220 *
221 * The function returns 1 if the key was added to the database, taking
222 * ownership of the SDS string, otherwise 0 is returned, and is up to the
223 * caller to free the SDS string. */
224int dbAddRDBLoad(redisDb *db, sds key, robj *val) {
225    dictEntry *de = dictAddRaw(db->dict, key, NULL);
226    if (de == NULL) return 0;
227    initObjectLRUOrLFU(val);
228    dictSetVal(db->dict, de, val);
229    if (server.cluster_enabled) slotToKeyAddEntry(de, db);
230    return 1;
231}

233/* Overwrite an existing key with a new value. Incrementing the reference
234 * count of the new value is up to the caller.
235 * This function does not modify the expire time of the existing key.
236 *
237 * The 'overwrite' flag is an indication whether this is done as part of a
238 * complete replacement of their key, which can be thought as a deletion and
239 * replacement (in which case we need to emit deletion signals), or just an
240 * update of a value of an existing key (when false).
241 *
242 * The dictEntry input is optional, can be used if we already have one.
243 *
244 * The program is aborted if the key was not already present. */
245static void dbSetValue(redisDb *db, robj *key, robj *val, int overwrite, dictEntry *de) {
246    if (!de) de = dictFind(db->dict,key->ptr);
247    serverAssertWithInfo(NULL,key,de != NULL);
248    robj *old = dictGetVal(de);
249
250    val->lru = old->lru;
251
252    if (overwrite) {
253        /* RM_StringDMA may call dbUnshareStringValue which may free val, so we
254         * need to incr to retain old */
255        incrRefCount(old);
256        /* Although the key is not really deleted from the database, we regard
257         * overwrite as two steps of unlink+add, so we still need to call the unlink
258         * callback of the module. */
259        moduleNotifyKeyUnlink(key,old,db->id,DB_FLAG_KEY_OVERWRITE);
260        /* We want to try to unblock any module clients or clients using a blocking XREADGROUP */
261        signalDeletedKeyAsReady(db,key,old->type);
262        decrRefCount(old);
263        /* Because of RM_StringDMA, old may be changed, so we need get old again */
264        old = dictGetVal(de);
265    }
266    dictSetVal(db->dict, de, val);
267
268    if (server.lazyfree_lazy_server_del) {
269        freeObjAsync(key,old,db->id);
270    } else {
271        /* This is just decrRefCount(old); */
272        db->dict->type->valDestructor(db->dict, old);
273    }
274}

282/* High level Set operation. This function can be used in order to set
283 * a key, whatever it was existing or not, to a new object.
284 *
285 * 1) The ref count of the value object is incremented.
286 * 2) clients WATCHing for the destination key notified.
287 * 3) The expire time of the key is reset (the key is made persistent),
288 *    unless 'SETKEY_KEEPTTL' is enabled in flags.
289 * 4) The key lookup can take place outside this interface outcome will be
290 *    delivered with 'SETKEY_ALREADY_EXIST' or 'SETKEY_DOESNT_EXIST'
291 *
292 * All the new keys in the database should be created via this interface.
293 * The client 'c' argument may be set to NULL if the operation is performed
294 * in a context where there is no clear client performing the operation. */
295void setKey(client *c, redisDb *db, robj *key, robj *val, int flags) {
296    int keyfound = 0;
297
298    if (flags & SETKEY_ALREADY_EXIST)
299        keyfound = 1;
300    else if (flags & SETKEY_ADD_OR_UPDATE)
301        keyfound = -1;
302    else if (!(flags & SETKEY_DOESNT_EXIST))
303        keyfound = (lookupKeyWrite(db,key) != NULL);
304
305    if (!keyfound) {
306        dbAdd(db,key,val);
307    } else if (keyfound<0) {
308        dbAddInternal(db,key,val,1);
309    } else {
310        dbSetValue(db,key,val,1,NULL);
311    }
312    incrRefCount(val);
313    if (!(flags & SETKEY_KEEPTTL)) removeExpire(db,key);
314    if (!(flags & SETKEY_NO_SIGNAL)) signalModifiedKey(c,db,key);
315}

317/* Return a random key, in form of a Redis object.
318 * If there are no keys, NULL is returned.
319 *
320 * The function makes sure to return keys not already expired. */
321robj *dbRandomKey(redisDb *db) {
322    dictEntry *de;
323    int maxtries = 100;
324    int allvolatile = dictSize(db->dict) == dictSize(db->expires);
325
326    while(1) {
327        sds key;
328        robj *keyobj;
329
330        de = dictGetFairRandomKey(db->dict);
331        if (de == NULL) return NULL;
332
333        key = dictGetKey(de);
334        keyobj = createStringObject(key,sdslen(key));
335        if (dictFind(db->expires,key)) {
336            if (allvolatile && server.masterhost && --maxtries == 0) {
337                /* If the DB is composed only of keys with an expire set,
338                 * it could happen that all the keys are already logically
339                 * expired in the slave, so the function cannot stop because
340                 * expireIfNeeded() is false, nor it can stop because
341                 * dictGetFairRandomKey() returns NULL (there are keys to return).
342                 * To prevent the infinite loop we do some tries, but if there
343                 * are the conditions for an infinite loop, eventually we
344                 * return a key name that may be already expired. */
345                return keyobj;
346            }
347            if (expireIfNeeded(db,keyobj,0)) {
348                decrRefCount(keyobj);
349                continue; /* search for another key. This expired. */
350            }
351        }
352        return keyobj;
353    }
354}

356/* Helper for sync and async delete. */
357int dbGenericDelete(redisDb *db, robj *key, int async, int flags) {
358    dictEntry **plink;
359    int table;
360    dictEntry *de = dictTwoPhaseUnlinkFind(db->dict,key->ptr,&plink,&table);
361    if (de) {
362        robj *val = dictGetVal(de);
363        /* RM_StringDMA may call dbUnshareStringValue which may free val, so we
364         * need to incr to retain val */
365        incrRefCount(val);
366        /* Tells the module that the key has been unlinked from the database. */
367        moduleNotifyKeyUnlink(key,val,db->id,flags);
368        /* We want to try to unblock any module clients or clients using a blocking XREADGROUP */
369        signalDeletedKeyAsReady(db,key,val->type);
370        /* We should call decr before freeObjAsync. If not, the refcount may be
371         * greater than 1, so freeObjAsync doesn't work */
372        decrRefCount(val);
373        if (async) {
374            /* Because of dbUnshareStringValue, the val in de may change. */
375            freeObjAsync(key, dictGetVal(de), db->id);
376            dictSetVal(db->dict, de, NULL);
377        }
378        if (server.cluster_enabled) slotToKeyDelEntry(de, db);
379
380        /* Deleting an entry from the expires dict will not free the sds of
381        * the key, because it is shared with the main dictionary. */
382        if (dictSize(db->expires) > 0) dictDelete(db->expires,key->ptr);
383        dictTwoPhaseUnlinkFree(db->dict,de,plink,table);
384        return 1;
385    } else {
386        return 0;
387    }
388}

407/* Prepare the string object stored at 'key' to be modified destructively
408 * to implement commands like SETBIT or APPEND.
409 *
410 * An object is usually ready to be modified unless one of the two conditions
411 * are true:
412 *
413 * 1) The object 'o' is shared (refcount > 1), we don't want to affect
414 *    other users.
415 * 2) The object encoding is not "RAW".
416 *
417 * If the object is found in one of the above conditions (or both) by the
418 * function, an unshared / not-encoded copy of the string object is stored
419 * at 'key' in the specified 'db'. Otherwise the object 'o' itself is
420 * returned.
421 *
422 * USAGE:
423 *
424 * The object 'o' is what the caller already obtained by looking up 'key'
425 * in 'db', the usage pattern looks like this:
426 *
427 * o = lookupKeyWrite(db,key);
428 * if (checkType(c,o,OBJ_STRING)) return;
429 * o = dbUnshareStringValue(db,key,o);
430 *
431 * At this point the caller is ready to modify the object, for example
432 * using an sdscat() call to append some data, or anything else.
433 */
434robj *dbUnshareStringValue(redisDb *db, robj *key, robj *o) {
435    serverAssert(o->type == OBJ_STRING);
436    if (o->refcount != 1 || o->encoding != OBJ_ENCODING_RAW) {
437        robj *decoded = getDecodedObject(o);
438        o = createRawStringObject(decoded->ptr, sdslen(decoded->ptr));
439        decrRefCount(decoded);
440        dbReplaceValue(db,key,o);
441    }
442    return o;
443}

445/* Remove all keys from the database(s) structure. The dbarray argument
446 * may not be the server main DBs (could be a temporary DB).
447 *
448 * The dbnum can be -1 if all the DBs should be emptied, or the specified
449 * DB index if we want to empty only a single database.
450 * The function returns the number of keys removed from the database(s). */
451long long emptyDbStructure(redisDb *dbarray, int dbnum, int async,
452                           void(callback)(dict*))
453{
454    long long removed = 0;
455    int startdb, enddb;
456
457    if (dbnum == -1) {
458        startdb = 0;
459        enddb = server.dbnum-1;
460    } else {
461        startdb = enddb = dbnum;
462    }
463
464    for (int j = startdb; j <= enddb; j++) {
465        removed += dictSize(dbarray[j].dict);
466        if (async) {
467            emptyDbAsync(&dbarray[j]);
468        } else {
469            dictEmpty(dbarray[j].dict,callback);
470            dictEmpty(dbarray[j].expires,callback);
471        }
472        /* Because all keys of database are removed, reset average ttl. */
473        dbarray[j].avg_ttl = 0;
474        dbarray[j].expires_cursor = 0;
475    }
476
477    return removed;
478}

480/* Remove all data (keys and functions) from all the databases in a
481 * Redis server. If callback is given the function is called from
482 * time to time to signal that work is in progress.
483 *
484 * The dbnum can be -1 if all the DBs should be flushed, or the specified
485 * DB number if we want to flush only a single Redis database number.
486 *
487 * Flags are be EMPTYDB_NO_FLAGS if no special flags are specified or
488 * EMPTYDB_ASYNC if we want the memory to be freed in a different thread
489 * and the function to return ASAP. EMPTYDB_NOFUNCTIONS can also be set
490 * to specify that we do not want to delete the functions.
491 *
492 * On success the function returns the number of keys removed from the
493 * database(s). Otherwise -1 is returned in the specific case the
494 * DB number is out of range, and errno is set to EINVAL. */
495long long emptyData(int dbnum, int flags, void(callback)(dict*)) {
496    int async = (flags & EMPTYDB_ASYNC);
497    int with_functions = !(flags & EMPTYDB_NOFUNCTIONS);
498    RedisModuleFlushInfoV1 fi = {REDISMODULE_FLUSHINFO_VERSION,!async,dbnum};
499    long long removed = 0;
500
501    if (dbnum < -1 || dbnum >= server.dbnum) {
502        errno = EINVAL;
503        return -1;
504    }
505
506    /* Fire the flushdb modules event. */
507    moduleFireServerEvent(REDISMODULE_EVENT_FLUSHDB,
508                          REDISMODULE_SUBEVENT_FLUSHDB_START,
509                          &fi);
510
511    /* Make sure the WATCHed keys are affected by the FLUSH* commands.
512     * Note that we need to call the function while the keys are still
513     * there. */
514    signalFlushedDb(dbnum, async);
515
516    /* Empty redis database structure. */
517    removed = emptyDbStructure(server.db, dbnum, async, callback);
518
519    /* Flush slots to keys map if enable cluster, we can flush entire
520     * slots to keys map whatever dbnum because only support one DB
521     * in cluster mode. */
522    if (server.cluster_enabled) slotToKeyFlush(server.db);
523
524    if (dbnum == -1) flushSlaveKeysWithExpireList();
525
526    if (with_functions) {
527        serverAssert(dbnum == -1);
528        functionsLibCtxClearCurrent(async);
529    }
530
531    /* Also fire the end event. Note that this event will fire almost
532     * immediately after the start event if the flush is asynchronous. */
533    moduleFireServerEvent(REDISMODULE_EVENT_FLUSHDB,
534                          REDISMODULE_SUBEVENT_FLUSHDB_END,
535                          &fi);
536
537    return removed;
538}

540/* Initialize temporary db on replica for use during diskless replication. */
541redisDb *initTempDb(void) {
542    redisDb *tempDb = zcalloc(sizeof(redisDb)*server.dbnum);
543    for (int i=0; i<server.dbnum; i++) {
544        tempDb[i].dict = dictCreate(&dbDictType);
545        tempDb[i].expires = dictCreate(&dbExpiresDictType);
546        tempDb[i].slots_to_keys = NULL;
547    }
548
549    if (server.cluster_enabled) {
550        /* Prepare temp slot to key map to be written during async diskless replication. */
551        slotToKeyInit(tempDb);
552    }
553
554    return tempDb;
555}

557/* Discard tempDb, this can be slow (similar to FLUSHALL), but it's always async. */
558void discardTempDb(redisDb *tempDb, void(callback)(dict*)) {
559    int async = 1;
560
561    /* Release temp DBs. */
562    emptyDbStructure(tempDb, -1, async, callback);
563    for (int i=0; i<server.dbnum; i++) {
564        dictRelease(tempDb[i].dict);
565        dictRelease(tempDb[i].expires);
566    }
567
568    if (server.cluster_enabled) {
569        /* Release temp slot to key map. */
570        slotToKeyDestroy(tempDb);
571    }
572
573    zfree(tempDb);
574}

583long long dbTotalServerKeyCount(void) {
584    long long total = 0;
585    int j;
586    for (j = 0; j < server.dbnum; j++) {

813/* Data used by the dict scan callback. */
814typedef struct {
815    list *keys;   /* elements that collect from dict */
816    robj *o;      /* o must be a hash/set/zset object, NULL means current db */
817    long long type; /* the particular type when scan the db */
818    sds pattern;  /* pattern string, NULL means no pattern */
819    long sampled; /* cumulative number of keys sampled */
820} scanData;

822/* Helper function to compare key type in scan commands */
823int objectTypeCompare(robj *o, long long target) {
824    if (o->type != OBJ_MODULE) {
825        if (o->type != target) 
826            return 0;
827        else 
828            return 1;
829    }
830    /* module type compare */
831    long long mt = (long long)REDISMODULE_TYPE_SIGN(((moduleValue *)o->ptr)->type->id);
832    if (target != -mt)
833        return 0;
834    else 
835        return 1;
836}
837/* This callback is used by scanGenericCommand in order to collect elements
838 * returned by the dictionary iterator into a list. */
839void scanCallback(void *privdata, const dictEntry *de) {
840    scanData *data = (scanData *)privdata;
841    list *keys = data->keys;
842    robj *o = data->o;
843    sds val = NULL;
844    sds key = NULL;
845    data->sampled++;
846
847    /* o and typename can not have values at the same time. */
848    serverAssert(!((data->type != LLONG_MAX) && o));
849
850    /* Filter an element if it isn't the type we want. */
851    /* TODO: uncomment in redis 8.0
852    if (!o && data->type != LLONG_MAX) {
853        robj *rval = dictGetVal(de);
854        if (!objectTypeCompare(rval, data->type)) return;
855    }*/
856
857    /* Filter element if it does not match the pattern. */
858    sds keysds = dictGetKey(de);
859    if (data->pattern) {
860        if (!stringmatchlen(data->pattern, sdslen(data->pattern), keysds, sdslen(keysds), 0)) {
861            return;
862        }
863    }
864
865    if (o == NULL) {
866        key = keysds;
867    } else if (o->type == OBJ_SET) {
868        key = keysds;
869    } else if (o->type == OBJ_HASH) {
870        key = keysds;
871        val = dictGetVal(de);
872    } else if (o->type == OBJ_ZSET) {
873        char buf[MAX_LONG_DOUBLE_CHARS];
874        int len = ld2string(buf, sizeof(buf), *(double *)dictGetVal(de), LD_STR_AUTO);
875        key = sdsdup(keysds);
876        val = sdsnewlen(buf, len);
877    } else {
878        serverPanic("Type not handled in SCAN callback.");
879    }
880
881    listAddNodeTail(keys, key);
882    if (val) listAddNodeTail(keys, val);
883}

885/* Try to parse a SCAN cursor stored at object 'o':
886 * if the cursor is valid, store it as unsigned integer into *cursor and
887 * returns C_OK. Otherwise return C_ERR and send an error to the
888 * client. */
889int parseScanCursorOrReply(client *c, robj *o, unsigned long *cursor) {
890    char *eptr;
891
892    /* Use strtoul() because we need an *unsigned* long, so
893     * getLongLongFromObject() does not cover the whole cursor space. */
894    errno = 0;
895    *cursor = strtoul(o->ptr, &eptr, 10);
896    if (isspace(((char*)o->ptr)[0]) || eptr[0] != '\0' || errno == ERANGE)
897    {
898        addReplyError(c, "invalid cursor");
899        return C_ERR;
900    }
901    return C_OK;
902}

944/* This command implements SCAN, HSCAN and SSCAN commands.
945 * If object 'o' is passed, then it must be a Hash, Set or Zset object, otherwise
946 * if 'o' is NULL the command will operate on the dictionary associated with
947 * the current database.
948 *
949 * When 'o' is not NULL the function assumes that the first argument in
950 * the client arguments vector is a key so it skips it before iterating
951 * in order to parse options.
952 *
953 * In the case of a Hash object the function returns both the field and value
954 * of every element on the Hash. */
955void scanGenericCommand(client *c, robj *o, unsigned long cursor) {
956    int i, j;
957    listNode *node;
958    long count = 10;
959    sds pat = NULL;
960    sds typename = NULL;
961    long long type = LLONG_MAX;
962    int patlen = 0, use_pattern = 0;
963    dict *ht;
964
965    /* Object must be NULL (to iterate keys names), or the type of the object
966     * must be Set, Sorted Set, or Hash. */
967    serverAssert(o == NULL || o->type == OBJ_SET || o->type == OBJ_HASH ||
968                o->type == OBJ_ZSET);
969
970    /* Set i to the first option argument. The previous one is the cursor. */
971    i = (o == NULL) ? 2 : 3; /* Skip the key argument if needed. */
972
973    /* Step 1: Parse options. */
974    while (i < c->argc) {
975        j = c->argc - i;
976        if (!strcasecmp(c->argv[i]->ptr, "count") && j >= 2) {
977            if (getLongFromObjectOrReply(c, c->argv[i+1], &count, NULL)
978                != C_OK)
979            {
980                return;
981            }
982
983            if (count < 1) {
984                addReplyErrorObject(c,shared.syntaxerr);
985                return;
986            }
987
988            i += 2;
989        } else if (!strcasecmp(c->argv[i]->ptr, "match") && j >= 2) {
990            pat = c->argv[i+1]->ptr;
991            patlen = sdslen(pat);
992
993            /* The pattern always matches if it is exactly "*", so it is
994             * equivalent to disabling it. */
995            use_pattern = !(patlen == 1 && pat[0] == '*');
996
997            i += 2;
998        } else if (!strcasecmp(c->argv[i]->ptr, "type") && o == NULL && j >= 2) {
999            /* SCAN for a particular type only applies to the db dict */
1000            typename = c->argv[i+1]->ptr;
1001            type = getObjectTypeByName(typename);
1002            if (type == LLONG_MAX) {
1003                /* TODO: uncomment in redis 8.0
1004                addReplyErrorFormat(c, "unknown type name '%s'", typename);
1005                return; */
1006            }
1007            i+= 2;
1008        } else {
1009            addReplyErrorObject(c,shared.syntaxerr);
1010            return;
1011        }
1012    }
1013
1014    /* Step 2: Iterate the collection.
1015     *
1016     * Note that if the object is encoded with a listpack, intset, or any other
1017     * representation that is not a hash table, we are sure that it is also
1018     * composed of a small number of elements. So to avoid taking state we
1019     * just return everything inside the object in a single call, setting the
1020     * cursor to zero to signal the end of the iteration. */
1021
1022    /* Handle the case of a hash table. */
1023    ht = NULL;
1024    if (o == NULL) {
1025        ht = c->db->dict;
1026    } else if (o->type == OBJ_SET && o->encoding == OBJ_ENCODING_HT) {
1027        ht = o->ptr;
1028    } else if (o->type == OBJ_HASH && o->encoding == OBJ_ENCODING_HT) {
1029        ht = o->ptr;
1030    } else if (o->type == OBJ_ZSET && o->encoding == OBJ_ENCODING_SKIPLIST) {
1031        zset *zs = o->ptr;
1032        ht = zs->dict;
1033    }
1034
1035    list *keys = listCreate();
1036    /* Set a free callback for the contents of the collected keys list.
1037     * For the main keyspace dict, and when we scan a key that's dict encoded
1038     * (we have 'ht'), we don't need to define free method because the strings
1039     * in the list are just a shallow copy from the pointer in the dictEntry.
1040     * When scanning a key with other encodings (e.g. listpack), we need to
1041     * free the temporary strings we add to that list.
1042     * The exception to the above is ZSET, where we do allocate temporary
1043     * strings even when scanning a dict. */
1044    if (o && (!ht || o->type == OBJ_ZSET)) {
1045        listSetFreeMethod(keys, (void (*)(void*))sdsfree);
1046    }
1047
1048    if (ht) {
1049        /* We set the max number of iterations to ten times the specified
1050         * COUNT, so if the hash table is in a pathological state (very
1051         * sparsely populated) we avoid to block too much time at the cost
1052         * of returning no or very few elements. */
1053        long maxiterations = count*10;
1054
1055        /* We pass scanData which have three pointers to the callback:
1056         * 1. data.keys: the list to which it will add new elements;
1057         * 2. data.o: the object containing the dictionary so that
1058         * it is possible to fetch more data in a type-dependent way;
1059         * 3. data.type: the specified type scan in the db, LLONG_MAX means
1060         * type matching is no needed;
1061         * 4. data.pattern: the pattern string
1062         * 5. data.sampled: the maxiteration limit is there in case we're
1063         * working on an empty dict, one with a lot of empty buckets, and
1064         * for the buckets are not empty, we need to limit the spampled number
1065         * to prevent a long hang time caused by filtering too many keys*/
1066        scanData data = {
1067            .keys = keys,
1068            .o = o,
1069            .type = type,
1070            .pattern = use_pattern ? pat : NULL,
1071            .sampled = 0,
1072        };
1073        do {
1074            cursor = dictScan(ht, cursor, scanCallback, &data);
1075        } while (cursor && maxiterations-- && data.sampled < count);
1076    } else if (o->type == OBJ_SET) {
1077        char *str;
1078        char buf[LONG_STR_SIZE];
1079        size_t len;
1080        int64_t llele;
1081        setTypeIterator *si = setTypeInitIterator(o);
1082        while (setTypeNext(si, &str, &len, &llele) != -1) {
1083            if (str == NULL) {
1084                len = ll2string(buf, sizeof(buf), llele);
1085            }
1086            char *key = str ? str : buf;
1087            if (use_pattern && !stringmatchlen(pat, sdslen(pat), key, len, 0)) {
1088                continue;
1089            }
1090            listAddNodeTail(keys, sdsnewlen(key, len));
1091        }
1092        setTypeReleaseIterator(si);
1093        cursor = 0;
1094    } else if ((o->type == OBJ_HASH || o->type == OBJ_ZSET) &&
1095               o->encoding == OBJ_ENCODING_LISTPACK)
1096    {
1097        unsigned char *p = lpFirst(o->ptr);
1098        unsigned char *str;
1099        int64_t len;
1100        unsigned char intbuf[LP_INTBUF_SIZE];
1101
1102        while(p) {
1103            str = lpGet(p, &len, intbuf);
1104            /* point to the value */
1105            p = lpNext(o->ptr, p);
1106            if (use_pattern && !stringmatchlen(pat, sdslen(pat), (char *)str, len, 0)) {
1107                /* jump to the next key/val pair */
1108                p = lpNext(o->ptr, p);
1109                continue;
1110            }
1111            /* add key object */
1112            listAddNodeTail(keys, sdsnewlen(str, len));
1113            /* add value object */
1114            str = lpGet(p, &len, intbuf);
1115            listAddNodeTail(keys, sdsnewlen(str, len));
1116            p = lpNext(o->ptr, p);
1117        }
1118        cursor = 0;
1119    } else {
1120        serverPanic("Not handled encoding in SCAN.");
1121    }
1122
1123    /* Step 3: Filter the expired keys */
1124    if (o == NULL && listLength(keys)) {
1125        robj kobj;
1126        listIter li;
1127        listNode *ln;
1128        listRewind(keys, &li);
1129        while ((ln = listNext(&li))) {
1130            sds key = listNodeValue(ln);
1131            initStaticStringObject(kobj, key);
1132            /* Filter an element if it isn't the type we want. */
1133            /* TODO: remove this in redis 8.0 */
1134            if (typename) {
1135                robj* typecheck = lookupKeyReadWithFlags(c->db, &kobj, LOOKUP_NOTOUCH|LOOKUP_NONOTIFY);
1136                if (!typecheck || !objectTypeCompare(typecheck, type)) {
1137                    listDelNode(keys, ln);
1138                }
1139                continue;
1140            }
1141            if (expireIfNeeded(c->db, &kobj, 0)) {
1142                listDelNode(keys, ln);
1143            }
1144        }
1145    }
1146
1147    /* Step 4: Reply to the client. */
1148    addReplyArrayLen(c, 2);
1149    addReplyBulkLongLong(c,cursor);
1150
1151    addReplyArrayLen(c, listLength(keys));
1152    while ((node = listFirst(keys)) != NULL) {
1153        sds key = listNodeValue(node);
1154        addReplyBulkCBuffer(c, key, sdslen(key));
1155        listDelNode(keys, node);
1156    }
1157
1158    listRelease(keys);
1159}

1182void shutdownCommand(client *c) {
1183    int flags = SHUTDOWN_NOFLAGS;
1184    int abort = 0;
1185    for (int i = 1; i < c->argc; i++) {
1186        if (!strcasecmp(c->argv[i]->ptr,"nosave")) {
1187            flags |= SHUTDOWN_NOSAVE;
1188        } else if (!strcasecmp(c->argv[i]->ptr,"save")) {
1189            flags |= SHUTDOWN_SAVE;
1190        } else if (!strcasecmp(c->argv[i]->ptr, "now")) {
1191            flags |= SHUTDOWN_NOW;
1192        } else if (!strcasecmp(c->argv[i]->ptr, "force")) {
1193            flags |= SHUTDOWN_FORCE;
1194        } else if (!strcasecmp(c->argv[i]->ptr, "abort")) {
1195            abort = 1;
1196        } else {
1197            addReplyErrorObject(c,shared.syntaxerr);
1198            return;
1199        }
1200    }
1201    if ((abort && flags != SHUTDOWN_NOFLAGS) ||
1202        (flags & SHUTDOWN_NOSAVE && flags & SHUTDOWN_SAVE))
1203    {
1204        /* Illegal combo. */
1205        addReplyErrorObject(c,shared.syntaxerr);
1206        return;
1207    }
1208
1209    if (abort) {
1210        if (abortShutdown() == C_OK)
1211            addReply(c, shared.ok);
1212        else
1213            addReplyError(c, "No shutdown in progress.");
1214        return;
1215    }
1216
1217    if (!(flags & SHUTDOWN_NOW) && c->flags & CLIENT_DENY_BLOCKING) {
1218        addReplyError(c, "SHUTDOWN without NOW or ABORT isn't allowed for DENY BLOCKING client");
1219        return;
1220    }
1221
1222    if (!(flags & SHUTDOWN_NOSAVE) && isInsideYieldingLongCommand()) {
1223        /* Script timed out. Shutdown allowed only with the NOSAVE flag. See
1224         * also processCommand where these errors are returned. */
1225        if (server.busy_module_yield_flags && server.busy_module_yield_reply) {
1226            addReplyErrorFormat(c, "-BUSY %s", server.busy_module_yield_reply);
1227        } else if (server.busy_module_yield_flags) {
1228            addReplyErrorObject(c, shared.slowmoduleerr);
1229        } else if (scriptIsEval()) {
1230            addReplyErrorObject(c, shared.slowevalerr);
1231        } else {
1232            addReplyErrorObject(c, shared.slowscripterr);
1233        }
1234        return;
1235    }
1236
1237    blockClientShutdown(c);
1238    if (prepareForShutdown(flags) == C_OK) exit(0);
1239    /* If we're here, then shutdown is ongoing (the client is still blocked) or
1240     * failed (the client has received an error). */
1241}

1355void copyCommand(client *c) {
1356    robj *o;
1357    redisDb *src, *dst;
1358    int srcid, dbid;
1359    long long expire;
1360    int j, replace = 0, delete = 0;
1361
1362    /* Obtain source and target DB pointers 
1363     * Default target DB is the same as the source DB 
1364     * Parse the REPLACE option and targetDB option. */
1365    src = c->db;
1366    dst = c->db;
1367    srcid = c->db->id;
1368    dbid = c->db->id;
1369    for (j = 3; j < c->argc; j++) {
1370        int additional = c->argc - j - 1;
1371        if (!strcasecmp(c->argv[j]->ptr,"replace")) {
1372            replace = 1;
1373        } else if (!strcasecmp(c->argv[j]->ptr, "db") && additional >= 1) {
1374            if (getIntFromObjectOrReply(c, c->argv[j+1], &dbid, NULL) != C_OK)
1375                return;
1376
1377            if (selectDb(c, dbid) == C_ERR) {
1378                addReplyError(c,"DB index is out of range");
1379                return;
1380            }
1381            dst = c->db;
1382            selectDb(c,srcid); /* Back to the source DB */
1383            j++; /* Consume additional arg. */
1384        } else {
1385            addReplyErrorObject(c,shared.syntaxerr);
1386            return;
1387        }
1388    }
1389
1390    if ((server.cluster_enabled == 1) && (srcid != 0 || dbid != 0)) {
1391        addReplyError(c,"Copying to another database is not allowed in cluster mode");
1392        return;
1393    }
1394
1395    /* If the user select the same DB as
1396     * the source DB and using newkey as the same key
1397     * it is probably an error. */
1398    robj *key = c->argv[1];
1399    robj *newkey = c->argv[2];
1400    if (src == dst && (sdscmp(key->ptr, newkey->ptr) == 0)) {
1401        addReplyErrorObject(c,shared.sameobjecterr);
1402        return;
1403    }
1404
1405    /* Check if the element exists and get a reference */
1406    o = lookupKeyRead(c->db, key);
1407    if (!o) {
1408        addReply(c,shared.czero);
1409        return;
1410    }
1411    expire = getExpire(c->db,key);
1412
1413    /* Return zero if the key already exists in the target DB. 
1414     * If REPLACE option is selected, delete newkey from targetDB. */
1415    if (lookupKeyWrite(dst,newkey) != NULL) {
1416        if (replace) {
1417            delete = 1;
1418        } else {
1419            addReply(c,shared.czero);
1420            return;
1421        }
1422    }
1423
1424    /* Duplicate object according to object's type. */
1425    robj *newobj;
1426    switch(o->type) {
1427        case OBJ_STRING: newobj = dupStringObject(o); break;
1428        case OBJ_LIST: newobj = listTypeDup(o); break;
1429        case OBJ_SET: newobj = setTypeDup(o); break;
1430        case OBJ_ZSET: newobj = zsetDup(o); break;
1431        case OBJ_HASH: newobj = hashTypeDup(o); break;
1432        case OBJ_STREAM: newobj = streamDup(o); break;
1433        case OBJ_MODULE:
1434            newobj = moduleTypeDupOrReply(c, key, newkey, dst->id, o);
1435            if (!newobj) return;
1436            break;
1437        default:
1438            addReplyError(c, "unknown type object");
1439            return;
1440    }
1441
1442    if (delete) {
1443        dbDelete(dst,newkey);
1444    }
1445
1446    dbAdd(dst,newkey,newobj);
1447    if (expire != -1) setExpire(c, dst, newkey, expire);
1448
1449    /* OK! key copied */
1450    signalModifiedKey(c,dst,c->argv[2]);
1451    notifyKeyspaceEvent(NOTIFY_GENERIC,"copy_to",c->argv[2],dst->id);
1452
1453    server.dirty++;
1454    addReply(c,shared.cone);
1455}

1457/* Helper function for dbSwapDatabases(): scans the list of keys that have
1458 * one or more blocked clients for B[LR]POP or other blocking commands
1459 * and signal the keys as ready if they are of the right type. See the comment
1460 * where the function is used for more info. */
1461void scanDatabaseForReadyKeys(redisDb *db) {
1462    dictEntry *de;
1463    dictIterator *di = dictGetSafeIterator(db->blocking_keys);
1464    while((de = dictNext(di)) != NULL) {
1465        robj *key = dictGetKey(de);
1466        dictEntry *kde = dictFind(db->dict,key->ptr);
1467        if (kde) {
1468            robj *value = dictGetVal(kde);
1469            signalKeyAsReady(db, key, value->type);
1470        }
1471    }
1472    dictReleaseIterator(di);
1473}

1475/* Since we are unblocking XREADGROUP clients in the event the
1476 * key was deleted/overwritten we must do the same in case the
1477 * database was flushed/swapped. */
1478void scanDatabaseForDeletedKeys(redisDb *emptied, redisDb *replaced_with) {
1479    dictEntry *de;
1480    dictIterator *di = dictGetSafeIterator(emptied->blocking_keys);
1481    while((de = dictNext(di)) != NULL) {
1482        robj *key = dictGetKey(de);
1483        int existed = 0, exists = 0;
1484        int original_type = -1, curr_type = -1;
1485
1486        dictEntry *kde = dictFind(emptied->dict, key->ptr);
1487        if (kde) {
1488            robj *value = dictGetVal(kde);
1489            original_type = value->type;
1490            existed = 1;
1491        }
1492
1493        if (replaced_with) {
1494            dictEntry *kde = dictFind(replaced_with->dict, key->ptr);
1495            if (kde) {
1496                robj *value = dictGetVal(kde);
1497                curr_type = value->type;
1498                exists = 1;
1499            }
1500        }
1501        /* We want to try to unblock any client using a blocking XREADGROUP */
1502        if ((existed && !exists) || original_type != curr_type)
1503            signalDeletedKeyAsReady(emptied, key, original_type);
1504    }
1505    dictReleaseIterator(di);
1506}

1508/* Swap two databases at runtime so that all clients will magically see
1509 * the new database even if already connected. Note that the client
1510 * structure c->db points to a given DB, so we need to be smarter and
1511 * swap the underlying referenced structures, otherwise we would need
1512 * to fix all the references to the Redis DB structure.
1513 *
1514 * Returns C_ERR if at least one of the DB ids are out of range, otherwise
1515 * C_OK is returned. */
1516int dbSwapDatabases(int id1, int id2) {
1517    if (id1 < 0 || id1 >= server.dbnum ||
1518        id2 < 0 || id2 >= server.dbnum) return C_ERR;
1519    if (id1 == id2) return C_OK;
1520    redisDb aux = server.db[id1];
1521    redisDb *db1 = &server.db[id1], *db2 = &server.db[id2];
1522
1523    /* Swapdb should make transaction fail if there is any
1524     * client watching keys */
1525    touchAllWatchedKeysInDb(db1, db2);
1526    touchAllWatchedKeysInDb(db2, db1);
1527
1528    /* Try to unblock any XREADGROUP clients if the key no longer exists. */
1529    scanDatabaseForDeletedKeys(db1, db2);
1530    scanDatabaseForDeletedKeys(db2, db1);
1531
1532    /* Swap hash tables. Note that we don't swap blocking_keys,
1533     * ready_keys and watched_keys, since we want clients to
1534     * remain in the same DB they were. */
1535    db1->dict = db2->dict;
1536    db1->expires = db2->expires;
1537    db1->avg_ttl = db2->avg_ttl;
1538    db1->expires_cursor = db2->expires_cursor;
1539
1540    db2->dict = aux.dict;
1541    db2->expires = aux.expires;
1542    db2->avg_ttl = aux.avg_ttl;
1543    db2->expires_cursor = aux.expires_cursor;
1544
1545    /* Now we need to handle clients blocked on lists: as an effect
1546     * of swapping the two DBs, a client that was waiting for list
1547     * X in a given DB, may now actually be unblocked if X happens
1548     * to exist in the new version of the DB, after the swap.
1549     *
1550     * However normally we only do this check for efficiency reasons
1551     * in dbAdd() when a list is created. So here we need to rescan
1552     * the list of clients blocked on lists and signal lists as ready
1553     * if needed. */
1554    scanDatabaseForReadyKeys(db1);
1555    scanDatabaseForReadyKeys(db2);
1556    return C_OK;
1557}

1559/* Logically, this discards (flushes) the old main database, and apply the newly loaded
1560 * database (temp) as the main (active) database, the actual freeing of old database
1561 * (which will now be placed in the temp one) is done later. */
1562void swapMainDbWithTempDb(redisDb *tempDb) {
1563    if (server.cluster_enabled) {
1564        /* Swap slots_to_keys from tempdb just loaded with main db slots_to_keys. */
1565        clusterSlotToKeyMapping *aux = server.db->slots_to_keys;
1566        server.db->slots_to_keys = tempDb->slots_to_keys;
1567        tempDb->slots_to_keys = aux;
1568    }
1569
1570    for (int i=0; i<server.dbnum; i++) {
1571        redisDb aux = server.db[i];
1572        redisDb *activedb = &server.db[i], *newdb = &tempDb[i];
1573
1574        /* Swapping databases should make transaction fail if there is any
1575         * client watching keys. */
1576        touchAllWatchedKeysInDb(activedb, newdb);
1577
1578        /* Try to unblock any XREADGROUP clients if the key no longer exists. */
1579        scanDatabaseForDeletedKeys(activedb, newdb);
1580
1581        /* Swap hash tables. Note that we don't swap blocking_keys,
1582         * ready_keys and watched_keys, since clients 
1583         * remain in the same DB they were. */
1584        activedb->dict = newdb->dict;
1585        activedb->expires = newdb->expires;
1586        activedb->avg_ttl = newdb->avg_ttl;
1587        activedb->expires_cursor = newdb->expires_cursor;
1588
1589        newdb->dict = aux.dict;
1590        newdb->expires = aux.expires;
1591        newdb->avg_ttl = aux.avg_ttl;
1592        newdb->expires_cursor = aux.expires_cursor;
1593
1594        /* Now we need to handle clients blocked on lists: as an effect
1595         * of swapping the two DBs, a client that was waiting for list
1596         * X in a given DB, may now actually be unblocked if X happens
1597         * to exist in the new version of the DB, after the swap.
1598         *
1599         * However normally we only do this check for efficiency reasons
1600         * in dbAdd() when a list is created. So here we need to rescan
1601         * the list of clients blocked on lists and signal lists as ready
1602         * if needed. */
1603        scanDatabaseForReadyKeys(activedb);
1604    }
1605
1606    trackingInvalidateKeysOnFlush(1);
1607    flushSlaveKeysWithExpireList();
1608}

1692/* Propagate expires into slaves and the AOF file.
1693 * When a key expires in the master, a DEL operation for this key is sent
1694 * to all the slaves and the AOF file if enabled.
1695 *
1696 * This way the key expiry is centralized in one place, and since both
1697 * AOF and the master->slave link guarantee operation ordering, everything
1698 * will be consistent even if we allow write operations against expiring
1699 * keys.
1700 *
1701 * This function may be called from:
1702 * 1. Within call(): Example: Lazy-expire on key access.
1703 *    In this case the caller doesn't have to do anything
1704 *    because call() handles server.also_propagate(); or
1705 * 2. Outside of call(): Example: Active-expire, eviction.
1706 *    In this the caller must remember to call
1707 *    postExecutionUnitOperations, preferably just after a
1708 *    single deletion batch, so that DELs will NOT be wrapped
1709 *    in MULTI/EXEC */
1710void propagateDeletion(redisDb *db, robj *key, int lazy) {
1711    robj *argv[2];
1712
1713    argv[0] = lazy ? shared.unlink : shared.del;
1714    argv[1] = key;
1715    incrRefCount(argv[0]);
1716    incrRefCount(argv[1]);
1717
1718    /* If the master decided to expire a key we must propagate it to replicas no matter what..
1719     * Even if module executed a command without asking for propagation. */
1720    int prev_replication_allowed = server.replication_allowed;
1721    server.replication_allowed = 1;
1722    alsoPropagate(db->id,argv,2,PROPAGATE_AOF|PROPAGATE_REPL);
1723    server.replication_allowed = prev_replication_allowed;
1724
1725    decrRefCount(argv[0]);
1726    decrRefCount(argv[1]);
1727}

1729/* Check if the key is expired. */
1730int keyIsExpired(redisDb *db, robj *key) {
1731    /* Don't expire anything while loading. It will be done later. */
1732    if (server.loading) return 0;
1733
1734    mstime_t when = getExpire(db,key);
1735    mstime_t now;
1736
1737    if (when < 0) return 0; /* No expire for this key */
1738
1739    now = commandTimeSnapshot();
1740
1741    /* The key expired if the current (virtual or real) time is greater
1742     * than the expire time of the key. */
1743    return now > when;
1744}

1746/* This function is called when we are going to perform some operation
1747 * in a given key, but such key may be already logically expired even if
1748 * it still exists in the database. The main way this function is called
1749 * is via lookupKey*() family of functions.
1750 *
1751 * The behavior of the function depends on the replication role of the
1752 * instance, because by default replicas do not delete expired keys. They
1753 * wait for DELs from the master for consistency matters. However even
1754 * replicas will try to have a coherent return value for the function,
1755 * so that read commands executed in the replica side will be able to
1756 * behave like if the key is expired even if still present (because the
1757 * master has yet to propagate the DEL).
1758 *
1759 * In masters as a side effect of finding a key which is expired, such
1760 * key will be evicted from the database. Also this may trigger the
1761 * propagation of a DEL/UNLINK command in AOF / replication stream.
1762 *
1763 * On replicas, this function does not delete expired keys by default, but
1764 * it still returns 1 if the key is logically expired. To force deletion
1765 * of logically expired keys even on replicas, use the EXPIRE_FORCE_DELETE_EXPIRED
1766 * flag. Note though that if the current client is executing
1767 * replicated commands from the master, keys are never considered expired.
1768 *
1769 * On the other hand, if you just want expiration check, but need to avoid
1770 * the actual key deletion and propagation of the deletion, use the
1771 * EXPIRE_AVOID_DELETE_EXPIRED flag.
1772 *
1773 * The return value of the function is 0 if the key is still valid,
1774 * otherwise the function returns 1 if the key is expired. */
1775int expireIfNeeded(redisDb *db, robj *key, int flags) {
1776    if (server.lazy_expire_disabled) return 0;
1777    if (!keyIsExpired(db,key)) return 0;
1778
1779    /* If we are running in the context of a replica, instead of
1780     * evicting the expired key from the database, we return ASAP:
1781     * the replica key expiration is controlled by the master that will
1782     * send us synthesized DEL operations for expired keys. The
1783     * exception is when write operations are performed on writable
1784     * replicas.
1785     *
1786     * Still we try to return the right information to the caller,
1787     * that is, 0 if we think the key should be still valid, 1 if
1788     * we think the key is expired at this time.
1789     *
1790     * When replicating commands from the master, keys are never considered
1791     * expired. */
1792    if (server.masterhost != NULL) {
1793        if (server.current_client && (server.current_client->flags & CLIENT_MASTER)) return 0;
1794        if (!(flags & EXPIRE_FORCE_DELETE_EXPIRED)) return 1;
1795    }
1796
1797    /* In some cases we're explicitly instructed to return an indication of a
1798     * missing key without actually deleting it, even on masters. */
1799    if (flags & EXPIRE_AVOID_DELETE_EXPIRED)
1800        return 1;
1801
1802    /* If 'expire' action is paused, for whatever reason, then don't expire any key.
1803     * Typically, at the end of the pause we will properly expire the key OR we
1804     * will have failed over and the new primary will send us the expire. */
1805    if (isPausedActionsWithUpdate(PAUSE_ACTION_EXPIRE)) return 1;
1806
1807    /* The key needs to be converted from static to heap before deleted */
1808    int static_key = key->refcount == OBJ_STATIC_REFCOUNT;
1809    if (static_key) {
1810        key = createStringObject(key->ptr, sdslen(key->ptr));
1811    }
1812    /* Delete the key */
1813    deleteExpiredKeyAndPropagate(db,key);
1814    if (static_key) {
1815        decrRefCount(key);
1816    }
1817    return 1;
1818}

1824/* Prepare the getKeysResult struct to hold numkeys, either by using the
1825 * pre-allocated keysbuf or by allocating a new array on the heap.
1826 *
1827 * This function must be called at least once before starting to populate
1828 * the result, and can be called repeatedly to enlarge the result array.
1829 */
1830keyReference *getKeysPrepareResult(getKeysResult *result, int numkeys) {
1831    /* GETKEYS_RESULT_INIT initializes keys to NULL, point it to the pre-allocated stack
1832     * buffer here. */
1833    if (!result->keys) {
1834        serverAssert(!result->numkeys);
1835        result->keys = result->keysbuf;
1836    }
1837
1838    /* Resize if necessary */
1839    if (numkeys > result->size) {
1840        if (result->keys != result->keysbuf) {
1841            /* We're not using a static buffer, just (re)alloc */
1842            result->keys = zrealloc(result->keys, numkeys * sizeof(keyReference));
1843        } else {
1844            /* We are using a static buffer, copy its contents */
1845            result->keys = zmalloc(numkeys * sizeof(keyReference));
1846            if (result->numkeys)
1847                memcpy(result->keys, result->keysbuf, result->numkeys * sizeof(keyReference));
1848        }
1849        result->size = numkeys;
1850    }
1851
1852    return result->keys;
1853}

1855/* Returns a bitmask with all the flags found in any of the key specs of the command.
1856 * The 'inv' argument means we'll return a mask with all flags that are missing in at least one spec. */
1857int64_t getAllKeySpecsFlags(struct redisCommand *cmd, int inv) {
1858    int64_t flags = 0;
1859    for (int j = 0; j < cmd->key_specs_num; j++) {
1860        keySpec *spec = cmd->key_specs + j;
1861        flags |= inv? ~spec->flags : spec->flags;
1862    }
1863    return flags;
1864}

1866/* Fetch the keys based of the provided key specs. Returns the number of keys found, or -1 on error.
1867 * There are several flags that can be used to modify how this function finds keys in a command.
1868 * 
1869 * GET_KEYSPEC_INCLUDE_NOT_KEYS: Return 'fake' keys as if they were keys.
1870 * GET_KEYSPEC_RETURN_PARTIAL:   Skips invalid and incomplete keyspecs but returns the keys
1871 *                               found in other valid keyspecs. 
1872 */
1873int getKeysUsingKeySpecs(struct redisCommand *cmd, robj **argv, int argc, int search_flags, getKeysResult *result) {
1874    int j, i, last, first, step;
1875    keyReference *keys;
1876    serverAssert(result->numkeys == 0); /* caller should initialize or reset it */
1877
1878    for (j = 0; j < cmd->key_specs_num; j++) {
1879        keySpec *spec = cmd->key_specs + j;
1880        serverAssert(spec->begin_search_type != KSPEC_BS_INVALID);
1881        /* Skip specs that represent 'fake' keys */
1882        if ((spec->flags & CMD_KEY_NOT_KEY) && !(search_flags & GET_KEYSPEC_INCLUDE_NOT_KEYS)) {
1883            continue;
1884        }
1885
1886        first = 0;
1887        if (spec->begin_search_type == KSPEC_BS_INDEX) {
1888            first = spec->bs.index.pos;
1889        } else if (spec->begin_search_type == KSPEC_BS_KEYWORD) {
1890            int start_index = spec->bs.keyword.startfrom > 0 ? spec->bs.keyword.startfrom : argc+spec->bs.keyword.startfrom;
1891            int end_index = spec->bs.keyword.startfrom > 0 ? argc-1: 1;
1892            for (i = start_index; i != end_index; i = start_index <= end_index ? i + 1 : i - 1) {
1893                if (i >= argc || i < 1)
1894                    break;
1895                if (!strcasecmp((char*)argv[i]->ptr,spec->bs.keyword.keyword)) {
1896                    first = i+1;
1897                    break;
1898                }
1899            }
1900            /* keyword not found */
1901            if (!first) {
1902                continue;
1903            }
1904        } else {
1905            /* unknown spec */
1906            goto invalid_spec;
1907        }
1908
1909        if (spec->find_keys_type == KSPEC_FK_RANGE) {
1910            step = spec->fk.range.keystep;
1911            if (spec->fk.range.lastkey >= 0) {
1912                last = first + spec->fk.range.lastkey;
1913            } else {
1914                if (!spec->fk.range.limit) {
1915                    last = argc + spec->fk.range.lastkey;
1916                } else {
1917                    serverAssert(spec->fk.range.lastkey == -1);
1918                    last = first + ((argc-first)/spec->fk.range.limit + spec->fk.range.lastkey);
1919                }
1920            }
1921        } else if (spec->find_keys_type == KSPEC_FK_KEYNUM) {
1922            step = spec->fk.keynum.keystep;
1923            long long numkeys;
1924            if (spec->fk.keynum.keynumidx >= argc)
1925                goto invalid_spec;
1926
1927            sds keynum_str = argv[first + spec->fk.keynum.keynumidx]->ptr;
1928            if (!string2ll(keynum_str,sdslen(keynum_str),&numkeys) || numkeys < 0) {
1929                /* Unable to parse the numkeys argument or it was invalid */
1930                goto invalid_spec;
1931            }
1932
1933            first += spec->fk.keynum.firstkey;
1934            last = first + (int)numkeys-1;
1935        } else {
1936            /* unknown spec */
1937            goto invalid_spec;
1938        }
1939
1940        int count = ((last - first)+1);
1941        keys = getKeysPrepareResult(result, result->numkeys + count);
1942
1943        /* First or last is out of bounds, which indicates a syntax error */
1944        if (last >= argc || last < first || first >= argc) {
1945            goto invalid_spec;
1946        }
1947
1948        for (i = first; i <= last; i += step) {
1949            if (i >= argc || i < first) {
1950                /* Modules commands, and standard commands with a not fixed number
1951                 * of arguments (negative arity parameter) do not have dispatch
1952                 * time arity checks, so we need to handle the case where the user
1953                 * passed an invalid number of arguments here. In this case we
1954                 * return no keys and expect the command implementation to report
1955                 * an arity or syntax error. */
1956                if (cmd->flags & CMD_MODULE || cmd->arity < 0) {
1957                    continue;
1958                } else {
1959                    serverPanic("Redis built-in command declared keys positions not matching the arity requirements.");
1960                }
1961            }
1962            keys[result->numkeys].pos = i;
1963            keys[result->numkeys].flags = spec->flags;
1964            result->numkeys++;
1965        }
1966
1967        /* Handle incomplete specs (only after we added the current spec
1968         * to `keys`, just in case GET_KEYSPEC_RETURN_PARTIAL was given) */
1969        if (spec->flags & CMD_KEY_INCOMPLETE) {
1970            goto invalid_spec;
1971        }
1972
1973        /* Done with this spec */
1974        continue;
1975
1976invalid_spec:
1977        if (search_flags & GET_KEYSPEC_RETURN_PARTIAL) {
1978            continue;
1979        } else {
1980            result->numkeys = 0;
1981            return -1;
1982        }
1983    }
1984
1985    return result->numkeys;
1986}

1988/* Return all the arguments that are keys in the command passed via argc / argv. 
1989 * This function will eventually replace getKeysFromCommand.
1990 *
1991 * The command returns the positions of all the key arguments inside the array,
1992 * so the actual return value is a heap allocated array of integers. The
1993 * length of the array is returned by reference into *numkeys.
1994 * 
1995 * Along with the position, this command also returns the flags that are
1996 * associated with how Redis will access the key.
1997 *
1998 * 'cmd' must be point to the corresponding entry into the redisCommand
1999 * table, according to the command name in argv[0]. */
2000int getKeysFromCommandWithSpecs(struct redisCommand *cmd, robj **argv, int argc, int search_flags, getKeysResult *result) {
2001    /* The command has at least one key-spec not marked as NOT_KEY */
2002    int has_keyspec = (getAllKeySpecsFlags(cmd, 1) & CMD_KEY_NOT_KEY);
2003    /* The command has at least one key-spec marked as VARIABLE_FLAGS */
2004    int has_varflags = (getAllKeySpecsFlags(cmd, 0) & CMD_KEY_VARIABLE_FLAGS);
2005
2006    /* We prefer key-specs if there are any, and their flags are reliable. */
2007    if (has_keyspec && !has_varflags) {
2008        int ret = getKeysUsingKeySpecs(cmd,argv,argc,search_flags,result);
2009        if (ret >= 0)
2010            return ret;
2011        /* If the specs returned with an error (probably an INVALID or INCOMPLETE spec),
2012         * fallback to the callback method. */
2013    }
2014
2015    /* Resort to getkeys callback methods. */
2016    if (cmd->flags & CMD_MODULE_GETKEYS)
2017        return moduleGetCommandKeysViaAPI(cmd,argv,argc,result);
2018
2019    /* We use native getkeys as a last resort, since not all these native getkeys provide
2020     * flags properly (only the ones that correspond to INVALID, INCOMPLETE or VARIABLE_FLAGS do.*/
2021    if (cmd->getkeys_proc)
2022        return cmd->getkeys_proc(cmd,argv,argc,result);
2023    return 0;
2024}

2301/* Helper function to extract keys from the SORT command.
2302 *
2303 * SORT <sort-key> ... STORE <store-key> ...
2304 *
2305 * The first argument of SORT is always a key, however a list of options
2306 * follow in SQL-alike style. Here we parse just the minimum in order to
2307 * correctly identify keys in the "STORE" option. 
2308 * 
2309 * This command declares incomplete keys, so the flags are correctly set for this function */
2310int sortGetKeys(struct redisCommand *cmd, robj **argv, int argc, getKeysResult *result) {
2311    int i, j, num, found_store = 0;
2312    keyReference *keys;
2313    UNUSED(cmd);
2314
2315    num = 0;
2316    keys = getKeysPrepareResult(result, 2); /* Alloc 2 places for the worst case. */
2317    keys[num].pos = 1; /* <sort-key> is always present. */
2318    keys[num++].flags = CMD_KEY_RO | CMD_KEY_ACCESS;
2319
2320    /* Search for STORE option. By default we consider options to don't
2321     * have arguments, so if we find an unknown option name we scan the
2322     * next. However there are options with 1 or 2 arguments, so we
2323     * provide a list here in order to skip the right number of args. */
2324    struct {
2325        char *name;
2326        int skip;
2327    } skiplist[] = {
2328        {"limit", 2},
2329        {"get", 1},
2330        {"by", 1},
2331        {NULL, 0} /* End of elements. */
2332    };
2333
2334    for (i = 2; i < argc; i++) {
2335        for (j = 0; skiplist[j].name != NULL; j++) {
2336            if (!strcasecmp(argv[i]->ptr,skiplist[j].name)) {
2337                i += skiplist[j].skip;
2338                break;
2339            } else if (!strcasecmp(argv[i]->ptr,"store") && i+1 < argc) {
2340                /* Note: we don't increment "num" here and continue the loop
2341                 * to be sure to process the *last* "STORE" option if multiple
2342                 * ones are provided. This is same behavior as SORT. */
2343                found_store = 1;
2344                keys[num].pos = i+1; /* <store-key> */
2345                keys[num].flags = CMD_KEY_OW | CMD_KEY_UPDATE;
2346                break;
2347            }

2406/* Helper function to extract keys from following commands:
2407 * GEORADIUS key x y radius unit [WITHDIST] [WITHHASH] [WITHCOORD] [ASC|DESC]
2408 *                             [COUNT count] [STORE key|STOREDIST key]
2409 * GEORADIUSBYMEMBER key member radius unit ... options ...
2410 * 
2411 * This command has a fully defined keyspec, so returning flags isn't needed. */
2412int georadiusGetKeys(struct redisCommand *cmd, robj **argv, int argc, getKeysResult *result) {
2413    int i, num;
2414    keyReference *keys;
2415    UNUSED(cmd);
2416
2417    /* Check for the presence of the stored key in the command */
2418    int stored_key = -1;
2419    for (i = 5; i < argc; i++) {
2420        char *arg = argv[i]->ptr;
2421        /* For the case when user specifies both "store" and "storedist" options, the
2422         * second key specified would override the first key. This behavior is kept
2423         * the same as in georadiusCommand method.
2424         */
2425        if ((!strcasecmp(arg, "store") || !strcasecmp(arg, "storedist")) && ((i+1) < argc)) {
2426            stored_key = i+1;
2427            i++;
2428        }
2429    }
2430    num = 1 + (stored_key == -1 ? 0 : 1);
2431
2432    /* Keys in the command come from two places:
2433     * argv[1] = key,
2434     * argv[5...n] = stored key if present
2435     */
2436    keys = getKeysPrepareResult(result, num);
2437
2438    /* Add all key positions to keys[] */
2439    keys[0].pos = 1;
2440    keys[0].flags = 0;
2441    if(num > 1) {
2442         keys[1].pos = stored_key;
2443         keys[1].flags = 0;
2444    }
2445    result->numkeys = num;
2446    return num;
2447}

2449/* XREAD [BLOCK <milliseconds>] [COUNT <count>] [GROUP <groupname> <ttl>]
2450 *       STREAMS key_1 key_2 ... key_N ID_1 ID_2 ... ID_N
2451 *
2452 * This command has a fully defined keyspec, so returning flags isn't needed. */
2453int xreadGetKeys(struct redisCommand *cmd, robj **argv, int argc, getKeysResult *result) {
2454    int i, num = 0;
2455    keyReference *keys;
2456    UNUSED(cmd);
2457
2458    /* We need to parse the options of the command in order to seek the first
2459     * "STREAMS" string which is actually the option. This is needed because
2460     * "STREAMS" could also be the name of the consumer group and even the
2461     * name of the stream key. */
2462    int streams_pos = -1;
2463    for (i = 1; i < argc; i++) {
2464        char *arg = argv[i]->ptr;
2465        if (!strcasecmp(arg, "block")) {
2466            i++; /* Skip option argument. */
2467        } else if (!strcasecmp(arg, "count")) {
2468            i++; /* Skip option argument. */
2469        } else if (!strcasecmp(arg, "group")) {
2470            i += 2; /* Skip option argument. */
2471        } else if (!strcasecmp(arg, "noack")) {
2472            /* Nothing to do. */
2473        } else if (!strcasecmp(arg, "streams")) {
2474            streams_pos = i;
2475            break;
2476        } else {
2477            break; /* Syntax error. */
2478        }
2479    }
2480    if (streams_pos != -1) num = argc - streams_pos - 1;
2481
2482    /* Syntax error. */
2483    if (streams_pos == -1 || num == 0 || num % 2 != 0) {
2484        result->numkeys = 0;
2485        return 0;
2486    }
2487    num /= 2; /* We have half the keys as there are arguments because
2488                 there are also the IDs, one per key. */
2489
2490    keys = getKeysPrepareResult(result, num);
2491    for (i = streams_pos+1; i < argc-num; i++) {
2492        keys[i-streams_pos-1].pos = i;
2493        keys[i-streams_pos-1].flags = 0; 
2494    } 
2495    result->numkeys = num;
2496    return num;
2497}

2524/* Helper function to extract keys from the BITFIELD command, which may be
2525 * read-only if the BITFIELD GET subcommand is used. */
2526int bitfieldGetKeys(struct redisCommand *cmd, robj **argv, int argc, getKeysResult *result) {
2527    keyReference *keys;
2528    int readonly = 1;
2529    UNUSED(cmd);
2530
2531    keys = getKeysPrepareResult(result, 1);
2532    keys[0].pos = 1; /* We always know the position */
2533    result->numkeys = 1;
2534
2535    for (int i = 2; i < argc; i++) {
2536        int remargs = argc - i - 1; /* Remaining args other than current. */
2537        char *arg = argv[i]->ptr;
2538        if (!strcasecmp(arg, "get") && remargs >= 2) {
2539            i += 2;
2540        } else if ((!strcasecmp(arg, "set") || !strcasecmp(arg, "incrby")) && remargs >= 3) {
2541            readonly = 0;
2542            i += 3;
2543            break;
2544        } else if (!strcasecmp(arg, "overflow") && remargs >= 1) {
2545            i += 1;
2546        } else {
2547            readonly = 0; /* Syntax error. safer to assume non-RO. */
2548            break;
2549        }
2550    }
2551
2552    if (readonly) {
2553        keys[0].flags = CMD_KEY_RO | CMD_KEY_ACCESS;
2554    } else {
2555        keys[0].flags = CMD_KEY_RW | CMD_KEY_ACCESS | CMD_KEY_UPDATE;
2556    }
2557    return 1;
