45// Pool maintains a pool of connections. The application calls the Get method
46// to get a connection from the pool and the connection's Close method to
47// return the connection's resources to the pool.
48//
49// The following example shows how to use a pool in a web application. The
50// application creates a pool at application startup and makes it available to
51// request handlers using a package level variable. The pool configuration used
52// here is an example, not a recommendation.
53//
54//  func newPool(addr string) *redis.Pool {
55//    return &redis.Pool{
56//      MaxIdle: 3,
57//      IdleTimeout: 240 * time.Second,
58//      // Dial or DialContext must be set. When both are set, DialContext takes precedence over Dial.
59//      Dial: func () (redis.Conn, error) { return redis.Dial("tcp", addr) },
60//    }
61//  }
62//
63//  var (
64//    pool *redis.Pool
65//    redisServer = flag.String("redisServer", ":6379", "")
66//  )
67//
68//  func main() {
69//    flag.Parse()
70//    pool = newPool(*redisServer)
71//    ...
72//  }
73//
74// A request handler gets a connection from the pool and closes the connection
75// when the handler is done:
76//
77//  func serveHome(w http.ResponseWriter, r *http.Request) {
78//      conn := pool.Get()
79//      defer conn.Close()
80//      ...
81//  }
82//
83// Use the Dial function to authenticate connections with the AUTH command or
84// select a database with the SELECT command:
85//
86//  pool := &redis.Pool{
87//    // Other pool configuration not shown in this example.
88//    Dial: func () (redis.Conn, error) {
89//      c, err := redis.Dial("tcp", server)
90//      if err != nil {
91//        return nil, err
92//      }
93//      if _, err := c.Do("AUTH", password); err != nil {
94//        c.Close()
95//        return nil, err
96//      }
97//      if _, err := c.Do("SELECT", db); err != nil {
98//        c.Close()
99//        return nil, err
100//      }
101//      return c, nil
102//    },
103//  }
104//
105// Use the TestOnBorrow function to check the health of an idle connection
106// before the connection is returned to the application. This example PINGs
107// connections that have been idle more than a minute:
108//
109//  pool := &redis.Pool{
110//    // Other pool configuration not shown in this example.
111//    TestOnBorrow: func(c redis.Conn, t time.Time) error {
112//      if time.Since(t) < time.Minute {
113//        return nil
114//      }
115//      _, err := c.Do("PING")
116//      return err
117//    },
118//  }
119//
120type Pool struct {
121	// Dial is an application supplied function for creating and configuring a
122	// connection.
123	//
124	// The connection returned from Dial must not be in a special state
125	// (subscribed to pubsub channel, transaction started, ...).
126	Dial func() (Conn, error)
127
128	// DialContext is an application supplied function for creating and configuring a
129	// connection with the given context.
130	//
131	// The connection returned from DialContext must not be in a special state
132	// (subscribed to pubsub channel, transaction started, ...).
133	DialContext func(ctx context.Context) (Conn, error)
134
135	// TestOnBorrow is an optional application supplied function for checking
136	// the health of an idle connection before the connection is used again by
137	// the application. Argument lastUsed is the time when the connection was returned
138	// to the pool. If the function returns an error, then the connection is
139	// closed.
140	TestOnBorrow func(c Conn, lastUsed time.Time) error
141
142	// TestOnBorrowContext is an optional application supplied function
143	// for checking the health of an idle connection with the given context
144	// before the connection is used again by the application.
145	// Argument lastUsed is the time when the connection was returned
146	// to the pool. If the function returns an error, then the connection is
147	// closed.
148	TestOnBorrowContext func(ctx context.Context, c Conn, lastUsed time.Time) error
149
150	// Maximum number of idle connections in the pool.
151	MaxIdle int
152
153	// Maximum number of connections allocated by the pool at a given time.
154	// When zero, there is no limit on the number of connections in the pool.
155	MaxActive int
156
157	// Close connections after remaining idle for this duration. If the value
158	// is zero, then idle connections are not closed. Applications should set
159	// the timeout to a value less than the server's timeout.
160	IdleTimeout time.Duration
161
162	// If Wait is true and the pool is at the MaxActive limit, then Get() waits
163	// for a connection to be returned to the pool before returning.
164	Wait bool
165
166	// Close connections older than this duration. If the value is zero, then
167	// the pool does not close connections based on age.
168	MaxConnLifetime time.Duration
169
170	mu           sync.Mutex    // mu protects the following fields
171	closed       bool          // set to true when the pool is closed.
172	active       int           // the number of open connections in the pool
173	initOnce     sync.Once     // the init ch once func
174	ch           chan struct{} // limits open connections when p.Wait is true
175	idle         idleList      // idle connections
176	waitCount    int64         // total number of connections waited for.
177	waitDuration time.Duration // total time waited for new connections.
178}

187// Get gets a connection. The application must close the returned connection.
188// This method always returns a valid connection so that applications can defer
189// error handling to the first use of the connection. If there is an error
190// getting an underlying connection, then the connection Err, Do, Send, Flush
191// and Receive methods return that error.
192func (p *Pool) Get() Conn {
193	// GetContext returns errorConn in the first argument when an error occurs.
194	c, _ := p.GetContext(context.Background())
195	return c
196}
197
198// GetContext gets a connection using the provided context.
199//
200// The provided Context must be non-nil. If the context expires before the
201// connection is complete, an error is returned. Any expiration on the context
202// will not affect the returned connection.
203//
204// If the function completes without error, then the application must close the
205// returned connection.
206func (p *Pool) GetContext(ctx context.Context) (Conn, error) {
207	// Wait until there is a vacant connection in the pool.
208	waited, err := p.waitVacantConn(ctx)
209	if err != nil {
210		return errorConn{err}, err
211	}
212
213	p.mu.Lock()
214
215	if waited > 0 {
216		p.waitCount++
217		p.waitDuration += waited
218	}
219
220	// Prune stale connections at the back of the idle list.
221	if p.IdleTimeout > 0 {
222		n := p.idle.count
223		for i := 0; i < n && p.idle.back != nil && p.idle.back.t.Add(p.IdleTimeout).Before(nowFunc()); i++ {
224			pc := p.idle.back
225			p.idle.popBack()
226			p.mu.Unlock()
227			pc.c.Close()
228			p.mu.Lock()
229			p.active--
230		}
231	}
232
233	// Get idle connection from the front of idle list.
234	for p.idle.front != nil {
235		pc := p.idle.front
236		p.idle.popFront()
237		p.mu.Unlock()
238		if (p.TestOnBorrow == nil || p.TestOnBorrow(pc.c, pc.t) == nil) &&
239			(p.TestOnBorrowContext == nil || p.TestOnBorrowContext(ctx, pc.c, pc.t) == nil) &&
240			(p.MaxConnLifetime == 0 || nowFunc().Sub(pc.created) < p.MaxConnLifetime) {
241			return &activeConn{p: p, pc: pc}, nil
242		}
243		pc.c.Close()
244		p.mu.Lock()
245		p.active--
246	}
247
248	// Check for pool closed before dialing a new connection.
249	if p.closed {
250		p.mu.Unlock()
251		err := errors.New("redigo: get on closed pool")
252		return errorConn{err}, err
253	}
254
255	// Handle limit for p.Wait == false.
256	if !p.Wait && p.MaxActive > 0 && p.active >= p.MaxActive {
257		p.mu.Unlock()
258		return errorConn{ErrPoolExhausted}, ErrPoolExhausted
259	}
260
261	p.active++
262	p.mu.Unlock()
263	c, err := p.dial(ctx)
264	if err != nil {
265		p.mu.Lock()
266		p.active--
267		if p.ch != nil && !p.closed {
268			p.ch <- struct{}{}
269		}
270		p.mu.Unlock()
271		return errorConn{err}, err
272	}
273	return &activeConn{p: p, pc: &poolConn{c: c, created: nowFunc()}}, nil
274}

324// Close releases the resources used by the pool.
325func (p *Pool) Close() error {
326	p.mu.Lock()
327	if p.closed {
328		p.mu.Unlock()
329		return nil
330	}
331	p.closed = true
332	p.active -= p.idle.count
333	pc := p.idle.front
334	p.idle.count = 0
335	p.idle.front, p.idle.back = nil, nil
336	if p.ch != nil {
337		close(p.ch)
338	}
339	p.mu.Unlock()
340	for ; pc != nil; pc = pc.next {
341		pc.c.Close()
342	}
343	return nil
344}

346func (p *Pool) lazyInit() {
347	p.initOnce.Do(func() {
348		p.ch = make(chan struct{}, p.MaxActive)
349		if p.closed {
350			close(p.ch)
351		} else {
352			for i := 0; i < p.MaxActive; i++ {
353				p.ch <- struct{}{}
354			}
355		}
356	})
357}

359// waitVacantConn waits for a vacant connection in pool if waiting
360// is enabled and pool size is limited, otherwise returns instantly.
361// If ctx expires before that, an error is returned.
362//
363// If there were no vacant connection in the pool right away it returns the time spent waiting
364// for that connection to appear in the pool.
365func (p *Pool) waitVacantConn(ctx context.Context) (waited time.Duration, err error) {
366	if !p.Wait || p.MaxActive <= 0 {
367		// No wait or no connection limit.
368		return 0, nil
369	}
370
371	p.lazyInit()
372
373	// wait indicates if we believe it will block so its not 100% accurate
374	// however for stats it should be good enough.
375	wait := len(p.ch) == 0
376	var start time.Time
377	if wait {
378		start = time.Now()
379	}
380
381	select {
382	case <-p.ch:
383		// Additionally check that context hasn't expired while we were waiting,
384		// because `select` picks a random `case` if several of them are "ready".
385		select {
386		case <-ctx.Done():
387			p.ch <- struct{}{}
388			return 0, ctx.Err()
389		default:
390		}
391	case <-ctx.Done():
392		return 0, ctx.Err()
393	}
394
395	if wait {
396		return time.Since(start), nil
397	}
398	return 0, nil
399}

401func (p *Pool) dial(ctx context.Context) (Conn, error) {
402	if p.DialContext != nil {
403		return p.DialContext(ctx)
404	}
405	if p.Dial != nil {
406		return p.Dial()
407	}
408	return nil, errors.New("redigo: must pass Dial or DialContext to pool")
409}

411func (p *Pool) put(pc *poolConn, forceClose bool) error {
412	p.mu.Lock()
413	if !p.closed && !forceClose {
414		pc.t = nowFunc()
415		p.idle.pushFront(pc)
416		if p.idle.count > p.MaxIdle {
417			pc = p.idle.back
418			p.idle.popBack()
419		} else {
420			pc = nil
421		}
422	}
423
424	if pc != nil {
425		p.mu.Unlock()
426		pc.c.Close()
427		p.mu.Lock()
428		p.active--
429	}
430
431	if p.ch != nil && !p.closed {
432		p.ch <- struct{}{}
433	}
434	p.mu.Unlock()
435	return nil
436}

449func initSentinel() {
450	p := make([]byte, 64)
451	if _, err := rand.Read(p); err == nil {
452		sentinel = p
453	} else {
454		h := sha1.New()
455		io.WriteString(h, "Oops, rand failed. Use time instead.")       // nolint: errcheck
456		io.WriteString(h, strconv.FormatInt(time.Now().UnixNano(), 10)) // nolint: errcheck
457		sentinel = h.Sum(nil)
458	}
459}

461func (ac *activeConn) firstError(errs ...error) error {
462	for _, err := range errs[:len(errs)-1] {
463		if err != nil {
464			return err
465		}
466	}
467	return errs[len(errs)-1]
468}

470func (ac *activeConn) Close() (err error) {
471	pc := ac.pc
472	if pc == nil {
473		return nil
474	}
475	ac.pc = nil
476
477	if ac.state&connectionMultiState != 0 {
478		err = pc.c.Send("DISCARD")
479		ac.state &^= (connectionMultiState | connectionWatchState)
480	} else if ac.state&connectionWatchState != 0 {
481		err = pc.c.Send("UNWATCH")
482		ac.state &^= connectionWatchState
483	}
484	if ac.state&connectionSubscribeState != 0 {
485		err = ac.firstError(err,
486			pc.c.Send("UNSUBSCRIBE"),
487			pc.c.Send("PUNSUBSCRIBE"),
488		)
489		// To detect the end of the message stream, ask the server to echo
490		// a sentinel value and read until we see that value.
491		sentinelOnce.Do(initSentinel)
492		err = ac.firstError(err,
493			pc.c.Send("ECHO", sentinel),
494			pc.c.Flush(),
495		)
496		for {
497			p, err2 := pc.c.Receive()
498			if err2 != nil {
499				err = ac.firstError(err, err2)
500				break
501			}
502			if p, ok := p.([]byte); ok && bytes.Equal(p, sentinel) {
503				ac.state &^= connectionSubscribeState
504				break
505			}
506		}
507	}
508	_, err2 := pc.c.Do("")
509	return ac.firstError(
510		err,
511		err2,
512		ac.p.put(pc, ac.state != 0 || pc.c.Err() != nil),
513	)
514}

629type idleList struct {
630	count       int
631	front, back *poolConn
632}
633
634type poolConn struct {
635	c          Conn
636	t          time.Time
637	created    time.Time
638	next, prev *poolConn
639}
640
641func (l *idleList) pushFront(pc *poolConn) {
642	pc.next = l.front
643	pc.prev = nil
644	if l.count == 0 {
645		l.back = pc
646	} else {
647		l.front.prev = pc
648	}
649	l.front = pc
650	l.count++
651}
652
653func (l *idleList) popFront() {
654	pc := l.front
655	l.count--
656	if l.count == 0 {
657		l.front, l.back = nil, nil
658	} else {
659		pc.next.prev = nil
660		l.front = pc.next
661	}
662	pc.next, pc.prev = nil, nil
663}
664
665func (l *idleList) popBack() {
666	pc := l.back
667	l.count--
668	if l.count == 0 {
669		l.front, l.back = nil, nil
670	} else {
671		pc.prev.next = nil
672		l.back = pc.prev
673	}
674	pc.next, pc.prev = nil, nil
675}
