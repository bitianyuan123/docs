17// Provider 特征提供器.
18// 优先从 Kafka 实时缓存获取，未命中时回退到离线 JSON 文件.
19type Provider struct {
20	consumer      *Consumer
21	fallbackPath  string
22	fallbackCache map[string]map[string]interface{} // 离线特征缓存
23	once          sync.Once
24}
25
26// NewProvider 创建特征提供器.
27//
28// Args:
29//     consumer: Kafka 消费者，可为 nil（纯离线模式）
30//     fallbackPath: 离线 JSON 文件路径，作为降级使用
31//
32// Returns:
33//     特征提供器实例
34func NewProvider(consumer *Consumer, fallbackPath string) *Provider {
35	return &Provider{
36		consumer:      consumer,
37		fallbackPath:  fallbackPath,
38		fallbackCache: make(map[string]map[string]interface{}),
39	}
40}
41
42// GetUserHistory 获取用户历史点击序列.
43// 优先查实时缓存，未命中时回退到离线 JSON.
44//
45// Args:
46//     userID: 用户 ID
47//
48// Returns:
49//     物品 ID 列表，如果不存在返回 nil
50func (p *Provider) GetUserHistory(userID string) ([]int, error) {
51	// 1. 优先查实时缓存
52	if p.consumer != nil {
53		if features := p.consumer.Get(userID); features != nil {
54			fmt.Printf("[FeatureProvider] [REALTIME] User=%s, HistoryLen=%d, Source=KafkaCache\n",
55				userID, len(features.ClickHistory))
56			return features.ClickHistory, nil
57		}
58	}
59
60	// 2. 降级到离线 JSON
61	fmt.Printf("[FeatureProvider] [FALLBACK] User=%s not in realtime cache, using fallback file\n", userID)
62	history, err := p.getFromFallback(userID)
63	if err != nil {
64		return nil, err
65	}
66	fmt.Printf("[FeatureProvider] [FALLBACK] User=%s, HistoryLen=%d, Source=%s\n",
67		userID, len(history), p.fallbackPath)
68	return history, nil
69}

71// GetUserFeatures 获取完整用户特征（包含实时特征）.
72//
73// Args:
74//     userID: 用户 ID
75//
76// Returns:
77//     用户特征结构，如果不存在返回 nil
78func (p *Provider) GetUserFeatures(userID string) *UserFeatures {
79	// 优先查实时缓存
80	if p.consumer != nil {
81		if features := p.consumer.Get(userID); features != nil {
82			return features
83		}
84	}
85	return nil
86}

88// IsRealtimeAvailable 检查实时特征是否可用.
89func (p *Provider) IsRealtimeAvailable() bool {
90	return p.consumer != nil && p.consumer.IsHealthy()
91}
92
93// GetStats 获取统计信息.
94func (p *Provider) GetStats() map[string]interface{} {
95	stats := map[string]interface{}{
96		"fallback_path": p.fallbackPath,
97		"realtime_available": p.IsRealtimeAvailable(),
98	}
99
100	if p.consumer != nil {
101		stats["cache_size"] = p.consumer.GetCacheSize()
102	}
103
104	return stats
105}

107// getFromFallback 从离线 JSON 获取用户历史.
108func (p *Provider) getFromFallback(userID string) ([]int, error) {
109	p.once.Do(func() {
110		p.loadFallback()
111	})
112
113	userData, ok := p.fallbackCache[userID]
114	if !ok {
115		return nil, fmt.Errorf("user %s not found in fallback", userID)
116	}
117
118	historyStr := ""
119	if val, ok := userData["click_history"]; ok {
120		historyStr = fmt.Sprintf("%v", val)
121	}
122
123	return parseHistoryString(historyStr)
124}
125
126// loadFallback 加载离线 JSON 文件.
127func (p *Provider) loadFallback() {
128	data, err := os.ReadFile(p.fallbackPath)
129	if err != nil {
130		fmt.Printf("[FeatureProvider] Failed to load fallback: %v\n", err)
131		return
132	}
133
134	if err := json.Unmarshal(data, &p.fallbackCache); err != nil {
135		fmt.Printf("[FeatureProvider] Failed to parse fallback: %v\n", err)
136		return
137	}
138
139	fmt.Printf("[FeatureProvider] Loaded fallback: %d users from %s\n",
140		len(p.fallbackCache), p.fallbackPath)
141}
142
143// parseHistoryString 解析逗号分隔的历史字符串.
144func parseHistoryString(historyStr string) ([]int, error) {
145	if historyStr == "" {
146		return nil, nil
147	}
148
149	parts := strings.Split(historyStr, ",")
150	history := make([]int, 0, len(parts))
151
152	for _, part := range parts {
153		part = strings.TrimSpace(part)
154		if part == "" {
155			continue
156		}
157
158		id, err := strconv.Atoi(part)
159		if err != nil {
160			continue
161		}
162		history = append(history, id)
163	}
164
165	return history, nil
166}
