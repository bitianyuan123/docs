20// KafkaUserFeature 适配 recall-sink topic 的数据格式
21type KafkaUserFeature struct {
22	UserID        int    `json:"user_id"`        // 数字类型
23	Gender        int    `json:"gender"`         // 性别
24	Age           int    `json:"age"`            // 年龄
25	ClickHistory  string `json:"click_history"`  // JSON 字符串，如 "[\"3035268\"]"
26}

28// Consumer Kafka 特征消费者.
29type Consumer struct {
30	config        *KafkaConfig
31	reader        *kafka.Reader
32	cache         *sync.Map // map[string]*UserFeatures
33	ctx           context.Context
34	cancel        context.CancelFunc
35	maxMessages   int       // 最大消费消息数（0表示无限制）
36	msgCount      int       // 已消费消息计数
37}
38
39// NewConsumer 创建 Kafka 消费者.
40//
41// Args:
42//     config: Kafka 配置，包含 brokers、topic、group_id
43//     maxMessages: 最大消费消息数（0表示无限制）
44//
45// Returns:
46//     消费者实例
47func NewConsumer(config *KafkaConfig, maxMessages int) *Consumer {
48	ctx, cancel := context.WithCancel(context.Background())
49
50	reader := kafka.NewReader(kafka.ReaderConfig{
51		Brokers:     config.Brokers,
52		Topic:       config.Topic,
53		GroupID:     config.GroupID,
54		MinBytes:    10e3, // 10KB
55		MaxBytes:    10e6, // 10MB
56		MaxWait:     time.Second,
57		ReadBackoffMin: 100 * time.Millisecond,
58		ReadBackoffMax: time.Second,
59	})
60
61	return &Consumer{
62		config:      config,
63		reader:      reader,
64		cache:       &sync.Map{},
65		ctx:         ctx,
66		cancel:      cancel,
67		maxMessages: maxMessages,
68		msgCount:    0,
69	}
70}

72// Start 启动消费循环（非阻塞）.
73// 在后台 goroutine 中持续消费 Kafka 消息.
74func (c *Consumer) Start() error {
75	go c.consume()
76	log.Printf("[FeatureConsumer] Started, topic=%s, group=%s", c.config.Topic, c.config.GroupID)
77	return nil
78}

80// Stop 停止消费者.
81// 调用此方法会停止消费循环并关闭 Kafka 连接.
82func (c *Consumer) Stop() error {
83	c.cancel()
84	return c.reader.Close()
85}

87// Get 获取用户特征.
88//
89// Args:
90//     userID: 用户 ID
91//
92// Returns:
93//     用户特征，如果不存在返回 nil
94func (c *Consumer) Get(userID string) *UserFeatures {
95	if v, ok := c.cache.Load(userID); ok {
96		return v.(*UserFeatures)
97	}
98	return nil
99}

101// GetCacheSize 获取缓存中的用户数量（用于监控）.
102func (c *Consumer) GetCacheSize() int {
103	count := 0
104	c.cache.Range(func(_, _ interface{}) bool {
105		count++
106		return true
107	})
108	return count
109}

111// consume 消费循环.
112// 持续读取 Kafka 消息并更新本地缓存.
113func (c *Consumer) consume() {
114	for {
115		select {
116		case <-c.ctx.Done():
117			return
118		default:
119		}
120
121		// 检查是否达到最大消费数量
122		if c.maxMessages > 0 && c.msgCount >= c.maxMessages {
123			log.Printf("[FeatureConsumer] Reached max messages limit (%d), stopping consumption", c.maxMessages)
124			return
125		}
126
127		msg, err := c.reader.ReadMessage(c.ctx)
128		if err != nil {
129			if c.ctx.Err() != nil {
130				return
131			}
132			log.Printf("[FeatureConsumer] Read error: %v", err)
133			time.Sleep(time.Second)
134			continue
135		}
136
137		// 解析 recall-sink 格式的消息
138		var kafkaFeature KafkaUserFeature
139		if err := json.Unmarshal(msg.Value, &kafkaFeature); err != nil {
140			log.Printf("[FeatureConsumer] Unmarshal error: %v, raw=%s", err, string(msg.Value))
141			continue
142		}
143
144		// 转换为内部 UserFeatures 格式
145		features := c.convertToUserFeatures(&kafkaFeature)
146		if features == nil {
147			log.Printf("[FeatureConsumer] Failed to convert feature for user=%d", kafkaFeature.UserID)
148			continue
149		}
150
151		// 更新本地缓存（全量替换）
152		c.cache.Store(features.UserID, features)
153		c.msgCount++
154		
155		// 调试日志：显示详细的特征信息
156		log.Printf("[FeatureConsumer] [KAFKA→CACHE] User=%s, History=%v, HistoryLen=%d, Count=%d/%d",
157			features.UserID, 
158			features.ClickHistory,
159			len(features.ClickHistory),
160			c.msgCount, c.maxMessages)
161	}
162}

164// convertToUserFeatures 将 Kafka 格式转换为内部 UserFeatures 格式
165func (c *Consumer) convertToUserFeatures(kf *KafkaUserFeature) *UserFeatures {
166	// 1. 转换 user_id: int -> string
167	userID := strconv.Itoa(kf.UserID)
168
169	// 2. 解析 click_history: JSON 字符串 -> []int
170	// click_history 格式: "[]" 或 "[\"3035268\"]" 或 "[\"123\",\"456\"]"
171	var historyStrs []string
172	if err := json.Unmarshal([]byte(kf.ClickHistory), &historyStrs); err != nil {
173		log.Printf("[FeatureConsumer] Failed to parse click_history string: %v, raw=%s", err, kf.ClickHistory)
174		// 如果解析失败，返回空历史而不是 nil
175		historyStrs = []string{}
176	}
177
178	// 3. 字符串数组转 int 数组
179	clickHistory := make([]int, 0, len(historyStrs))
180	for _, s := range historyStrs {
181		if s == "" {
182			continue
183		}
184		id, err := strconv.Atoi(s)
185		if err != nil {
186			log.Printf("[FeatureConsumer] Failed to convert item_id to int: %s", s)
187			continue
188		}
189		clickHistory = append(clickHistory, id)
190	}
191
192	return &UserFeatures{
193		UserID:        userID,
194		ClickHistory:  clickHistory,
195		UpdateTime:    time.Now().Format(time.RFC3339), // 使用当前时间
196		RealtimeCTR:   0.0,
197		ShortTermTags: []string{},
198	}
199}

201// IsHealthy 检查消费者健康状态.
202func (c *Consumer) IsHealthy() bool {
203	// 简单检查：如果能获取到缓存大小，说明运行正常
204	_ = c.GetCacheSize()
205	return c.ctx.Err() == nil
206}
