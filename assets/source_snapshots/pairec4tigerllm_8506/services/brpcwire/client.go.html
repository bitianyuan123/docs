24// Client implements protobuf calls over brpc baidu_std/TCP. Attempts is the
25// total number of attempts, so Attempts=1 means zero retries.
26type Client struct {
27	endpoint string
28	service  string
29	timeout  time.Duration
30	attempts int
31	nextID   int64
32}
33
34// Session keeps one baidu_std TCP connection alive across sequential calls.
35// Calls on one session are serialized; concurrency is expressed with multiple
36// preconnected sessions.
37type Session struct {
38	client *Client
39	mu     sync.Mutex
40	conn   net.Conn
41}

43func NewClient(endpoint, service string, timeout time.Duration, attempts int) (*Client, error) {
44	endpoint = strings.TrimSpace(endpoint)
45	service = strings.TrimSpace(service)
46	if endpoint == "" || service == "" {
47		return nil, fmt.Errorf("brpc endpoint and service are required")
48	}
49	if timeout <= 0 {
50		return nil, fmt.Errorf("brpc timeout must be positive")
51	}
52	if attempts <= 0 {
53		return nil, fmt.Errorf("brpc attempts must be positive")
54	}
55	return &Client{endpoint: endpoint, service: service, timeout: timeout, attempts: attempts}, nil
56}

58func (c *Client) Call(ctx context.Context, method string, request, response proto.Message) error {
59	if strings.TrimSpace(method) == "" || request == nil || response == nil {
60		return fmt.Errorf("brpc method, request, and response are required")
61	}
62	var lastErr error
63	for attempt := 0; attempt < c.attempts; attempt++ {
64		attemptCtx, cancel := context.WithTimeout(ctx, c.timeout)
65		lastErr = c.callOnce(attemptCtx, method, request, response)
66		cancel()
67		if lastErr == nil {
68			return nil
69		}
70		if ctx.Err() != nil {
71			return ctx.Err()
72		}
73	}
74	return fmt.Errorf("brpc %s failed after %d attempt(s): %w", method, c.attempts, lastErr)
75}

79func (s *Session) Connect(ctx context.Context) error {
80	s.mu.Lock()
81	defer s.mu.Unlock()
82	if s.conn != nil {
83		return nil
84	}
85	var dialer net.Dialer
86	conn, err := dialer.DialContext(ctx, "tcp", s.client.endpoint)
87	if err != nil {
88		return fmt.Errorf("dial %s failed: %w", s.client.endpoint, err)
89	}
90	s.conn = conn
91	return nil
92}

94func (s *Session) Call(ctx context.Context, method string, request, response proto.Message) error {
95	if strings.TrimSpace(method) == "" || request == nil || response == nil {
96		return fmt.Errorf("brpc method, request, and response are required")
97	}
98	s.mu.Lock()
99	defer s.mu.Unlock()
100	var lastErr error
101	for attempt := 0; attempt < s.client.attempts; attempt++ {
102		attemptCtx, cancel := context.WithTimeout(ctx, s.client.timeout)
103		lastErr = nil
104		if s.conn == nil {
105			var dialer net.Dialer
106			s.conn, lastErr = dialer.DialContext(attemptCtx, "tcp", s.client.endpoint)
107		}
108		if lastErr == nil {
109			lastErr = s.client.callOnConn(attemptCtx, s.conn, method, request, response)
110		}
111		cancel()
112		if lastErr == nil {
113			return nil
114		}
115		_ = s.closeLocked()
116		if ctx.Err() != nil {
117			return ctx.Err()
118		}
119	}
120	return fmt.Errorf("brpc %s failed after %d attempt(s): %w", method, s.client.attempts, lastErr)
121}

125	defer s.mu.Unlock()

129func (s *Session) closeLocked() error {
130	if s.conn == nil {
131		return nil
132	}
133	err := s.conn.Close()
134	s.conn = nil

138func (c *Client) callOnce(ctx context.Context, method string, request, response proto.Message) error {
139	var dialer net.Dialer
140	conn, err := dialer.DialContext(ctx, "tcp", c.endpoint)
141	if err != nil {
142		return fmt.Errorf("dial %s failed: %w", c.endpoint, err)
143	}
144	defer conn.Close()
145	return c.callOnConn(ctx, conn, method, request, response)
146}

148func (c *Client) callOnConn(ctx context.Context, conn net.Conn, method string, request, response proto.Message) error {
149	if deadline, ok := ctx.Deadline(); ok {
150		if err := conn.SetDeadline(deadline); err != nil {
151			return fmt.Errorf("set deadline failed: %w", err)
152		}
153	}
154
155	payload, err := proto.Marshal(request)
156	if err != nil {
157		return fmt.Errorf("marshal %s request failed: %w", method, err)
158	}
159	id := atomic.AddInt64(&c.nextID, 1)
160	meta := &rpcMeta{
161		CompressType: proto.Int32(0), CorrelationID: proto.Int64(id),
162		AttachmentSize: proto.Int32(0), ContentType: proto.Int32(0),
163		ChecksumType: proto.Int32(0), Request: &requestMeta{
164			ServiceName: proto.String(c.service), MethodName: proto.String(method), LogID: proto.Int64(id),
165		},
166	}
167	metaBytes, err := proto.Marshal(meta)
168	if err != nil {
169		return fmt.Errorf("marshal brpc meta failed: %w", err)
170	}
171	frame, err := buildFrame(metaBytes, payload)
172	if err != nil {
173		return err
174	}
175	if _, err := conn.Write(frame); err != nil {
176		return fmt.Errorf("write brpc frame failed: %w", err)
177	}
178	responsePayload, err := readResponse(conn, id)
179	if err != nil {
180		return err
181	}
182	if err := proto.Unmarshal(responsePayload, response); err != nil {
183		return fmt.Errorf("unmarshal %s response failed: %w", method, err)
184	}
185	return nil
186}

188func buildFrame(meta, payload []byte) ([]byte, error) {
189	bodySize := len(meta) + len(payload)
190	if bodySize > maxBodySize {
191		return nil, fmt.Errorf("brpc request body too large: %d", bodySize)
192	}
193	frame := make([]byte, headerSize+bodySize)
194	copy(frame[:4], magic)
195	binary.BigEndian.PutUint32(frame[4:8], uint32(bodySize))
196	binary.BigEndian.PutUint32(frame[8:12], uint32(len(meta)))
197	copy(frame[headerSize:], meta)
198	copy(frame[headerSize+len(meta):], payload)
199	return frame, nil
200}

202func readResponse(reader io.Reader, expectedID int64) ([]byte, error) {
203	header := make([]byte, headerSize)
204	if _, err := io.ReadFull(reader, header); err != nil {
205		return nil, fmt.Errorf("read brpc header failed: %w", err)
206	}
207	if string(header[:4]) != magic {
208		return nil, fmt.Errorf("invalid brpc magic: %q", string(header[:4]))
209	}
210	bodySize := binary.BigEndian.Uint32(header[4:8])
211	metaSize := binary.BigEndian.Uint32(header[8:12])
212	if bodySize > maxBodySize || metaSize > bodySize {
213		return nil, fmt.Errorf("invalid brpc sizes: body=%d meta=%d", bodySize, metaSize)
214	}
215	body := make([]byte, int(bodySize))
216	if _, err := io.ReadFull(reader, body); err != nil {
217		return nil, fmt.Errorf("read brpc body failed: %w", err)
218	}
219	var meta rpcMeta
220	if err := proto.Unmarshal(body[:metaSize], &meta); err != nil {
221		return nil, fmt.Errorf("unmarshal brpc response meta failed: %w", err)
222	}
223	if meta.CorrelationID != nil && *meta.CorrelationID != 0 && *meta.CorrelationID != expectedID {
224		return nil, fmt.Errorf("brpc correlation mismatch: got=%d want=%d", *meta.CorrelationID, expectedID)
225	}
226	if meta.CompressType != nil && *meta.CompressType != noCompression {
227		return nil, fmt.Errorf("unsupported brpc compression type: %d", *meta.CompressType)
228	}
229	if meta.Response != nil && meta.Response.ErrorCode != nil && *meta.Response.ErrorCode != 0 {
230		text := ""
231		if meta.Response.ErrorText != nil {
232			text = *meta.Response.ErrorText
233		}
234		return nil, fmt.Errorf("brpc server error %d: %s", *meta.Response.ErrorCode, text)
235	}
236	attachment := int32(0)
237	if meta.AttachmentSize != nil {
238		attachment = *meta.AttachmentSize
239	}
240	if attachment < 0 || uint32(attachment) > bodySize-metaSize {
241		return nil, fmt.Errorf("invalid brpc attachment size: %d", attachment)
242	}
243	return body[metaSize : bodySize-uint32(attachment)], nil
