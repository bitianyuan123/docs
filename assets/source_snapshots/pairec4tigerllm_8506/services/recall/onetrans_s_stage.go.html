3// OneTransSStage —— 为 rank（NS 打分）服务的 OneTrans S 阶段（Stage I）预计算 触发器.
4//
5// 定位：S 阶段的产物是 UserKV，供 rank 阶段的 NS 打分读取。本对象负责在 rank 前
6// 通过 OneTrans nearline /ingest 提前算好并落库该用户的 UserKV。
7// 注意：其借用了召回收取节点位（实现 GetCandidateItems）以被 PaiRec 调度，但语义上
8// 属 rank 前置的预计算，不代表召回、也不产出候选。
9//
10// 设计要点：
11//  1. 触发 它是异步 fire-and-forget：丢入有界队列由独立 worker 消费，
12//     绝不阻塞、放大、拖死在线关键路径；OneTrans 不可用 / 队列满时可丢弃或同步兜底（可配）；
13//  2. 组件可替换/可 mock：历史构建 HistProvider、投递客户端 IngestClient 均抽成接口，便于离线 mock 与单测。
14//
15// 说明：OneTrans /ingest 请求体字段名以 OneTrans 侧 ingest_case.json 契约为准，
16// 此处 ingestRequest 为占位骨架，接入时请对齐（user_id / item_ids / timestamps 语义一致）。

41// ingestRequest OneTrans /ingest 请求体（字段名对齐 OneTrans server 的 IngestInput 契约）.
42// 注意：OneTrans 侧 parse_ingest_input 强制要求 timestamps 键（缺失返回 400），故不可 omitempty.
43type ingestRequest struct {
44	UserID     string  `json:"user_id"`
45	ItemIDs    []int64 `json:"item_ids"`   // S 序列：按时间升序（最旧在前）
46	Timestamps []int64 `json:"timestamps"` // 与 ItemIDs 等长；缺失会导致 OneTrans 400 拒绝
47
48	// latch 关联本请求的 S 阶段闸门：由消费该请求的 worker 在 /ingest 真正返回后闭锁放行，
49	// 供 NS 阶段 /rank 在 KV miss 时有界等待后重查（见 scachelatch）。
50	latch *scachelatch.Latch `json:"-"`
51}

53// HistProvider 从 user 抽取 OneTrans S 段所需的历史序列（与 generative 同源）.
54type HistProvider interface {
55	BuildIngest(user *module.User, ctx *context.RecommendContext) (*ingestRequest, error)
56}

58// IngestClient OneTrans S 阶段（nearline /ingest）投递客户端.
59type IngestClient interface {
60	Ingest(req *ingestRequest) error
61}

63// ---------------- 默认实现 ----------------

65// userHistProvider 从 feature.Provider 获取用户历史，
66// 与 GenerativeRecall 共用同一数据源（Kafka 实时优先 + 离线 JSON 降级），
67// 保证 S 序列与训练侧同源、字段名与逗号解析由 Provider 统一处理.
68type userHistProvider struct {
69	provider *feature.Provider
70}

72func (p userHistProvider) BuildIngest(user *module.User, ctx *context.RecommendContext) (*ingestRequest, error) {
73	if user == nil {
74		return nil, fmt.Errorf("onetrans_s_stage: nil user")
75	}
76	uid := string(user.Id)
77	req := &ingestRequest{UserID: uid, ItemIDs: []int64{}}
78
79	if p.provider == nil {
80		return req, nil
81	}
82	hist, err := p.provider.GetUserHistory(uid)
83	if err != nil {
84		// 拿不到历史不是致命错：记录后返回空序列，交由上层决定是否投递.
85		log.Error(fmt.Sprintf("module=OneTransSStage\tuser=%s\thistory_err=%v", uid, err))
86		return req, nil
87	}
88	for i, id := range hist {
89		req.ItemIDs = append(req.ItemIDs, int64(id))
90		// OneTrans 侧契约要求 timestamps 与 item_ids 等长且按时间升序（最旧在前）。
91		// S 序列本身即为升序历史，故用单调递增的序列位置作为时间占位即可满足约束。
92		req.Timestamps = append(req.Timestamps, int64(i))
93	}
94	return req, nil
95}

97// httpIngestClient POST 到 OneTrans /ingest（同步按需；上层队列负责异步化）.
98type httpIngestClient struct {
99	serverURL string
100	timeout   time.Duration
101	client    *http.Client
102}
103
104func (c *httpIngestClient) Ingest(req *ingestRequest) error {
105	body, err := json.Marshal(req)
106	if err != nil {
107		return err
108	}
109	httpReq, err := http.NewRequestWithContext(stdcontext.Background(),
110		http.MethodPost, strings.TrimRight(c.serverURL, "/")+"/ingest",
111		bytes.NewReader(body))
112	if err != nil {
113		return err
114	}
115	httpReq.Header.Set("Content-Type", "application/json")
116	resp, err := c.client.Do(httpReq)
117	if err != nil {
118		return err
119	}
120	defer resp.Body.Close()
121	if resp.StatusCode != http.StatusOK {
122		return fmt.Errorf("onetrans /ingest status=%d", resp.StatusCode)
123	}
124	return nil
125}

129// OneTransSStage 为 rank 服务的 S 阶段预计算触发器（借召回节点位调度，不产候选）.
130type OneTransSStage struct {
131	*recall.BaseRecall
132	histProvider HistProvider        // 历史构建
133	ingestClient IngestClient        // OneTrans-S 投递
134	queue        chan *ingestRequest // 有界异步队列
135	injectSync   bool                // true: OneTrans 不可用/队列满时同步兜底；false: 直接丢弃
136	configJSON   *onetransSStageConfigJSON
137}
138
139// onetransSStageConfigJSON 从 RecallAlgo 字段解析配置.
140type onetransSStageConfigJSON struct {
141	OnetransURL       string `json:"onetrans_url"`        // OneTrans /ingest 地址
142	OnetransTimeoutMs int    `json:"onetrans_timeout_ms"` // ingest 超时
143	InjectSync        bool   `json:"inject_sync"`         // 队列满/失败是否同步兜底
144	QueueSize         int    `json:"queue_size"`          // S 投递有界队列长度
145	WorkerNum         int    `json:"worker_num"`          // S 消费 worker 数
146	UserFeaturesPath  string `json:"user_features_path"`  // 离线用户特征文件路径（降级源）
147	// 历史数据源（与 GenerativeRecall 一致）
148	FeatureSource string   `json:"feature_source"` // "offline" 或 "kafka"
149	KafkaBrokers  []string `json:"kafka_brokers"`  // kafka 时必填
150	KafkaTopic    string   `json:"kafka_topic"`    // kafka 时必填
151	KafkaGroupID  string   `json:"kafka_group_id"` // kafka 时必填
152}

154// NewOneTransSStage 构造函数（PaiRec RecallConfig 驱动，与 NewMilvusRecall 同范式）.
155func NewOneTransSStage(conf recconf.RecallConfig) *OneTransSStage {
156	cfg := &onetransSStageConfigJSON{
157		OnetransURL:       "http://localhost:8000",
158		OnetransTimeoutMs: 2000,
159		InjectSync:        false,
160		QueueSize:         1024,
161		WorkerNum:         4,
162		UserFeaturesPath:  "../data/user_features.json",
163		FeatureSource:     "offline",
164	}
165	if conf.RecallAlgo != "" {
166		if err := json.Unmarshal([]byte(conf.RecallAlgo), &cfg); err != nil {
167			log.Error(fmt.Sprintf("module=OneTransSStage\tname=%s\terr=parse_recall_algo:%v", conf.Name, err))
168		}
169	}
170
171	r := &OneTransSStage{
172		BaseRecall:   recall.NewBaseRecall(conf),
173		histProvider: buildHistProvider(cfg),
174		ingestClient: &httpIngestClient{
175			serverURL: cfg.OnetransURL,
176			timeout:   time.Duration(cfg.OnetransTimeoutMs) * time.Millisecond,
177			client:    &http.Client{Timeout: time.Duration(cfg.OnetransTimeoutMs) * time.Millisecond},
178		},
179		queue:      make(chan *ingestRequest, cfg.QueueSize),
180		injectSync: cfg.InjectSync,
181		configJSON: cfg,
182	}
183
184	// 启动 S 投递消费者.
185	for i := 0; i < cfg.WorkerNum; i++ {
186		go r.consumer()
187	}
188
189	fmt.Printf("[OneTransSStage] init name=%s onetrans_url=%s inject_sync=%v queue=%d workers=%d feature_source=%s\n",
190		conf.Name, cfg.OnetransURL, cfg.InjectSync, cfg.QueueSize, cfg.WorkerNum, cfg.FeatureSource)
191	return r
192}

194// buildHistProvider 构造历史提供器：Kafka 实时优先（可选）+ 离线 JSON 降级，与 GenerativeRecall 同源.
195func buildHistProvider(cfg *onetransSStageConfigJSON) HistProvider {
196	provider := feature.NewProvider(nil, cfg.UserFeaturesPath)
197	if cfg.FeatureSource == "kafka" && len(cfg.KafkaBrokers) > 0 && cfg.KafkaTopic != "" {
198		consumer := feature.NewConsumer(&feature.KafkaConfig{
199			Brokers: cfg.KafkaBrokers,
200			Topic:   cfg.KafkaTopic,
201			GroupID: cfg.KafkaGroupID,
202		}, 0)
203		if err := consumer.Start(); err != nil {
204			log.Error(fmt.Sprintf("module=OneTransSStage\tkafka_start_err=%v\tfallback=offline", err))
205		} else {
206			provider = feature.NewProvider(consumer, cfg.UserFeaturesPath)
207		}
208	}
209	return userHistProvider{provider: provider}
210}

212// GetCandidateItems 实现 pairec Recall 接口，以便被 PaiRec 在召回收取节点位调度。
213// 本对象定位为 rank 前置的 S 阶段预计算：只触发近线 /ingest 提前算 UserKV，不产出候选，
214// 因此返回 nil（不参与召回 merge）。
215func (r *OneTransSStage) GetCandidateItems(user *module.User, ctx *context.RecommendContext) []*module.Item {
216	// panic 隔离：OneTrans-S 预计算失败绝不能影响在线链路.
217	defer func() {
218		if e := recover(); e != nil {
219			log.Error(fmt.Sprintf("module=OneTransSStage\trecover=%v", e))
220		}
221	}()
222
223	// 触发 OneTrans-S 预计算（异步、不阻塞）.
224	r.deliverS(user, ctx)
225	return nil
226}
227
228// deliverS 构建 OneTrans ingest 请求并异步投递，同时为"可能到来的 S"开一把闸门。
229// 闭锁时机：worker 在 /ingest 真正返回后调用 -> 保证 NS 阶段取到的是"已写完"的缓存。
230func (r *OneTransSStage) deliverS(user *module.User, ctx *context.RecommendContext) {
231	defer func() {
232		if e := recover(); e != nil {
233			log.Error(fmt.Sprintf("module=OneTransSStage\tdeliverS_recover=%v", e))
234		}
235	}()
236	req, err := r.histProvider.BuildIngest(user, ctx)
237	if err != nil || len(req.ItemIDs) == 0 {
238		// 无历史 / 取历史失败：本就不该产 S，也不再放闸门，NS 侧看到"无闸门"直接降级。
239		return
240	}
241	// 开一把按用户闸门：S 是否在途一目了然；写入完成后由投递路径闭锁放行。
242	latch := scachelatch.New()
243	req.latch = latch
244	ctx.AddContextParam(scachelatch.ContextKey, latch)
245
246	select {
247	case r.queue <- req:
248		// 进队列，worker 异步消费；闭锁时机在 consumer（Ingest 返回后）。
249	default:
250		// 队列满：按配置决定同步兜底或丢弃；但必须闭锁，避免 NS 对一把永不放行的闸门空等。
251		if r.injectSync {
252			_ = r.ingestClient.Ingest(req)
253		}
254		latch.Done()
255	}
256}
257
258// consumer 消费有界队列里的 ingest 请求，发往 OneTrans /ingest（fire-and-forget）.
259// 无论投递成功或失败，都在 Ingest 返回后闭锁：成功则 NS 重查命中，失败则 NS 用均匀分降级。
260func (r *OneTransSStage) consumer() {
261	for req := range r.queue {
262		if err := r.ingestClient.Ingest(req); err != nil {
263			// OneTrans 不可达：记录后丢弃（不影响在线链路）.
264			log.Error(fmt.Sprintf("module=OneTransSStage\tuser=%s\tingest_err=%v", req.UserID, err))
265		}
266		if req.latch != nil {
267			req.latch.Done()
268		}
269	}
