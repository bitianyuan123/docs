1// Package scachelatch 提供一把"按用户"的闸门，用于保证 OneTrans 两阶段的一致性：
2//
3//	S 阶段（/ingest）预计算的 UserKV 缓存在写入完成后闭锁放行；
4//	NS 阶段（/rank）在 KV miss 时做一次有届等待后重查，从而让"NS 读取"严格发生在
5//	"S 落盘"之后，消除冷启动竞态下的无区分度打分，全程不需要在线补算。
6//
7// 两个约定（务必遵守）：
8//  1. 闭锁时机 Done() 必须在 OneTrans store_.put 真正完成后调用，否则 NS 等到的是
9//     "仅被接受、尚未写完"的信号，重查仍会 miss。
10//  2. 等待 Wait() 必须带超时上限；对"无闸门（冷用户/无历史）"的场景不要执行等待，
11//     避免空等一个永不发生的写入而挂户。
12package scachelatch
13
14import (
15	"sync"
16	"time"
17)
18
19// ContextKey 用于在 PaiRec 单请求共享的 RecommendContext 中携带闸门，
20// 使 S 阶段（recall）设置的闸门能被 NS 阶段（sort）同步取到。
21const ContextKey = "onetrans.s_cache_latch"
22
23// Latch 是一把幂等的按用户闸门。
24type Latch struct {
25	ch   chan struct{}
26	once sync.Once
27}
28
29// New 创建一把初始关闭（未就绪）的闸门。
30func New() *Latch { return &Latch{ch: make(chan struct{})} }
31
32// Done 标记 S 已成功写入缓存并放行所有等待者。可被并发安全地多次调用（幂等）。
33func (l *Latch) Done() { l.once.Do(func() { close(l.ch) }) }
34
35// Wait 有届等待闸门放行。
36//   - 返回 true：S 已就绪（缓存已写入），可放心重查；
37//   - 返回 false：超时（S 失败/尚未完成），调用方应降级而不是无限挂起。
38func (l *Latch) Wait(timeout time.Duration) bool {
39	select {
40	case <-l.ch:
41		return true
42	case <-time.After(timeout):
43		return false
44	}
45}
