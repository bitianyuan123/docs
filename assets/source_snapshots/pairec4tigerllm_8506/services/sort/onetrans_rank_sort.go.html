28// OneTransRankConfig 是 OneTransRankSort 的配置项。
29// HTTP POST /rank 精排：PaiRec 只传 user_id + item_id[]，特征由 OneTrans 自装配。
30type OneTransRankConfig struct {
31	Name               string `json:"name"`
32	Protocol           string `json:"protocol"`
33	ServerURL          string `json:"server_url"`
34	TimeoutMS          int    `json:"timeout_ms"`
35	ExpectedCandidates int    `json:"expected_candidates"`
36	RequiredModelRole  string `json:"required_model_role"`
37}
38
39type oneTransUserDefineConfig struct {
40	OneTransRankSorts []OneTransRankConfig `json:"OneTransRankSorts"`
41}

43// OneTransRankSort 实现 ISort 接口，请求协议固定为 HTTP（复用 rankRequest / rankResponse /
44// validateResponse 契约，与 DeepFMRankSort 完全对齐）。
45type OneTransRankSort struct {
46	config OneTransRankConfig
47	client *http.Client
48}
49
50func NewOneTransRankSort(config OneTransRankConfig) (*OneTransRankSort, error) {
51	if config.Name == "" {
52		return nil, errors.New("OneTrans rank sort name is required")
53	}
54	if config.Protocol == "" {
55		config.Protocol = "http"
56	}
57	if config.Protocol != "http" {
58		return nil, fmt.Errorf("OneTrans rank protocol must be http, got %q", config.Protocol)
59	}
60	if config.ServerURL == "" {
61		return nil, errors.New("OneTrans rank server_url is required")
62	}
63	if config.TimeoutMS <= 0 {
64		return nil, errors.New("OneTrans rank timeout_ms must be positive")
65	}
66	if config.ExpectedCandidates <= 0 {
67		return nil, errors.New("OneTrans expected_candidates must be positive")
68	}
69	if config.RequiredModelRole == "" {
70		return nil, errors.New("OneTrans required_model_role is required")
71	}
72	transport := &http.Transport{
73		Proxy: nil,
74		DialContext: (&net.Dialer{
75			Timeout: time.Duration(config.TimeoutMS) * time.Millisecond,
76		}).DialContext,
77	}
78	return &OneTransRankSort{
79		config: config,
80		client: &http.Client{
81			Transport: transport,
82			Timeout:   time.Duration(config.TimeoutMS) * time.Millisecond,
83		},
84	}, nil
85}

87// RegisterOneTransRankSortFromConfig 从 UserDefineConfs.OneTransRankSorts 注册所有 OneTrans 精排器。
88func RegisterOneTransRankSortFromConfig() error {
89	if recconf.Config == nil || len(recconf.Config.UserDefineConfs) == 0 {
90		return nil
91	}
92	var userConfig oneTransUserDefineConfig
93	if err := json.Unmarshal(recconf.Config.UserDefineConfs, &userConfig); err != nil {
94		return fmt.Errorf("parse OneTrans rank config: %w", err)
95	}
96	for _, config := range userConfig.OneTransRankSorts {
97		instance, err := NewOneTransRankSort(config)
98		if err != nil {
99			return err
100		}
101		pairecsort.RegisterSort(config.Name, instance)
102		fmt.Printf("Registering OneTransRankSort: %s protocol=HTTP endpoint=%s timeout_ms=%d candidates=%d model_role=%s\n",
103			config.Name, config.ServerURL, config.TimeoutMS, config.ExpectedCandidates, config.RequiredModelRole)
104	}
105	return nil
106}

108func (s *OneTransRankSort) Sort(sortData *pairecsort.SortData) error {
109	items, ok := sortData.Data.([]*module.Item)
110	if !ok {
111		return s.fail(sortData, errors.New("sort data is not []*module.Item"))
112	}
113	if sortData.Context == nil || sortData.User == nil {
114		return s.fail(sortData, errors.New("rank context and user are required"))
115	}
116	if len(items) == 0 {
117		return s.fail(sortData, errors.New("rank candidates empty"))
118	}
119	if len(items) > s.config.ExpectedCandidates {
120		return s.fail(sortData, fmt.Errorf("exceeded %d candidates, got %d",
121			s.config.ExpectedCandidates, len(items)))
122	}
123
124	requestItems := make([]requestItem, 0, len(items))
125	originalOrder := make([]string, 0, len(items))
126	inputSet := make(map[string]struct{}, len(items))
127	for _, item := range items {
128		if item == nil || item.Id == "" {
129			return s.fail(sortData, errors.New("candidate has empty item_id"))
130		}
131		itemID := string(item.Id)
132		if _, duplicate := inputSet[itemID]; duplicate {
133			return s.fail(sortData, fmt.Errorf("duplicate candidate item_id=%s", itemID))
134		}
135		inputSet[itemID] = struct{}{}
136		originalOrder = append(originalOrder, itemID)
137		requestItems = append(requestItems, requestItem{ItemID: itemID})
138	}
139
140	// PaiRec 侧能给的信息只有 request_id + user_id + item_id[]，
141	// 其余特征（uid_sparse/user_dense/artist_ids/album_ids/dense）由 OneTrans 侧自装配。
142	payload := rankRequest{
143		RequestID: sortData.Context.RecommendId,
144		Context: map[string]interface{}{
145			"request_id":     sortData.Context.RecommendId,
146			"span_id":        "onetrans-rank",
147			"parent_span_id": "sort",
148			"sampled":        true,
149		},
150		UserID: string(sortData.User.Id),
151		Items:  requestItems,
152	}
153	started := time.Now()
154	ranked, err := s.call(payload)
155	if err != nil {
156		return s.fail(sortData, err)
157	}
158
159	// 冷启动竞态收敛：S 段 KV 首查 miss 时，若召回阶段确有在途 S（有闸门），
160	// 则做一次有界等待后重查，保证"S 真正落盘后 NS 再读取"，全程无在线补算。
161	// - 无闸门（冷用户/无历史/被丢弃）：跳过等待，直接用首查结果（均匀分降级），不空等。
162	// - 有闸门但超时：Wait 返回 false，仍用首查结果降级，不挂起。
163	if !ranked.Trace.KvHit {
164		if latch, ok := sortData.Context.GetContextParam(scachelatch.ContextKey).(*scachelatch.Latch); ok {
165			latch.Wait(time.Duration(s.config.TimeoutMS) * time.Millisecond)
166			ranked, err = s.call(payload)
167			if err != nil {
168				return s.fail(sortData, err)
169			}
170		}
171	}
172	if err := validateResponse(ranked, payload, inputSet, s.config.RequiredModelRole); err != nil {
173		return s.fail(sortData, err)
174	}
175
176	scores := make(map[string]float64, len(ranked.Items))
177	for _, rankedItem := range ranked.Items {
178		scores[rankedItem.ItemID] = rankedItem.Score
179	}
180	for _, item := range items {
181		item.Score = scores[string(item.Id)]
182		item.AddAlgoScore("onetrans", item.Score)
183		item.AddProperty("onetrans_score", item.Score)
184	}
185	gosort.SliceStable(items, func(left, right int) bool {
186		return items[left].Score > items[right].Score
187	})
188	reordered := false
189	for index, item := range items {
190		if string(item.Id) != originalOrder[index] {
191			reordered = true
192			break
193		}
194	}
195	sortData.Data = items
196
197	clientTotalMS := float64(time.Since(started).Microseconds()) / 1000
198	sortData.Context.AddContextParam(OneTransRankTraceContextKey, map[string]interface{}{
199		"protocol":            "http",
200		"model_version":       ranked.ModelVersion,
201		"model_role":          ranked.ModelRole,
202		"candidate_count":     len(items),
203		"service_total_ms":    ranked.Trace.TotalMS,
204		"client_total_ms":     clientTotalMS,
205		"score_unique_count":  ranked.Trace.ScoreUniqueCount,
206		"business_payload_ok": true,
207		"reordered":           reordered,
208	})
209	if os.Getenv("PAIREC_TRACE_STDOUT") == "1" {
210		encoded, _ := json.Marshal(map[string]interface{}{
211			"event":                 "onetrans_rank_complete",
212			"request_id":            payload.RequestID,
213			"model_version":         ranked.ModelVersion,
214			"model_role":            ranked.ModelRole,
215			"candidate_count":       len(items),
216			"service_total_ms":      ranked.Trace.TotalMS,
217			"client_total_ms":       clientTotalMS,
218			"score_unique_count":    ranked.Trace.ScoreUniqueCount,
219			"reordered":             reordered,
220			"business_payload_bytes": ranked.Trace.BusinessPayloadBytes,
221		})
222		fmt.Println(string(encoded))
223	}
224
225	// 记入 PaiRec pipeline span，进 service_phase_duration_seconds 直方图
226	//（phase 取自 attributes 的 feature_us / compute_us / backend_rpc_us，见 observability.Record）
227	observability.RecordDuration(sortData.Context, "onetrans_rank", "onetrans_rank",
228		s.config.Protocol, "sort", false, started, "ok", map[string]interface{}{
229			"candidate_count":  len(items),
230			"model_version":    ranked.ModelVersion,
231			"service_total_us": ranked.Trace.TotalUS,
232			"feature_us":       ranked.Trace.FeatureUS,
233			"compute_us":       ranked.Trace.ComputeUS,
234			"backend_rpc_us":   ranked.Trace.BackendRPCUS,
235			"lookup_us":        ranked.Trace.LookupUS,
236			"encode_us":        ranked.Trace.EncodeUS,
237			"kv_us":            ranked.Trace.KvUS,
238			"batch_wait_us":    ranked.Trace.BatchWaitUS,
239			"batch_compute_us": ranked.Trace.BatchComputeUS,
240			"batch_size":       int(ranked.Trace.BatchSize),
241		})
242	return nil
243}

245func (s *OneTransRankSort) call(payload rankRequest) (rankResponse, error) {
246	body, err := json.Marshal(payload)
247	if err != nil {
248		return rankResponse{}, fmt.Errorf("encode rank request: %w", err)
249	}
250	req, err := http.NewRequest(http.MethodPost, s.config.ServerURL+"/rank", bytes.NewReader(body))
251	if err != nil {
252		return rankResponse{}, fmt.Errorf("create rank request: %w", err)
253	}
254	req.Header.Set("Content-Type", "application/json")
255	response, err := s.client.Do(req)
256	if err != nil {
257		return rankResponse{}, fmt.Errorf("call OneTrans rank service: %w", err)
258	}
259	defer response.Body.Close()
260	if response.StatusCode != http.StatusOK {
261		return rankResponse{}, fmt.Errorf("OneTrans rank service HTTP status=%d", response.StatusCode)
262	}
263	responseBody, err := io.ReadAll(io.LimitReader(response.Body, 1<<20))
264	if err != nil {
265		return rankResponse{}, fmt.Errorf("read OneTrans rank response: %w", err)
266	}
267	var ranked rankResponse
268	if err := json.Unmarshal(responseBody, &ranked); err != nil {
269		return rankResponse{}, fmt.Errorf("decode OneTrans rank response: %w", err)
270	}
271	return ranked, nil
272}

274func (s *OneTransRankSort) fail(sortData *pairecsort.SortData, err error) error {
275	if sortData.Context != nil {
276		sortData.Context.AddContextParam(OneTransRankErrorContextKey, err.Error())
277		sortData.Context.LogError("module=OneTransRankSort\terr=" + err.Error())
278		if os.Getenv("PAIREC_TRACE_STDOUT") == "1" {
279			encoded, _ := json.Marshal(map[string]interface{}{
280				"event":      "onetrans_rank_error",
281				"request_id": sortData.Context.RecommendId,
282				"error":      err.Error(),
283			})
284			fmt.Println(string(encoded))
285		}
286	}
287	sortData.Data = []*module.Item{}
288	return err
289}
