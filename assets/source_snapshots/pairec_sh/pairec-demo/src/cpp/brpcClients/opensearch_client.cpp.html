48std::string Opensearch_Clientor::Base64_encode(const std::string& input) {
49    BIO* b64 = BIO_new(BIO_f_base64());
50    BIO* mem = BIO_new(BIO_s_mem());
51    b64 = BIO_push(b64, mem);
52    BIO_set_flags(b64, BIO_FLAGS_BASE64_NO_NL);
53    BIO_write(b64, input.c_str(), input.length());
54    BIO_flush(b64);
55    BUF_MEM* bptr = nullptr;
56    BIO_get_mem_ptr(b64, &bptr);
57    std::string result(bptr->data, bptr->length);
58    BIO_free_all(b64);
59    return result;
60}

62int Opensearch_Clientor::PostRequest(std::string indexName, std::string queryInfo)
63{
64    brpc::Controller cntl;
65    cntl.http_request().set_method(brpc::HTTP_METHOD_POST);
66    cntl.http_request().uri() = "/" + indexName + "/_search?pretty";
67    cntl.http_request().SetHeader("Authorization",
68        "Basic " + m_userInfo);
69
70    cntl.http_request().SetHeader("Content-Type", "application/json");
71    // Query body
72    // std::string body = R"({"query":{"match":{"content":"Python 编程语言"}},"explain":false})";
73    std::string body = R"({"query":{"match":{"content":")" + queryInfo + R"("}},"explain":false})";
74    cntl.request_attachment().append(body);
75
76    m_channel.CallMethod(nullptr, &cntl, nullptr, nullptr, nullptr);
77    if (cntl.Failed()) {
78        LOG(ERROR) << "Request failed: " << cntl.ErrorText();
79        return -1;
80    }
81
82    LOG(INFO) << "Response: " << cntl.response_attachment();
83    return 0;
84}
