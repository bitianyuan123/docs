36int Redis_Clientor::SendRequest(brpc::RedisRequest* pRequest, brpc::RedisResponse* pResponse)
37{
38    brpc::Controller cntl;
39    cntl.set_timeout_ms(1000);
40
41    m_channel.CallMethod(NULL, &cntl, pRequest, pResponse, NULL);
42    if (cntl.Failed()) {
43        LOG(ERROR) << "Fail to get key, error=" << cntl.ErrorText();
44        return -1;
45    }
46
47    const brpc::RedisReply* pReply = &(pResponse->reply(0));
48    if (pReply->is_error()) {
49        LOG(ERROR) << "Redis error reply=" << pReply->error_message();
50        return -1;
51    }
52    if (pReply->is_nil()) {
53        return 1;
54    }
55
56    return 0;
57}

59int Redis_Clientor::Get(const std::string& key, std::string* pValue) {
60    brpc::RedisRequest request;
61    brpc::RedisResponse response;
62    
63    std::vector<butil::StringPiece> components;
64    components.push_back("GET");
65    components.push_back(key);
66    if (!request.AddCommandByComponents(components.data(), components.size())) {
67        LOG(ERROR) << "Fail to add GET command, key=" << key;
68        return -1;
69    }
70
71    int res = SendRequest(&request, &response);
72    if (res) {
73        return res;
74    }  
75
76    const brpc::RedisReply& reply = response.reply(0);
77    if (pValue != NULL) {
78        pValue->assign(reply.c_str(), reply.size());
79    }
80    return 0;

111int Redis_Clientor::MGet(const std::vector<std::string>& keys, std::vector<std::string>* pValues) {
112    if (keys.empty()) {
113        return 0;
114    }
115
116    brpc::RedisRequest request;
117    brpc::RedisResponse response;
118    brpc::Controller cntl;
119
120    std::vector<butil::StringPiece> components;
121    components.push_back("MGET");
122    for (size_t i = 0; i < keys.size(); ++i) {
123        components.push_back(keys[i]);
124    }
125    if (!request.AddCommandByComponents(components.data(), components.size())) {
126        LOG(ERROR) << "Fail to add MGET command, count=" << keys.size();
127        return -1;
128    }
129
130    m_channel.CallMethod(NULL, &cntl, &request, &response, NULL);
131    if (cntl.Failed()) {
132        LOG(ERROR) << "Fail to mget keys, error=" << cntl.ErrorText();
133        return -1;
134    }
135
136    if (response.reply_size() != (int)keys.size()) {
137        LOG(ERROR) << "MGET reply count mismatch, expect=" << keys.size()
138                   << ", actual=" << response.reply_size();
139        return -1;
140    }
141
142    pValues->clear();
143    pValues->resize(keys.size());
144    for (int i = 0; i < response.reply_size(); ++i) {
145        const brpc::RedisReply& reply = response.reply(i);
146        if (reply.is_nil() || reply.is_error()) {
147            continue;
148        }
149        (*pValues)[i].assign(reply.c_str(), reply.size());
150    }
151    return 0;
