63int Clis_RedisMGet(ConnectClients_T* p, char** keys, int n, char*** pValues) {
64    if (p == NULL || keys == NULL || pValues == NULL || n <= 0) {
65        return -1;
66    }
67    std::vector<std::string> vkeys;
68    vkeys.reserve(n);
69    for (int i = 0; i < n; ++i) {
70        vkeys.push_back(std::string(keys[i] != NULL ? keys[i] : ""));
71    }
72    std::vector<std::string> values;
73    if (p->impl.RedisMGet(vkeys, &values) != 0) {
74        return -1;
75    }
76
77    char** arr = (char**)malloc(sizeof(char*) * (size_t)n);
78    if (arr == NULL) {
79        return -1;
80    }
81    for (int i = 0; i < n; ++i) {
82        arr[i] = strdup(values[i].c_str());
83        if (arr[i] == NULL) {
84            Clis_FreeStringArray(arr, i);
85            return -1;
86        }
87    }
88    *pValues = arr;
89    return 0;
90}
