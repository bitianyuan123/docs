3// Package stageclient bridges the C++ ConnectClients (see include/ConnectClients.h)
4// to Go, so that redis data is fetched via the brpc c++ client, and opensearch
5// searches are sent via the brpc http c++ client.
6//
7// The underlying C++ clients are singletons, all clients created by
8// NewClient/NewOpensearchClient share one pooled connection to the addr passed
9// at creation.
10//
11// Before building with CGO_ENABLED=1, build the static library first (linux):
12//
13//	cd src/cpp && make
14//
15// which produces libstageclients.a in lib/.

32// Client is a stage client which talks to redis via the brpc c++ client
33type Client struct {
34	clis *C.ConnectClients_T
35}
36
37// NewClient connects to the redis server at addr, example: "127.0.0.1:6379"
38func NewRedisClient(addr string) (*Client, error) {
39	clis := C.Clis_New()
40	if clis == nil {
41		return nil, fmt.Errorf("stageClient: new connect clients error")
42	}
43	cAddr := C.CString(addr)
44	defer C.free(unsafe.Pointer(cAddr))
45	if rc := C.Clis_ConnectRedis(clis, cAddr); rc != 0 {
46		C.Clis_Delete(clis)
47		return nil, fmt.Errorf("stageClient: connect redis error, addr=%s, rc=%d", addr, int(rc))
48	}
49	return &Client{clis: clis}, nil
50}

52// NewOpensearchClient connects to the opensearch server at addr,
53// example: "127.0.0.1:9200", user and password may be empty when
54// the server has no basic authentication
55func NewOpensearchClient(addr, user, password string) (*Client, error) {
56	clis := C.Clis_New()
57	if clis == nil {
58		return nil, fmt.Errorf("stageClient: new connect clients error")
59	}
60	cAddr := C.CString(addr)
61	defer C.free(unsafe.Pointer(cAddr))
62	cUser := C.CString(user)
63	defer C.free(unsafe.Pointer(cUser))
64	cPassword := C.CString(password)
65	defer C.free(unsafe.Pointer(cPassword))
66	if rc := C.Clis_ConnectOpensearch(clis, cAddr, cUser, cPassword); rc != 0 {
67		C.Clis_Delete(clis)
68		return nil, fmt.Errorf("stageClient: connect opensearch error, addr=%s, rc=%d", addr, int(rc))
69	}
70	return &Client{clis: clis}, nil
71}

73// Close releases the c++ connect clients
74func (c *Client) Close() {
75	if c != nil && c.clis != nil {
76		C.Clis_Delete(c.clis)
77		c.clis = nil
78	}
79}

81// Get returns the value of the key, found is false when the key does not exist
82func (c *Client) Get(key string) (value string, found bool, err error) {
83	cKey := C.CString(key)
84	defer C.free(unsafe.Pointer(cKey))
85
86	var cValue *C.char
87	rc := C.Clis_RedisGet(c.clis, cKey, &cValue)
88	switch rc {
89	case 0:
90		defer C.Clis_FreeString(cValue)
91		return C.GoString(cValue), true, nil
92	case 1:
93		return "", false, nil
94	default:
95		return "", false, fmt.Errorf("stageClient: redis get error, key=%s, rc=%d", key, int(rc))
96	}
97}

99// Set stores the value of the key

112// MGet returns the values of the keys in one pipelined command,
113// values[i] is empty when the key does not exist
114func (c *Client) MGet(keys []string) ([]string, error) {
115	if len(keys) == 0 {
116		return []string{}, nil
117	}
118
119	cKeys := make([]*C.char, len(keys))
120	for i, key := range keys {
121		cKeys[i] = C.CString(key)
122	}
123	defer func() {
124		for _, cKey := range cKeys {
125			C.free(unsafe.Pointer(cKey))
126		}
127	}()
128
129	var cValues **C.char
130	if rc := C.Clis_RedisMGet(c.clis, &cKeys[0], C.int(len(keys)), &cValues); rc != 0 {
131		return nil, fmt.Errorf("stageClient: redis mget error, count=%d, rc=%d", len(keys), int(rc))
132	}
133	defer C.Clis_FreeStringArray(cValues, C.int(len(keys)))
134
135	values := make([]string, len(keys))
136	for i := range values {
137		p := *(**C.char)(unsafe.Pointer(uintptr(unsafe.Pointer(cValues)) + uintptr(i)*unsafe.Sizeof(*cValues)))
138		values[i] = C.GoString(p)
139	}
140	return values, nil
141}

143// Search posts the query DSL body to /{index}/_search and returns
144// the response body json, an empty result set is still a success
145func (c *Client) Search(index string, body string) (string, error) {
146	cIndex := C.CString(index)
147	defer C.free(unsafe.Pointer(cIndex))
148	cBody := C.CString(body)
149	defer C.free(unsafe.Pointer(cBody))
150
151	var cValue *C.char
152	if rc := C.Clis_OpensearchSearch(c.clis, cIndex, cBody, &cValue); rc != 0 {
153		return "", fmt.Errorf("stageClient: opensearch search error, index=%s, rc=%d", index, int(rc))
154	}
155	defer C.Clis_FreeString(cValue)
156	return C.GoString(cValue), nil
