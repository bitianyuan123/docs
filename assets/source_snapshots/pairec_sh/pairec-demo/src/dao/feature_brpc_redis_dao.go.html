27type BrpcRedisFeatureDaoConfig struct {
28	// redis server address, example: "127.0.0.1:6379"
29	Addr string
30	// user or item
31	FeatureStore string
32	// feature key, example: "user:uid" or "item:id"
33	FeatureKey string
34	// redis key prefix
35	KeyPrefix string
36	// value format: csv (name:value,name:value) or json
37	FieldType string
38	// csv value delimiter
39	Delimiter string
40}

42// FeatureBrpcRedisDao loads user/item features from redis via the brpc c++ client
43type FeatureBrpcRedisDao struct {
44	client       *stageclient.Client
45	featureStore string
46	featureKey   string
47	keyPrefix    string
48	fieldType    string
49	delimiter    string
50}

52func NewFeatureBrpcRedisDao(conf BrpcRedisFeatureDaoConfig) (*FeatureBrpcRedisDao, error) {
53	if conf.Addr == "" {
54		conf.Addr = "127.0.0.1:6379"
55	}
56	if conf.FeatureStore == "" {
57		conf.FeatureStore = module.Feature_Store_User
58	}
59	if conf.FeatureKey == "" {
60		conf.FeatureKey = "user:uid"
61	}
62	if conf.FieldType == "" {
63		conf.FieldType = BrpcRedisFieldTypeJSON
64	}
65	if conf.Delimiter == "" {
66		conf.Delimiter = ","
67	}
68
69	client, err := stageclient.NewRedisClient(conf.Addr)
70	if err != nil {
71		return nil, err
72	}
73
74	return &FeatureBrpcRedisDao{
75		client:       client,
76		featureStore: conf.FeatureStore,
77		featureKey:   conf.FeatureKey,
78		keyPrefix:    conf.KeyPrefix,
79		fieldType:    conf.FieldType,
80		delimiter:    conf.Delimiter,
81	}, nil
82}

84// FeatureFetch implements module.FeatureDao
85func (d *FeatureBrpcRedisDao) FeatureFetch(user *module.User, items []*module.Item, context *context.RecommendContext) {
86	if d.featureStore == module.Feature_Store_User {
87		d.userFeatureFetch(user, context)
88	} else {
89		d.itemsFeatureFetch(items, context)
90	}
91}

93func (d *FeatureBrpcRedisDao) userFeatureFetch(user *module.User, context *context.RecommendContext) {
94	key := d.userKey(user)
95	if key == "" {
96		log.Error(fmt.Sprintf("requestId=%s\tmodule=FeatureBrpcRedisDao\tuid=%s\terror=feature key empty", context.RecommendId, user.Id))
97		return
98	}
99
100	value, found, err := d.client.Get(key)
101	if err != nil {
102		log.Error(fmt.Sprintf("requestId=%s\tmodule=FeatureBrpcRedisDao\tuid=%s\terror=get user feature error(%v), key=%s", context.RecommendId, user.Id, err, key))
103		return
104	}
105	if !found {
106		log.Info(fmt.Sprintf("requestId=%s\tmodule=FeatureBrpcRedisDao\tuid=%s\tmsg=user feature empty, key=%s", context.RecommendId, user.Id, key))
107		return
108	}
109
110	properties := d.parseProperties(value)
111	log.Info(fmt.Sprintf("properties=%v", properties))
112	if len(properties) == 0 {
113		return
114	}
115	user.AddProperties(properties)
116
117	if context.Debug {
118		log.Info(fmt.Sprintf("requestId=%s\tmodule=FeatureBrpcRedisDao\tuid=%s\tmsg=user features loaded, key=%s, features=%v", context.RecommendId, user.Id, key, properties))
119	}
120}

122func (d *FeatureBrpcRedisDao) itemsFeatureFetch(items []*module.Item, context *context.RecommendContext) {
123	if len(items) == 0 {
124		return
125	}
126
127	keys := make([]string, len(items))
128	for i, item := range items {
129		keys[i] = d.keyPrefix + d.itemKey(item)
130	}
131
132	for start := 0; start < len(keys); start += brpcRedisMGetBatchSize {
133		end := min(start+brpcRedisMGetBatchSize, len(keys))
134		batchKeys := keys[start:end]
135		batchItems := items[start:end]
136
137		values, err := d.client.MGet(batchKeys)
138		if err != nil {
139			log.Error(fmt.Sprintf("requestId=%s\tmodule=FeatureBrpcRedisDao\terror=mget item feature error(%v)", context.RecommendId, err))
140			continue
141		}
142		for i, value := range values {
143			if value == "" {
144				continue
145			}
146			properties := d.parseProperties(value)
147			if len(properties) == 0 {
148				continue
149			}
150			batchItems[i].AddProperties(properties)
151		}
152	}
153}

155func (d *FeatureBrpcRedisDao) userKey(user *module.User) string {
156	comms := strings.Split(d.featureKey, ":")
157	if len(comms) < 2 {
158		return string(user.Id)
159	}
160	if comms[1] == "uid" {
161		return d.keyPrefix + string(user.Id)
162	}
163	key := user.StringProperty(comms[1])
164	if key == "" {
165		return ""
166	}
167	return d.keyPrefix + key
168}

170func (d *FeatureBrpcRedisDao) itemKey(item *module.Item) string {
171	if d.featureKey == "item:id" {
172		return string(item.Id)
173	}
174	return item.StringProperty(d.featureKey)
175}

177func (d *FeatureBrpcRedisDao) parseProperties(value string) map[string]interface{} {
178	properties := make(map[string]interface{})
179	if d.fieldType == BrpcRedisFieldTypeJSON {
180		if err := json.Unmarshal([]byte(value), &properties); err != nil {
181			return nil
182		}
183		return properties
184	}
185
186	for _, pair := range strings.Split(value, d.delimiter) {
187		idx := strings.Index(pair, ":")
188		if idx <= 0 {
189			continue
190		}
191		name := pair[:idx]
192		val := pair[idx+1:]
193		if f, err := strconv.ParseFloat(val, 64); err == nil {
194			properties[name] = f
195		} else {
196			properties[name] = val
197		}
198	}
199	return properties
200}

202// RegisterBrpcRedisFeatureDao registers the brpc redis feature dao to the feature
203// service of the scene, so that user features of the scene are fetched via brpc redis.
204// The redis address can be set by the env BRPC_REDIS_ADDR, default "127.0.0.1:6379".
205func RegisterBrpcRedisFeatureDao(sceneName string) error {
206	return RegisterBrpcRedisFeatureDaoWithConfig(sceneName, BrpcRedisFeatureDaoConfig{
207		Addr:         os.Getenv("BRPC_REDIS_ADDR"),
208		FeatureStore: module.Feature_Store_User,
209		FeatureKey:   "user:uid",
210		KeyPrefix:    "pairec:feature:",
211		FieldType:    BrpcRedisFieldTypeJSON,
212		Delimiter:    ",",
213	})
214}

227// RegisterBrpcRedisFeatureDaoWithConfig registers a brpc redis feature dao with
228// a custom config, set FeatureStore to module.Feature_Store_Item to fetch item features
229func RegisterBrpcRedisFeatureDaoWithConfig(sceneName string, conf BrpcRedisFeatureDaoConfig) error {
230	if conf.Addr == "" {
231		conf.Addr = "127.0.0.1:6379"
232	}
233	dao, err := NewFeatureBrpcRedisDao(conf)
234	if err != nil {
235		return fmt.Errorf("init brpc redis feature dao: %w", err)
236	}
237
238	feature.RegisterLoadFeatureFunc(sceneName, dao.FeatureFetch)
239	log.Info(fmt.Sprintf("module=FeatureBrpcRedisDao\tmsg=brpc redis feature dao registered, scene=%s, addr=%s, store=%s, prefix=%s", sceneName, conf.Addr, conf.FeatureStore, conf.KeyPrefix))
240	return nil
241}
