26// BrpcSparseRecall retrieves candidate items from opensearch with a
27// sparse query built from the user sparse tokens. The user sparse tokens are
28// turned into a bool should query of term clauses, the token weight becomes the
29// term boost, so the hit score approximates the weighted sum of the matched
30// item tokens (sparse vector dot product). It implements the framework
31// recall.Recall interface.
32type BrpcSparseRecall struct {
33	client      *stageclient.Client
34	modelName   string
35	indexName   string
36	itemIdField string
37	recallCount int
38}
39
40func NewBrpcSparseRecall(name string, conf BrpcSparseRecallConfig) (*BrpcSparseRecall, error) {
41	if name == "" {
42		return nil, fmt.Errorf("brpc opensearch sparse recall: name empty")
43	}
44	if conf.IndexName == "" {
45		return nil, fmt.Errorf("brpc opensearch sparse recall: index name empty")
46	}
47	if conf.ItemIdField == "" {
48		conf.ItemIdField = brpcSparseDefaultItemIdField
49	}
50
51	client, err := stageclient.NewOpensearchClient(conf.Addr, conf.User, conf.Password)
52	if err != nil {
53		return nil, err
54	}
55
56	return &BrpcSparseRecall{
57		client:      client,
58		modelName:   name,
59		indexName:   conf.IndexName,
60		itemIdField: conf.ItemIdField,
61		recallCount: conf.RecallCount,
62	}, nil
63}

65// GetCandidateItems implements recall.Recall
66func (r *BrpcSparseRecall) GetCandidateItems(user *module.User, context *context.RecommendContext) (ret []*module.Item) {
67	start := time.Now()
68	value := user.GetProperty("video_category")
69	category, ok := value.(string)
70	if !ok {
71    		log.Error(fmt.Sprintf("requestId=%s\tmodule=BrpcSparseRecall\tname=%s\terror=video_category type assertion failed, got %T", context.RecommendId, r.modelName, value))
72    		return
73	}
74
75	response, err := r.client.Search(r.indexName, category)
76	if err != nil {
77		log.Error(fmt.Sprintf("requestId=%s\tmodule=BrpcSparseRecall\tname=%s\terror=opensearch search error(%v), index=%s", context.RecommendId, r.modelName, err, r.indexName))
78		return
79	}
80
81	ret = r.parseResponse(response)
82	if r.recallCount > 0 && len(ret) > r.recallCount {
83		ret = ret[:r.recallCount]
84	}
85
86	log.Info(fmt.Sprintf("requestId=%s\tmodule=BrpcSparseRecall\tname=%s\tcount=%d\tcost=%d", context.RecommendId, r.modelName, len(ret), utils.CostTime(start)))
87	return
88}

90// parseResponse parses the opensearch search response into items,
91// the hit _score becomes the item score, the hit _source the item properties
92func (r *BrpcSparseRecall) parseResponse(response string) []*module.Item {
93	items := make([]*module.Item, 0)
94	if response == "" {
95		return items
96	}
97
98	var result brpcSparseResponse
99	if err := json.Unmarshal([]byte(response), &result); err != nil {
100		log.Error(fmt.Sprintf("module=BrpcSparseRecall\tname=%s\terror=parse search response error(%v)", r.modelName, err))
101		return items
102	}
103
104	for _, hit := range result.Hits.Hits {
105		id := brpcSparseItemID(hit.Source, hit.ID, r.itemIdField)
106		if id == "" {
107			continue
108		}
109		properties := make(map[string]interface{}, len(hit.Source))
110		for k, v := range hit.Source {
111			properties[k] = v
112		}
113		item := module.NewItemWithProperty(id, properties)
114		item.RetrieveId = r.modelName
115		item.Score = hit.Score
116		items = append(items, item)
117	}
118	return items
119}

121func brpcSparseItemID(source map[string]interface{}, hitID string, itemIdField string) string {
122	if source != nil {
123		if v, ok := source[itemIdField]; ok {
124			switch id := v.(type) {
125			case string:
126				return id
127			case float64:
128				return strconv.FormatFloat(id, 'f', -1, 64)
129			}
130		}
131	}
132	return hitID
133}

135// opensearch search response structs 
136type brpcSparseResponse struct { 
137	Hits struct { 
138 		Total struct { 
139 			Value int `json:"value"` 
140 		} `json:"total"` 
141 		Hits []struct { 
142 			ID     string                 `json:"_id"` 
143 			Score  float64                `json:"_score"` 
144 			Source map[string]interface{} `json:"_source"` 
145 		} `json:"hits"` 
146 	} `json:"hits"` 
147 }

149// RegisterBrpcSparseRecall registers a brpc opensearch sparse recall
150// with the given name, then the name can be referenced by SceneConfs RecallNames.
151// Do NOT add the name to RecallConfs, the framework recall.Load does not know
152// this RecallType and would panic on it.
153// The opensearch connection is set by the envs OPENSEARCH_ADDR, OPENSEARCH_USER
154// and OPENSEARCH_PASSWORD, default addr "127.0.0.1:9200".
155// The registration is skipped with a log when indexName is empty, so a cgo build
156// without opensearch data source still starts up.
157func RegisterBrpcSparseRecall(name string, indexName string) error {
158	if indexName == "" {
159		log.Info(fmt.Sprintf("module=BrpcSparseRecall\tmsg=opensearch index empty, skip registering, name=%s", name))
160		return nil
161	}
162	return RegisterBrpcSparseRecallWithConfig(name, BrpcSparseRecallConfig{
163		Addr:        "127.0.0.1:9200",
164		User:        "admin",
165		Password:    "[redacted]",
166		IndexName:   indexName,
167		RecallCount: brpcSparseDefaultQuerySize,
168		ItemIdField: brpcSparseDefaultItemIdField,
169	})
170}

172// RegisterBrpcSparseRecallWithConfig registers a brpc opensearch
173// sparse recall with a custom config
174func RegisterBrpcSparseRecallWithConfig(name string, conf BrpcSparseRecallConfig) error {
175	r, err := NewBrpcSparseRecall(name, conf)
176	if err != nil {
177		return fmt.Errorf("init brpc opensearch sparse recall: %w", err)
178	}
179
180	pairecrecall.RegisterRecall(name, r)
181	log.Info(fmt.Sprintf("module=BrpcSparseRecall\tmsg=brpc opensearch sparse recall registered, name=%s, addr=%s, index=%s", name, conf.Addr, conf.IndexName))
182	return nil
