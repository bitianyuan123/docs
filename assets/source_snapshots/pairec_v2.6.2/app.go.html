36func (app *App) Run() {
37	mode := os.Getenv("RUN_MODE")
38	if mode == "COMMAND" {
39		return
40	}
41
42	addr := fmt.Sprintf("%s:%d", recconf.Config.ListenConf.HttpAddr, recconf.Config.ListenConf.HttpPort)
43
44	if recconf.Config.PrometheusConfig.Enable {
45		config := recconf.Config.PrometheusConfig
46
47		var options []prometheus.PrometheusOption
48
49		if config.Subsystem != "" {
50			options = append(options, prometheus.WithSubsystem(config.Subsystem))
51		}
52		if len(config.ReqDurBuckets) > 0 {
53			options = append(options, prometheus.WithReqDurBuckets(config.ReqDurBuckets))
54		}
55		if len(config.ReqSizeBuckets) > 0 {
56			options = append(options, prometheus.WithReqSzBuckets(config.ReqSizeBuckets))
57		}
58		if len(config.RespSizeBuckets) > 0 {
59			options = append(options, prometheus.WithResSzBuckets(config.RespSizeBuckets))
60		}
61
62		p := prometheus.NewPrometheus(options...)
63
64		if config.PushGatewayURL != "" && config.PushIntervalSecs > 0 {
65			if config.Job == "" {
66				env := recconf.Config.RunMode
67				if os.Getenv("PAIREC_ENVIRONMENT") != "" {
68					env = os.Getenv("PAIREC_ENVIRONMENT")
69				}
70
71				config.Job = env
72			}
73			p.Push(config.PushGatewayURL, config.PushGatewayToken, config.PushIntervalSecs, config.Job)
74		}
75
76		app.Use(p.HandlerFunc)
77	}
78	app.Handlers.ApplyMiddlewares()
79
80	app.Server.Handler = app.Handlers
81	app.Server.Addr = addr
82	app.Server.ReadTimeout = 30 * time.Second
83	app.Server.WriteTimeout = 30 * time.Second
84	app.Server.MaxHeaderBytes = 1 << 20
85
86	wg := sync.WaitGroup{}
87	wg.Add(1)
88	go func() {
89		defer wg.Done()
90		if err := app.Server.ListenAndServe(); err != http.ErrServerClosed {
91			log.Error(fmt.Sprintf("server stop, err=%v", err))
92		}
93	}()
94
95	fmt.Println("server start")
96	wg.Wait()
97	log.Flush()
98}
