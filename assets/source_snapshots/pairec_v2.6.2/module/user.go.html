37func NewUserWithContext(id UID, context *context.RecommendContext) *User {
38	user := NewUser(string(id))
39
40	features := context.GetParameter("features")
41
42	properties := make(map[string]any, 64)
43	if featuresMap, ok := features.(map[string]interface{}); ok {
44		properties = make(map[string]any, len(featuresMap))
45		for k, v := range featuresMap {
46			if strValue, ok := v.(string); ok {
47				if strValue != "" {
48					properties[k] = v
49				}
50			} else {
51				properties[k] = v
52			}
53		}
54	}
55
56	properties["uid"] = string(id)
57	user.Properties = properties
58	return user
59}
60func (u *User) Clone() *User {
61	user := User{
62		Id:                       u.Id,
63		Properties:               make(map[string]interface{}),
64		cacheFeatures:            make(map[string]map[string]any),
65		featureAsyncLoadCh:       make(chan struct{}, 1),
66		featureAsyncLoadChClosed: true,
67	}
68	u.mutex.RLock()
69	defer u.mutex.RUnlock()
70	close(user.featureAsyncLoadCh)
71	for k, v := range u.Properties {
72		user.Properties[k] = v
73	}
74
75	for name, m := range u.cacheFeatures {
76		cloneMap := make(map[string]any, len(m))
77		for k, v := range m {
78			cloneMap[k] = v
79		}
80
81		user.cacheFeatures[name] = cloneMap
82
83	}
84
85	return &user
86}

100func (u *User) FloatProperty(key string) (float64, error) {
101	u.mutex.RLock()
102	defer u.mutex.RUnlock()
103	val, ok := u.Properties[key]
104	if !ok {
105		return float64(0), errors.New("property key not exist")
106	}
107
108	switch value := val.(type) {
109	case float64:
110		return value, nil
111	case int:
112		return float64(value), nil
113	case string:
114		f, err := strconv.ParseFloat(value, 64)
115		return f, err
116	default:
117		return float64(0), errors.New("unsupported type")
118	}
119}

137func (u *User) MakeUserFeatures() (features map[string]interface{}) {
138	u.mutex.RLock()
139	defer u.mutex.RUnlock()
140	features = make(map[string]interface{})
141	for k, v := range u.Properties {
142		if k == "type" {
143			continue
144		}
145		if s, ok := v.(float64); ok {
146			features[k] = s
147			continue
148		}
149		if str, ok := v.(string); ok {
150			if s, err := strconv.ParseFloat(str, 64); err == nil {
151				features[k] = s
152				continue
153			}
154		}
155
156		features[k] = v
157	}
158	return
159}

285func (u *User) DescFeatureAsyncLoadCount(count int32) {
286	u.mutex.Lock()
287	defer u.mutex.Unlock()
288	if atomic.LoadInt32(&u.featureAsyncLoadCount) < 1 {
289		panic("featureAsyncLoadCount not less than 0")
290	}
291	curr := atomic.AddInt32(&u.featureAsyncLoadCount, -1*count)
292	if curr == 0 {
293		if !u.featureAsyncLoadChClosed {
294			close(u.featureAsyncLoadCh)
295			u.featureAsyncLoadChClosed = true
296		}
297	}
298}
