38func NewFeatureRedisDao(config recconf.FeatureDaoConfig) *FeatureRedisDao {
39	dao := &FeatureRedisDao{
40		FeatureBaseDao: NewFeatureBaseDao(&config),
41		redisPrefix:    config.RedisPrefix,
42		redisDelimeter: config.RedisValueDelimeter,
43		redisDataType:  REDIS_DATA_TYPE_STRING,
44		redisFieldType: REDIS_FIELD_TYPE_CSV,
45	}
46	redis, err := redisdb.GetRedis(config.RedisName)
47	if err != nil {
48		log.Error(fmt.Sprintf("error=%v", err))
49		return nil
50	}
51	dao.redis = redis
52	if dao.redisDelimeter == "" {
53		dao.redisDelimeter = ","
54	}
55	if config.RedisDataType != "" {
56		dao.redisDataType = config.RedisDataType
57	}
58	if config.RedisFieldType != "" {
59		dao.redisFieldType = config.RedisFieldType
60	}
61
62	if config.UserSelectFields != "" && config.UserSelectFields != "*" {
63		fields := strings.Split(config.UserSelectFields, ",")
64		for _, f := range fields {
65			dao.userSelectFields = append(dao.userSelectFields, f)
66		}
67	}
68	if config.ItemSelectFields != "" && config.ItemSelectFields != "*" {
69		fields := strings.Split(config.ItemSelectFields, ",")
70		for _, f := range fields {
71			dao.itemSelectFields = append(dao.itemSelectFields, f)
72		}
73	}
74	return dao
75}

84func (d *FeatureRedisDao) userFeatureFetch(user *User, context *context.RecommendContext) {
85	comms := strings.Split(d.featureKey, ":")
86	if len(comms) < 2 {
87		log.Error(fmt.Sprintf("requestId=%s\tuid=%s\terror=featureKey error(%s)", context.RecommendId, user.Id, d.featureKey))
88		return
89	}
90
91	key := user.StringProperty(comms[1])
92	if key == "" {
93		log.Error(fmt.Sprintf("requestId=%s\tuid=%s\terror=property not found(%s)", context.RecommendId, user.Id, comms[1]))
94		return
95	}
96
97	key = d.redisPrefix + key
98
99	// hit user cache
100	if d.cache != nil {
101		if cacheValue, ok := d.cache.GetIfPresent(key); ok {
102			if d.cacheFeaturesName != "" {
103				user.AddCacheFeatures(d.cacheFeaturesName, cacheValue.(map[string]interface{}))
104			} else {
105				user.AddProperties(cacheValue.(map[string]interface{}))
106			}
107			if context.Debug {
108				log.Info(fmt.Sprintf("requestId=%s\tmodule=FeatureHologresDao\tmsg=hit cache(%s)", context.RecommendId, key))
109			}
110			return
111		}
112	}
113
114	conn := d.redis.Get()
115	defer conn.Close()
116	if d.redisDataType == REDIS_DATA_TYPE_STRING {
117		d.userFeatureFetchByString(user, context, conn, key)
118	} else if d.redisDataType == REDIS_DATA_TYPE_HASH {
119		d.userFeatureFetchByHash(user, context, conn, key)
120	}
121}
122func (d *FeatureRedisDao) userFeatureFetchByString(user *User, context *context.RecommendContext, conn redis.Conn, key string) error {
123	str, err := redis.String(conn.Do("GET", key))
124	if err != nil {
125		if errors.Is(err, redis.ErrNil) {
126			log.Info(fmt.Sprintf("requestId=%s\tuid=%s\tmsg=user feature empty", context.RecommendId, user.Id))
127		} else {
128			log.Error(fmt.Sprintf("requestId=%s\tuid=%s\terror=get user feature error(%v)", context.RecommendId, user.Id, err))
129		}
130		return err
131	}
132	properties := make(map[string]interface{})
133	if d.redisFieldType == REDIS_FIELD_TYPE_CSV {
134		keyParis := strings.Split(str, d.redisDelimeter)
135		if len(keyParis) == 0 {
136			return nil
137		}
138		for _, pair := range keyParis {
139			idx := strings.Index(pair, ":")
140			if idx > 0 {
141				name := pair[:idx]
142				value := pair[idx+1:]
143				properties[name] = value
144			}
145		}
146	} else if d.redisFieldType == REDIS_FIELD_TYPE_JSON {
147		err := json.Unmarshal([]byte(str), &properties)
148		if err != nil {
149			log.Error(fmt.Sprintf("requestId=%s\tuid=%s\terror=get user feature error(%v)", context.RecommendId, user.Id, err))
150			return err
151		}
152	}
153	if d.cacheFeaturesName != "" {
154		user.AddCacheFeatures(d.cacheFeaturesName, properties)
155	} else {
156		user.AddProperties(properties)
157	}
158	if d.cache != nil {
159		d.cache.Put(key, properties)
160	}
161
162	return nil
163}
164func (d *FeatureRedisDao) userFeatureFetchByHash(user *User, context *context.RecommendContext, conn redis.Conn, key string) error {
165	properties := make(map[string]interface{})
166	if len(d.userSelectFields) == 0 {
167		// get all fields
168		strs, err := redis.Strings(conn.Do("HGETALL", key))
169		if err != nil {
170			log.Error(fmt.Sprintf("requestId=%s\tuid=%s\terror=get user feature error(%v)", context.RecommendId, user.Id, err))
171			return err
172		}
173		for i := 0; i < len(strs); i += 2 {
174			properties[strs[i]] = strs[i+1]
175		}
176
177	} else {
178		var params []interface{}
179		params = append(params, key)
180		params = append(params, d.userSelectFields...)
181		strs, err := redis.Strings(conn.Do("HMGET", params...))
182		if err != nil {
183			log.Error(fmt.Sprintf("requestId=%s\tuid=%s\terror=get user feature error(%v)", context.RecommendId, user.Id, err))
184			return err
185		}
186		for i, val := range strs {
187			if val == "" {
188				continue
189			}
190			properties[d.userSelectFields[i].(string)] = val
191		}
192
193	}
194
195	if d.cacheFeaturesName != "" {
196		user.AddCacheFeatures(d.cacheFeaturesName, properties)
197	} else {
198		user.AddProperties(properties)
199	}
200	if d.cache != nil {
201		d.cache.Put(key, properties)
202	}
203	return nil
204}

206func (d *FeatureRedisDao) itemsFeatureFetch(items []*Item, context *context.RecommendContext) {
207	fk := d.featureKey
208	if fk != "item:id" {
209		comms := strings.Split(d.featureKey, ":")
210		if len(comms) < 2 {
211			log.Error(fmt.Sprintf("requestId=%s\tevent=itemsFeatureFetch\terror=featureKey error(%s)", context.RecommendId, d.featureKey))
212			return
213		}
214
215		fk = comms[1]
216	}
217
218	cpuCount := utils.MaxInt(int(len(items)/100), 1)
219	maps := make(map[int][]*Item)
220	for i, item := range items {
221		maps[i%cpuCount] = append(maps[i%cpuCount], item)
222	}
223
224	requestCh := make(chan []*Item, cpuCount)
225	defer close(requestCh)
226
227	for _, itemlist := range maps {
228		requestCh <- itemlist
229	}
230
231	var wg sync.WaitGroup
232	for i := 0; i < cpuCount; i++ {
233		wg.Add(1)
234		go func() {
235			defer wg.Done()
236			select {
237			case itemlist := <-requestCh:
238				var keys []interface{}
239				for _, item := range itemlist {
240					var key string
241					if fk == "item:id" {
242						key = string(item.Id)
243					} else {
244						key = item.StringProperty(fk)
245					}
246					key = d.redisPrefix + key
247					if d.cache != nil {
248						if cacheValue, ok := d.cache.GetIfPresent(key); ok {
249							item.AddProperties(cacheValue.(map[string]interface{}))
250							if context.Debug {
251								item.AddProperty("__debug_cache_hit__", true)
252							}
253							continue
254						}
255					}
256
257					keys = append(keys, key)
258				}
259				if len(keys) == 0 {
260					return
261				}
262
263				conn := d.redis.Get()
264				defer conn.Close()
265				if d.redisDataType == REDIS_DATA_TYPE_STRING {
266					d.itemFeatureFetchByString(itemlist, context, conn, keys)
267				} else if d.redisDataType == REDIS_DATA_TYPE_HASH {
268					d.itemFeatureFetchByHash(itemlist, context, conn, keys)
269				}
270			default:
271			}
272		}()
273	}
274	wg.Wait()
275}

277func (d *FeatureRedisDao) itemFeatureFetchByString(items []*Item, context *context.RecommendContext, conn redis.Conn, keys []interface{}) error {
278	values, err := redis.Strings(conn.Do("MGET", keys...))
279	if err != nil {
280		log.Error(fmt.Sprintf("requestId=%s\tmodule=FeatureRedisDao\terror=%v", context.RecommendId, err))
281		return err
282	}
283
284	for i, str := range values {
285		if str == "" {
286			continue
287		}
288		item := items[i]
289		properties := make(map[string]interface{})
290
291		if d.redisFieldType == REDIS_FIELD_TYPE_CSV {
292			keyParis := strings.Split(str, d.redisDelimeter)
293			if len(keyParis) == 0 {
294				continue
295			}
296
297			for _, pair := range keyParis {
298				keyValues := strings.Split(pair, ":")
299				if len(keyValues) == 2 {
300					name := keyValues[0]
301					val := keyValues[1]
302					f, err := strconv.ParseFloat(val, 64)
303					if err == nil {
304						properties[name] = f
305					} else {
306						properties[name] = val
307					}
308				}
309			}
310
311		} else if d.redisFieldType == REDIS_FIELD_TYPE_JSON {
312			if err := json.Unmarshal([]byte(str), &properties); err != nil {
313				continue
314			}
315		}
316		item.AddProperties(properties)
317		if d.cache != nil {
318			d.cache.Put(keys[i], properties)
319		}
320	}
321
322	return nil
323}

325func (d *FeatureRedisDao) itemFeatureFetchByHash(items []*Item, context *context.RecommendContext, conn redis.Conn, keys []interface{}) error {
326	if len(d.itemSelectFields) == 0 {
327		// get all fields
328		for _, key := range keys {
329			conn.Send("HGETALL", key)
330		}
331		conn.Flush()
332
333		for i, key := range keys {
334			strs, err := redis.Strings(conn.Receive())
335			if err != nil {
336				log.Error(fmt.Sprintf("requestId=%s\tmodule=FeatureRedisDao\terror=%v", context.RecommendId, err))
337				continue
338			}
339			item := items[i]
340			properties := make(map[string]interface{})
341			for i := 0; i < len(strs); i += 2 {
342				properties[strs[i]] = strs[i+1]
343			}
344			item.AddProperties(properties)
345			if d.cache != nil {
346				d.cache.Put(key, properties)
347			}
348
349		}
350	} else {
351		for _, key := range keys {
352			var params []interface{}
353			params = append(params, key)
354			params = append(params, d.itemSelectFields...)
355			conn.Send("HMGET", params...)
356		}
357		conn.Flush()
358		for i, key := range keys {
359			strs, err := redis.Strings(conn.Receive())
360			if err != nil {
361				log.Error(fmt.Sprintf("requestId=%s\tmodule=FeatureRedisDao\terror=%v", context.RecommendId, err))
362				continue
363			}
364			item := items[i]
365			properties := make(map[string]interface{})
366			for k, val := range strs {
367				if val == "" {
368					continue
369				}
370				properties[d.itemSelectFields[k].(string)] = val
371			}
372			item.AddProperties(properties)
373			if d.cache != nil {
374				d.cache.Put(key, properties)
375			}
376
377		}
378
379	}
380
381	return nil
