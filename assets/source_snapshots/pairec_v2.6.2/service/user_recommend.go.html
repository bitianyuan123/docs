45func (r *UserRecommendService) Recommend(context *context.RecommendContext) []*module.Item {
46	start := time.Now()
47
48	var scene, expId string
49	if context.ExperimentResult != nil {
50		scene = context.ExperimentResult.SceneName
51		//expId = context.ExperimentResult.ExpId
52	} else {
53		scene, _ = context.Param.GetParameter("scene").(string)
54	}
55
56	userId := r.GetUID(context)
57	user := module.NewUserWithContext(userId, context)
58
59	//loadFeatureStart := time.Now()
60
61	// load user features
62	r.userFeatureService.LoadUserFeatures(user, context)
63
64	//if metrics.Enabled() {
65	//metrics.LoadFeatureDurSecs.WithLabelValues(scene, expId, "before_recall").Observe(time.Since(loadFeatureStart).Seconds())
66	//}
67
68	debugService := debug.NewDebugService(user, context)
69
70	var pipelineItems []*module.Item
71
72	var wg sync.WaitGroup
73	wg.Add(1)
74	go func() {
75		defer wg.Done()
76		pipelineItems = pipeline.Recommend(user, context, debugService)
77	}()
78
79	recallStart := time.Now()
80
81	items := r.recallService.GetItems(user, context)
82
83	if metrics.Enabled() {
84		metrics.RecallDurSecs.WithLabelValues(scene, expId).Observe(time.Since(recallStart).Seconds())
85		metrics.RecallCountTotal.WithLabelValues(scene).Add(float64(len(items)))
86
87		recallCountMap := map[string]int{}
88		for _, item := range items {
89			recallCountMap[item.RetrieveId]++
90		}
91
92		for src, count := range recallCountMap {
93			//metrics.RecallItemsPercentage.WithLabelValues(src).Set(float64(count) / float64(len(items)))
94			metrics.RecallCount.WithLabelValues(scene, src).Add(float64(count))
95		}
96	}
97
98	debugService.WriteRecallLog(user, items, context)
99
100	filterStart := time.Now()
101
102	// filter
103	items = r.Filter(user, items, context)
104
105	if metrics.Enabled() {
106		metrics.FilterDurSecs.WithLabelValues(scene, expId).Observe(time.Since(filterStart).Seconds())
107	}
108
109	debugService.WriteFilterLog(user, items, context)
110
111	generalRankStart := time.Now()
112
113	// general rank
114	items = r.generalRankService.Rank(user, items, context)
115
116	if metrics.Enabled() {
117		metrics.GeneralRankDurSecs.WithLabelValues(scene, expId).Observe(time.Since(generalRankStart).Seconds())
118	}
119
120	debugService.WriteGeneralLog(user, items, context)
121
122	//loadFeatureStart = time.Now()
123
124	// load user or item features
125	// can load data from datasource(holo, ots, redis)
126	// after load data, use feature engine to create or modify features
127	items = r.featureService.LoadFeatures(user, items, context)
128
129	//if metrics.Enabled() {
130	//metrics.LoadFeatureDurSecs.WithLabelValues(scene, expId, "before_rank").Observe(time.Since(loadFeatureStart).Seconds())
131	//}
132
133	rankStart := time.Now()
134
135	r.rankService.Rank(user, items, context)
136
137	if metrics.Enabled() {
138		metrics.RankDurSecs.WithLabelValues(scene, expId).Observe(time.Since(rankStart).Seconds())
139	}
140
141	wg.Wait()
142	items = r.mergePipelineItems(items, pipelineItems)
143
144	debugService.WriteRankLog(user, items, context)
145
146	sortStart := time.Now()
147
148	// sort items
149	items = r.Sort(user, items, context)
150
151	if metrics.Enabled() {
152		metrics.SortDurSecs.WithLabelValues(scene, expId).Observe(time.Since(sortStart).Seconds())
153	}
154	debugService.WriteSortLog(user, items, context)
155
156	size := context.Size
157	if size > len(items) {
158		size = len(items)
159		log.Warning(fmt.Sprintf("requestId=%s\tmodule=recommend\tevent=filter\tuid=%s\tmsg=length of items less than size\tcount=%d", context.RecommendId, userId, len(items)))
160
161		if metrics.Enabled() {
162			metrics.SizeNotEnoughTotal.WithLabelValues(scene, expId).Inc()
163		}
164	}
165
166	items = items[:size]
167	go feature_log.FeatureLog(user, items, context)
168	go r.featureConsistencyJobService.LogSampleResult(user, items, context)
169	debugService.WriteRecommendLog(user, items, context)
170	// asynchronous clean hook func
171	for _, hf := range hook.RecommendCleanHooks {
172		go hf(context, user, items)
173	}
174
175	if metrics.Enabled() {
176		metrics.RecTotal.WithLabelValues(scene, expId).Inc()
177		metrics.RecDurSecs.WithLabelValues(scene, expId).Observe(time.Since(start).Seconds())
178	}
179
180	return items
181}

183func (r *UserRecommendService) mergePipelineItems(items []*module.Item, pipelineItems []*module.Item) []*module.Item {
184	itemMap := make(map[module.ItemId]*module.Item, len(items))
185	for _, item := range items {
186		itemMap[item.Id] = item
187	}
188	// need to merge item properties of different pipelines
189	for _, item := range pipelineItems {
190		if exist, ok := itemMap[item.Id]; ok {
191			exist.AddProperties(item.GetProperties())
192		} else {
193			itemMap[item.Id] = item
194			items = append(items, item)
195		}
196	}
197	return items
198}

200func (r *UserRecommendService) TryRecommendWithFallback(context *context.RecommendContext) []*module.Item {
201	start := time.Now()
202	scene, _ := context.Param.GetParameter("scene").(string)
203
204	f := fallback.DefaultFallbackService().GetFallback(scene)
205	if f == nil {
206		return r.Recommend(context)
207	}
208
209	fallbackTimer := f.GetTimer()
210
211	tryResult := make(chan []*module.Item, 1)
212
213	go func() {
214		tryResult <- r.Recommend(context)
215	}()
216
217	select {
218	case <-fallbackTimer.C:
219		f.PutTimer(fallbackTimer)
220		fallbackResult := f.Recommend(context)
221		log.Warning(fmt.Sprintf("requestId=%s\tmodule=recommend\tevent=fallback\tcause=timeout\tcost=%d", context.RecommendId, utils.CostTime(start)))
222		return fallbackResult
223	case ret := <-tryResult:
224		f.PutTimer(fallbackTimer)
225
226		if f.CompleteItemsIfNeed() && len(ret) < context.Size {
227			originRetMap := make(map[module.ItemId]bool)
228			for _, item := range ret {
229				originRetMap[item.Id] = true
230			}
231
232			fallbackResult := f.Recommend(context)
233			for i := 0; i < len(fallbackResult); i++ {
234				fallbackItem := fallbackResult[i]
235
236				if !originRetMap[fallbackItem.Id] { // must not appear in origin result
237					ret = append(ret, fallbackItem)
238
239					if len(ret) == context.Size {
240						break
241					}
242				}
243			}
244
245			log.Warning(fmt.Sprintf("requestId=%s\tmodule=recommend\tevent=fallback\tcause=itemNotEnough\tcost=%d", context.RecommendId, utils.CostTime(start)))
246			return ret
247		} else {
248			return ret
249		}
250	}
