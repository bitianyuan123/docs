127func (t *FileOutput) getCurrFile() (*os.File, error) {
128	files, err := t.getSortedFiles()
129	if err != nil {
130		return nil, err
131	}
132	var f *os.File
133	if len(files) == 0 {
134		f, err = t.newFile()
135		if err != nil {
136			return nil, err
137		}
138	} else {
139		f, err = os.OpenFile(filepath.Join(t.path, files[len(files)-1].Name()), os.O_WRONLY|os.O_CREATE|os.O_APPEND, 0644)
140		if err != nil {
141			return nil, err
142		}
143		fileInfo, err := f.Stat()
144		if err != nil {
145			return nil, err
146		}
147		if fileInfo.Size() > 1024*1024*1024 {
148			f.Close()
149
150			if len(files) >= t.maxFileNum {
151				for i := 0; i < len(files)-t.maxFileNum+1; i++ {
152					err := os.Remove(filepath.Join(t.path, files[i].Name()))
153					if err != nil {
154						return nil, err
155					}
156				}
157			}
158
159			f, err = t.newFile()
160			if err != nil {
161				return nil, err
162			}
163		} else {
164			if len(files) > t.maxFileNum {
165				for i := 0; i < len(files)-t.maxFileNum; i++ {
166					err := os.Remove(filepath.Join(t.path, files[i].Name()))
167					if err != nil {
168						return nil, err
169					}
170				}
171			}
172		}
173	}
174	return f, nil
175}

186func (t *FileOutput) getSortedFiles() ([]os.FileInfo, error) {
187	files := []os.FileInfo{}
188	err := filepath.Walk(t.path, func(path string, info os.FileInfo, err error) error {
189		if info.IsDir() && path != t.path {
190			return filepath.SkipDir
191		}
192		if !info.IsDir() {
193			if strings.HasPrefix(info.Name(), "pairec.debug.") && strings.HasSuffix(info.Name(), ".log") {
194				files = append(files, info)
195			}
196		}
197		return nil
198	})
199	if err != nil {
200		return nil, err
201	}
202	sort.Slice(files, func(i, j int) bool {
203		return t.getFileTimestamp(files[i]) < t.getFileTimestamp(files[j])
204	})
205	return files, nil
206}

209	timestampStr := strings.TrimSuffix(strings.TrimPrefix(file.Name(), "pairec.debug."), ".log")
210	timestamp, _ := strconv.ParseInt(timestampStr, 10, 64)
211	return timestamp

226func NewDebugService(user *module.User, context *context.RecommendContext) *DebugService {
227	service := DebugService{
228		logFlag: false,
229	}
230	var debugConfig recconf.DebugConfig
231
232	found := false
233	if context.ExperimentResult != nil {
234		data := context.ExperimentResult.GetExperimentParams().GetString("debugConfig", "")
235		if data != "" {
236			if err := json.Unmarshal([]byte(data), &debugConfig); err == nil {
237				found = true
238			}
239		}
240	}
241
242	if !found {
243		scene := context.GetParameter("scene").(string)
244		if config, ok := context.Config.DebugConfs[scene]; ok {
245			found = true
246			debugConfig = config
247		}
248	}
249
250	if !found {
251		// not found debug config, not set logflag
252		return &service
253	}
254
255	if debugConfig.Rate == 100 {
256		service.logFlag = true
257	} else {
258		if len(debugConfig.DebugUsers) > 0 {
259			for _, uid := range debugConfig.DebugUsers {
260				if uid == string(user.Id) {
261					service.logFlag = true
262					break
263				}
264			}
265		}
266
267		if !service.logFlag {
268			if rand.Intn(100) < debugConfig.Rate {
269				service.logFlag = true
270			}
271		}
272	}
273
274	if service.logFlag {
275		service.requestTime = time.Now().Unix()
276		service.init(&debugConfig)
277	}
278	return &service
279}

281func (d *DebugService) init(config *recconf.DebugConfig) {
282
283	switch config.OutputType {
284	case "console":
285		d.logOutputer = new(ConsoleOutput)
286
287	case "datahub":
288		d.logOutputer = NewDatahubOutput(config)
289
290	case "file":
291		d.logOutputer = NewFileOutput(config)
292
293	case "kafka":
294		d.logOutputer = NewKafkaOutput(config)
295
296	default:
297		d.logOutputer = new(EmptyOutput)
298	}
