60func (s *UserFeatureService) GetLoadFeaturesStageNames(context *context.RecommendContext) (stageNames []string) {
61	// if not have experiment, direct return scene name
62	if context.ExperimentResult == nil {
63		sceneName := context.GetParameter("scene").(string)
64		stageNames = append(stageNames, sceneName)
65		return
66	}
67
68	useStage := false
69	stageFlage := context.ExperimentResult.GetExperimentParams().Get("user_features.multistage.on", false)
70	useStage = utils.ToBool(stageFlage, false)
71
72	// use mutlti stage to load features
73	if useStage {
74		if name := context.ExperimentResult.GetExperimentParams().GetString("user_features.stage.base.scene.name", ""); name != "" {
75			stageNames = append(stageNames, name)
76		}
77
78		if name := context.ExperimentResult.GetExperimentParams().GetString("user_features.stage.recall.scene.name", ""); name != "" {
79			stageNames = append(stageNames, name)
80		}
81
82		if name := context.ExperimentResult.GetExperimentParams().GetString("user_features.stage.generalrank.scene.name", ""); name != "" {
83			stageNames = append(stageNames, name)
84		}
85
86		if name := context.ExperimentResult.GetExperimentParams().GetString("user_features.stage.rank.scene.name", ""); name != "" {
87			stageNames = append(stageNames, name)
88		}
89		if len(stageNames) > 0 {
90			return
91		}
92
93	}
94
95	sceneName := context.ExperimentResult.GetExperimentParams().GetString("user_features.scene.name", "")
96	if sceneName == "" {
97		sceneName = context.GetParameter("scene").(string)
98	}
99
100	stageNames = append(stageNames, sceneName)
101
102	return
103}

106func (s *UserFeatureService) LoadUserFeatures(user *module.User, context *context.RecommendContext) {
107	start := time.Now()
108	stageNames := s.GetLoadFeaturesStageNames(context)
109
110	for _, stageName := range stageNames {
111		if features, ok := s.FeatureSceneMap[stageName]; ok {
112			// for close featureAsyncLoadCh
113			user.IncrementFeatureAsyncLoadCount(1)
114			defer user.DescFeatureAsyncLoadCount(1)
115
116			var wg sync.WaitGroup
117			featureFunc, featureFuncExist := s.FeatureFuncMap[stageName]
118			if featureFuncExist {
119				var wg2 sync.WaitGroup
120				for _, fea := range features {
121					if s.FeatureAsyncLoadMap[fea] {
122						user.IncrementFeatureAsyncLoadCount(1)
123						wg2.Add(1)
124						go func(fea *Feature) {
125							defer wg2.Done()
126							defer user.DescFeatureAsyncLoadCount(1)
127							fea.LoadFeatures(user, nil, context)
128						}(fea)
129
130					} else {
131						wg.Add(1)
132						wg2.Add(1)
133						go func(fea *Feature) {
134							defer wg2.Done()
135							defer wg.Done()
136							fea.LoadFeatures(user, nil, context)
137						}(fea)
138					}
139				}
140				// custom user feature load
141				if loadFeatureFunc, exist := s.LoadFeatureFuncMap[stageName]; exist {
142					wg.Add(1)
143					wg2.Add(1)
144					go func() {
145						defer wg2.Done()
146						defer wg.Done()
147						loadFeatureFunc(user, nil, context)
148					}()
149				}
150
151				go func() {
152					wg2.Wait()
153					featureFunc(user, nil, context)
154				}()
155			} else {
156				for _, fea := range features {
157					if s.FeatureAsyncLoadMap[fea] {
158						user.IncrementFeatureAsyncLoadCount(1)
159						go func(fea *Feature) {
160							defer user.DescFeatureAsyncLoadCount(1)
161							fea.LoadFeatures(user, nil, context)
162						}(fea)
163
164					} else {
165						wg.Add(1)
166						go func(fea *Feature) {
167							defer wg.Done()
168							fea.LoadFeatures(user, nil, context)
169						}(fea)
170					}
171				}
172				if loadFeatureFunc, exist := s.LoadFeatureFuncMap[stageName]; exist {
173					wg.Add(1)
174					go func() {
175						defer wg.Done()
176						loadFeatureFunc(user, nil, context)
177					}()
178				}
179
180			}
181
182			wg.Wait()
183			log.Info(fmt.Sprintf("requestId=%s\tmodule=UserLoadFeatures\tname=%s\tcost=%d", context.RecommendId, stageName, utils.CostTime(start)))
184		}
185
186	}
187
188}

190func (s *UserFeatureService) LoadUserFeaturesForCallback(user *module.User, context *context.RecommendContext) {
191	start := time.Now()
192
193	stageNames := s.GetLoadFeaturesStageNames(context)
194	for _, stageName := range stageNames {
195		if features, ok := s.FeatureSceneMap[stageName]; ok {
196			// for close featureAsyncLoadCh
197			user.IncrementFeatureAsyncLoadCount(1)
198			defer user.DescFeatureAsyncLoadCount(1)
199
200			var syncFeatures []*Feature
201			for _, feature := range features {
202				if s.FeatureAsyncLoadMap[feature] {
203					user.IncrementFeatureAsyncLoadCount(1)
204					go func(fea *Feature) {
205						defer user.DescFeatureAsyncLoadCount(1)
206						fea.LoadFeatures(user, nil, context)
207					}(feature)
208				} else {
209					syncFeatures = append(syncFeatures, feature)
210				}
211			}
212
213			go func() {
214				if len(syncFeatures) > 0 {
215					user.IncrementFeatureAsyncLoadCount(int32(len(syncFeatures)))
216					defer user.DescFeatureAsyncLoadCount(int32(len(syncFeatures)))
217					for _, feature := range syncFeatures {
218						feature.LoadFeatures(user, nil, context)
219					}
220				}
221
222			}()
223
224			log.Info(fmt.Sprintf("requestId=%s\tmodule=UserLoadFeatures\tname=%s\tcost=%d", context.RecommendId, stageName, utils.CostTime(start)))
225		}
226
227	}
