76// LoadFeatures load user or item feature use feature.Feature
77func (s *FeatureService) LoadFeatures(user *module.User, items []*module.Item, context *context.RecommendContext) []*module.Item {
78	start := time.Now()
79	var sceneName string
80	if context.ExperimentResult != nil {
81		sceneName = context.ExperimentResult.GetExperimentParams().GetString("features.scene.name", "")
82	}
83	if sceneName == "" {
84		sceneName = context.GetParameter("scene").(string)
85	}
86
87	if features, ok := s.FeatureSceneMap[sceneName]; ok {
88		async := s.FeatureSceneAsyncMap[sceneName]
89		if async {
90			var wg sync.WaitGroup
91			for _, fea := range features {
92				wg.Add(1)
93				go func(fea *Feature) {
94					defer wg.Done()
95					fea.LoadFeatures(user, items, context)
96				}(fea)
97			}
98			if loadFeatureFunc, exist := s.LoadFeatureFuncMap[sceneName]; exist {
99				wg.Add(1)
100				go func() {
101					defer wg.Done()
102					loadFeatureFunc(user, items, context)
103				}()
104			}
105
106			wg.Wait()
107		} else {
108			for _, feature := range features {
109				feature.LoadFeatures(user, items, context)
110			}
111
112			if loadFeatureFunc, exist := s.LoadFeatureFuncMap[sceneName]; exist {
113				loadFeatureFunc(user, items, context)
114			}
115
116		}
117
118		featureFunc, ok := s.FeatureFuncMap[sceneName]
119		if ok {
120			items = featureFunc(user, items, context)
121		}
122	}
123
124	if metrics.Enabled() {
125		metrics.LoadFeatureDurSecs.WithLabelValues(sceneName, "", "before_rank").Observe(time.Since(start).Seconds())
126	}
127
128	log.Info(fmt.Sprintf("requestId=%s\tmodule=LoadFeatures\tscene=%s\tcost=%d", context.RecommendId, sceneName, utils.CostTime(start)))
129
130	return items
131}

195func LoadFeatureConfig(config *recconf.RecommendConfig) {
196	for name, sceneConf := range config.FeatureConfs {
197		if _, ok := featureService.FeatureSceneMap[name]; ok {
198			if signOfFeatureLoadConfs(sceneConf.FeatureLoadConfs) == featureService.FeatureSceneSigns[name] {
199				continue
200			}
201		}
202
203		var features []*Feature
204		for _, conf := range sceneConf.FeatureLoadConfs {
205			f := LoadWithConfig(conf)
206			features = append(features, f)
207		}
208
209		featureService.FeatureSceneAsyncMap[name] = false
210		if sceneConf.AsynLoadFeature {
211			featureService.FeatureSceneAsyncMap[name] = true
212		}
213		featureService.FeatureSceneMap[name] = features
214		featureService.FeatureSceneSigns[name] = signOfFeatureLoadConfs(sceneConf.FeatureLoadConfs)
215	}
216}

218func signOfFeatureLoadConfs(confs []recconf.FeatureLoadConfig) string {
219	var signs string
220	for _, conf := range confs {
221		sign, _ := json.Marshal(conf)
222
223		signs += string(sign)
224	}
225
226	return utils.Md5(signs)
