9// Feature is a type for load user or item feature
10type Feature struct {
11	// featureDao use for load feature from specific dao, example:redis or hologres
12	featureDao module.FeatureDao
13	// featureTrans use for transform feature, specific operator by FeatureOp
14	featureTrans []FeatureTrans
15}

17func LoadWithConfig(config recconf.FeatureLoadConfig) *Feature {
18	f := &Feature{
19		featureDao:   module.NewFeatureDao(config.FeatureDaoConf),
20		featureTrans: make([]FeatureTrans, 0),
21	}
22
23	for _, conf := range config.Features {
24		source := conf.FeatureSource
25		if conf.FeatureValue != "" {
26			source = conf.FeatureValue
27		}
28		ft := NewFeatureTrans(conf.FeatureType, conf.FeatureName, conf.FeatureStore, source, conf.RemoveFeatureSource, conf.Normalizer, conf.Expression)
29		f.featureTrans = append(f.featureTrans, ft)
30	}
31
32	return f
33}
34
35func (f *Feature) LoadFeatures(user *module.User, items []*module.Item, context *context.RecommendContext) {
36	f.featureDao.FeatureFetch(user, items, context)
37	for _, trans := range f.featureTrans {
38		trans.FeatureTran(user, items, context)
39	}
40}

42const (
43	SOURCE_USER = "user"
44	SOURCE_ITEM = "item"
45)
46
47type FeatureTrans interface {
48	FeatureTran(user *module.User, items []*module.Item, context *context.RecommendContext)
49}
50
51type featureTrans struct {
52	// featureType         string
53	featureOp           FeatureOp
54	featureName         string
55	featureSource       string
56	removeFeatureSource bool   // delete feature source
57	featureStore        string // user or item
58	normalizer          Normalizer
59}
60
61func NewFeatureTrans(t, name, store string, source string, remove bool, normalizer, expression string) FeatureTrans {
62	return &featureTrans{
63		// featureType: t,
64		featureOp:           NewFeatureOp(t),
65		featureName:         name,
66		featureStore:        store,
67		featureSource:       source,
68		removeFeatureSource: remove,
69		normalizer:          NewNormalizer(normalizer, expression),
70	}
71}
72
73func (f *featureTrans) FeatureTran(user *module.User, items []*module.Item, context *context.RecommendContext) {
74	if f.featureStore == SOURCE_ITEM {
75		for _, item := range items {
76			f.featureOp.ItemTransOp(f.featureName, f.featureSource, f.removeFeatureSource, f.normalizer, user, item, context)
77		}
78
79	} else {
80		f.featureOp.UserTransOp(f.featureName, f.featureSource, f.removeFeatureSource, f.normalizer, user, context)
81	}
82}
