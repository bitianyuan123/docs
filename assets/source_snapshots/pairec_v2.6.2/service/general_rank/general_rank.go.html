129// GetGeneralRank return the GeneralRank by the context
130// If not found, return nil
131func (r *GeneralRankService) GetGeneralRankByContext(context *context.RecommendContext) *GeneralRank {
132	scene := context.GetParameter("scene").(string)
133	// find rank config
134	var rankConfig recconf.GeneralRankConfig
135	found := false
136	if context.ExperimentResult != nil {
137		rankconf := context.ExperimentResult.GetExperimentParams().Get("generalRankConf", "")
138		if rankconf != "" {
139			d, _ := json.Marshal(rankconf)
140			if err := json.Unmarshal(d, &rankConfig); err == nil {
141				found = true
142			}
143		}
144	}
145
146	if !found {
147		if rankConfigs, ok := recconf.Config.GeneralRankConfs[scene]; ok {
148			rankConfig = rankConfigs
149			found = true
150		}
151	}
152
153	// not find any generalRankconf, return  nil
154	if !found {
155		return nil
156	}
157	d, _ := json.Marshal(rankConfig)
158	id := scene + "#" + utils.Md5(string(d))
159	if generalRank, ok := r.generalRanks[id]; ok {
160		return generalRank
161	}
162
163	generalRank, err := NewGeneralRankWithConfig(scene, rankConfig)
164	if err != nil {
165		log.Error(fmt.Sprintf("create generalRank error, err:%v", err))
166		return nil
167	}
168
169	r.generalRanks[id] = generalRank
170
171	return generalRank
172}

216func (r *GeneralRankService) Rank(user *module.User, items []*module.Item, context *context.RecommendContext) (ret []*module.Item) {
217	start := time.Now()
218
219	generalRank := r.GetGeneralRankByContext(context)
220	if generalRank == nil {
221		return items
222	}
223
224	var (
225		coldStartItems []*module.Item
226		rankItems      []*module.Item
227
228		coldStartRankResult []*module.Item
229		generalRankResult   []*module.Item
230	)
231
232	coldGeneralRank := r.GetColdStartGeneralRankByContext(context)
233	if coldGeneralRank == nil {
234		rankItems = items
235	} else {
236		for _, item := range items {
237			if coldGeneralRank.Filter(user, item, context) {
238				coldStartItems = append(coldStartItems, item)
239			} else {
240				rankItems = append(rankItems, item)
241			}
242		}
243	}
244
245	var wg sync.WaitGroup
246	if coldGeneralRank != nil {
247		wg.Add(1)
248		go func() {
249			defer wg.Done()
250			coldStartRankResult = coldGeneralRank.Rank(user, coldStartItems, context)
251		}()
252	}
253
254	generalRankResult = generalRank.Rank(user, rankItems, context)
255
256	wg.Wait()
257
258	ret = append(ret, generalRankResult...)
259	ret = append(ret, coldStartRankResult...)
260
261	log.Info(fmt.Sprintf("requestId=%s\tmodule=GeneralRank\tcount=%d\tcost=%d", context.RecommendId, len(ret), utils.CostTime(start)))
262	return
