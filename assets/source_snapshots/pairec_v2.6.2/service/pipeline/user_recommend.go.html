42func (r *UserRecommendService) Recommend(user *module.User, context *context.RecommendContext, debugService *debug.DebugService) []*module.Item {
43
44	start := time.Now()
45	var items []*module.Item
46	items = r.recallService.GetItems(user, context)
47	if len(items) == 0 {
48		log.Info(fmt.Sprintf("requestId=%s\tmodule=pipeline\tpipeline=%s\tcount=0", context.RecommendId, r.pipelineName))
49		return items
50	}
51
52	debugService.WriteRecallLog(user, items, context)
53
54	// filter
55	items = r.Filter(user, items, context)
56
57	debugService.WriteFilterLog(user, items, context)
58
59	// general rank
60	items = r.generalRankService.Rank(user, items, context)
61
62	debugService.WriteGeneralLog(user, items, context)
63
64	// load user or item features
65	// can load data from datasource(holo, ots, redis)
66	// after load data, use feature engine to create or modify features
67	items = r.featureService.LoadFeatures(user, items, context)
68
69	r.rankService.Rank(user, items, context)
70
71	r.coldStartRankService.Rank(user, items, context)
72
73	items = r.Sort(user, items, context)
74	log.Info(fmt.Sprintf("requestId=%s\tmodule=pipeline\tpipeline=%s\tcount=%d\tcost=%d", context.RecommendId, r.pipelineName, len(items), utils.CostTime(start)))
75	return items
76}
77func (s *UserRecommendService) Filter(user *module.User, items []*module.Item, context *context.RecommendContext) []*module.Item {
78	start := time.Now()
79	filterData := filter.FilterData{Data: items, Uid: user.Id, Context: context, PipelineName: s.pipelineName, User: user}
80
81	s.filterService.Filter(&filterData)
82	log.Info(fmt.Sprintf("requestId=%s\tmodule=Filter\tpipeline=%s\tcost=%d", context.RecommendId, s.pipelineName, utils.CostTime(start)))
83	return filterData.Data.([]*module.Item)
84}
85
86func (s *UserRecommendService) Sort(user *module.User, items []*module.Item, context *context.RecommendContext) []*module.Item {
87	start := time.Now()
88	sortData := sort.SortData{Data: items, Context: context, User: user, PipelineName: s.pipelineName}
89
90	s.sortService.Sort(&sortData)
91	log.Info(fmt.Sprintf("requestId=%s\tmodule=Sort\tpipeline=%s\tcost=%d", context.RecommendId, s.pipelineName, utils.CostTime(start)))
92	return sortData.Data.([]*module.Item)
