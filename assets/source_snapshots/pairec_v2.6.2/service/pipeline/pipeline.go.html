32func LoadPipelineConfigs(conf *recconf.RecommendConfig) {
33	userRecommendSceneMap := make(map[string][]*UserRecommendService)
34
35	for sceneName, configs := range conf.PipelineConfs {
36		for _, config := range configs {
37			userRecommendService := NewUserRecommendService(&config)
38
39			userRecommendSceneMap[sceneName] = append(userRecommendSceneMap[sceneName], userRecommendService)
40		}
41	}
42
43	pipelineService.userRecommendSceneMap = userRecommendSceneMap
44}
45func Recommend(user *module.User, context *context.RecommendContext, debugService *debug.DebugService) (ret []*module.Item) {
46	scene := context.GetParameter("scene").(string)
47
48	services, ok := pipelineService.userRecommendSceneMap[scene]
49	if !ok {
50		return
51	}
52
53	ch := make(chan []*module.Item, len(services))
54
55	for _, service := range services {
56		go func(userRecommendService *UserRecommendService) {
57			ch <- userRecommendService.Recommend(user, context, debugService)
58		}(service)
59	}
60
61	for i := 0; i < len(services); i++ {
62		items := <-ch
63		ret = append(ret, items...)
64	}
65	close(ch)
66
67	return
