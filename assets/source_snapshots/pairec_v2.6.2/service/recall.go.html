24func getRecallFromABConfig(recallConfig map[string]interface{}, name, recallNewName string) recall.Recall {
25
26	oldRecall, err := recall.GetRecall(name)
27	if err != nil {
28		return nil
29	}
30
31	mutex.Lock()
32	if newRecall, err := recall.GetRecall(recallNewName); err == nil {
33		mutex.Unlock()
34		return newRecall
35	}
36
37	if m := reflect.ValueOf(oldRecall).MethodByName("CloneWithConfig"); m.IsValid() {
38		if callValues := m.Call([]reflect.Value{reflect.ValueOf(recallConfig)}); len(callValues) == 1 {
39			i := callValues[0].Interface()
40			if newRecall, ok := i.(recall.Recall); ok {
41				recall.RegisterRecall(recallNewName, newRecall)
42				log.Info("register recall :" + recallNewName)
43				mutex.Unlock()
44				return newRecall
45			}
46		}
47	}
48
49	mutex.Unlock()
50	return nil
51}
