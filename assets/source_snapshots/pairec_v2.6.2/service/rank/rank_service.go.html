49func (r *RankService) AddRanks(sceneName string, ranks ...IRank) {
50	var rankInters []IRank
51	if preRanks, ok := r.rankInters[sceneName]; ok {
52		rankInters = preRanks
53	}
54
55	for _, rank := range ranks {
56		rankInters = append(rankInters, rank)
57	}
58
59	r.rankInters[sceneName] = rankInters
60}

62func (r *RankService) GetRanks(sceneName string, context *context.RecommendContext) (ret []IRank) {
63	// first find experiment info
64	if context.ExperimentResult != nil {
65		rankconf := context.ExperimentResult.GetExperimentParams().Get("coldStartRankConf", "")
66		if rankconf != "" {
67			var rankConfig recconf.ColdStartRankConfig
68			d, _ := json.Marshal(rankconf)
69			if err := json.Unmarshal(d, &rankConfig); err == nil {
70				rank := NewColdStartRank(&rankConfig)
71				ret = append(ret, rank)
72				// when rank is not ColdStartRank, add it
73				if ranks, exist := r.rankInters[sceneName]; exist {
74					for _, rank := range ranks {
75						if _, ok := rank.(*ColdStartRank); !ok {
76							ret = append(ret, rank)
77						}
78					}
79				}
80				return
81			} else {
82				context.LogError(fmt.Sprintf("Unmarshal rank config error\terr=%v\tconfig=%s", err, rankconf))
83			}
84		}
85	}
86
87	if ranks, ok := r.rankInters[sceneName]; ok {
88		ret = ranks
89	}
90
91	return
92}

102func (r *RankService) Rank(user *module.User, items []*module.Item, context *context.RecommendContext) {
103	start := time.Now()
104	if context.Debug {
105		data, _ := json.Marshal(user)
106		size := len(data)
107		for i := 0; i < size; {
108			end := i + 4096
109			if end >= size {
110				end = size
111			} else {
112				for end > i {
113					if data[end] == ',' {
114						end++
115						break
116					}
117					end--
118				}
119				if end == i {
120					end = i + 4096
121				}
122			}
123			log.Info(fmt.Sprintf("requestId=%s\tuser=%s", context.RecommendId, string(data[i:end])))
124			i = end
125		}
126	}
127
128	rankItems := items
129	algoDataList := make([]IAlgoData, 0)
130	scene := context.GetParameter("scene").(string)
131
132	i := 0
133	var customRanks []*customRank
134	for _, rank := range r.GetRanks(scene, context) {
135		customRank := newCustomRank(rank)
136		customRanks = append(customRanks, customRank)
137	}
138
139	// find rank config
140	var rankConfig recconf.RankConfig
141	var rankscore string
142	found := false
143	if context.ExperimentResult != nil {
144		rankscore = context.ExperimentResult.GetExperimentParams().GetString("rankscore", "")
145		rankconf := context.ExperimentResult.GetExperimentParams().Get("rankconf", "")
146		if rankconf != "" {
147			d, _ := json.Marshal(rankconf)
148			if err := json.Unmarshal(d, &rankConfig); err == nil {
149				found = true
150			}
151		}
152	}
153	if !found {
154		if rankConfigs, ok := recconf.Config.RankConf[scene]; ok {
155			rankConfig = rankConfigs
156		}
157	}
158
159	if rankscore != "" {
160		rankConfig.RankScore = rankscore
161	}
162
163	batchCount := 100
164	if rankConfig.BatchCount > 0 {
165		batchCount = rankConfig.BatchCount
166	}
167	algoGenerator := CreateAlgoDataGenerator(rankConfig.Processor, rankConfig.ContextFeatures)
168
169	var userFeatures map[string]interface{}
170
171	if rankConfig.Processor == eas.Eas_Processor_EASYREC {
172		userFeatures = user.MakeUserFeatures2()
173		algoGenerator.SetItemFeatures(rankConfig.ItemFeatures)
174	} else {
175		userFeatures = user.MakeUserFeatures()
176	}
177
178	var filter bool
179	for _, item := range rankItems {
180		filter = false
181		if len(customRanks) > 0 {
182			for _, rank := range customRanks {
183				if rank.rankInter.Filter(user, item, context) {
184					filter = true
185					if _, ok := rank.rankInter.(*ColdStartRank); ok {
186						rank.appendFeature(nil, item, context)
187					} else {
188						rank.appendFeature(userFeatures, item, context)
189					}
190					break
191				}
192			}
193		}
194
195		if filter {
196			continue
197		}
198
199		var features map[string]any
200		if rankConfig.Processor == eas.Eas_Processor_EASYREC {
201			if len(rankConfig.ContextFeatures) > 0 || len(rankConfig.ItemFeatures) > 0 {
202				features = item.GetFeatures()
203			}
204		} else {
205			features = item.GetFeatures()
206		}
207		algoGenerator.AddFeatures(item, features, userFeatures)
208		i++
209		if i%batchCount == 0 {
210			var algoData IAlgoData
211			if context.Debug {
212				algoData = algoGenerator.GeneratorAlgoDataDebugWithLevel(100)
213			} else {
214				algoData = algoGenerator.GeneratorAlgoData()
215			}
216			algoDataList = append(algoDataList, algoData)
217		}
218	}
219
220	if algoGenerator.HasFeatures() {
221		var algoData IAlgoData
222		if context.Debug {
223			algoData = algoGenerator.GeneratorAlgoDataDebugWithLevel(100)
224		} else {
225			algoData = algoGenerator.GeneratorAlgoData()
226		}
227		algoDataList = append(algoDataList, algoData)
228	}
229
230	var rankWG sync.WaitGroup
231	// invoke custom rank
232	if len(customRanks) > 0 {
233		for _, rank := range customRanks {
234			rankWG.Add(1)
235			go func(customRank *customRank) {
236				defer rankWG.Done()
237				customRank.rankInter.Rank(user, customRank.items, customRank.requestData, context)
238			}(rank)
239		}
240	}
241
242	if len(algoDataList) == 0 {
243		if len(customRanks) > 0 {
244			rankWG.Wait()
245		}
246		return
247	}
248	requestCh := make(chan IAlgoData, len(algoDataList))
249	responseCh := make(chan IAlgoData, len(algoDataList))
250	defer close(requestCh)
251	defer close(responseCh)
252
253	for _, data := range algoDataList {
254		requestCh <- data
255	}
256
257	gCount := len(algoDataList)
258	for i := 0; i < gCount; i++ {
259		go func() {
260			algoData := <-requestCh
261
262			var wg sync.WaitGroup
263			for _, algoName := range rankConfig.RankAlgoList {
264				wg.Add(1)
265				go func(algo string) {
266					defer wg.Done()
267					ret, err := algorithm.Run(algo, algoData.GetFeatures())
268					if err != nil {
269						log.Error(fmt.Sprintf("requestId=%s\terror=run algorithm error(%v)", context.RecommendId, err))
270						algoData.SetError(err)
271					} else {
272						if result, ok := ret.([]response.AlgoResponse); ok {
273							algoData.SetAlgoResult(algo, result)
274						}
275					}
276
277				}(algoName)
278
279			}
280			wg.Wait()
281			responseCh <- algoData
282		}()
283	}
284
285	exprAst, err := ast.GetExpASTWithType(rankConfig.RankScore, rankConfig.ASTType)
286	if err != nil {
287		log.Error(fmt.Sprintf("requestId=%s\tmodule=rank\trankscore=%s\terror=%v", context.RecommendId, rankConfig.RankScore, err))
288	}
289
290	var scoreRewriteAst map[string]ast.ExprAST
291	if len(rankConfig.ScoreRewrite) > 0 {
292		scoreRewriteAst = make(map[string]ast.ExprAST, len(rankConfig.ScoreRewrite))
293		for source, sourceExpr := range rankConfig.ScoreRewrite {
294			ast, err := ast.GetExpASTWithType(sourceExpr, rankConfig.ASTType)
295			if err != nil {
296				log.Info(fmt.Sprintf("requestId=%s\tmodule=rank\tscorerewrite=%s\terror=%v", context.RecommendId, rankConfig.RankScore, err))
297				continue
298			}
299			scoreRewriteAst[source] = ast
300		}
301	}
302
303	for i := 0; i < gCount; i++ {
304		algoData := <-responseCh
305		if algoData.Error() == nil && algoData.GetAlgoResult() != nil {
306			for name, algoResult := range algoData.GetAlgoResult() {
307				itemList := algoData.GetItems()
308				for j := 0; j < len(algoResult) && j < len(itemList); j++ {
309					if algoResult[j].GetModuleType() {
310						arr_score := algoResult[j].GetScoreMap()
311						for k, v := range arr_score {
312							itemList[j].AddAlgoScore(name+"_"+k, v)
313						}
314					} else if resp, ok := algoResult[j].(response.AlgoMultiClassifyResponse); ok {
315						arr_score := resp.GetClassifyMap()
316						for k, scores := range arr_score {
317							if len(scores) == 1 {
318								itemList[j].AddAlgoScore(name+"_"+k, scores[0])
319							} else if len(scores) > 1 {
320								for i, score := range scores {
321									itemList[j].AddAlgoScore(name+"_"+k+"_"+strconv.Itoa(i), score)
322								}
323								itemList[j].AddProperty(name+"_"+k, scores)
324							}
325						}
326					} else {
327						itemList[j].AddAlgoScore(name, algoResult[j].GetScore())
328					}
329				}
330			}
331		}
332
333		if rankConfig.RankScore != "" {
334			itemList := algoData.GetItems()
335			for k := range itemList {
336				// score rewrite 重写
337				if len(rankConfig.ScoreRewrite) > 0 {
338					scores := make(map[string]float64, len(rankConfig.ScoreRewrite))
339					for source := range rankConfig.ScoreRewrite {
340						if exprAst, ok := scoreRewriteAst[source]; ok {
341							scores[source] = ast.ExprASTResultWithType(exprAst, AstParameterData{context, itemList[k]}, rankConfig.ASTType)
342						} else {
343							scores[source] = 0
344						}
345					}
346					itemList[k].AddAlgoScores(scores)
347				}
348
349				if exprAst != nil {
350					itemList[k].Score = ast.ExprASTResultWithType(exprAst, AstParameterData{context, itemList[k]}, rankConfig.ASTType)
351				}
352
353				if boostScoreFunc != nil {
354					itemList[k].Score = boostScoreFunc(itemList[k].Score, user, itemList[k], context)
355				}
356			}
357		}
358	}
359
360	if len(customRanks) > 0 {
361		rankWG.Wait()
362	}
363
364	go r.featureConsistencyJobService.LogRankResult(user, items, context)
365	log.Info(fmt.Sprintf("requestId=%s\tmodule=rank\tcost=%d", context.RecommendId, utils.CostTime(start)))
366}
