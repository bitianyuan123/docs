104func (g *AlgoDataGenerator) AddFeatures(item *module.Item, itemFeatures map[string]interface{}, userFeatures map[string]interface{}) {
105	if item != nil {
106		g.requestItem = append(g.requestItem, item)
107	}
108
109	features := make(map[string]interface{}, len(itemFeatures)+len(userFeatures))
110	for k, v := range userFeatures {
111		features[k] = v
112	}
113	for k, v := range itemFeatures {
114		features[k] = v
115	}
116
117	g.requestData = append(g.requestData, features)
118}
119func (g *AlgoDataGenerator) GeneratorAlgoData() IAlgoData {

131	}
132
133	g.requestData = g.requestData[:0]
134	g.requestItem = g.requestItem[:0]
135	return algoData

149type feature struct {
150	name      string
151	valueType reflect.Type
152}
153
154func (f *feature) defaultValue() interface{} {
155	switch f.valueType.Kind() {
156	case reflect.Int, reflect.Int8, reflect.Int16:
157		return int(0)
158	case reflect.Int32:
159		return int32(0)
160	case reflect.Int64:
161		return int64(0)
162	case reflect.Float64:
163		return float64(0)
164	case reflect.Float32:
165		return float32(0)
166	case reflect.String:
167		return ""
168	default:
169		return ""
170	}
171}

184func NewEasyrecAlgoDataGenerator(contextFeatures []string) *EasyrecAlgoDataGenerator {
185	generator := &EasyrecAlgoDataGenerator{
186		requestItem:     make([]*module.Item, 0, 100),
187		contextFeatures: make(map[string][]interface{}, 8),

199	}

204// SetItemFeatures implements IAlgoDataGenerator.
205func (generator *EasyrecAlgoDataGenerator) SetItemFeatures(inputItemFeatures []string) {
206	if len(inputItemFeatures) > 0 {
207		generator.inputItemFeatureMap = make(map[string][]interface{}, 8)
208		if inputItemFeatures[0] != "*" {
209			generator.parseInputItemFeature = true
210			for _, featureName := range inputItemFeatures {
211				feature := &feature{
212					name:      featureName,
213					valueType: reflect.TypeOf(""),
214				}
215				generator.inputItemFeatures = append(generator.inputItemFeatures, feature)
216			}
217		}
218	} else {
219		generator.parseInputItemFeature = true
220	}
221}

223func (g *EasyrecAlgoDataGenerator) AddFeatures(item *module.Item, itemFeatures map[string]interface{}, userFeatures map[string]interface{}) {
224	if item != nil {
225		g.requestItem = append(g.requestItem, item)
226	}
227	if !g.parseFeature {
228		for k, v := range itemFeatures {
229			feature := &feature{
230				name:      k,
231				valueType: reflect.TypeOf(v),
232			}
233			g.itemFeatures = append(g.itemFeatures, feature)
234		}
235
236		g.userFeatures = userFeatures
237		g.parseFeature = true
238	}
239	if !g.parseInputItemFeature {
240		var contextItemFeatureNames []string
241		for _, v := range g.itemFeatures {
242			contextItemFeatureNames = append(contextItemFeatureNames, v.name)
243		}
244		for k, v := range itemFeatures {
245			if utils.IndexOfArray(contextItemFeatureNames, k) == -1 {
246				feature := &feature{
247					name:      k,
248					valueType: reflect.TypeOf(v),
249				}
250				g.inputItemFeatures = append(g.inputItemFeatures, feature)
251			}
252		}
253		g.parseInputItemFeature = true
254
255	}
256	if len(g.userFeatures) == 0 {
257		g.userFeatures = userFeatures
258	}
259
260	for _, f := range g.itemFeatures {
261		if v, ok := itemFeatures[f.name]; ok {
262			g.contextFeatures[f.name] = append(g.contextFeatures[f.name], v)
263		} else {
264			g.contextFeatures[f.name] = append(g.contextFeatures[f.name], f.defaultValue())
265		}
266	}
267	for _, f := range g.inputItemFeatures {
268		if v, ok := itemFeatures[f.name]; ok {
269			g.inputItemFeatureMap[f.name] = append(g.inputItemFeatureMap[f.name], v)
270		} else {
271			g.inputItemFeatureMap[f.name] = append(g.inputItemFeatureMap[f.name], f.defaultValue())
272		}
273	}
274}

276func (g *EasyrecAlgoDataGenerator) GeneratorAlgoData() IAlgoData {
277	copyItems := make([]*module.Item, len(g.requestItem))
278	copy(copyItems, g.requestItem)
279
280	builder := easyrec.NewEasyrecRequestBuilder()
281	for k, v := range g.userFeatures {
282		builder.AddUserFeature(k, v)
283	}
284	for _, item := range g.requestItem {
285		builder.AddItemId(string(item.Id))
286	}
287	for k, v := range g.contextFeatures {
288		builder.AddContextFeature(k, v)
289		g.contextFeatures[k] = g.contextFeatures[k][:0]
290	}
291	for k, v := range g.inputItemFeatureMap {
292		builder.AddItemFeature(k, v)
293		g.inputItemFeatureMap[k] = g.inputItemFeatureMap[k][:0]
294	}
295
296	algoData := &EasyrecAlgoData{
297		AlgoDataBase: &AlgoDataBase{
298			Items:      copyItems,
299			AlgoResult: make(map[string][]response.AlgoResponse),
300		},
301		easyrecRequest: builder.EasyrecRequest(),
302	}
303
304	g.requestItem = g.requestItem[:0]
305	return algoData
306}
