28type EasModel struct {

34func NewEasModel(name string) *EasModel {

37func (m *EasModel) Init(conf *recconf.AlgoConfig) error {
38	if conf.EasConf.Processor == Eas_Processor_PMML {
39		req := PMMLRequest{}
40		req.SetUrl(conf.EasConf.Url)
41		req.SetAuth(conf.EasConf.Auth)
42		req.SetTimeout(conf.EasConf.Timeout)
43		req.SetResponseFunc(conf.EasConf.ResponseFuncName)
44		m.request = &req
45
46		if conf.EasConf.RetryTimes > 0 {
47			m.retryTimes = conf.EasConf.RetryTimes
48		}
49		return nil
50	} else if conf.EasConf.Processor == Eas_Processor_TF {
51		req := TFRequest{}
52		req.SetUrl(conf.EasConf.Url)
53		req.SetAuth(conf.EasConf.Auth)
54		req.SetSignatureName(conf.EasConf.SignatureName)
55		req.SetTimeout(conf.EasConf.Timeout)
56		req.SetResponseFunc(conf.EasConf.ResponseFuncName)
57		if len(conf.EasConf.Outputs) > 0 {
58			req.SetOutputs(conf.EasConf.Outputs)
59		}
60		m.request = &req
61
62		if conf.EasConf.RetryTimes > 0 {
63			m.retryTimes = conf.EasConf.RetryTimes
64		}
65		return nil
66	} else if conf.EasConf.Processor == Eas_Processor_FM {
67		req := NewFMRequest()
68		req.SetUrl(conf.EasConf.Url)
69		req.SetAuth(conf.EasConf.Auth)
70		req.SetTimeout(conf.EasConf.Timeout)
71		req.SetResponseFunc(conf.EasConf.ResponseFuncName)
72		m.request = req
73
74		if conf.EasConf.RetryTimes > 0 {
75			m.retryTimes = conf.EasConf.RetryTimes
76		}
77		return nil
78
79	} else if conf.EasConf.Processor == Eas_Processor_EASYREC && conf.EasConf.EndpointType == eas.EndpointTypeDocker {
80		req := HttpEasyrecRequest{}
81		req.SetUrl(conf.EasConf.Url)
82		req.SetAuth(conf.EasConf.Auth)
83		req.SetTimeout(conf.EasConf.Timeout)
84		req.SetResponseFunc(conf.EasConf.ResponseFuncName)
85		m.request = &req
86
87		if conf.EasConf.RetryTimes > 0 {
88			m.retryTimes = conf.EasConf.RetryTimes
89		}
90		return nil
91	} else if conf.EasConf.Processor == Eas_Processor_EASYREC {
92		req := EasyrecRequest{}
93		req.SetUrl(conf.EasConf.Url)
94		req.SetAuth(conf.EasConf.Auth)
95		req.SetTimeout(conf.EasConf.Timeout)
96		req.SetResponseFunc(conf.EasConf.ResponseFuncName)
97
98		var client *eas.PredictClient
99		if conf.EasConf.EndpointType == eas.EndpointTypeDirect {
100			url := strings.ReplaceAll(conf.EasConf.Url, "http://", "")
101			index := strings.Index(url, "/api/predict/")
102			endpoint := url[:index]
103			strs := strings.Split(endpoint, ".")
104			region := strs[2]
105			for i := 0; i < len(strs); i++ {
106				if strs[i] == "pai-eas" {
107					region = strs[i-1]
108					break
109				}
110			}
111			name := url[index+len("/api/predict/"):]
112			client = eas.NewPredictClient(fmt.Sprintf("pai-eas-vpc.%s.aliyuncs.com", region), name)
113			client.SetEndpointType(eas.EndpointTypeDirect)
114		} else {
115			url := strings.ReplaceAll(conf.EasConf.Url, "http://", "")
116			index := strings.Index(url, "/api/predict/")
117			endpoint := url[:index]
118			name := url[index+len("/api/predict/"):]
119			client = eas.NewPredictClient(endpoint, name)
120		}
121		client.SetToken(conf.EasConf.Auth)
122		client.SetTimeout(conf.EasConf.Timeout)
123		client.SetHttpTransport(&http.Transport{
124			MaxConnsPerHost:       2000,
125			MaxIdleConnsPerHost:   300,
126			MaxIdleConns:          300,
127			TLSHandshakeTimeout:   100 * time.Millisecond,
128			ExpectContinueTimeout: 200 * time.Millisecond,
129			DialContext: (&net.Dialer{
130				Timeout:   100 * time.Millisecond, // 100ms
131				KeepAlive: 10 * time.Minute,
132			}).DialContext,
133		})
134		if conf.EasConf.RetryTimes > 0 {
135			m.retryTimes = conf.EasConf.RetryTimes
136		}
137		client.SetRetryCount(m.retryTimes - 1)
138		// if use eas sdk, we should not retry
139		m.retryTimes = 1
140		client.Init()
141		req.EasClient = client
142		m.request = &req
143
144		return nil
145	} else if conf.EasConf.Processor == Eas_Processor_LINUCB {
146		req := LincubRequest{}
147		req.SetUrl(conf.EasConf.Url)
148		req.SetAuth(conf.EasConf.Auth)
149		req.SetTimeout(conf.EasConf.Timeout)
150		req.SetResponseFunc(conf.EasConf.ResponseFuncName)
151		m.request = &req
152
153		if conf.EasConf.RetryTimes > 0 {
154			m.retryTimes = conf.EasConf.RetryTimes
155		}
156		return nil
157
158	} else if conf.EasConf.Processor == Eas_Processor_TFServing {
159		req := TFServingRequest{}
160		req.SetUrl(conf.EasConf.Url)
161		req.SetModelName(conf.EasConf.ModelName)
162		req.SetAuth(conf.EasConf.Auth)
163		req.SetSignatureName(conf.EasConf.SignatureName)
164		req.SetTimeout(conf.EasConf.Timeout)
165		req.SetResponseFunc(conf.EasConf.ResponseFuncName)
166		if len(conf.EasConf.Outputs) > 0 {
167			req.SetOutputs(conf.EasConf.Outputs)
168		}
169
170		var opts []grpc.DialOption
171		opts = append(opts, grpc.WithTransportCredentials(insecure.NewCredentials()))
172
173		conn, err := grpc.Dial(req.url, opts...)
174		if err != nil {
175			panic(fmt.Sprintf("fail to dial: %v", err))
176		}
177
178		client := tensorflow_serving.NewPredictionServiceClient(conn)
179		req.Client = client
180		m.request = &req
181
182		if conf.EasConf.RetryTimes > 0 {
183			m.retryTimes = conf.EasConf.RetryTimes
184		}
185		return nil
186	}
187
188	return errors.New("not found eas Processor:" + conf.EasConf.Processor)
189}
190func (m *EasModel) Run(algoData interface{}) (interface{}, error) {
191	retryTimes := m.retryTimes
192
193	var (
194		data interface{}
195		err  error
196	)
197
198	for {
199		retryTimes--
200		data, err = m.request.Invoke(algoData)
201		if err != nil && retryTimes == 0 {
202			return data, err
203		} else if err == nil {
204			break
205		} else if err != nil {
206			log.Warning("eas request fail :" + err.Error())
207		}
208	}
209
210	if m.request.GetResponseFunc() != nil {
211		return m.request.GetResponseFunc()(data)
212	}
213
214	return data, err
215}
