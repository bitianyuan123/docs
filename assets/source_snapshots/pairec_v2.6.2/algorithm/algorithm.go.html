28type IAlgorithm interface {
29	Init(conf *recconf.AlgoConfig) error
30	Run(algoData interface{}) (interface{}, error)
31}

33type AlgorithmFactory struct {
34	algorithms       map[string]IAlgorithm
35	requestDataFuncs map[string]RequestDataFunc
36	mutex            sync.RWMutex
37	algorithmSigns   map[string]string
38}

40func NewAlgorithmFactory() *AlgorithmFactory {
41	factory := AlgorithmFactory{
42		algorithmSigns: make(map[string]string, 2),
43	}
44	factory.algorithms = make(map[string]IAlgorithm)
45	factory.requestDataFuncs = make(map[string]RequestDataFunc)
46
47	return &factory
48}
49func (a *AlgorithmFactory) Init(algoConfs []recconf.AlgoConfig) {
50	a.mutex.Lock()
51	defer a.mutex.Unlock()
52
53	for _, conf := range algoConfs {
54		sign, _ := json.Marshal(conf)
55		if _, ok := a.algorithms[conf.Name]; ok {
56			if utils.Md5(string(sign)) == a.algorithmSigns[conf.Name] {
57				continue
58			}
59		}
60		algo, err := a.initAlgo(conf)
61		if err != nil {
62			log.Error(err.Error())
63			continue
64		}
65		a.algorithms[conf.Name] = algo
66		a.algorithmSigns[conf.Name] = utils.Md5(string(sign))
67	}
68}
69func (a *AlgorithmFactory) initAlgo(conf recconf.AlgoConfig) (IAlgorithm, error) {
70	var algo IAlgorithm
71	if conf.Type == "EAS" {
72		algo = eas.NewEasModel(conf.Name)
73		err := algo.Init(&conf)
74		if err != nil {
75			return nil, fmt.Errorf("init algorithm error, name:%s, err:%v", conf.Name, err)
76		}
77	} else if conf.Type == "FAISS" {
78		algo = faiss.NewFaissModel(conf.Name)
79		err := algo.Init(&conf)
80		if err != nil {
81			return nil, fmt.Errorf("init algorithm error, name:%s, err:%v", conf.Name, err)
82		}
83	} else if conf.Type == "LOOKUP" {
84		algo = NewLookupPolicy()
85		err := algo.Init(&conf)
86		if err != nil {
87			return nil, fmt.Errorf("init algorithm error, name:%s, err:%v", conf.Name, err)
88		}
89	} else if conf.Type == "SELDON" {
90		algo = new(seldon.Model)
91		err := algo.Init(&conf)
92		if err != nil {
93			return nil, fmt.Errorf("init algorithm error, name:%s, err:%v", conf.Name, err)
94		}
95	} else if conf.Type == "TFSERVING" {
96		algo = tfserving.NewTFservingModel(conf.Name)
97		err := algo.Init(&conf)
98		if err != nil {
99			return nil, fmt.Errorf("init algorithm error, name:%s, err:%v", conf.Name, err)
100		}
101
102	} else {
103		return nil, fmt.Errorf("algorithm type not support , type:%s", conf.Type)
104	}
105	return algo, nil
106}
107func (a *AlgorithmFactory) Run(name string, algoData interface{}) (interface{}, error) {
108	a.mutex.RLock()
109	algo, found := a.algorithms[name]
110	f, funcFound := a.requestDataFuncs[name]
111	a.mutex.RUnlock()
112	if !found {
113		return nil, errors.New("not found algorithm, name:" + name)
114	}
115	// if find request data func
116	if funcFound {
117		return algo.Run(f(name, algoData))
118	}
119	return algo.Run(algoData)
120}

129func AddAlgo(conf recconf.AlgoConfig) {
130	algoFactory.mutex.Lock()
131	defer algoFactory.mutex.Unlock()
132	_, found := algoFactory.algorithms[conf.Name]
133	if found {
134		return
135	}
136	algo, err := algoFactory.initAlgo(conf)
137	if err != nil {
138		log.Error(err.Error())
139		return
140	}
141	algoFactory.algorithms[conf.Name] = algo
142}
143func AddAlgoWithSign(conf recconf.AlgoConfig) {
144	algoFactory.mutex.Lock()
145	defer algoFactory.mutex.Unlock()
146	var signStr string
147	sign, _ := json.Marshal(conf)
148	signStr = utils.Md5(string(sign))
149	_, found := algoFactory.algorithms[conf.Name]
150	if found {
151		if signStr == algoFactory.algorithmSigns[conf.Name] {
152			return
153		}
154	}
155	algo, err := algoFactory.initAlgo(conf)
156	if err != nil {
157		log.Error(err.Error())
158		return
159	}
160	algoFactory.algorithms[conf.Name] = algo
161	algoFactory.algorithmSigns[conf.Name] = signStr
162}
