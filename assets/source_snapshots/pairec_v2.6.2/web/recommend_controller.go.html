33func (r *RecommendParam) GetParameter(name string) interface{} {
34	if name == "uid" {
35		return r.Uid
36	} else if name == "scene" {
37		return r.SceneId
38	} else if name == "category" {
39		if r.Category != "" {
40			return r.Category
41		}
42		return "default"
43	} else if name == "features" {
44		return r.Features
45	}
46
47	return nil
48}

72func (c *RecommendController) Process(w http.ResponseWriter, r *http.Request) {
73	c.Start = time.Now()
74	var err error
75	c.RequestBody, err = io.ReadAll(r.Body)
76	if err != nil {
77		c.SendError(w, ERROR_PARAMETER_CODE, "read parammeter error")
78		return
79	}
80	if len(c.RequestBody) == 0 {
81		c.SendError(w, ERROR_PARAMETER_CODE, "request body empty")
82		return
83	}
84	c.RequestId = utils.UUID()
85	c.LogRequestBegin(r)
86	if err := c.CheckParameter(); err != nil {
87		c.SendError(w, ERROR_PARAMETER_CODE, err.Error())
88		return
89	}
90	c.doProcess(w, r)
91	c.End = time.Now()
92	c.LogRequestEnd(r)
93}
94func (r *RecommendController) CheckParameter() error {
95	if err := json.Unmarshal(r.RequestBody, &r.param); err != nil {
96		return err
97	}
98
99	if len(r.param.Uid) == 0 {
100		return errors.New("uid not empty")
101	}
102	if r.param.Size <= 0 {
103		r.param.Size = Default_Size
104	}
105	if r.param.SceneId == "" {
106		r.param.SceneId = "default_scene"
107	}
108	if r.param.Category == "" {
109		r.param.Category = "default"
110	}
111
112	return nil
113}
114func (c *RecommendController) doProcess(w http.ResponseWriter, r *http.Request) {
115	c.makeRecommendContext()
116	userRecommendService := service.NewUserRecommendService()
117	items := userRecommendService.Recommend(c.context)
118	data := make([]*ItemData, 0)
119	for _, item := range items {
120		if c.param.Debug {
121			fmt.Println(item)
122		}
123
124		idata := &ItemData{
125			ItemId:     string(item.Id),
126			ItemType:   item.ItemType,
127			RetrieveId: item.RetrieveId,
128		}
129
130		data = append(data, idata)
131	}
132
133	if len(data) < c.param.Size {
134		response := RecommendResponse{
135			Size:  len(data),
136			Items: data,
137			Response: Response{
138				RequestId: c.RequestId,
139				Code:      299,
140				Message:   "items size not enough",
141			},
142		}
143		io.WriteString(w, response.ToString())
144		return
145	}
146
147	response := RecommendResponse{
148		Size:  len(data),
149		Items: data,
150		Response: Response{
151			RequestId: c.RequestId,
152			Code:      200,
153			Message:   "success",
154		},
155	}
156	io.WriteString(w, response.ToString())
157}
158func (c *RecommendController) makeRecommendContext() {
159	c.context = context.NewRecommendContext()
160	c.context.Size = c.param.Size
161	c.context.Debug = c.param.Debug
162	c.context.Param = &c.param
163	c.context.RecommendId = c.RequestId
164	c.context.Config = recconf.Config
165
166	abcontext := model.ExperimentContext{
167		Uid:          c.param.Uid,
168		RequestId:    c.RequestId,
169		FilterParams: map[string]interface{}{},
170	}
171
172	if abtest.GetExperimentClient() != nil {
173		c.context.ExperimentResult = abtest.GetExperimentClient().MatchExperiment(c.param.SceneId, &abcontext)
174		log.Info(c.context.ExperimentResult.Info())
175	}
