45func (ss *SortService) AddSort(scene string, s ISort) {
46	sorts, ok := ss.SortStrategies[scene]
47	if ok {
48		for _, sort := range sorts {
49			if sort == s {
50				return
51			}
52		}
53
54		sorts = append(sorts, s)
55		ss.SortStrategies[scene] = sorts
56	} else {
57		sorts := []ISort{s}
58		ss.SortStrategies[scene] = sorts
59	}
60}

65func (ss *SortService) Sort(data *SortData, tag string) {
66	var scene string
67	ctx := data.Context
68	s := ctx.GetParameter("scene")
69	if _, ok := s.(string); ok {
70		scene = s.(string)
71	}
72	var categoryName string
73	c := ctx.GetParameter("category")
74	if _, ok := c.(string); ok {
75		categoryName = c.(string)
76	} else {
77		categoryName = "default"
78	}
79
80	sorts := make([]ISort, 0)
81	if ctx.ExperimentResult != nil {
82		names := ctx.ExperimentResult.GetExperimentParams().Get(categoryName+".SortNames", nil)
83		if names != nil {
84			if values, ok := names.([]interface{}); ok {
85				for _, v := range values {
86					if name, okay := v.(string); okay {
87						if sort, found := sortMapping[name]; found {
88							sorts = append(sorts, sort)
89						}
90					}
91				}
92			}
93		}
94	}
95
96	if len(sorts) == 0 {
97		scene = scene + tag
98		var ok bool
99		sorts, ok = ss.SortStrategies[scene]
100		if !ok {
101			sorts, ok = ss.SortStrategies[categoryName]
102		}
103		if !ok || sorts == nil || len(sorts) == 0 {
104			sorts = make([]ISort, 1)
105			sorts[0] = NewItemRankScoreSort()
106			ctx.LogInfo(fmt.Sprintf("defaultSort=ItemRankScore\tscene=%s", scene))
107		}
108	}
109
110	for _, sort := range sorts {
111		newSort := sort
112		if cloneSort, ok := sort.(ICloneSort); ok && ctx.ExperimentResult != nil {
113			sortConfig := ctx.ExperimentResult.GetExperimentParams().Get("sort."+cloneSort.GetSortName(), nil)
114			if sortConfig != nil {
115				if params, ok := sortConfig.(map[string]interface{}); ok {
116					if sortInstance := cloneSort.CloneWithConfig(params); !utils.IsNil(sortInstance) {
117						newSort = sortInstance
118					}
119				}
120			}
121		}
122
123		newSort.Sort(data)
124	}
125}

127func Load(config *recconf.RecommendConfig) {
128	for scene, names := range config.SortNames {
129		var sorts []ISort

162func RegisterSortWithConfig(config *recconf.RecommendConfig) {
163	for _, conf := range config.SortConfs {
164		if _, ok := sortMapping[conf.Name]; ok {
165			sign, _ := json.Marshal(&conf)
166			if utils.Md5(string(sign)) == sortSigns[conf.Name] {
167				continue
168			}
169		}
170
171		var s ISort
172		if conf.SortType == "DPPSort" {
173			s = NewDPPSort(conf.DPPConf)
174		} else if conf.SortType == "SSDSort" {
175			s = NewSSDSort(conf.SSDConf)
176		} else if conf.SortType == "MultiRecallMixSort" {
177			s = NewMultiRecallMixSort(conf)
178		} else if conf.SortType == "BoostScoreSort" {
179			s = NewBoostScoreSort(conf)
180		} else if conf.SortType == "DiversityRuleSort" {
181			s = NewDiversityRuleSort(conf)
182		} else if conf.SortType == "AlgoScoreSort" {
183			s = NewAlgoScoreSort(conf)
184		} else if conf.SortType == "TrafficControlSort" {
185			s = NewTrafficControlSort(conf)
186		} else if conf.SortType == "BoostScoreByWeight" {
187			s = NewBoostScoreByWeight(conf)
188		} else if conf.SortType == "DistinctIdSort" {
189			s = NewDistinctIdSort(conf)
190		}
191
192		if s == nil {
193			panic("Sort is nil, name:" + conf.Name)
194		}
195
196		sign, _ := json.Marshal(&conf)
197		registerSortWithSign(conf.Name, s, utils.Md5(string(sign)))
198	}
199}
200func registerSortWithSign(name string, sort ISort, sign string) {
201	sortMapping[name] = sort
202	sortSigns[name] = sign
203}
