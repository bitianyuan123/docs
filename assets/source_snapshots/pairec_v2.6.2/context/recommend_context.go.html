39func (r *RecommendContext) GetParameterByPath(path string) interface{} {
40	if strings.Contains(path, ".") {
41		pos := strings.Index(path, ".")
42		root := path[:pos]
43		features := r.Param.GetParameter(root)
44		if features == nil {
45			return nil
46		}
47		newPath := path[pos+1:]
48		if newPath == "" {
49			return features
50		}
51		if dict, ok := features.(map[string]interface{}); ok {
52			b, err := json.Marshal(dict)
53			if err != nil {
54				r.LogError(fmt.Sprintf("GetParameterByPath Marshal fail: %v", err))
55				return nil
56			}
57			value := gjson.Get(string(b), newPath)
58			return value.String()
59		} else if feature, okay := features.(string); okay {
60			value := gjson.Get(feature, newPath)
61			return value.String()
62		} else {
63			r.LogError("GetParameterByPath fail: " + path)
64			return nil
65		}
66	}
67	return r.Param.GetParameter(path)
68}
