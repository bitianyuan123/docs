43func (r *Redis) Init() error {
44
45	dialFunc := func() (redis.Conn, error) {
46		var (
47			ip  string
48			err error
49		)
50
51		ip, err = netutil.GetAddrByHost(r.Host)
52		if err != nil {
53			ip = r.Host
54		}
55		addr := fmt.Sprintf("%s:%d", ip, r.Port)
56		conn, err := redis.DialTimeout("tcp", addr, r.connectTimeout, r.readTimeout, r.writeTimeout)
57		if err != nil {
58			// use Host:Port try again
59			addr = fmt.Sprintf("%s:%d", r.Host, r.Port)
60			conn, err = redis.DialTimeout("tcp", addr, r.connectTimeout, r.readTimeout, r.writeTimeout)
61			if err != nil {
62				return nil, err
63			}
64		}
65
66		if len(r.Password) != 0 {
67			_, err = conn.Do("AUTH", r.Password)
68			if err != nil {
69				conn.Close()
70				return nil, err
71			}
72		}
73		_, selecterr := conn.Do("SELECT", r.DbNum)
74		if selecterr != nil {
75			conn.Close()
76			return nil, selecterr
77		}
78		return conn, nil
79	}
80
81	r.Pool = &redis.Pool{
82		MaxIdle:     r.MaxIdle,
83		IdleTimeout: 180 * time.Second,
84		Dial:        dialFunc,
85		TestOnBorrow: func(c redis.Conn, t time.Time) error {
86			if time.Since(t) < time.Minute {
87				return nil
88			}
89			_, err := c.Do("PING")
90			return err
91		},
92	}
93
94	c := r.Pool.Get()
95	defer c.Close()
96
97	return c.Err()
98}
99func (r *Redis) Get() redis.Conn {
100	return r.Pool.Get()

102func Load(config *recconf.RecommendConfig) {
103	for name, conf := range config.RedisConfs {
104		redisConfs[name] = &conf
105		if _, ok := redisPools[name]; ok {
106			continue
107		}
108
109		r := &Redis{
110			Host:     conf.Host,
111			Password: conf.Password,
112			Port:     conf.Port,
113			MaxIdle:  conf.MaxIdle,
114			DbNum:    conf.DbNum,
115		}
116
117		r.connectTimeout = time.Millisecond * time.Duration(50)
118		if conf.ConnectTimeout != 0 {
119			r.connectTimeout = time.Millisecond * time.Duration(conf.ConnectTimeout)
120		}
121
122		r.readTimeout = time.Millisecond * time.Duration(100)
123		if conf.ReadTimeout != 0 {
124			r.readTimeout = time.Millisecond * time.Duration(conf.ReadTimeout)
125		}
126
127		r.writeTimeout = time.Millisecond * time.Duration(100)
128		if conf.WriteTimeout != 0 {
129			r.writeTimeout = time.Millisecond * time.Duration(conf.WriteTimeout)
130		}
131
132		err := r.Init()
133		if err != nil {
134			panic(fmt.Sprintf("name=%s, err=%v", name, err))
135		}
136		redisPools[name] = r
137	}
