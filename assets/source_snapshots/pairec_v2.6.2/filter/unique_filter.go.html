26func (f *UniqueFilter) doFilter(filterData *FilterData) error {
27	items := filterData.Data.([]*module.Item)
28	newItems := make([]*module.Item, 0, len(items))
29	uniq := make(map[module.ItemId]*module.Item, len(items))
30
31	for _, item := range items {
32		if exist, ok := uniq[item.Id]; !ok {
33			uniq[item.Id] = item
34			newItems = append(newItems, item)
35		} else {
36			algoScores := item.GetAlgoScores()
37			for name, score := range algoScores {
38				exist.AddAlgoScore(name, score)
39			}
40			if exist.RecallScores == nil {
41				exist.RecallScores = map[string]float64{exist.RetrieveId: exist.Score}
42			}
43			exist.RecallScores[item.RetrieveId] = item.Score
44		}
45	}
46
47	filterData.Data = newItems
48	return nil
49}
