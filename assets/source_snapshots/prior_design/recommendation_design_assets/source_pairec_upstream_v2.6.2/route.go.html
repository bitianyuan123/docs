22func Route(pattern string, c ControllerInterface) {
23	reflectVal := reflect.ValueOf(c)
24	t := reflect.Indirect(reflectVal).Type()
25	info := RouteInfo{
26		pattern:        pattern,
27		controllerType: t,
28	}
29
30	info.initialize = func() ControllerInterface {
31		vc := reflect.New(info.controllerType)
32		execController, ok := vc.Interface().(ControllerInterface)
33		if !ok {
34			panic("controller is not ControllerInterface")
35		}
36
37		elemVal := reflect.ValueOf(c).Elem()
38		elemType := reflect.TypeOf(c).Elem()
39		execElem := reflect.ValueOf(execController).Elem()
40
41		numOfFields := elemVal.NumField()
42		for i := 0; i < numOfFields; i++ {
43			fieldType := elemType.Field(i)
44			elemField := execElem.FieldByName(fieldType.Name)
45			if elemField.CanSet() {
46				fieldVal := elemVal.Field(i)
47				elemField.Set(fieldVal)
48			}
49		}
50
51		return execController
52	}
53
54	PairecApp.Handlers.Register(&info)
55}

66func Forward(method, url, body string) *http.Response {
67	readBuf := bytes.NewBufferString(body)
68	req := httptest.NewRequest(method, url, readBuf)
69	w := httptest.NewRecorder()
70	PairecApp.Handlers.ServeHTTP(w, req)
